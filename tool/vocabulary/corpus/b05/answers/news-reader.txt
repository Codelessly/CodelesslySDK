### Saved
project savedIds=[] saved
project offlineIds=[] saved
state filter="All"
state stories=[{"id":"s1","sec":"World","title":"Coastal cities sign landmark flood-defence pact","mins":6,"size":"1.2 MB"},{"id":"s2","sec":"Business","title":"Rail freight rebounds as ports clear backlog","mins":4,"size":"0.9 MB"},{"id":"s3","sec":"Science","title":"A cheaper way to read ancient ink without opening the scroll","mins":5,"size":"1.5 MB"},{"id":"s5","sec":"Culture","title":"The quiet revival of the neighbourhood cinema","mins":7,"size":"2.1 MB"}]
App
  Sidebar
    Brand "Broadsheet" icon="newspaper"
    NavItem "Front page" icon="home" onTap={go front_page}
    NavItem "Saved" icon="bookmark" badge="{length(savedIds)}" selected={true}
    Spacer grow
    Text.muted "{length(offlineIds)} stories offline"
  Main
    Nav
      Heading "Saved for later" level=1
      Button.ghost "Clear all" onTap={savedIds=[]; toast "Cleared"}
    Column pad=24 gap=16 scroll
      SegmentedButton bind=filter items=["All","Offline","Online only"]
      if length(savedIds)==0
        EmptyState "Nothing saved yet" "Tap the bookmark on any story to read it later." icon="bookmark_border"
          Button "Browse the front page" onTap={go front_page}
      each s in filter=="Offline" ? filter(stories,x=>contains(offlineIds,x.id) && contains(savedIds,x.id)) : filter=="Online only" ? filter(stories,x=>!contains(offlineIds,x.id) && contains(savedIds,x.id)) : filter(stories,x=>contains(savedIds,x.id))
        ListTile "{s.title}" "{s.sec} · {s.mins} min" onTap={go article}
          if contains(offlineIds,s.id)
            Badge.success "Offline · {s.size}"
          if !contains(offlineIds,s.id)
            Badge.neutral "Online only"
          IconButton "more_vert" tooltip="More" onTap={sel=s.id; open rowMenu}
  Menu#rowMenu end
    MenuItem "Download for offline" icon="download_for_offline" onTap={offlineIds=append(offlineIds,sel); toast "Downloaded"}
    MenuItem "Remove download" icon="cloud_off" onTap={offlineIds=without(offlineIds,sel)}
    MenuItem "Remove from saved" icon="delete" onTap={savedIds=without(savedIds,sel); offlineIds=without(offlineIds,sel)}
