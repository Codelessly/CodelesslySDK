App
  Sidebar
    Brand "Longwave" icon="radio"
    NavItem "Stations" icon="podcasts" onTap={go stations}
    NavItem "Now playing" icon="graphic_eq" onTap={go now-playing}

  Main
    Nav
      Heading "Alarm" level=1
      Toggle "Alarm on" @alarmOn

      Card gap=16
        Text.fieldLabel "Wake at"
        TimePicker @alarmTime
        Text.fieldLabel "Repeat"

        Select @alarmStation items=["KEXP 90.3","FIP","Radio Paradise","NTS 1","Jazz24","Radio Swiss Classic"]

        Row gap=12 middle
          Icon "volume_up"
          Slider @alarmVolume min=0 max=100 grow
        Text.muted "Rings {join(days,', ')} at {alarmTime} to {alarmStation}, volume {alarmVolume}%."
        Row gap=8
          Button.outline "Test now" onTap={nowStation=find(stations, s=>s.name==alarmStation); playing=true; go now-playing}
          Button "Save alarm" onTap={emit saveAlarm {"time":alarmTime,"days":days,"station":alarmStation}; toast "Alarm set for {alarmTime}"}
