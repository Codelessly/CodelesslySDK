      Grid cols=2 gap=16
        Card gap=12
          Text.panelTitle "Foreground"
          Select "Palette color" items={palette} @fg
          Input "Hex" @fg
        Card gap=12
          Text.panelTitle "Background"
          Select "Palette color" items={palette} @bg
          Input "Hex" @bg
      Card gap=16
        Box fill={bg} radius=12 pad=24 gap=8
          Text "The quick brown fox jumps over the lazy dog" size=28 weight=700 color={fg}
          Text "Small body copy to check readability at 14px." color={fg}
        ContrastRatio fg={fg} bg={bg} @ratio
        Text.heading "{fixed(ratio,2)} : 1"
        Row gap=8
          Badge.success "AA normal text" hidden={ratio < 4.5}
          Badge.error "AA normal text" hidden={ratio >= 4.5}
          Badge.success "AA large text" hidden={ratio < 3}
          Badge.error "AA large text" hidden={ratio >= 3}
          Badge.success "AAA normal text" hidden={ratio < 7}
          Badge.error "AAA normal text" hidden={ratio >= 7}
          Badge.success "AAA large text" hidden={ratio < 4.5}
          Badge.error "AAA large text" hidden={ratio >= 4.5}
      Column#shades gap=12
        Row between
          Heading "Shades" level=3
          Select "Base color" items={palette} @base w=200
        Row h=96
          each step in [50,100,200,300,400,500,600,700,800,900]
            Column grow center middle fill={shade(base,step)} onTap={copy shade(base,step); toast "Copied {shade(base,step)}"}
              Text.small "{step}" color=contrast(shade(base,step))
        Text.muted "Tap a step to copy its hex."
