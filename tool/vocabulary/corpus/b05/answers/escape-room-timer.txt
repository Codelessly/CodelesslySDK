### Game master
project seconds=3600
project running=false
project status="playing"
project roomHint=""
project hintQueue=[{"id":1,"team":"Team Ravens","text":"We found the key but not the lock. Any ideas?","at":"41:12"},{"id":2,"team":"Team Ravens","text":"What do the numbers on the globe mean?","at":"38:50"}]
project puzzles=[{"id":1,"name":"Library cipher","sensor":"Bookshelf reed switch","done":true},{"id":2,"name":"Globe coordinates","sensor":"Globe tilt sensor","done":true},{"id":3,"name":"Pressure plate floor","sensor":"Floor mat 4 of 4","done":false},{"id":4,"name":"Radio frequency","sensor":"Radio dial encoder","done":false},{"id":5,"name":"Vault dial","sensor":"Vault door latch","done":false}]
project cams=[{"id":1,"name":"Library","live":true},{"id":2,"name":"Study","live":true},{"id":3,"name":"Cellar","live":true},{"id":4,"name":"Corridor","live":false}]
project presets=["Look closely at the bookshelf","Try the globe again","Count the tiles on the floor","Not everything is a number"]
state typed=""
state sel=0
every 1s when running && seconds>0 onTick={seconds=seconds-1}
every 1s when running && seconds==0 onTick={running=false; status="failed"}
if count(puzzles,p=>p.done)==length(puzzles)
  every 1s onTick={running=false; status="won"}
App
  Sidebar
    Brand "Lockbox" icon="lock"
    NavItem "Control" icon="timer" selected={true}
    NavItem "Room screen" icon="tv" onTap={go room-screen}
    NavItem "Leaderboard" icon="emoji_events" onTap={go leaderboard}
    Spacer grow
    Text.muted "Room 2: The Archivist"
    Text.muted "Game master: Dana"
  Main
    Nav
      Heading "Room 2: The Archivist" level=1
      Row gap=8
        Badge.success "Team Ravens" hidden={status!="playing"}
        Badge.success "Escaped" hidden={status!="won"}
        Badge.error "Time up" hidden={status!="failed"}
        Button.outline "Reset room" onTap={open resetConfirm}
    Column pad=24 gap=16 scroll
      Row gap=16 top
        Card grow gap=12
          Text.caption "TIME REMAINING"
          Text "{duration(seconds)}" size=96 weight=700 color={seconds<300 ? "#FFD32F2F" : "#FF1B1B1B"}
          Progress {seconds/3600}
          Row gap=8
            Button "Start" icon="play_arrow" hidden={running} onTap={running=true; status="playing"}
            Button.secondary "Pause" icon="pause" hidden={!running} onTap={running=false}
            Button.outline "+1 min" onTap={seconds=seconds+60}
            Button.outline "+5 min" onTap={seconds=seconds+300}
            Button.outline "-1 min" onTap={seconds=max(0,seconds-60)}
        Card w=360 gap=8
          Text.strong "Puzzles"
          Progress {count(puzzles,p=>p.done)/length(puzzles)}
          Text.muted "{count(puzzles,p=>p.done)} of {length(puzzles)} solved"
          each p in puzzles
            ListTile "{p.name}" subtitle="{p.sensor}"
              Checkbox @p.done
              Badge.success "Fired" hidden={!p.done}
      Row gap=16 top
        Card grow gap=12
          Row between
            Text.strong "Hint queue"
            Badge.warning "{length(hintQueue)} waiting" hidden={isEmpty(hintQueue)}
          if isEmpty(hintQueue)
            Text.muted "No requests from the room."
          each h in hintQueue
            ListTile "{h.team}, {h.at}" subtitle="{h.text}"
              Button.outline "Answer" onTap={sel=h.id; open answer}
              IconButton "close" tooltip="Dismiss" onTap={hintQueue=removeWhere(hintQueue,x=>x.id==h.id)}
          Divider
          Text.caption "Send a preset"
          Row gap=8
            each pr in presets
              Chip "{pr}" onTap={roomHint=pr; toast "Hint sent to the room screen"}
          Input "Type a hint" "Shown on the room screen" @typed multiline
          Row between
            Text.muted "{length(typed)} characters"
            Button "Send to room" icon="send" onTap={roomHint=typed; typed=""; toast "Hint sent to the room screen"}
        Card grow gap=12
          Text.strong "Cameras"
          Grid cols=2 gap=8
            each c in cams
              Stack
                CameraFeed "{c.name}" live={c.live} h=130
                Badge.error "LIVE" hidden={!c.live}
                Badge.neutral "Offline" hidden={c.live}
  Dialog#answer "Answer hint request" "Choose what the room sees"
    Input "Your hint" @typed multiline
    Button "Send" onTap={roomHint=typed; hintQueue=removeWhere(hintQueue,x=>x.id==sel); typed=""; close}
  Dialog#resetConfirm "Reset the room?" "The clock goes back to 60:00 and every puzzle unticks."
    Row end gap=8
      Button.outline "Cancel" onTap={close}
      Button.destructive "Reset" onTap={seconds=3600; running=false; status="playing"; roomHint=""; puzzles=map(puzzles,p=>with(p,"done",false)); close}
