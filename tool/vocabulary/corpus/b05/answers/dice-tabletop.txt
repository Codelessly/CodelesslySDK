          Row gap=8 center

          Text.muted "Roll to see the dice here."
      Card gap=12

    Footer
      Row between middle

  Pane w=320 pad=16 gap=12

      Text.strong "History"
      Button.ghost "Clear" onTap={history=[]}
    Column gap=8 scroll
      each h in history
        ListTile "{h.label}" "{h.expr}  ({h.detail})" trailing="{h.total}" onTap={formula=h.expr}
          Text.caption "{h.time}"

  Text.muted "Formula: {formula}"
  Row end gap=8
    Button.outline "Cancel" onTap={close}
    Button "Save" onTap={newName!="" ? {saved=append(saved,{"name":newName,"expr":formula}); newName=""; close} : toast "Give it a name"}

### Initiative
state order=[{"id":1,"name":"Kira Voss","init":19,"hp":34,"max":34,"kind":"pc"},{"id":2,"name":"Goblin Boss","init":16,"hp":9,"max":21,"kind":"foe"},{"id":3,"name":"Brother Tal","init":14,"hp":22,"max":28,"kind":"pc"},{"id":4,"name":"Mirelle","init":11,"hp":7,"max":24,"kind":"pc"},{"id":5,"name":"Goblin Archer","init":8,"hp":0,"max":7,"kind":"foe"}]
state turn=0
state round=1
state sel=0
state dmg=5
App
  Sidebar
    Brand "Critical" icon="casino"
    NavItem "Dice tray" icon="casino" onTap={go dice-tray}
    NavItem "Initiative" icon="swords" selected={true}
    NavItem "Party health" icon="favorite" onTap={go party}
    Spacer grow
    Text.muted "Round {round}"
  Main
    Nav
      Heading "Initiative" level=1
      Row gap=8
        Badge.primary "Round {round}"
        Button.outline "Sort" icon="sort" onTap={order=sortByDesc(order,c=>c.init)}
        Button "Add combatant" icon="person_add" onTap={open addCombatant}
    Column pad=24 gap=12 scroll
      Card gap=4
        Text.muted "Current turn"
        Text "{order[turn].name}" size=32 weight=700
      each c in order
        Card gap=8 drag=init(c.id) drop=init onDrop={order=sortByDesc(updateWhere(order,x=>x.id==result.data,x=>with(x,"init",c.init+0.5)),x=>x.init)} menu=combatantMenu
          Row gap=12 middle
            Icon "play_arrow" hidden={index!=turn}
            Avatar "{c.name}" size=40
            Column grow gap=4
              Row between
                Text.strong "{c.name}"
                Row gap=8 middle
                  Badge.neutral "Init {c.init}"
                  IconButton "more_vert" tooltip="Actions" onTap={sel=c.id; open combatantMenu}
              Progress {c.hp/c.max}
              Text.caption "{c.hp} / {c.max} HP {c.hp==0 ? ' - down' : ''}"
    Footer
      Row between middle
        Button.outline "Previous" icon="chevron_left" onTap={turn=turn==0 ? length(order)-1 : turn-1}
        Text.strong "{order[turn].name}'s turn"
        Button "Next turn" icon="chevron_right" onTap={turn=turn+1>=length(order) ? 0 : turn+1; round=turn+1>=length(order) ? round+1 : round}
  Pane w=320 pad=16 gap=12
    Text.strong "Damage and healing"
    Input "Amount" type=number bind=dmg
    Row gap=8
      Button.destructive "Damage" grow onTap={order=updateWhere(order,c=>c.id==sel,c=>with(c,"hp",max(0,c.hp-dmg)))}
      Button.secondary "Heal" grow onTap={order=updateWhere(order,c=>c.id==sel,c=>with(c,"hp",min(c.max,c.hp+dmg)))}
    Text.muted "Select a combatant with the menu, then apply."
    Divider
    Text.strong "Order"
    each c in order
      Text "{index+1}. {c.name}" strike={c.hp==0}
Menu#combatantMenu
  MenuItem "Set as current turn" icon="play_arrow" onTap={turn=indexOf(map(order,c=>c.id),sel)}
  MenuItem "Select for damage" icon="bolt" onTap={toast "Selected"}
  MenuItem "Remove" icon="delete" onTap={order=removeWhere(order,c=>c.id==sel)}
Dialog#addCombatant "Add combatant" "Roll or type their initiative"
  Input "Name" bind=newName
  Input "Initiative" type=number bind=newInit
  Input "Max HP" type=number bind=newHp
  Button "Add" onTap={order=sortByDesc(append(order,{"id":length(order)+10,"name":newName,"init":newInit,"hp":newHp,"max":newHp,"kind":"foe"}),c=>c.init); close}
