### Alignment
state tab="align"
state step=0
state stars=[{"name":"Vega","done":false,"err":0},{"name":"Arcturus","done":false,"err":0},{"name":"Capella","done":false,"err":0}]
state centered=false
App
  Sidebar
    Brand "Astrolabe Remote" icon="travel_explore"
    NavItem "Sky" icon="public" selected={tab=="sky"} onTap={tab="sky"; go sky}
    NavItem "Imaging" icon="photo_camera" selected={tab=="imaging"} onTap={tab="imaging"; go imaging}
    NavItem "Alignment" icon="my_location" selected={tab=="align"} onTap={tab="align"}
    NavItem "Night plan" icon="event_note" selected={tab=="plan"} onTap={tab="plan"; go night-plan}
  Main
    Nav
      Heading "Three star alignment" level=1
      Text.muted "Step {min(step+1,3)} of 3"
    Column pad=24 gap=16 maxw=720
      Stepper @step steps=["Vega","Arcturus","Capella"]
      each s in stars
        Card gap=8
          Row between middle
            Row gap=8 middle
              Icon "check_circle" visible={s.done}
              Icon "radio_button_unchecked" visible={!s.done}
              Text.strong "{s.name}"
            if s.done
              Badge.success "Error {s.err} arcmin"
            if !s.done
              Badge.neutral "Waiting"
      if step<3
        Card gap=12
          Text.strong "Slew to {stars[step].name}"
          Text.muted "Center the star in the crosshair, then confirm."
          Row gap=8
            Button.outline "Slew to star" icon="gps_fixed" onTap={toast "Slewing to {stars[step].name}"}
            Toggle "Star is centered" @centered
          Button "Confirm star" disabled={!centered} onTap={stars=setAt(stars,step,with(with(stars[step],"done",true),"err",round(abs(2.4-step*0.7),1))); step=step+1; centered=false}
      if step==3
        Card gap=8
          Text.success "Alignment complete"
          Text.muted "Mean error {round(sum(stars,s=>s.err)/3,1)} arcmin"
          Row gap=8
            Button "Go to sky" onTap={go sky}
            Button.outline "Redo alignment" onTap={step=0; stars=map(stars,s=>with(with(s,"done",false),"err",0))}

    MenuItem "Go to now" icon="gps_fixed" onTap={emit goto sel; go sky}
    MenuItem "Move up" icon="arrow_upward" onTap={plan=updateWhere(plan,p=>p.id==sel,p=>with(p,"alt",p.alt+10))}
    MenuItem "Remove" icon="delete" onTap={plan=removeWhere(plan,p=>p.id==sel)}
  Dialog#addTarget "Add target" "Pick an object for tonight"

    Input "Frames" type=number @newFrames
    Button "Add" onTap={newName=="" ? toast "Enter a name" : {plan=append(plan,{"id":lower(newName),"name":newName,"alt":40,"start":"03:00","frames":newFrames}); newName=""; close}}
