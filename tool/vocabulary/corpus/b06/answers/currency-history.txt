### Alerts
state pairA="USD/EUR"
state target=0.9300
state direction="rises above"
state notify=true
state email=false
state alerts=[{"id":1,"pair":"GBP/USD","dir":"falls below","target":1.2500,"on":true},{"id":2,"pair":"USD/JPY","dir":"rises above","target":153.00,"on":true}]
state offline=true
App
  Sidebar
    Brand "Rate Room" icon="currency_exchange"
    NavItem "Convert" icon="swap_vert" onTap={go convert}
    NavItem "Rate history" icon="show_chart" onTap={go rate-history}
    NavItem "Favourites" icon="star" badge="4" onTap={go favourites}
    NavItem "Alerts" icon="notifications" badge="{count(alerts,a=>a.on)}" selected={true} onTap={go alerts}
    Spacer grow
    if offline
      Badge.warning "Offline"
    Text.muted "Updated Today, 09:41"
  Main
    Nav
      Heading "Rate alerts"
    Column pad=24 gap=16 scroll
      Card gap=12
        Heading "New alert" level=3
        Select "Pair" @pairA items=["USD/EUR","GBP/USD","USD/JPY","EUR/CHF"]
        SegmentedButton @direction items=["rises above","falls below"]
        Input "Target rate" @target type=number
        Text.muted "Now 0.9234. Your target is {fixed(abs(target-0.9234)/0.9234*100,2)}% away."
        Toggle "Push notification" @notify
        Toggle "Email me" @email
        Button "Create alert" onTap={target>0 ? {alerts=append(alerts,{"id":length(alerts)+10,"pair":pairA,"dir":direction,"target":target,"on":true}); toast "Alert set for {pairA}"} : toast "Enter a target rate"}
      Heading "Active alerts" level=3
      if isEmpty(alerts)
        Text.muted "No alerts yet."
      each a in alerts
        Card
          Row gap=12 middle
            Column grow gap=2
              Text.strong "{a.pair}"
              Text.muted "When it {a.dir} {a.target}"
            Toggle @a.on
            IconButton "delete" tooltip="Delete alert" onTap={alerts=removeWhere(alerts,x=>x.id==a.id); toast "Alert deleted"}
