### Quiz
state screen="quiz"
state round=0
state score=0
state checked=false
state rounds=[{"prompt":"Put these in order, earliest first","cards":[{"id":"q1","title":"Magna Carta","year":1215},{"id":"q2","title":"Printing press","year":1450},{"id":"q3","title":"Battle of Waterloo","year":1815},{"id":"q4","title":"Moon landing","year":1969}]},{"prompt":"Now these","cards":[{"id":"q5","title":"Fall of Rome","year":476},{"id":"q6","title":"Black Death","year":1347},{"id":"q7","title":"Mona Lisa painted","year":1503}]}]
state order=[]
state tray=[{"id":"q3","title":"Battle of Waterloo","year":1815},{"id":"q1","title":"Magna Carta","year":1215},{"id":"q4","title":"Moon landing","year":1969},{"id":"q2","title":"Printing press","year":1450}]
App
  Sidebar
    Brand "Epoch" icon="hourglass_top"
    NavItem "Timeline" icon="timeline" selected={screen=="timeline"} onTap={go timeline}
    NavItem "Compare regions" icon="compare_arrows" selected={screen=="compare"} onTap={go compare}
    NavItem "Quiz" icon="quiz" selected={screen=="quiz"} onTap={go quiz}
  Main
    Nav
      Heading "Quiz" level=1
      Row gap=12
        Text.muted "Round {round+1} of {length(rounds)}"
        Badge.success "Score {score}"
    Column pad=24 gap=16
      Progress {round / length(rounds)}
      Text.heading "{rounds[round].prompt}"
      Text.caption "Drag the cards into the slots below"
      Row gap=12 scroll
        each c in tray
          Card drag=card(c.id) w=180 gap=4
            Text.strong "{c.title}"
            Text.muted "{checked ? c.year : '?'}"
      Column gap=8 drop=card onDrop={order=append(order,result.data); tray=removeWhere(tray,c=>c.id==result.data)}
        Text.fieldLabel "Your order"
        each id in order
          ListTile "{index+1}. {find(rounds[round].cards,c=>c.id==id).title}" icon={checked ? (id == sortBy(rounds[round].cards,c=>c.year)[index].id ? "check_circle" : "cancel") : "drag_indicator"}
        if isEmpty(order)
          Text.muted "Drop the first card here"
      Row gap=8
        Button.outline "Reset" onTap={order=[]; checked=false; tray=rounds[round].cards}
        Button "Check order" onTap={checked=true; score=score+count(order,id=>id==sortBy(rounds[round].cards,c=>c.year)[indexOf(order,id)].id)}
        Spacer grow
        if checked
          Button "Next round" onTap={round<length(rounds)-1 ? {round=round+1; order=[]; checked=false; tray=rounds[round+1].cards} : open done}
  Dialog#done "Quiz complete" "You scored {score}"
    Text "Nice work. Come back tomorrow for a new set."
    Row gap=8
      Button.outline "Review timeline" onTap={close; go timeline}
      Button "Play again" onTap={round=0; score=0; order=[]; checked=false; tray=rounds[0].cards; close}
