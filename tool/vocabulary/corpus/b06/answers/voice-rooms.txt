### Room
state muted=true
state handRaised=false
state active="Dev Patel"
state speakers=[{"name":"Maya Reyes","host":true},{"name":"Dev Patel","host":false},{"name":"Lena Fischer","host":false},{"name":"Tomas Ruiz","host":false}]
state listeners=[{"name":"Alex Morgan"},{"name":"Ines Costa"},{"name":"Ben Okafor"},{"name":"Yuki Tanaka"},{"name":"Rosa Mendes"},{"name":"Sam Lee"},{"name":"Noah Kim"},{"name":"Amara Diallo"}]
state queue=[{"name":"Ines Costa"},{"name":"Ben Okafor"}]
state reactions=[]
state isHost=true
every 3s onTick={active=speakers[(indexOf(map(speakers,s=>s.name),active)+1)%length(speakers)].name}
App
  Main
    Nav
      Row gap=8 middle
        IconButton "arrow_back" tooltip="Leave quietly" onTap={emit leaveQuietly; go rooms}
        Column gap=2
          Heading "Shipping a Flutter app solo" level=3
          Text.muted "{length(listeners)+length(speakers)} in the room"
      Row gap=8
        if isHost
          Button.outline "Host panel" icon="admin_panel_settings" badge="{length(queue)}" onTap={open host}
        Button.ghost "Leave quietly" icon="logout" onTap={emit leaveQuietly; go rooms}
    Column pad=24 gap=24 scroll
      Text.sectionTitle "Speakers"
      SpeakerCircle items={speakers} active={active}
        each s in speakers
          Column center gap=4
            Avatar "{s.name}" size=72 animate={active==s.name ? "pulse" : ""}
            Row gap=4 middle
              Text.strong "{s.name}"
              if s.host
                Badge.primary "Host"
            if active==s.name
              Waveform active size=20
      Divider
      Text.sectionTitle "Listeners"
      Grid cols=4 gap=16
        each l in listeners
          Column center gap=4
            Avatar "{l.name}" size=48
            Text.caption "{l.name}"
            if any(queue, q=>q.name==l.name)
              Icon "front_hand" color=#FFF5A623
      FloatingReactions items={reactions} onDone={reactions=[]}
    Footer
      Row between middle pad=12
        Row gap=8
          IconButton "{muted ? 'mic_off' : 'mic'}" tooltip={muted ? "Unmute" : "Mute"} selected={!muted} onTap={muted=!muted; toast muted ? "Muted" : "You are live"}
          Button.outline "{handRaised ? 'Lower hand' : 'Raise hand'}" icon="front_hand" onTap={handRaised ? {handRaised=false; queue=removeWhere(queue,q=>q.name=="Alex Morgan"); emit lowerHand} : {handRaised=true; queue=append(queue,{"name":"Alex Morgan"}); emit raiseHand; toast "The host can see you in the queue"}}
        Row gap=4
          IconButton "favorite" tooltip="Love" onTap={reactions=append(reactions,{"emoji":"❤️"}); emit react "heart"}
          IconButton "celebration" tooltip="Cheer" onTap={reactions=append(reactions,{"emoji":"🎉"}); emit react "cheer"}
          IconButton "sentiment_very_satisfied" tooltip="Laugh" onTap={reactions=append(reactions,{"emoji":"😂"}); emit react "laugh"}
          IconButton "thumb_up" tooltip="Agree" onTap={reactions=append(reactions,{"emoji":"👍"}); emit react "agree"}
  Drawer#host "Host panel" right
    Text.sectionTitle "Hand-raise queue ({length(queue)})"
    if length(queue)==0
      Text.muted "Nobody has a hand up."
    each q in queue
      ListTile "{q.name}" subtitle="Wants to speak" leading={Avatar "{q.name}" size=36}
        Row gap=4
          Button "Bring up" size=small onTap={speakers=append(speakers,{"name":q.name,"host":false}); listeners=removeWhere(listeners,l=>l.name==q.name); queue=removeWhere(queue,x=>x.name==q.name); emit moveUp q.name; toast "{q.name} is a speaker"}
          IconButton "close" tooltip="Dismiss" onTap={queue=removeWhere(queue,x=>x.name==q.name)}
    Divider
    Text.sectionTitle "On stage"
    each s in speakers
      ListTile "{s.name}" subtitle={s.host ? "Host" : "Speaker"} leading={Avatar "{s.name}" size=36}
        if !s.host
          Button.outline "Move down" size=small onTap={listeners=append(listeners,{"name":s.name}); speakers=removeWhere(speakers,x=>x.name==s.name); emit moveDown s.name}
    Divider
    Button.outline "Invite someone" icon="person_add" onTap={open invite}
  Dialog#invite "Invite to speak" "Send an invite to someone in the room"
    Select "Person" items={map(listeners,l=>l.name)} bind=inviteWho
    Button "Send invite" onTap={emit invite inviteWho; toast "Invite sent to {inviteWho}"; close}
