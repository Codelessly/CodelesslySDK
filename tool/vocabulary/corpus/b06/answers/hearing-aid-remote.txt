### Find my aid
state finding=false
state side="left"
state signal=0
state ticks=0
state lastSeen="Today, 8:12 AM near Kitchen"
App
  Sidebar
    Brand "Whisper" icon="hearing"
    NavItem "Hearing" icon="hearing" onTap={go hearing}
    NavItem "Find my aid" icon="location_searching" selected={true} onTap={go find_my_aid}
    NavItem "Streaming" icon="bluetooth_audio" onTap={go streaming}
    NavItem "Soothing sounds" icon="spa" onTap={go soothing_sounds}
  Main
    Nav
      Heading "Find my aid" level=1
      Row gap=8
        Button.outline "Open map" onTap={emit openMap}
    Column pad=24 gap=20 center
      SegmentedButton @side items=["left","right"]
      Card gap=16 center maxw=520
        Gauge value={signal} min=0 max=100 unit="% signal" h=240
        Text.heading "{signal < 20 ? 'Out of range' : (signal < 50 ? 'Getting warmer' : (signal < 80 ? 'Close' : 'Very close'))}"
        Text.muted "Walk slowly. The {side} aid beeps faster as you get closer."
        Beeper rate={signal} active={finding}
        Row gap=12
          Button "Start searching" hide={finding} onTap={finding=true; signal=12; emit startFind side}
          Button.destructive "Stop" hidden={!finding} onTap={finding=false; signal=0}
          Button.outline "Play loud tone" onTap={emit playTone side}
        Sparkline values={history}
      Card gap=8
        Text.strong "Last seen"
        Text.muted "{lastSeen}"
        Row gap=8
          Chip "Kitchen"
          Chip "Bedside table"
          Chip "Car"
  every 1s when finding onTick={signal=clamp(signal+8,0,100); ticks=ticks+1}

### Streaming
state sources=[{"id":"tv","name":"Living room TV","kind":"tv","connected":true,"vol":70},{"id":"phone","name":"Ray's phone","kind":"smartphone","connected":false,"vol":50},{"id":"mic","name":"Partner microphone","kind":"mic","connected":false,"vol":60},{"id":"pc","name":"Office laptop","kind":"laptop","connected":false,"vol":40}]
state active="tv"
state mixOutside=30
state newName=""
App
  Sidebar
    Brand "Whisper" icon="hearing"
    NavItem "Hearing" icon="hearing" onTap={go hearing}
    NavItem "Find my aid" icon="location_searching" onTap={go find_my_aid}
    NavItem "Streaming" icon="bluetooth_audio" selected={true} onTap={go streaming}
    NavItem "Soothing sounds" icon="spa" onTap={go soothing_sounds}
  Main
    Nav
      Heading "Streaming" level=1
      Button.outline "Pair new device" icon="add" onTap={open pair}
    Column pad=24 gap=16 scroll
      Text.sectionTitle "Sources"
      each s in sources
        ListTile "{s.name}" subtitle="{s.connected ? 'Streaming to both ears' : 'Available'}" icon={s.kind} selected={active==s.id}
          Row gap=8
            if s.connected
              Badge.success "Live"
            Switch @s.connected onChange={active=s.id; emit stream s.id}
            IconButton "more_vert" tooltip="More" onTap={sel=s.id; open sourceMenu}
      Card gap=8
        Text.heading "Mix with surroundings"
        Row gap=8
          Text.muted "Source only"
          Slider @mixOutside min=0 max=100 grow
          Text.muted "Room sound"
        Text.muted "{mixOutside}% of the room is let in."
    Footer
      Row between
        Row gap=8
          Icon "bluetooth_audio"
          Text.strong "{find(sources, x=>x.id==active).name}"
        Button.outline "Stop streaming" onTap={sources=map(sources, x=>with(x,"connected",false)); toast "Streaming stopped"}
  Dialog#pair "Pair new device" "Hold the device's pair button until it blinks."
    Input "Device name" @newName
    Button "Search" onTap={emit pairSearch newName; close}
  Menu#sourceMenu end
    MenuItem "Rename" icon="edit" onTap={emit rename sel}
    MenuItem "Forget device" icon="delete" onTap={sources=removeWhere(sources, x=>x.id==sel)}

### Soothing sounds
state playing=false
state layers=[{"name":"Ocean waves","icon":"waves","on":true,"vol":60},{"name":"Rain","icon":"water_drop","on":false,"vol":40},{"name":"White noise","icon":"graphic_eq","on":true,"vol":25},{"name":"Soft chimes","icon":"notifications","on":false,"vol":30},{"name":"Forest","icon":"park","on":false,"vol":45}]
state timer="30 min"
state pitch=50
App
  Sidebar
    Brand "Whisper" icon="hearing"
    NavItem "Hearing" icon="hearing" onTap={go hearing}
    NavItem "Find my aid" icon="location_searching" onTap={go find_my_aid}
    NavItem "Streaming" icon="bluetooth_audio" onTap={go streaming}
    NavItem "Soothing sounds" icon="spa" selected={true} onTap={go soothing_sounds}
  Main
    Nav
      Heading "Soothing sounds" level=1
      Row gap=8
        Select @timer items=["15 min","30 min","1 hour","All night"]
        Button "{playing ? 'Pause' : 'Play'}" icon={playing ? "pause" : "play_arrow"} onTap={playing=!playing; emit soothe playing}
    Column pad=24 gap=16 scroll
      Card gap=8
        Text.heading "Your mix"
        Waveform values={map(filter(layers, l=>l.on), l=>l.vol)} active={playing} h=80
        Text.muted "{count(layers, l=>l.on)} layers playing for {timer}"
      each l in layers
        Card gap=8
          Row between
            Row gap=8
              IconTile "{l.icon}"
              Text.strong "{l.name}"
            Toggle @l.on
          Slider @l.vol min=0 max=100 hidden={!l.on}
      Card gap=8
        Text.heading "Tone matching"
        Row gap=8
          Text.muted "Low"
          Slider @pitch min=0 max=100 grow
          Text.muted "High"
        Text.muted "Shape the mix to sit under your tinnitus pitch."
      Row gap=8
        Button.outline "Save as preset" onTap={emit savePreset layers; toast "Preset saved"}
        Button.ghost "Reset" onTap={layers=map(layers, l=>with(l,"on",false))}
