### Map
state tab="map"
state q=""
state showConservation=true
state showFlood=false
state statuses=["received","approved","refused"]
state selected=""
state suggestions=[{"id":"a1","text":"14 Mill Lane, Hartley"},{"id":"a2","text":"14 Mill Road, Fenwick"},{"id":"a3","text":"14 Millbank Close, Hartley"}]
state apps=[{"id":"24/0312","address":"21 Orchard Row","status":"received","proposal":"Two-storey rear extension","comments":14},{"id":"24/0287","address":"Former Dairy, Mill Lane","status":"approved","proposal":"Conversion to 6 flats","comments":52},{"id":"24/0301","address":"8 Brook Street","status":"refused","proposal":"Loft conversion with dormer","comments":3}]
state alertAddress=""
state alertRadius=""
state alertEmail=""
state alertFreq=""
App
  Sidebar
    Brand "Zoned" icon="map"
    NavItem "Map" icon="map" selected={tab=="map"} onTap={tab="map"}
    NavItem "Applications" icon="description" badge="3" selected={tab=="list"} onTap={tab="list"}
    NavItem "My alerts" icon="notifications" onTap={open alerts}
    Spacer grow
    Text.muted "Hartley District Council"
  Main
    Nav
      Heading "Planning map"
      Row gap=8
        Input "Search by address" @q w=280 icon="search"
        Button.outline "Alerts" icon="notifications" onTap={open alerts}
    Stack
      Map center=[51.512,-0.118] zoom=15 onSelect={selected=result.id}
        Polygon each=apps status={app.status} onTap={selected=app.id}
        if showConservation
          Layer "Conservation areas" kind=conservation fill=#33388E3C
        if showFlood
          Layer "Flood zones" kind=flood fill=#332196F3
      if q!=""
        Card gap=4 w=320 top left
          each s in filter(suggestions,s=>contains(lower(s.text),lower(q)))
            ListTile "{s.text}" icon="place" onTap={q=s.text; emit locate s.id}
      Card gap=12 w=260 top right
        Text.panelTitle "Layers"
        Toggle "Conservation areas" @showConservation
        Toggle "Flood zones" @showFlood
        Divider
        Text.fieldLabel "Status"
        FilterChip "Received" selected={contains(statuses,"received")} onTap={statuses=contains(statuses,"received") ? without(statuses,"received") : append(statuses,"received")}
        FilterChip "Approved" selected={contains(statuses,"approved")} onTap={statuses=contains(statuses,"approved") ? without(statuses,"approved") : append(statuses,"approved")}
        FilterChip "Refused" selected={contains(statuses,"refused")} onTap={statuses=contains(statuses,"refused") ? without(statuses,"refused") : append(statuses,"refused")}
      if selected!=""
        Card gap=8 w=340 bottom left
          each a in filter(apps,a=>a.id==selected)
            Row between
              Text.strong "{a.id}"
              Badge.info "{a.status}"
            Text "{a.address}"
            Text.muted "{a.proposal}"
            Row between
              Text.caption "{a.comments} comments"
              Button "View application" onTap={go application}
  Dialog#alerts "Alert me about new applications" "Within the radius you draw"
    Input "Postcode or address" @alertAddress
    Map h=220 center=[51.512,-0.118] zoom=14
      Circle @alertRadius center=[51.512,-0.118] draggable
    Slider @alertRadius min=100 max=2000 step=100
    Text.muted "Radius: {alertRadius} m"
    Input "Email" type=email @alertEmail
    Select "Frequency" items=["Instantly","Daily digest","Weekly"] @alertFreq
    Button "Sign up for alerts" onTap={alertEmail!="" ? emit signUpAlerts; toast "Alerts on for {alertRadius} m"; close : toast "Enter your email"}
