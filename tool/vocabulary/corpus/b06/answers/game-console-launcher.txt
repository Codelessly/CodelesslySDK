### Downloads
state downloads=[{"id":"d1","title":"Void Sparrow","size":"38.2 GB","done":0.64,"paused":false},{"id":"d2","title":"Ember Reach: Ashen Crown","size":"6.4 GB","done":0.12,"paused":false},{"id":"d3","title":"Tidal Rush update 1.4","size":"1.1 GB","done":0,"paused":true}]
state allPaused=false
state sel=0
every 1s onTick={downloads=map(downloads,d=>d.paused ? d : with(d,"done",min(1,d.done+0.004)))}
App
  Main
    Nav
      IconButton "arrow_back" tooltip="Back" onTap={go home}
      Heading "Downloads" level=1
      Row gap=8
        Button.outline "{allPaused ? \"Resume all\" : \"Pause all\"}" icon="{allPaused ? \"play_arrow\" : \"pause\"}" onTap={allPaused=!allPaused; downloads=map(downloads,d=>with(d,"paused",allPaused))}
    Column grow pad=24 gap=16 onKeyDown={sel=min(length(downloads)-1,sel+1); play tick} onKeyUp={sel=max(0,sel-1); play tick} onKeyA={downloads=updateWhere(downloads,d=>d.id==downloads[sel].id,d=>with(d,"paused",!d.paused))} onKeyX={downloads=removeAt(downloads,sel)}
      if isEmpty(downloads)
        Column center middle gap=8
          Icon "cloud_done"
          Text.muted "Nothing in the queue"
      each d in downloads
        Card gap=8 border-left={index==sel ? #FFFF7A1A : #00000000} onTap={sel=index; play tick}
          Row between
            Column gap=2
              Text.strong "{d.title}"
              Text.caption "{d.size} · {d.paused ? \"Paused\" : (d.done>=1 ? \"Ready\" : round(d.done*100,0) + \"% · \" + duration(round((1-d.done)/0.004,0)) + \" left\")}"
            Row gap=8
              IconButton "{d.paused ? \"play_arrow\" : \"pause\"}" tooltip="{d.paused ? \"Resume\" : \"Pause\"}" onTap={downloads=updateWhere(downloads,x=>x.id==d.id,x=>with(x,"paused",!x.paused))}
              IconButton "close" tooltip="Cancel download" onTap={downloads=removeWhere(downloads,x=>x.id==d.id); toast "Removed {d.title}"}
          Progress {d.done}
    Footer
      Row gap=24 center
        Row gap=8
          Badge.neutral "A"
          Text.muted "Pause or resume"
        Row gap=8
          Badge.neutral "X"
          Text.muted "Cancel"
        Row gap=8
          Badge.neutral "▲ ▼"
          Text.muted "Select"
        Row gap=8
          Badge.neutral "B"
          Text.muted "Back"
