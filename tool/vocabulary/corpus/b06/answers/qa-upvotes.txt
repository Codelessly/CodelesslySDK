### Host
state tab="live"
state sel=0
state running=true
state left=600
every 1s when running onTick={left=max(0,left-1)}
App
  Sidebar
    Brand "Ask Away" icon="forum"
    NavItem "Live questions" icon="question_answer" badge="{count(questions,q=>!q.answered && !q.hidden)}" selected={tab=="live"} onTap={tab="live"}
    NavItem "Answered" icon="task_alt" selected={tab=="answered"} onTap={tab="answered"}
    NavItem "Hidden" icon="visibility_off" selected={tab=="hidden"} onTap={tab="hidden"}
    NavItem "Projector view" icon="cast" onTap={go projector}
    Spacer grow
    Text.muted "Host: Dana Whitfield"
  Main
    Nav
      Heading "Spring Launch Q&A" level=1
      Row gap=8
        Badge.warning "Round {duration(left)}"
        Button.outline "{running ? 'Pause' : 'Resume'}" onTap={running=!running}
        Button.outline "Reset 10:00" onTap={left=600; running=true}
        Button.outline "5 min" onTap={left=300}
        Button "Open projector" onTap={go projector}
    Column pad=24 gap=16 scroll
      if tab=="live"
        each q in sortByDesc(filter(questions,x=>!x.answered && !x.hidden),x=>(x.pinned ? 1000 : 0)+x.votes)
          Card gap=8 animate=fade
            Row gap=16
              Column center gap=2
                Icon "arrow_upward"
                Text.strong "{q.votes}"
              Column grow gap=4
                Text "{q.text}"
                Text.caption "{q.author}"
                if q.mergeInto>0
                  Badge.info "Looks similar to question {q.mergeInto}"
            Row gap=8 end
              if q.mergeInto>0
                Button.outline "Merge" icon="call_merge" onTap={questions=removeWhere(updateWhere(questions,x=>x.id==q.mergeInto,x=>with(with(x,"votes",x.votes+q.votes),"merged",x.merged+1)),x=>x.id==q.id); toast "Merged into question {q.mergeInto}"}
              Button.outline "{q.pinned ? 'Unpin' : 'Pin'}" icon="push_pin" onTap={questions=map(questions,x=>with(x,"pinned",x.id==q.id ? !q.pinned : false))}
              Button "Answered" icon="check" onTap={questions=updateWhere(questions,x=>x.id==q.id,x=>with(with(x,"answered",true),"pinned",false))}
              Button.destructive "Hide" icon="visibility_off" onTap={sel=q.id; open confirmHide}
      if tab=="answered"
        each q in filter(questions,x=>x.answered)
          Card
            Row between
              Column gap=2
                Text "{q.text}"
                Text.caption "{q.author} · {q.votes} upvotes"
              Button.outline "Reopen" onTap={questions=updateWhere(questions,x=>x.id==q.id,x=>with(x,"answered",false))}
      if tab=="hidden"
        if isEmpty(filter(questions,x=>x.hidden))
          Text.muted "Nothing hidden."
        each q in filter(questions,x=>x.hidden)
          Card
            Row between
              Text "{q.text}"
              Button.outline "Restore" onTap={questions=updateWhere(questions,x=>x.id==q.id,x=>with(x,"hidden",false))}
Dialog#confirmHide "Hide this question?" "It disappears from the audience list and the projector."
  Row gap=8 end
    Button.outline "Cancel" onTap={close}
    Button.destructive "Hide" onTap={questions=updateWhere(questions,x=>x.id==sel,x=>with(with(x,"hidden",true),"pinned",false)); close; toast "Question hidden"}
