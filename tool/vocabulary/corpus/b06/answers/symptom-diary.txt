### Today
state tab="today"
state kind="pain"
state spots=[{"area":"lower back","kind":"ache","sev":5},{"area":"left knee","kind":"pain","sev":3}]
state picked=""
state severity=4
state triggers=["Stress","Poor sleep"]
state foods=["Coffee","Dairy"]
state mood="Okay"
state sleepHours=6.5
state sleepQuality="Restless"
state saved=false
state exportRange=""
state exportFormat=""
state exportExtras=false
App
  Sidebar
    Brand "Daybook Health" icon="healing"
    NavItem "Today" icon="today" selected={tab=="today"} onTap={tab="today"}
    NavItem "Month" icon="calendar_month" selected={tab=="month"} onTap={go month}
    NavItem "Patterns" icon="insights" selected={tab=="patterns"} onTap={go patterns}
    NavItem "Doctor note" icon="description" selected={tab=="note"} onTap={go doctor-note}
    Spacer grow
    Text.muted "Maya Reyes"
  Main
    Nav
      Heading "Today, {formatDate(today(),'EEEE MMM d')}" level=1
      Row gap=8
        Button.outline "Export" icon="ios_share" onTap={open export}
        Button "Save entry" onTap={saved=true; toast "Entry saved"; emit saveEntry}
    Column scroll pad=24 gap=16
      Grid cols=2 gap=16
        Card gap=12
          Text.strong "Where does it hurt?"
          SegmentedButton @kind items=["pain","itch","ache"]
          Stack center middle
            BodyMap view="front" @spots mode={kind} onTapSpot={picked=result.area; spots=append(spots,{"area":result.area,"kind":kind,"sev":severity})} h=360
          Text.muted "Tap a spot on the outline to mark it as {kind}."
          if picked!=""
            Text.small "Last marked: {picked}"
        Column gap=16
          Card gap=12
            Text.strong "Marked today"
            if isEmpty(spots)
              Text.muted "Nothing marked yet."
            each s in spots
              ListTile "{s.area}" subtitle="{s.kind}, {s.sev} of 10"
                IconButton "close" tooltip="Remove" onTap={spots=removeWhere(spots,x=>x.area==s.area)}
          Card gap=12
            Row between
              Text.strong "How bad is it?"
              Text.strong "{severity} of 10"
            Slider @severity min=0 max=10 step=1 faces
            Row between
              Icon "sentiment_very_satisfied"
              Icon "sentiment_neutral"
              Icon "sentiment_very_dissatisfied"
          Card gap=12
            Text.strong "Triggers"
            Row gap=8 wrap
              each t in ["Stress","Poor sleep","Weather","Exercise","Screen time","Alcohol","Skipped meal"]
                FilterChip "{t}" selected={contains(triggers,t)} onTap={triggers=contains(triggers,t) ? without(triggers,t) : append(triggers,t)}
            Text.strong "Food"
            Row gap=8 wrap
              each f in ["Coffee","Dairy","Gluten","Sugar","Spicy","Eggs","Nuts"]
                FilterChip "{f}" selected={contains(foods,f)} onTap={foods=contains(foods,f) ? without(foods,f) : append(foods,f)}
          Card gap=12
            Row middle gap=12
              Text.strong "Mood" w=60
              Row gap=8 grow
                ChoiceChip "Low" selected={mood=="Low"} onTap={mood="Low"}
                ChoiceChip "Okay" selected={mood=="Okay"} onTap={mood="Okay"}
                ChoiceChip "Good" selected={mood=="Good"} onTap={mood="Good"}
            Row middle gap=12
              Text.strong "Sleep" w=60
              Input "Hours" type=number @sleepHours w=90
              Select @sleepQuality items=["Deep","Okay","Restless","Barely slept"] grow
          if saved
            Text.success "Saved. Today is logged."

  Dialog#export "Export your diary" "Choose what goes into the file"
    SegmentedButton @exportRange items=["2 weeks","Month","All"]
    Select "Format" @exportFormat items=["PDF","CSV","Plain text"]
    Toggle "Include food and triggers" @exportExtras
    Row end gap=8
      Button.outline "Copy as text" onTap={copy summary; toast "Copied"}
      Button "Download" onTap={emit export {"range":exportRange,"format":exportFormat,"extras":exportExtras}; toast "Preparing file"; close}

  Sidebar
    Brand "Daybook Health" icon="healing"
    NavItem "Today" icon="today" selected={tab=="today"} onTap={go today}
    NavItem "Month" icon="calendar_month" selected={tab=="month"} onTap={go month}
    NavItem "Patterns" icon="insights" selected={tab=="patterns"} onTap={tab="patterns"}
    NavItem "Doctor note" icon="description" selected={tab=="note"} onTap={go doctor-note}
    Spacer grow
    Text.muted "Maya Reyes"
  Main
    Nav
      Heading "Patterns" level=1
      SegmentedButton @range items=["14 days","30 days","90 days"]
    Column scroll pad=24 gap=16
      Card gap=12
        Text.strong "Share of bad days with each trigger"
        Text.muted "A bad day is severity 6 or higher. Bars show how many bad days included the trigger."
        BarChart horizontal values={map(triggerNames,n=>count(filter(days,d=>d.sev>=6),d=>contains(d.t,n)))} labels={triggerNames} h=260
      Grid cols=2 gap=16
        Card gap=12
          Text.strong "Average severity with the trigger"
          each n in sortByDesc(triggerNames,n=>sum(filter(days,d=>contains(d.t,n)),d=>d.sev)/max(1,count(days,d=>contains(d.t,n))))
            Row middle gap=12
              Text n w=110
              Progress {sum(filter(days,d=>contains(d.t,n)),d=>d.sev)/max(1,count(days,d=>contains(d.t,n)))/10} grow
              Text.strong "{fixed(sum(filter(days,d=>contains(d.t,n)),d=>d.sev)/max(1,count(days,d=>contains(d.t,n))),1)}" w=40
        Card gap=12
          Text.strong "Severity over time"
          LineChart values={map(days,d=>d.sev)} h=200
          Text.muted "Poor sleep and stress show up together on your worst days."
      Row gap=12
        Button.outline "Write the doctor note" icon="description" onTap={go doctor-note}
        Button.ghost "Export" icon="ios_share" onTap={emit exportPatterns}
