### Record a hike
state running=true
state secs=0
state dist=0
state climb=0
state ended=false
every 1s when running onTick={secs=secs+1; dist=dist+0.0031; climb=climb+0.2}

          Text.muted "Climb"
          Text.strong "{round(climb,0)} m" size=28
      Card maxw=720 gap=8
        TrailMap route=[[47.6101,-121.8802],[47.6134,-121.8751],[47.6188,-121.8719]] live=true follow=true h=240
      Row gap=16 center
        Button.outline "Pause" icon="pause" size=large show={running} onTap={running=false}
        Button "Resume" icon="play_arrow" size=large show={!running} onTap={running=true}
        Button.destructive "Finish" icon="stop" size=large onTap={running=false; open finish}
  Dialog#finish "Finish this hike?" "You walked {fixed(dist,2)} km in {duration(secs)}."
    Input "Name this hike" bind=hikeName
    Row end gap=8
      Button.ghost "Keep going" onTap={running=true; close}
      Button.destructive "Discard" onTap={secs=0; dist=0; climb=0; close; go trail}
      Button "Save hike" onTap={emit saveHike {"name":hikeName,"secs":secs,"dist":dist}; toast "Hike saved"; secs=0; dist=0; climb=0; close; go trail}

### Offline maps
state regions=[{"id":"r1","name":"Cascade Foothills","size":412,"pct":1,"state":"done"},{"id":"r2","name":"Mount Rainier NP","size":688,"pct":0.42,"state":"loading"},{"id":"r3","name":"Olympic Peninsula","size":534,"pct":0.1,"state":"loading"},{"id":"r4","name":"North Cascades","size":301,"pct":0,"state":"idle"}]
state sel=""
every 1s onTick={regions=map(regions,r=>r.state=="loading" ? with(with(r,"pct",min(1,r.pct+0.03)),"state",r.pct+0.03>=1 ? "done" : "loading") : r)}
App
  Sidebar
    Brand "Switchback" icon="landscape"
    NavItem "Trails" icon="terrain" onTap={go trail}
    NavItem "Record" icon="fiber_manual_record" onTap={go record}
    NavItem "Offline maps" icon="download_for_offline" badge="2" selected onTap={go offline}
    Spacer grow
    Row gap=8 middle
      Avatar "Dana Whitfield" size=32
      Text.muted "Dana Whitfield"
  Main
    Nav
      Heading "Offline maps" level=1
      Button "Add region" icon="add" onTap={open addRegion}
    Column pad=24 gap=16 scroll
      Card gap=8
        Row between
          Text.strong "Storage"
          Text.muted "{round(sum(regions,r=>r.size*r.pct),0)} MB of 2,048 MB used"
        Progress {sum(regions,r=>r.size*r.pct)/2048}
      each r in regions
        Card gap=8
          Row between middle
            Column gap=2
              Text.strong "{r.name}"
              Text.caption "{r.size} MB"
            Row gap=8 middle
              Badge.success "Ready" show={r.state=="done"}
              Badge.info "{round(r.pct*100,0)}%" show={r.state=="loading"}
              Button.outline "Download" icon="download" show={r.state=="idle"} onTap={regions=updateWhere(regions,x=>x.id==r.id,x=>with(x,"state","loading"))}
              IconButton "pause" tooltip="Pause" show={r.state=="loading"} onTap={regions=updateWhere(regions,x=>x.id==r.id,x=>with(x,"state","idle"))}
              IconButton "more_vert" tooltip="More" onTap={sel=r.id; open regionMenu}
          Progress {r.pct} show={r.state!="idle"}
  Menu#regionMenu end
    MenuItem "Update map" icon="refresh" onTap={regions=updateWhere(regions,x=>x.id==sel,x=>with(with(x,"pct",0),"state","loading"))}
    MenuItem "Remove download" icon="delete" onTap={regions=updateWhere(regions,x=>x.id==sel,x=>with(with(x,"pct",0),"state","idle")); toast "Removed"}
  Dialog#addRegion "Add a region" "Choose an area to save for offline use"
    Input "Search places" bind=placeQuery
    Select "Detail" items=["Standard","High (topo + trails)"] bind=detail
    Row end gap=8
      Button.ghost "Cancel" onTap={close}
      Button "Download" onTap={regions=append(regions,{"id":"r{length(regions)+1}","name":coalesce(placeQuery,"New region"),"size":250,"pct":0,"state":"loading"}); toast "Download started"; close}
