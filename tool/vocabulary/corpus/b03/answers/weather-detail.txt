### Places
state places=[{"id":"sea","name":"Seattle","region":"WA","temp":14,"sky":"Light rain"},{"id":"pdx","name":"Portland","region":"OR","temp":16,"sky":"Cloudy"},{"id":"sfo","name":"San Francisco","region":"CA","temp":18,"sky":"Clear"},{"id":"van","name":"Vancouver","region":"BC","temp":12,"sky":"Rain"}]
state q=""
state sel=""
state newCity=""
App
  Sidebar
    Brand "Barometer" icon="speed"
    NavItem "Today" icon="wb_sunny" onTap={go today}
    NavItem "Radar" icon="radar" onTap={go radar}
    NavItem "Places" icon="place" badge="{length(places)}" selected={true}
  Main
    Nav
      Heading "Saved places" level=1
      Button "Add place" icon="add" onTap={open addPlace}
    Column pad=24 gap=16
      Input "Search places" @q
      Text.muted "Drag a place or use the arrows to reorder. The first one opens by default."
      Column gap=8 drop=place onDrop={places=filter(places,p=>true)}
        each p in places
          if q=="" || contains(lower(p.name),lower(q))
            ListTile drag=place(p.id) menu=rowMenu
              Icon "drag_indicator"
              Column grow
                Text.strong "{p.name}, {p.region}"
                Text.muted "{p.sky}"
              Text.heading "{p.temp}°"
              IconButton "arrow_upward" tooltip="Move up" hidden={index==0} onTap={places=setAt(setAt(places,index,places[index-1]),index-1,places[index])}
              IconButton "arrow_downward" tooltip="Move down" hidden={index==length(places)-1} onTap={places=setAt(setAt(places,index,places[index+1]),index+1,places[index])}
              IconButton "more_vert" tooltip="More" onTap={sel=p.id; open rowMenu}
  Menu#rowMenu end
    MenuItem "Open" icon="open_in_new" onTap={go today}
    MenuItem "Move to top" icon="vertical_align_top" onTap={places=concat([find(places,p=>p.id==sel)],removeWhere(places,p=>p.id==sel))}
    MenuItem "Remove" icon="delete" onTap={places=removeWhere(places,p=>p.id==sel); toast "Place removed"}
  Dialog#addPlace "Add place" "Search a city or a postcode"
    Input "City" "e.g. Tacoma" @newCity
    Button "Add" onTap={places=append(places,{"id":lower(newCity),"name":newCity,"region":"","temp":15,"sky":"Cloudy"}); newCity=""; close}
