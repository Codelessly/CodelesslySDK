### Bumped history
state bumped=[{"id":"T409","table":"3","time":"7:41","by":"Dee"},{"id":"T410","table":"9","time":"7:44","by":"Marta"},{"id":"T411","table":"Bar 1","time":"7:46","by":"Dee"}]
state sel=""
App
  Sidebar
    Brand "Pass" icon="restaurant"
    NavItem "Expo board" icon="view_column" onTap={go expo-board}
    NavItem "All-day counts" icon="format_list_numbered" onTap={go all-day-counts}
    NavItem "Bumped history" icon="history" selected={true} badge="{length(bumped)}"
  Main
    Nav
      Heading "Bumped history" level=1
      Button "Recall last" size=large icon="undo" hotkey="r" onTap={length(bumped)>0 ? {emit recall last(bumped).id; bumped=removeAt(bumped,length(bumped)-1); toast "Ticket recalled"} : toast "Nothing to recall"}
    Column pad=24 gap=12
      each b in reverse(bumped)
        Card
          Row between middle
            Row gap=16 middle
              Text "Table {b.table}" size=28 weight=700
              Text.muted "{b.id}" size=20
            Row gap=16 middle
              Text.muted "Bumped {b.time} by {b.by}" size=20
              Button.outline "Recall" onTap={sel=b.id; open confirm}
  Dialog#confirm "Recall ticket {sel}" "It goes back to the board with its original timer."
    Row gap=8 end
      Button.ghost "Cancel" onTap={close}
      Button "Recall" onTap={emit recall sel; bumped=removeWhere(bumped,x=>x.id==sel); toast "Recalled {sel}"; close}
