    Nav
      Heading "Schedule" level=1

    Column pad=24 gap=16 scroll
      Row scroll gap=8
        each d in days
          ChoiceChip "{d.d} {d.n}" selected={day==index} onTap={day=index}
      if length(booked)>0
        Card gap=8 fill=#FFE8F5EE
          Row between
            Column gap=2
              Text.strong "You are booked: Power Mat Pilates, 8:00 AM"
              Text.muted "Free cancel window closes in"
            Text.strong "{duration(cancelLeft)}"
          Row gap=8
            Button.outline "Cancel booking" onTap={cancelLeft>0 ? {booked=[]; credits=credits+1; toast "Cancelled, credit returned"} : toast "Cancel window has closed"}
            Button.ghost "View class" onTap={sel="c2"; go class}
      each c in classes
        Card gap=8
          Row between
            Column gap=2 grow
              Text.caption "{c.time}"
              Text.heading "{c.name}"
              Row gap=8
                Avatar "{c.coach}" size=24
                Text.muted "{c.coach}"
            Column end gap=4
              Dots count={c.cap} filled={c.cap-c.spots} size=6
              if c.spots>0
                Text.small "{c.spots} spots left"
              if c.spots==0
                Badge.warning "Waitlist"
          Row gap=8 middle
            Button.ghost "Details" onTap={sel=c.id; go class}
            Spacer grow
            if any(waitlisted,w=>w.id==c.id)
              Text.muted "You are #{find(waitlisted,w=>w.id==c.id).pos} on the waitlist"
              Button.outline "Leave" onTap={waitlisted=removeWhere(waitlisted,w=>w.id==c.id); toast "Left the waitlist"}
            if !any(waitlisted,w=>w.id==c.id) && c.spots==0
              Button.secondary "Join waitlist" onTap={waitlisted=append(waitlisted,{"id":c.id,"pos":4}); toast "You are #4 on the waitlist"}
            if c.spots>0 && !any(booked,b=>b.id==c.id)
              Button "Book, 1 credit" onTap={credits>0 ? {booked=append(booked,{"id":c.id,"cancelBy":5400}); credits=credits-1; cancelLeft=5400; toast "Booked {c.name}"} : open nocredits}
            if any(booked,b=>b.id==c.id)
              Badge.success "Booked"
  Sheet#filters "Filters"
    Column gap=12
      SegmentedButton bind=kindFilter items=["All","Bike","Mat"]
      Select "Coach" items=["Anyone","Dani Ortiz","Lena Park","Marcus Bell"] bind=coachFilter

  Dialog#nocredits "Out of credits" "Top up to keep booking"
    Text.muted "You have no credits left this month."
    Button "Buy a 5 class pack" onTap={credits=credits+5; close; toast "5 credits added"}
