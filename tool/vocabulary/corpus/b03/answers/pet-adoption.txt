### Favourites
App
  Sidebar
    Brand "Paws & Co" icon="pets"
    NavItem "Discover" icon="style" onTap={go discover}
    NavItem "Favourites" icon="favorite" badge="{length(favs)}" selected={true}
    NavItem "Apply" icon="assignment" onTap={go application}
    NavItem "My application" icon="timeline" onTap={go status}
  Main
    Nav
      Heading "Favourites" level=1
      Text.muted "{length(favs)} saved"
    Column pad=24 gap=12
      if length(favs)==0
        Card gap=8 center
          Icon "favorite_border"
          Text.muted "Nothing saved yet. Swipe right on a pet you like."
          Button "Browse pets" onTap={go discover}
      each id in favs
        ListTile "{find(deck, p=>p.id==id).name}" subtitle="{find(deck, p=>p.id==id).breed} · {find(deck, p=>p.id==id).dist} km away"
          Avatar src="{find(deck, p=>p.id==id).photo}" size=48
          Row gap=8
            Button.outline "Apply" onTap={go application}
            IconButton "delete" tooltip="Remove" onTap={favs=without(favs,id); toast "Removed"}
