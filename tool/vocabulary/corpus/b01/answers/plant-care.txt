### Today
state tab="today"
state tasks=[{"id":1,"plant":"Pothos","task":"Water, soil is dry","time":"8:00","icon":"water_drop","done":false},{"id":2,"plant":"Calathea","task":"Mist the leaves","time":"10:30","icon":"cloud","done":false},{"id":3,"plant":"Fiddle Leaf Fig","task":"Rotate a quarter turn","time":"18:00","icon":"autorenew","done":false},{"id":4,"plant":"Monstera","task":"Feed with fertilizer","time":"19:00","icon":"science","done":true}]
App
  Sidebar
    Brand "Fernwise" icon="eco"
    NavItem "My plants" icon="grid_view" onTap={go my-plants}
    NavItem "Today" icon="task_alt" badge="{count(tasks,t=>!t.done)}" selected={tab=="today"} onTap={tab="today"}
    NavItem "Identify" icon="photo_camera" onTap={go identify}
  Main
    Nav
      Heading "Today" level=1
      Text.muted "{formatDate(today(),'EEEE, MMM d')}"
    Column pad=24 gap=12 scroll maxw=640
      Row between middle
        Text.strong "{count(tasks,t=>t.done)} of {length(tasks)} done"
        Text.caption "Swipe a card left to finish it"
      Progress {count(tasks,t=>t.done)/length(tasks)}
      each t in tasks
        Card gap=8 onSwipeLeft={tasks=updateWhere(tasks,x=>x.id==t.id,x=>with(x,"done",true)); toast "Done: {t.task}"} animate=slide-up stagger=80
          Row gap=12 middle
            IconTile "{t.icon}"
            Column grow gap=2
              Text.strong "{t.plant}" strike={t.done}
              Text.muted "{t.task}" strike={t.done}
            Column gap=2 end
              Text.caption "{t.time}"
              if t.done
                Badge.success "Done"
              if !t.done
                Badge.neutral "To do"
      if count(tasks,t=>!t.done)==0
        Card center gap=8
          Icon "check_circle" size=48
          Text.heading "All plants are happy"
          Text.muted "Nothing left for today."
