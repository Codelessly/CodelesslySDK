state qi=0
state running=true
state left=45
state code="KQTMR"

      Avatar "Dana Whitfield" size=32

        Badge.success "{viewers} connected"

          Text.heading "{questions[qi].title}"

            each o in questions[qi].options
              Column gap=4
                Row between
                  Text.strong "{o}"

            Text.heading "Average {fixed(sum(range(5),i=>(i+1)*questions[qi].votes[i])/max(1,sum(questions[qi].votes,v=>v)),1)} / 5"
        Row gap=8
          Button.destructive "Close voting" icon="stop" onTap={running=false; left=0; emit closePoll questions[qi].id}
          Button.outline "Reopen" icon="replay" hidden={running} onTap={running=true; left=45; emit reopenPoll questions[qi].id}

      Pane w=320 pad=16 gap=12

          QRCode "https://showhands.app/{code}" size=200
          Text "Code" center

          Button.outline "Copy join link" icon="content_copy" onTap={copy "https://showhands.app/{code}"; toast "Link copied"}
        Card gap=8
          Text.strong "Time left"
          Text "{left}s" size=48 weight=700
          Progress {left/45}
  Dialog#newq "New question" "Choice, word cloud or rating"
    Input "Question" bind=draftTitle

### Audience vote
state choice=-1
state submitted=false
state closed=false
state options=["Provider","Riverpod","Bloc","setState"]
state results=[18,31,12,9]
Page
  Column pad=20 gap=16 maxw=420
    Row between middle
      Badge.neutral "KQTMR"
      Badge.success "Live"
    if !submitted && !closed
      Text.heading "Which Flutter state tool do you reach for first?"
      each o in options
        Card onTap={choice=index; submitted=true; emit vote index} animate=slide-up stagger=60
          Row gap=12 middle
            Icon "radio_button_unchecked"
            Text.strong "{o}"
      Text.muted "One tap, and your vote counts" center
    if submitted && !closed
      Column gap=12 center middle animate=pulse
        Icon "check_circle" size=64 color=#FF2E9E5B
        Text.heading "Vote in: {options[choice]}" center
        Text.muted "Waiting for the presenter to close voting" center
        Progress animate=shimmer
        Button.outline "Change my vote" onTap={submitted=false}
    if closed
      Text.heading "Results"
      each o in options
        Column gap=4
          Row between
            Text.strong "{o}"
            Text "{results[index]}"
          Progress {results[index]/max(1,sum(results,v=>v))}
      Text.muted "You voted {options[choice]}" center
    Button.ghost "See the results (demo)" hide=phone onTap={closed=true}
  Announce "Voting closed" when={closed}

### Audience word cloud and rating
state word=""
state sent=[]
state stars=0
state emoji=["😠","🙁","😐","🙂","🤩"]
Page
  Column pad=20 gap=24 maxw=420
    Card gap=12
      Text.heading "One word for how you feel about code review"
      Input "Your word" "e.g. necessary" bind=word maxlength=20
      Row gap=8
        Button "Send" grow onTap={sent=append(sent,lower(trim(word))); emit word lower(trim(word)); word=""; toast "Sent"} disabled={length(trim(word))==0}
      if length(sent)>0
        Text.caption "You sent {length(sent)} of 3"
        Row gap=8 wrap
          each w in sent
            Chip "{w}" onDelete={sent=without(sent,w)}
    Card gap=12
      Text.heading "How was today's talk?"
      Row between
        each e in emoji
          Text "{e}" size={stars==index+1?48:32} onTap={stars=index+1} animate=pop
      Rating bind=stars max=5
      Text.muted "{stars==0?'Tap a face':'You gave '+stars+' of 5'}" center
      Button "Submit rating" disabled={stars==0} onTap={emit rate stars; toast "Thanks!"}
