### Schedule

App
  Sidebar
    Brand "Crumb" icon="bakery_dining"
    NavItem "Formula" icon="scale" onTap={go formula}
    NavItem "Schedule" icon="schedule" selected={true} onTap={go schedule}

    Column pad=24 gap=16 scroll
      Card gap=12
        Row between middle

          Column

          Column
            Text.fieldLabel "Starts"

          Column
            Text.fieldLabel "Ready at"
            Text size=32 weight=700 color=#FF2E7D32 "{floor(((mode==\"start\"?ah*60+am+sum(steps,s=>s.mins):ah*60+am)+1440)%1440/60)}:{((mode==\"start\"?ah*60+am+sum(steps,s=>s.mins):ah*60+am)+1440)%60<10?\"0\":\"\"}{((mode==\"start\"?ah*60+am+sum(steps,s=>s.mins):ah*60+am)+1440)%60}"
      Card gap=12
        Text.strong "Timeline"
        Timeline
          each step in steps
            TimelineItem "{step.name}" time="{step.mins} min" icon="{step.name==\"Bake\"?\"local_fire_department\":\"circle\"}"
              Row gap=8 middle
                Text.muted "at +{sum(take(steps,index),s=>s.mins)} min"
                Spacer grow
                IconButton "remove" tooltip="Shorter" onTap={steps=updateWhere(steps,s=>s.id==step.id,s=>with(s,"mins",max(5,s.mins-5)))}
                Input bind=step.mins type=number w=80
                IconButton "add" tooltip="Longer" onTap={steps=updateWhere(steps,s=>s.id==step.id,s=>with(s,"mins",s.mins+5))}
        Row gap=8
          Button.outline "Add step" icon="add" onTap={steps=append(steps,{"id":length(steps)+100,"name":"Rest","mins":15})}
          Button.ghost "Remove last" onTap={steps=removeAt(steps,length(steps)-1)}
      Card gap=8
        Text.strong "Time per step"
        BarChart values={map(steps,s=>s.mins)} labels={map(steps,s=>s.name)} h=160

### Timers
App
  Sidebar
    Brand "Crumb" icon="bakery_dining"
    NavItem "Formula" icon="scale" onTap={go formula}
    NavItem "Schedule" icon="schedule" onTap={go schedule}
    NavItem "Timers" icon="timer" badge="{count(timers,t=>t.running)}" selected={true} onTap={go timers}
    Spacer grow
    Text.muted "Country loaf, 2 loaves"
  Main
    Nav
      Heading "Timers" level=1
      Row gap=8
        Button.outline "Pause all" onTap={timers=map(timers,t=>with(t,"running",false))}
        Button "Start all" onTap={timers=map(timers,t=>with(t,"running",t.left>0))}
    Column pad=24 gap=16 scroll
      every 1s onTick={timers=updateWhere(timers,t=>t.running&&t.left>0,t=>with(t,"left",t.left-1))}
      Grid cols=2 gap=16
        each t in timers
          Card gap=12
            Row between middle
              Text.heading3 "{t.name}"
              if t.left==0
                Badge.error "Time's up" animate=pulse
              if t.left>0&&t.running
                Badge.info "Running"
              if t.left>0&&!t.running
                Badge.neutral "Paused"
            Stack center middle
              Ring {1-t.left/t.full}
              Text "{duration(t.left)}" size=28 weight=700
            Sound "bell" playing={t.left==0&&!t.muted} loop
            Row gap=8 center
              Button "{t.running?\"Pause\":\"Start\"}" onTap={timers=updateWhere(timers,x=>x.id==t.id,x=>with(x,"running",!x.running))}
              Button.outline "Restart" onTap={timers=updateWhere(timers,x=>x.id==t.id,x=>with(with(x,"left",x.full),"muted",false))}
              if t.left==0
                Button.destructive "Silence" icon="notifications_off" onTap={timers=updateWhere(timers,x=>x.id==t.id,x=>with(x,"muted",true))}
              IconButton "add" tooltip="Add 5 minutes" onTap={timers=updateWhere(timers,x=>x.id==t.id,x=>with(with(x,"left",x.left+300),"muted",false))}
