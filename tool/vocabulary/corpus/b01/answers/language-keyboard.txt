### Practice
state typed=""
state running=false
state elapsed=0
state errors=0
state lessonId=3
state showFingers=true
state result={"wpm":0,"acc":100}
App
  Sidebar
    Brand "Glyph" icon="keyboard"
    NavItem "Lessons" icon="school" onTap={go lessons}
    NavItem "Practice" icon="keyboard" selected={true}
    NavItem "Results" icon="insights" onTap={go results}
  Main
    Nav
      Heading "Lesson {lessonId}: {find(lessons,l=>l.id==lessonId).name}" level=1
      Row gap=8
        Toggle "Finger guide" bind=showFingers
        Button.outline "Restart" icon="restart_alt" onTap={typed=""; running=false; elapsed=0; errors=0}
    Column pad=24 gap=16 scroll
      every 1s when running onTick={elapsed=elapsed+1}
      Grid cols=3 gap=16
        Card gap=4
          Text.muted "Words per minute"
          Text "{elapsed>0 ? round(length(typed)/5/(elapsed/60),0) : 0}" size=48 weight=700
        Card gap=4
          Text.muted "Accuracy"
          Text "{length(typed)>0 ? round(100*(length(typed)-errors)/length(typed),0) : 100}%" size=48 weight=700
        Card gap=4
          Text.muted "Time"
          Text "{duration(elapsed)}" size=48 weight=700
      Card gap=12
        TypingText target={find(lessons,l=>l.id==lessonId).text} bind=typed autofocus size=28 correctColor=#FF2E7D32 wrongColor=#FFC62828 onKeyDown={running=true} onError={errors=errors+1} onComplete={running=false; result={"wpm":round(length(typed)/5/(elapsed/60),0),"acc":round(100*(length(typed)-errors)/length(typed),0)}; open done}
        Progress {length(typed)/length(find(lessons,l=>l.id==lessonId).text)}
      Keyboard layout="qwerty" typed={typed} target={find(lessons,l=>l.id==lessonId).text} highlightNext showFingers={showFingers}
      Row gap=8 center
        Icon "back_hand"
        Text.strong "Next key: {substring(find(lessons,l=>l.id==lessonId).text,length(typed),length(typed)+1)}"
        Text.muted "use your {fingerFor(substring(find(lessons,l=>l.id==lessonId).text,length(typed),length(typed)+1))}"
  Dialog#done "Lesson complete" "Here is how that went."
    Row gap=24 center
      Column center gap=4
        Text "{result.wpm}" size=48 weight=700
        Text.muted "WPM"
      Column center gap=4
        Text "{result.acc}%" size=48 weight=700
        Text.muted "Accuracy"
    if result.wpm>=30 && result.acc>=92
      Badge.success "Passed: the next lesson is unlocked"
      Button "Next lesson" onTap={lessons=updateWhere(lessons,l=>l.id==lessonId,l=>with(with(with(l,"passed",true),"best",max(l.best,result.wpm)),"acc",max(l.acc,result.acc))); lessonId=lessonId+1; typed=""; elapsed=0; errors=0; close}
    if !(result.wpm>=30 && result.acc>=92)
      Badge.warning "Not yet: aim for 30 WPM and 92% accuracy"
      Button "Try again" onTap={typed=""; elapsed=0; errors=0; close}
    Button.ghost "See results" onTap={close; go results}
