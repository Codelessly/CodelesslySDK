### Lights and dosing
state sunrise="07:00"
state sunset="21:30"
state peak=80
state pumps=[{"id":1,"name":"Alkalinity","ml":12.0,"times":["08:00","20:00"],"on":true,"left":420},{"id":2,"name":"Calcium","ml":8.0,"times":["09:00","21:00"],"on":true,"left":610},{"id":3,"name":"Magnesium","ml":5.0,"times":["10:00"],"on":false,"left":300}]
state sel=0
state manual=2
state newTime=""
App
  Sidebar
    Brand "Reefkeeper" icon="waves"
    NavItem "Dashboard" icon="dashboard" onTap={go dashboard}
    NavItem "Lights and dosing" icon="light_mode" selected onTap={go lights-dosing}
    NavItem "Livestock" icon="set_meal" onTap={go livestock}
    NavItem "Tests and alarms" icon="science" badge="2" onTap={go tests-alarms}
    Spacer grow
    Text.muted "Reef 240 L, Living room"
  Main
    Nav
      Heading "Lights and dosing" level=1
      Button.outline "Reset schedule" onTap={sunrise="07:00"; sunset="21:30"; peak=80; toast "Schedule reset"}
    Column pad=24 gap=16 scroll
      Card gap=12
        Row between
          Text.panelTitle "Daylight curve"
          Text.muted "Sunrise {sunrise}, sunset {sunset}, peak {peak}%"
        DayCurve h=200 sunrise=sunrise sunset=sunset peak=peak draggable onChange={sunrise=result.sunrise; sunset=result.sunset; peak=result.peak}
        Row gap=12
          TimePicker "Sunrise" @sunrise
          TimePicker "Sunset" @sunset
          Slider @peak min=10 max=100
        Button "Save schedule" onTap={emit saveLights {"sunrise":sunrise,"sunset":sunset,"peak":peak}; toast "Light schedule saved"}
      Text.sectionTitle "Dosing pumps"
      each p in pumps
        Card gap=12
          Row between
            Column gap=4
              Text.strong "{p.name}"
              Text.muted "{p.ml} ml per dose at {join(p.times,', ')}"
            Toggle @p.on
          Row between
            Text.caption "Reservoir {p.left} ml"
            Progress {p.left/1000}
          Row gap=8
            Button.outline "Edit schedule" onTap={sel=p.id; open schedule}
            Button.secondary "Manual dose" onTap={sel=p.id; open dose}
  Sheet#schedule "Dosing schedule"
    Text.muted "{find(pumps,x=>x.id==sel).name}"
    Row gap=8 wrap
      each t in find(pumps,x=>x.id==sel).times
        Chip "{t}" onTap={pumps=updateWhere(pumps,x=>x.id==sel,x=>with(x,"times",without(x.times,t)))}
    Row gap=8
      TimePicker "Add time" @newTime grow
      Button "Add" onTap={newTime!="" ? {pumps=updateWhere(pumps,x=>x.id==sel,x=>with(x,"times",append(x.times,newTime))); newTime=""} : toast "Pick a time"}
    Button "Done" onTap={emit savePump find(pumps,x=>x.id==sel); close}
  Dialog#dose "Manual dose" "Adds to the tank now"
    Text.strong "{find(pumps,x=>x.id==sel).name}"
    Input "Amount in ml" type=number @manual
    Row gap=8 end
      Button.ghost "Cancel" onTap={close}
      Button "Dose now" onTap={manual>0 ? {emit dose {"pump":sel,"ml":manual}; pumps=updateWhere(pumps,x=>x.id==sel,x=>with(x,"left",x.left-manual)); toast "Dosed {manual} ml"; close} : toast "Enter an amount"}

App
  Sidebar
    Brand "Reefkeeper" icon="waves"
    NavItem "Dashboard" icon="dashboard" onTap={go dashboard}
    NavItem "Lights and dosing" icon="light_mode" onTap={go lights-dosing}
    NavItem "Livestock" icon="set_meal" selected onTap={go livestock}

    Spacer grow

    Nav
      Heading "Livestock" level=1
      Row gap=8
        Input "Search" @q w=220
        Button "Add animal" icon="add" onTap={fname=""; fcount=1; open addAnimal}
    Column pad=24 gap=12 scroll
      Text.muted "{sum(animals,a=>a.count)} animals in {length(animals)} species"
      each a in filter(animals,x=>contains(lower(x.name),lower(q)))
        Card
          Row gap=16 middle
            Image "{a.photo}" w=96 h=72 radius=8 onTap={at=index; open shots}
            Column grow gap=4
              Text.strong "{a.name}"
              Text.muted "{a.note}"
              Text.caption "Added {formatDate(a.added,'MMM d, yyyy')}"
            Badge.neutral "x{a.count}"
            IconButton "more_vert" tooltip="More" onTap={sel=a.id; open rowMenu}
  Menu#rowMenu end
    MenuItem "Add one" icon="add" onTap={animals=updateWhere(animals,x=>x.id==sel,x=>with(x,"count",x.count+1))}
    MenuItem "Remove one" icon="remove" onTap={animals=updateWhere(animals,x=>x.id==sel,x=>with(x,"count",max(1,x.count-1)))}
    MenuItem "Delete" icon="delete" onTap={animals=removeWhere(animals,x=>x.id==sel); toast "Removed"}
  Lightbox#shots photos={map(animals,a=>({"url":a.photo,"caption":a.name}))} @at
  Dialog#addAnimal "Add animal"
    Input "Name" @fname
    Input "Count" type=number @fcount
    Button "Add" onTap={fname!="" ? {animals=append(animals,{"id":length(animals)+10,"name":fname,"count":fcount,"added":today(),"photo":"","note":""}); close; toast "Added"} : toast "Give it a name"}
