### Compare and Quiz
state tab="compare"
state a="earth"
state b="jupiter"
state planets=[{"id":"mercury","name":"Mercury","diameter":4879},{"id":"venus","name":"Venus","diameter":12104},{"id":"earth","name":"Earth","diameter":12742},{"id":"mars","name":"Mars","diameter":6779},{"id":"jupiter","name":"Jupiter","diameter":139820},{"id":"saturn","name":"Saturn","diameter":116460},{"id":"uranus","name":"Uranus","diameter":50724},{"id":"neptune","name":"Neptune","diameter":49244}]
state questions=[{"q":"How many moons does Mars have?","options":["0","2","14","79"],"answer":1},{"q":"Which planet has the most known moons?","options":["Jupiter","Neptune","Saturn","Earth"],"answer":2},{"q":"Which planet has no moons at all, along with Mercury?","options":["Mars","Venus","Uranus","Earth"],"answer":1},{"q":"What is the name of Earth's only natural satellite?","options":["Luna, the Moon","Phobos","Titan","Europa"],"answer":0}]
state step=0
state score=0
state picked=-1
state done=false
App
  Sidebar
    Brand "Orbit School" icon="public"
    NavItem "Solar system" icon="brightness_5" onTap={go solar-system}
    NavItem "Compare" icon="compare" selected={tab=="compare"} onTap={tab="compare"}
    NavItem "Moon quiz" icon="quiz" selected={tab=="quiz"} onTap={tab="quiz"}
    NavItem "Night sky" icon="nightlight" onTap={go night-sky}
    NavItem "Glossary" icon="menu_book" onTap={go glossary}
  Main
    Nav
      Heading "{tab=="compare" ? "Compare two planets" : "Moon quiz"}" level=1
    Column pad=24 gap=16 scroll
      if tab=="compare"
        Row gap=16
          Select "First planet" bind=a items=["mercury","venus","earth","mars","jupiter","saturn","uranus","neptune"] grow
          Select "Second planet" bind=b items=["mercury","venus","earth","mars","jupiter","saturn","uranus","neptune"] grow
        Card gap=16
          Text.strong "Drawn to the same scale"
          Row center bottom gap=48
            Column center gap=8
              Box w={clamp(find(planets,p=>p.id==a).diameter/700,6,200)} h={clamp(find(planets,p=>p.id==a).diameter/700,6,200)} radius=999 fill=#FF4F8FD9
              Text.strong "{upper(a)}"
              Text.muted "{find(planets,p=>p.id==a).diameter} km"
            Column center gap=8
              Box w={clamp(find(planets,p=>p.id==b).diameter/700,6,200)} h={clamp(find(planets,p=>p.id==b).diameter/700,6,200)} radius=999 fill=#FFD98A4F
              Text.strong "{upper(b)}"
              Text.muted "{find(planets,p=>p.id==b).diameter} km"
        Card gap=4
          Text "{upper(b)} is {fixed(find(planets,p=>p.id==b).diameter/find(planets,p=>p.id==a).diameter,2)} times the width of {upper(a)}."
          Text.muted "About {round(pow3,0)} of one would fit inside the other by volume."
      if tab=="quiz"
        if done
          Card gap=12 center
            Text "{score} of {length(questions)}" size=48 weight=700
            Text.muted "{score==length(questions) ? "Perfect, a real astronomer." : "Good start. Try again to beat it."}"
            Button "Try again" onTap={step=0; score=0; picked=-1; done=false}
        if !done
          Card gap=12
            Progress {(step)/length(questions)}
            Text.caption "Question {step+1} of {length(questions)}"
            Text.heading "{questions[step].q}"
            Column gap=8
              each o in questions[step].options
                ChoiceChip "{o}" selected={picked==index} onTap={picked=index}
            Button "{step+1==length(questions) ? "Finish" : "Next"}" onTap={picked<0 ? toast "Pick an answer" : {score=score+(picked==questions[step].answer ? 1 : 0); picked=-1; step+1==length(questions) ? done=true : step=step+1}}
