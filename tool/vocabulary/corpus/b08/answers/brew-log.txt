### Brew Day
state boil=3600
state running=false
state mashleft=3600
state mashing=false
state steps=[{"t":"Sanitize fermenter and airlock","done":true},{"t":"Heat strike water to 163 F","done":true},{"t":"Mash in at 152 F","done":false},{"t":"Sparge and collect 7 gal","done":false},{"t":"Boil 60 minutes","done":false},{"t":"Chill to 68 F","done":false},{"t":"Transfer, aerate and pitch yeast","done":false}]
state adds=[{"label":"Magnum 0.5 oz","at":60,"rung":false},{"label":"Cascade 1 oz","at":15,"rung":false},{"label":"Citra 1 oz","at":5,"rung":false},{"label":"Citra 2 oz, flameout","at":0,"rung":false}]
every 1s when running onTick={boil=max(0,boil-1); adds=updateWhere(adds,a=>a.at*60>=boil&&!a.rung,a=>with(a,"rung",true)); boil==0 ? {running=false; emit boilDone; play bell; toast "Boil complete"} : }
every 1s when mashing onTick={mashleft=max(0,mashleft-1); mashleft==0 ? {mashing=false; play bell; toast "Mash is done, start the sparge"} : }
App
  Sidebar
    Brand "Wort" icon="sports_bar"
    NavItem "Batches" icon="list_alt" onTap={go batches}
    NavItem "Recipe" icon="menu_book" onTap={go recipe}
    NavItem "Brew day" icon="timer" selected={true} onTap={go brew-day}
    NavItem "Fermentation" icon="show_chart" onTap={go fermentation}
  Main
    Nav
      Heading "Brew day: Harbor Pale Ale" level=1
      Text.muted "{count(steps,s=>s.done)} of {length(steps)} steps"
    Column pad=24 gap=16 scroll
      Progress {count(steps,s=>s.done)/length(steps)}
      Row gap=16 top
        Card grow gap=8
          Heading "Checklist" level=3
          each s in steps
            Checkbox "{s.t}" @s.done
        Column grow gap=16
          Card gap=12
            Heading "Mash timer" level=3
            Text "{duration(mashleft)}" size=40 weight=700
            Progress {1-mashleft/3600}
            Row gap=8
              Button "{mashing ? 'Pause' : 'Start mash'}" onTap={mashing=!mashing}
              Button.outline "Reset" onTap={mashing=false; mashleft=3600}
          Card gap=12
            Heading "Boil timer" level=3
            Text "{duration(boil)}" size=40 weight=700 color={boil<=300 ? "#FFC62828" : "#FF212121"}
            Progress {1-boil/3600}
            Row gap=8
              Button "{running ? 'Pause' : 'Start boil'}" onTap={running=!running}
              Button.outline "Reset" onTap={running=false; boil=3600; adds=map(adds,a=>with(a,"rung",false))}
          Card gap=8
            Heading "Hop additions" level=3
            each a in adds
              Row between middle
                Column gap=2
                  Text.strong "{a.label}"
                  Text.muted "{a.at==0 ? 'at flameout' : 'at ' + a.at + ' min left'}"
                if a.rung
                  Badge.success "Rang" animate=pulse
                if !a.rung
                  Text.muted "in {duration(max(0,boil-a.at*60))}"
