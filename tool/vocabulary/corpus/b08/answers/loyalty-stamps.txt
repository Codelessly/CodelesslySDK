### Redeem
state sec=60
state redeemed=false
state slid=0
every 1s when !redeemed && sec>0 onTick={sec=sec-1}
App
  Sidebar
    Brand "Stampbook" icon="loyalty"
    NavItem "Wallet" icon="wallet" onTap={go wallet}
    NavItem "Find shops" icon="map" onTap={go shop-finder}
    NavItem "History" icon="history" onTap={go history}
  Main
    Nav
      IconButton "arrow_back" tooltip="Back" onTap={go wallet}
      Heading "Redeem reward" level=1
    Column pad=24 gap=16 center maxw=480
      Card gap=12 center
        IconTile "local_cafe"
        Heading "Daybreak Coffee" level=3
        Text.strong "Free large drink"
        Text.muted "10 of 10 stamps collected"
      if !redeemed
        Text.muted "Slide to show your barcode to the cashier. It stays valid for one minute."
        Slider bind=slid min=0 max=100 onChange={result>=100 ? sec=60 : slid=0}
        Text.caption center "Slide all the way to the right to reveal"
        if slid>=100
          Card gap=12 center animate=scale
            Barcode "DBC-4821-7730-1956" type=code128 h=120
            Row gap=8 center middle
              Icon "timer"
              Text.strong "Expires in {duration(sec)}"
            Progress {sec/60}
            if sec==0
              Text.danger "Code expired"
              Button.outline "Get a new code" onTap={sec=60}
            Button "Cashier scanned it" onTap={redeemed=true; toast "Reward redeemed"}
      if redeemed
        Card gap=12 center animate=pop
          Icon "check_circle" size=64
          Heading "Enjoy your drink" level=3
          Text.muted "Your card starts fresh at 0 of 10."
          Button "Back to wallet" onTap={go wallet}
