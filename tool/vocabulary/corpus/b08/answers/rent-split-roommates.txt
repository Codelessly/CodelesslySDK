        Card grow gap=12
          Row between
            Heading "{month}" level=3
            Text.muted "{money(sum(recurring,r=>r.amount),'$',2)} a month"
          Calendar month="2026-11" marked={map(recurring,r=>r.day)} bind=picked onSelect={picked=result.day}
        Column w=320 gap=12
          Card gap=8
            Heading "Day {picked}" level=3
            if count(recurring,r=>r.day==picked)==0
              Text.muted "Nothing due on this day."
            each r in filter(recurring,x=>x.day==picked)
              Row between
                Text.strong "{r.title}"
                Text "{money(r.amount,'$',2)}"
          Card gap=8
            Heading "Coming up" level=3
            each r in sortBy(filter(recurring,x=>x.day>=picked),x=>x.day)
              ListTile "{r.title}" subtitle="Due on the {r.day}th"
                Text.strong "{money(r.amount,'$',2)}"
                IconButton "close" tooltip="Stop repeating" onTap={recurring=removeWhere(recurring,x=>x.id==r.id)}
  Dialog#newbill "New recurring bill"
    Input "Bill" bind=newTitle
    Input "Amount" type=number bind=newAmount
    Slider bind=newDay min=1 max=28 step=1
    Text.muted "Repeats on day {newDay} of every month"
    Button "Add" onTap={newTitle!="" ? {recurring=append(recurring,{"id":length(recurring)+10,"title":newTitle,"day":newDay,"amount":newAmount}); newTitle=""; close} : toast "Name the bill"}
