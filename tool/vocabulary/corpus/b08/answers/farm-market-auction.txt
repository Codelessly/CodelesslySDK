### Live Auction
state tab="auction"
state clock=28800
state lots=[{"id":1,"name":"Crate of Bramley apples","photo":"https://images.example.com/lots/apples.jpg","qty":1,"unit":"crate","kg":12,"reserve":40,"reserveMet":false,"price":28,"bids":3,"seller":"Hollins Orchard"},{"id":2,"name":"Three Light Sussex hens","photo":"https://images.example.com/lots/hens.jpg","qty":3,"unit":"hens","kg":8.5,"reserve":60,"reserveMet":true,"price":65,"bids":7,"seller":"Dunmore Farm"},{"id":3,"name":"Sack of Maris Piper potatoes","photo":"https://images.example.com/lots/potatoes.jpg","qty":1,"unit":"sack","kg":25,"reserve":15,"reserveMet":true,"price":18,"bids":2,"seller":"Tilley and Sons"}]
state at=0
state window=30
state running=false
state count=0
state sold=false
state step=2
state myBids=[]
every 1s onTick={clock=clock+1}
every 1s when running onTick={window=window-1; window<=0 ? {running=false; count=0; sold=true; toast "Sold"} : {}}
every 1s when running && window<=3 onTick={count=count+1}
App
  Sidebar
    Brand "Barrow Bids" icon="gavel"
    NavItem "Live auction" icon="gavel" selected={tab=="auction"} onTap={go live-auction}
    NavItem "Sell a lot" icon="add_a_photo" selected={tab=="sell"} onTap={go sell-a-lot}
    NavItem "My wins" icon="emoji_events" badge="2" selected={tab=="wins"} onTap={go my-wins}
    Spacer grow
    Text.caption "Market day"
    Text.strong "{formatDate(today(),"EEE d MMM")}"
    Clock "{clock}" format="HH:mm:ss" start=28800
  Main
    Nav
      Heading "Lot {at+1} of {length(lots)}" level=1
      Row gap=8
        Badge.info "Window {window}s"
        IconButton "volume_up" tooltip="Bell on"
        Button.outline "Next lot" onTap={at=(at+1)%length(lots); window=30; running=false; count=0; sold=false}
    Column pad=24 gap=16 scroll
      Row gap=24 top
        Card grow gap=12
          Image "{lots[at].photo}" h=280 radius=12
          Row between
            Text.heading "{lots[at].name}"
            if lots[at].reserveMet
              Badge.success "Reserve met"
            if !lots[at].reserveMet
              Badge.warning "Reserve not met"
          Row gap=24
            Column gap=2
              Text.caption "Quantity"
              Text.strong "{lots[at].qty} {lots[at].unit}"
            Column gap=2
              Text.caption "Weight"
              Text.strong "{lots[at].kg} kg"
            Column gap=2
              Text.caption "Seller"
              Text.strong "{lots[at].seller}"
        Column w=320 gap=16
          Card gap=12 center
            Text.caption "Current bid"
            Text "{money(lots[at].price,"£",2)}" size=48 weight=700
            Text.muted "{lots[at].bids} bids"
            Row gap=8 center
              Icon "timer"
              Text.heading "{window}s"
            Progress {window/30}
            if count>0 && !sold
              Text "Going {count == 1 ? "once" : count == 2 ? "twice" : "gone"}..." size=28 weight=700 color=#FFB3261E animate=pulse
            if sold
              Row gap=8 center animate=pop
                Icon "notifications_active"
                Text.success "Sold to the room"
          Card gap=12
            SegmentedButton bind=step items=[1,2,5]
            Button "Bid +£{step}" size=large disabled={sold} onTap={lots=updateWhere(lots,l=>l.id==lots[at].id,l=>with(with(with(l,"price",l.price+step),"bids",l.bids+1),"reserveMet",l.price+step>=l.reserve)); myBids=append(myBids,{"lot":lots[at].id,"amount":lots[at].price+step}); running=true; window=30; count=0}
            Button.outline "Start the window" disabled={running} onTap={running=true; window=30; sold=false}
            Button.tertiary "Ring the bell" icon="notifications" onTap={sold=true; running=false; toast "Bell rung, lot {at+1} sold"}
      Card gap=8
        Text.strong "Your bids this session"
        if isEmpty(myBids)
          Text.muted "No bids yet."
        each b in myBids
          Row between
            Text "Lot {b.lot}"
            Text.strong "{money(b.amount,"£",2)}"

    MenuItem "Withdraw lot" icon="delete" onTap={entered=removeWhere(entered,l=>l.id==sel); toast "Lot withdrawn"}

### My Wins
state tab="wins"
state wins=[{"id":1,"name":"Three Light Sussex hens","price":65,"slot":"","paid":false},{"id":2,"name":"Sack of Maris Piper potatoes","price":18,"slot":"11:30","paid":false},{"id":3,"name":"Crate of Bramley apples","price":41,"slot":"","paid":true}]
state slots=["10:30","11:00","11:30","12:00","12:30"]
state pick=0
App
  Sidebar
    Brand "Barrow Bids" icon="gavel"
    NavItem "Live auction" icon="gavel" selected={tab=="auction"} onTap={go live-auction}
    NavItem "Sell a lot" icon="add_a_photo" selected={tab=="sell"} onTap={go sell-a-lot}
    NavItem "My wins" icon="emoji_events" badge="{count(wins,w=>!w.paid)}" selected={tab=="wins"} onTap={go my-wins}
    Spacer grow
    Text.muted "Buyer 214"
  Main
    Nav
      Heading "Won lots" level=1
      Text.muted "Market closes at 13:00"
    Column pad=24 gap=16 scroll
      each w in wins
        Card gap=8
          Row between middle
            Column gap=2
              Text.heading "{w.name}"
              Text.muted "Hammer price {money(w.price,"£",2)}"
            if w.paid
              Badge.success "Paid"
            if !w.paid
              Badge.warning "Payment due"
          Row gap=12 middle
            Icon "schedule"
            if w.slot==""
              Text.muted "No collect slot yet"
              Spacer grow
              Button.outline "Choose slot" onTap={pick=w.id; open slotSheet}
            if w.slot!=""
              Text.strong "Collect at {w.slot}, stall 6"
              Spacer grow
              Button.ghost "Change" onTap={pick=w.id; open slotSheet}
      Footer
        Row between middle
          Column gap=2
            Text.caption "To pay"
            Text "{money(sum(filter(wins,w=>!w.paid),w=>w.price),"£",2)}" size=28 weight=700
          Button "Pay now" size=large disabled={count(wins,w=>!w.paid)==0} onTap={open payDialog}
  Sheet#slotSheet "Collect slot"
    Grid cols=3 gap=8
      each s in slots
        ChoiceChip "{s}" onTap={wins=updateWhere(wins,w=>w.id==pick,w=>with(w,"slot",s)); toast "Slot booked for {s}"; close}
  Dialog#payDialog "Pay for your lots" "Total {money(sum(filter(wins,w=>!w.paid),w=>w.price),"£",2)}"
    Column gap=12
      Text "Pay at the market office or by card."
      Button "Pay by card" onTap={wins=map(wins,w=>with(w,"paid",true)); emit pay; toast "Paid, thank you"; close}
      Button.outline "Pay at the office" onTap={emit payAtOffice; close}
