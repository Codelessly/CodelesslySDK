          ChoiceChip "20%" selected={tipPct==20 && customTip==""} onTap={tipPct=20; customTip=""}

        Slider @tipPct min=0 max=40
      Card gap=12
        Text.strong "Tax"
        Input "Tax on the receipt" type=number @tax
        Text.muted "Tax is shared in proportion to what each person ordered."
      Card gap=8
        Row between
          Text "Subtotal"
          Text "{money(sum(items,i=>i.price),"$",2)}"
        Row between
          Text "Tax"
          Text "{money(tax,"$",2)}"

          Text "Tip ({tipPct}%)"
          Text "{money(sum(items,i=>i.price)*tipPct/100,"$",2)}"

        Row between

          Text.strong "{money(sum(items,i=>i.price)*(1+tipPct/100)+tax,"$",2)}"

### Totals
state people=[{"id":1,"name":"Maya Reyes"},{"id":2,"name":"Jon Park"},{"id":3,"name":"Priya Nair"}]
state items=[{"id":1,"name":"Margherita pizza","price":18.5,"who":[1,2]},{"id":2,"name":"Truffle fries","price":9,"who":[1,2,3]},{"id":3,"name":"Burrata salad","price":13,"who":[3]},{"id":4,"name":"Aperol spritz","price":12,"who":[2]},{"id":5,"name":"Tiramisu","price":8.5,"who":[1,3]}]
state tipPct=18
state tax=5.4
state who=1
state settled=false
state message=""
App
  Sidebar
    Brand "Halves" icon="call_split"
    NavItem "Receipt" icon="receipt_long" onTap={go receipt}
    NavItem "Tip and tax" icon="percent" onTap={go tip-and-tax}
    NavItem "Totals" icon="groups" selected={true}
    Spacer grow
    Text.muted "Dinner at Lupa, Fri"
  Main
    Nav
      Heading "Who owes what"
      Button "Send pay request" icon="send" onTap={open request}
    Column pad=24 gap=16
      if settled
        Badge.success "SETTLED" animate=pop size=large
      Grid cols=3 gap=16
        each p in people
          Card gap=8 center
            Avatar "{p.name}" size=56
            Text.strong "{p.name}"
            Text "{money(sum(items,i=>contains(i.who,p.id) ? i.price/length(i.who) : 0)*(1+(tax+sum(items,i=>i.price)*tipPct/100)/sum(items,i=>i.price)),"$",2)}" size=28 weight=700
            Text.caption "items {money(sum(items,i=>contains(i.who,p.id) ? i.price/length(i.who) : 0),"$",2)} + tax and tip"
            Button.outline "Request" onTap={who=p.id; open request}
      Card gap=8
        Text.strong "Share of the bill"
        Donut values={map(people,p=>sum(items,i=>contains(i.who,p.id) ? i.price/length(i.who) : 0))} labels={map(people,p=>p.name)}
      Row gap=8
        Button.outline "Reset" onTap={settled=false}
        Button "Mark all settled" onTap={settled=true; toast "All settled up"}
  Dialog#request "Pay request" "Edit it before you send"
    Input "Message" multiline @message
    Text.muted "Hi {find(people,p=>p.id==who).name}, your share of dinner at Lupa is {money(sum(items,i=>contains(i.who,who) ? i.price/length(i.who) : 0)*(1+(tax+sum(items,i=>i.price)*tipPct/100)/sum(items,i=>i.price)),"$",2)} including tax and a {tipPct}% tip. Thanks!"
    Row gap=8
      Button.outline "Copy" icon="content_copy" onTap={copy message; toast "Copied"}
      Button "Share" icon="share" onTap={emit sharePayRequest who; close}
