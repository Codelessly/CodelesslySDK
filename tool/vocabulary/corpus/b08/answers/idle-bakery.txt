### Bakery
project dough=0 saved
project total=0 saved
project stars=0 saved
project items=[{"id":"cook","name":"Apprentice Baker","icon":"person","base":15,"cps":0.5,"owned":0},{"id":"oven","name":"Brick Oven","icon":"local_fire_department","base":120,"cps":4,"owned":0},{"id":"mill","name":"Windmill","icon":"air","base":900,"cps":22,"owned":0},{"id":"fact","name":"Bread Factory","icon":"factory","base":8000,"cps":150,"owned":0},{"id":"lab","name":"Yeast Lab","icon":"science","base":65000,"cps":900,"owned":0}] saved
project qty=1
state pops=[]
state clock=0
state crumb=0
state crumbX=0
state perTap=1
state away=true
state awayGain=18400
every 1s onTick={dough=dough+sum(items,i=>i.cps*i.owned)*(1+stars*0.1); total=total+sum(items,i=>i.cps*i.owned)*(1+stars*0.1); clock=clock+1; crumb=crumb>0?crumb-1:(clock%25==0?6:0); pops=take(pops,3)}
every 500 when crumb>0 onTick={crumbX=crumbX+6}
App
  Sidebar
    Brand "Dough Empire" icon="bakery_dining"
    NavItem "Bakery" icon="cookie" selected={true}
    NavItem "Shop" icon="storefront" onTap={go shop}
    NavItem "Ascend" icon="auto_awesome" badge={floor(total/1000000)>0?"new":""} onTap={go ascend}
    Spacer grow
    Text.muted "Lifetime {total>=1000000?fixed(total/1000000,2)+'M':(total>=1000?fixed(total/1000,1)+'K':fixed(total,0))}"
  Main
    Nav
      Heading "Bakery" level=1
      Row gap=8
        Badge.warning "{stars} golden stars"
        Button.outline "Shop" icon="storefront" onTap={go shop}
    Column pad=24 gap=16 center
      Card gap=4 center
        Text.muted "Loaves baked"
        Text "{dough>=1000000?fixed(dough/1000000,2)+'M':(dough>=1000?fixed(dough/1000,1)+'K':fixed(dough,0))}" size=64 weight=700
        Text.muted "{fixed(sum(items,i=>i.cps*i.owned)*(1+stars*0.1),1)} per second, +{perTap*(1+stars*0.1)} per tap"
      if crumb>0
        Row
          Spacer w={crumbX}
          IconButton "stars" tooltip="Golden crumb" animate=pulse onTap={dough=dough+max(100,dough*0.15); total=total+max(100,dough*0.15); crumb=0; crumbX=0; toast "Golden crumb! +15%"}
          Text.caption "{crumb}s"
      Stack center middle
        Box fill=#FFE8B86B radius=140 w=240 h=240 animate=scale onTap={dough=dough+perTap*(1+stars*0.1); total=total+perTap*(1+stars*0.1); pops=append(pops,perTap*(1+stars*0.1))}
          Icon "bakery_dining" size=120
        each p in pops
          Text "+{p}" size=24 weight=700 color=#FFB8860B animate=slide-up duration=600
      Text.muted "Tap the loaf"
      Card gap=8
        Row between
          Text.strong "Golden crumb"
          Text.muted "appears every 25s, fades in 6s"
        Progress {crumb/6}
      Card gap=8
        Text.strong "Your helpers"
        each it in items
          if it.owned>0
            Row between
              Row gap=8
                Icon "{it.icon}"
                Text "{it.name}"
              Text.strong "x{it.owned}"
  Dialog#welcome "Welcome back" "Your bakers kept working while you were away"
    Column gap=12 center
      Icon "schedule" size=48
      Text "You were away 3h 12m"
      Text "+{awayGain} loaves" size=40 weight=700 color=#FFB8860B
      Button "Collect" onTap={dough=dough+awayGain; total=total+awayGain; away=false; close}
every 1s when away onTick={away=false; open welcome}
