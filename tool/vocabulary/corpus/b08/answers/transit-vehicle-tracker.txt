      Heading "Fleet" level=1

        Text.muted "{count(fleet, f=>f.delay>0)} of {length(fleet)} trams running late"
      Card gap=0
        Row gap=12 pad=12 border-bottom=#FFE0E0E0
          Text.strong "Tram" w=70 onTap={sortKey=="id" ? asc=!asc : {sortKey="id"; asc=true}}
          Text.strong "Line" w=60 onTap={sortKey=="line" ? asc=!asc : {sortKey="line"; asc=true}}
          Text.strong "Driver" grow onTap={sortKey=="driver" ? asc=!asc : {sortKey="driver"; asc=true}}
          Text.strong "Crowding" w=90 onTap={sortKey=="load" ? asc=!asc : {sortKey="load"; asc=false}}
          Text.strong "Delay" w=80 onTap={sortKey=="delay" ? asc=!asc : {sortKey="delay"; asc=false}}
          Text.strong "Speed" w=80 onTap={sortKey=="speed" ? asc=!asc : {sortKey="speed"; asc=true}}
        each f in (asc ? sortBy(filter(fleet, x=>(!onlyLate || x.delay>0) && (q=="" || contains(lower(x.id+x.driver), lower(q)))), x=>x[sortKey]) : sortByDesc(filter(fleet, x=>(!onlyLate || x.delay>0) && (q=="" || contains(lower(x.id+x.driver), lower(q)))), x=>x[sortKey]))
          Row gap=12 pad=12 middle border-bottom=#FFF0F0F0 onTap={go live-map}
            Text.strong "{f.id}" w=70
            Badge.neutral "{f.line}" w=60
            Text "{f.driver}" grow
            Row gap=2 w=90
              each i in [1,2,3]
                Icon "person" size=16 color={i<=f.load ? "#FFE53935" : "#FFBDBDBD"}
            Text.danger "+{f.delay} min" w=80 visible={f.delay>0}
            Text.success "On time" w=80 visible={f.delay==0}
            Text.muted "{f.speed} km/h" w=80
      if length(filter(fleet, x=>!onlyLate || x.delay>0))==0
        Text.muted "No trams match."

state from="Dam Square"
state to="Muiderpoort"
state reason="Track work"
state closed=[{"line":"4","from":"Leidseplein","to":"Vondelpark","reason":"Burst water main"}]
state alts=[]
state stopNames=["Central Station","Dam Square","Rembrandtplein","Waterlooplein","Muiderpoort","Science Park","University"]
App
  Sidebar
    Brand "Liveline" icon="tram"
    NavItem "Live map" icon="map" onTap={go live-map}
    NavItem "Fleet" icon="table_rows" onTap={go fleet}
    NavItem "Disruptions" icon="warning" badge="{length(closed)}" selected={true}
    Spacer grow
    Text.muted "Control room, shift B"
  Main
    Nav
      Heading "Disruption editor" level=1
      Button "Close stretch" icon="block" onTap={open confirmClose}
    Row gap=16 pad=24 top
      Card grow gap=12
        Heading "Close a stretch" level=3
        Select "Line" items=["1","4","7","9"] @line
        Row gap=12
          Select "From stop" items={stopNames} @from grow
          Select "To stop" items={stopNames} @to grow
        SegmentedButton items=["Track work","Accident","Event"] @reason
        Button.outline "Show alternatives" onTap={alts=[{"name":"Bus 77 shuttle","detail":"{from} to {to}, every 6 min","extra":"+8 min"},{"name":"Line 9 via Riverside","detail":"Change at Rembrandtplein","extra":"+12 min"},{"name":"Metro M52","detail":"Walk 4 min to Dam station","extra":"+5 min"}]}
        if length(alts)>0
          Text.fieldLabel "Alternatives for riders"
          each a in alts
            ListTile "{a.name}" "{a.detail}" icon="alt_route"
              Badge.warning "{a.extra}"
      Pane w=320 pad=16 gap=12
        Heading "Active closures" level=3
        if length(closed)==0
          Text.muted "No stretches closed."
        each c in closed
          Card gap=8
            Row between
              Text.strong "Line {c.line}"
              Badge.error "{c.reason}"
            Text "{c.from} to {c.to}"
            Button.ghost "Reopen" onTap={closed=removeWhere(closed, x=>x.from==c.from && x.line==c.line); toast "Stretch reopened"}
  Dialog#confirmClose "Close this stretch?" "Line {line}, {from} to {to}"
    Text.muted "Riders are notified in the app and trams are held at the last open stop."
    Row gap=8 end
      Button.outline "Cancel" onTap={close}
      Button.destructive "Close stretch" onTap={from==to ? toast "Pick two different stops" : {closed=append(closed, {"line":line,"from":from,"to":to,"reason":reason}); emit closeStretch {"line":line,"from":from,"to":to}; toast "Stretch closed"; close}}
