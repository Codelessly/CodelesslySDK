            Row gap=8
              Text.mono "{find(notes,n=>n.id==selected).pitch}"
              Text.muted "start {find(notes,n=>n.id==selected).start + 1}, length {find(notes,n=>n.id==selected).len}"
            Row gap=8
              Button.outline "Shorter" onTap={notes=updateWhere(notes,n=>n.id==selected,n=>with(n,"len",max(1,n.len-1)))}
              Button.outline "Longer" onTap={notes=updateWhere(notes,n=>n.id==selected,n=>with(n,"len",min(16-n.start,n.len+1)))}
              Button.destructive "Delete" onTap={notes=removeWhere(notes,n=>n.id==selected); selected=0}
        Card grow gap=8
          Text.strong "Notes"
          Text.heading "{length(notes)}"
          Text.muted "{sum(notes,n=>n.len)} steps filled of 16"
