### Predictions
state angle=40
state friction=0.1
state guess=""
state revealed=false
state outcome=""
state history=[]
App
  Sidebar
    Brand "Lab Bubble" icon="science"
    NavItem "Ramp" icon="trending_down" onTap={go lab}
    NavItem "Challenge" icon="flag" onTap={go challenge}
    NavItem "Predictions" icon="psychology" selected={true}
    NavItem "Notebook" icon="menu_book" onTap={go notebook}
  Main
    Nav
      Heading "Predict first" level=1
      Text.muted "{count(history,h=>h.correct)} of {length(history)} right"
    Column pad=24 gap=16 maxw=720
      Card gap=12
        Text.heading3 "A ball rolls down a {angle} degree ramp with friction {friction}."
        Text.muted "Before you run it, what will happen when you double the mass?"
        Column gap=8
          ChoiceChip "It reaches the bottom faster" selected={guess=="faster"} onTap={guess="faster"}
          ChoiceChip "It reaches the bottom at the same time" selected={guess=="same"} onTap={guess="same"}
          ChoiceChip "It reaches the bottom slower" selected={guess=="slower"} onTap={guess="slower"}
        Button "Run it and compare" disabled={guess==""} onTap={outcome="same"; revealed=true; history=append(history,{"guess":guess,"actual":"same","correct":guess=="same"})}
      if revealed
        Card gap=12 animate=slide-up
          Row between
            Text.strong "Your prediction: {guess}"
            Text.strong "Actual: {outcome}"
          Badge.success "Correct" visible={guess==outcome}
          Badge.warning "Not quite" visible={guess!=outcome}
          Text.muted "Mass cancels out: both gravity's pull and the friction grow with mass, so the acceleration stays the same."
          Ramp angle={angle} friction={friction} mass=4 h=200 autoplay
          Row gap=8
            Button.outline "Try another" onTap={guess=""; revealed=false}
            Button "Open in the lab" onTap={go lab}
      if length(history)>0
        Card gap=8
          Text.strong "Your guesses"
          each h in history
            ListTile "{h.guess}" subtitle="Actual: {h.actual}" icon="{h.correct ? 'check' : 'close'}"
