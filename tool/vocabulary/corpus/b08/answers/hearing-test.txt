### Tone test
state ear="left"
state freqs=[250,500,1000,2000,4000,8000]
state fi=0
state level=50
state playing=true
state heard=[]
state secs=0
state misses=0
every 1s when playing onTick={secs=secs+1; level=level-5}
every 1s when playing onTick={level<=0 ? {heard=append(heard,{"ear":ear,"freq":freqs[fi],"db":0,"found":false}); level=50; fi=fi+1} : {}}
App
  Main
    Nav
      Brand "Earful" icon="hearing"
      Row gap=8 middle
        Badge.info "{upper(ear)} ear"
        Text.muted "{fi+1} of {length(freqs)} pitches"
        IconButton "close" tooltip="End test" onTap={playing=false; open endTest}
    Column pad=24 gap=20 center maxw=560
      Progress {(fi + (ear=="right" ? length(freqs) : 0)) / (length(freqs)*2)}
      Text.heading "Press the button each time you hear the tone"
      Text.muted "It gets quieter and quieter. Press as soon as you hear it, even if it is very faint."
      Stack center middle h=240
        PulseRing active={playing} color=#FF3B82F6
        Column center gap=4
          Text "{freqs[fi]} Hz" size=32 weight=700
          Text.caption "{level} dB"
      Button "I hear it" size=large w=fill h=96 onTap={heard=append(heard,{"ear":ear,"freq":freqs[fi],"db":level,"found":true}); level=50; (fi<length(freqs)-1) ? {fi=fi+1} : {fi=0; ear=(ear=="left" ? "right" : "left")}}
      Row gap=12 center
        Button.outline playing ? "Pause" : "Resume" icon="pause" onTap={playing=!playing}
        Button.ghost "Replay tone" icon="replay" onTap={play tone ear=ear freq=freqs[fi] level=level}
      if length(heard) >= length(freqs)*2
        Button "See results" w=fill onTap={playing=false; go word-in-noise}
      Text.caption "{count(heard, h => h.found)} tones heard so far"
  Dialog#endTest "End the test?" "Your answers so far will be thrown away."
    Row end gap=8
      Button.outline "Keep going" onTap={playing=true; close}
      Button.destructive "End test" onTap={heard=[]; fi=0; level=50; close; go headphone-check}
