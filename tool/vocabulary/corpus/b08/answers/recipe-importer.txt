### Cook Mode
state step=0
state steps=["Bring a large pot of salted water to a boil.","Melt the butter in a wide skillet over medium heat until it foams and smells nutty, about 4 minutes.","Add the sage and fry until crisp, about 1 minute.","Boil the gnocchi until they float, then lift them into the skillet.","Toss with Parmesan and serve at once."]
state timer=0
state running=false
every 1s when running onTick={timer=max(0,timer-1); timer==0 ? running=false : toast ""}

  Nav
    IconButton "close" tooltip="Leave cook mode" onTap={go recipe}
    Text.strong "Brown Butter Gnocchi"
    Text.muted "Step {step+1} of {length(steps)}"
  Section center
    Column gap=24 center
      Progress "{(step+1)/length(steps)}"
      Pages @step h=260
        each s in steps
          Text "{s}" size=40 weight=700 center
      Dots @step count=5
      if timer>0
        Text "{duration(timer)}" size=64 weight=700
      Row gap=8 center
        Button.outline "1 min" icon="timer" onTap={timer=60; running=true}
        Button.outline "4 min" icon="timer" onTap={timer=240; running=true}
        Button.ghost "Stop" onTap={running=false; timer=0}
      Row gap=16 center
        Button.outline "Back" size=large icon="arrow_back" onTap={step=max(0,step-1)}

### Collection
state tag="all"
state q=""
state recipes=[{"id":1,"title":"Brown Butter Gnocchi","time":"35 min","tags":["pasta","weeknight"]},{"id":2,"title":"Miso Roasted Carrots","time":"40 min","tags":["vegetarian","side"]},{"id":3,"title":"Lemon Ricotta Pancakes","time":"25 min","tags":["breakfast"]},{"id":4,"title":"Sheet Pan Chicken Thighs","time":"50 min","tags":["weeknight","dinner"]},{"id":5,"title":"Cacio e Pepe","time":"20 min","tags":["pasta","weeknight"]}]
state tags=["all","pasta","weeknight","vegetarian","breakfast","dinner","side"]
state sel=0
state grocery=[{"name":"potato gnocchi","amount":"1 lb","got":false},{"name":"unsalted butter","amount":"4 tbsp","got":false},{"name":"sage leaves","amount":"1/4 cup","got":true}]
state newTag=""
App
  Sidebar
    Brand "Clip a Dish" icon="restaurant_menu"
    NavItem "Import" icon="link" onTap={go import}
    NavItem "Collection" icon="collections_bookmark" badge="{length(recipes)}" selected={true}
    Spacer grow
    Text.muted "Maya Reyes"
  Main
    Nav
      Heading "Collection" level=1
      Row gap=8
        Input "Search recipes" @q w=220
        Button "Import" icon="add" onTap={go import}
    Column pad=24 gap=16
      Row gap=8 scroll
        each t in tags
          FilterChip "{t}" selected={tag==t} onTap={tag=t}
      Grid cols=3 gap=16
        each r in filter(recipes, x=>(tag=="all" || contains(x.tags,tag)) && contains(lower(x.title),lower(q)))
          Card gap=8 menu=rowMenu onTap={go recipe}
            Image "https://images.example.com/recipe-{r.id}.jpg" h=120 radius=8
            Row between
              Text.strong "{r.title}"
              IconButton "more_vert" tooltip="More" onTap={sel=r.id; open rowMenu}
            Row gap=4
              Badge.neutral "{r.time}"
              each t in r.tags
                Chip "{t}"
    Footer
      Row between middle
        Text.strong "Grocery list ({count(grocery,g=>!g.got)} to get)"
        Button.outline "Open" icon="shopping_basket" onTap={open groceryList}
  Menu#rowMenu
    MenuItem "Open" icon="open_in_new" onTap={go recipe}
    MenuItem "Add tag" icon="label" onTap={open tagDialog}
    MenuItem "Add to grocery list" icon="add_shopping_cart" onTap={emit groceryAdd sel; toast "Added to grocery list"}
    MenuItem "Delete" icon="delete" onTap={recipes=removeWhere(recipes,x=>x.id==sel); toast "Recipe removed"}
  Dialog#tagDialog "Add a tag" "Tags help you find it later"
    Input "Tag" @newTag
    Button "Add" onTap={newTag=="" ? toast "Type a tag" : {recipes=updateWhere(recipes,x=>x.id==sel,x=>with(x,"tags",append(x.tags,newTag))); tags=unique(append(tags,newTag)); newTag=""; close}}
  Drawer#groceryList "Grocery list" right
    each g in grocery
      Row gap=8 middle
        Checkbox @g.got
        Text "{g.amount} {g.name}" grow strike={g.got}
    Button.outline "Clear checked" onTap={grocery=removeWhere(grocery,g=>g.got)}
