### Live
state tab="live"
state running=true
state secs=0
state lapStart=0
state laps=[]
state banner="Warm-up: keep it easy, 6:10 per km"
state pace="6:08"
state hr=142
every 1s when running onTick={secs=secs+1; banner=secs==30?"Rep 1 in 10 seconds: get ready":banner}
App
  Main
    Nav
      Row gap=8
        IconButton "close" tooltip="End workout" onTap={open endWorkout}
        Heading "4 x 4 min intervals" level=1
      Badge.success "GPS locked"
    Column pad=24 gap=16 center
      Card gap=4
        Row gap=12
          Icon "volume_up"
          Column grow gap=2
            Text.caption "Audio cue"
            Text.strong "{banner}"
          IconButton "close" tooltip="Dismiss" onTap={banner=""}
      Text "{duration(secs)}" size=72 weight=700 center
      Row gap=16
        Card grow gap=4
          Text.caption "Pace"
          Text.heading "{pace}"
        Card grow gap=4
          Text.caption "Heart rate"
          Text.heading "{hr}"
        Card grow gap=4
          Text.caption "Distance"
          Text.heading "{fixed(secs/370,2)} km"
      Progress {clamp(secs/1500,0,1)}
      Row gap=12 center
        Button.outline "{running ? 'Pause' : 'Resume'}" icon={running ? "pause" : "play_arrow"} onTap={running=!running}
        Button "Lap" icon="flag" size=large onTap={laps=append(laps,{"n":length(laps)+1,"t":secs-lapStart}); lapStart=secs; toast "Lap {length(laps)+1}"}
      if length(laps)>0
        Card gap=8
          Text.strong "Laps"
          each l in reverse(laps)
            Row between
              Text "Lap {l.n}"
              Text.strong "{duration(l.t)}"
  Dialog#endWorkout "End this workout?" "Your run so far will be saved."
    Button "Save and finish" onTap={emit finishRun secs; close; go load}
    Button.destructive "Discard" onTap={emit discardRun; close; go plan}
    Button.ghost "Keep running" onTap={close}
