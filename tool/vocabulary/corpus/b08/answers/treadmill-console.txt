### Workout
state elapsed=0
state total=1800
state running=true
state segment=0
state segLen=120
state speed=6.5
state incline=3
state bpm=128
state profile=[2,3,5,6,8,6,4,3,5,7,9,6,3,2]
state phase="run"
every 1s when running onTick={elapsed=elapsed+1; segment=min(length(profile)-1,floor(elapsed/segLen)); incline=profile[segment]; phase=elapsed>=total-300 ? "cooldown" : "run"; elapsed>=total ? (running=false; go summary) : ""}
App
  Sidebar
    Brand "Stride" icon="directions_run"
    NavItem "Console" icon="speed" onTap={go console}
    NavItem "Programs" icon="terrain" onTap={go programs}
    NavItem "Workout" icon="timer" selected={true}
    NavItem "Summary" icon="emoji_events" onTap={go summary}
  Main
    Nav
      Row gap=8 middle
        Heading "Rolling hills"
        Badge.info "Cool-down" hidden={phase!="cooldown"}
        Badge.success "Running" hidden={phase=="cooldown" || !running}
        Badge.warning "Paused" hidden={running}
      Button.destructive "Emergency stop" icon="stop_circle" onTap={open stop}
    Column pad=24 gap=16
      if phase=="cooldown"
        Card gap=8 fill=#FFE8F1FB
          Text.strong "Cool-down"
          Text.muted "Speed eases to 3.0 mph. Keep walking until the belt stops."
      Grid cols=4 gap=16
        Card gap=4
          Text.caption "SPEED"
          Text "{fixed(speed,1)}" size=56 weight=700
          Text.muted "mph"
        Card gap=4
          Text.caption "INCLINE"
          Text "{incline}" size=56 weight=700
          Text.muted "%"
        Card gap=4
          Text.caption "HEART RATE"
          Text "{bpm}" size=56 weight=700
          Text.muted "bpm"
        Card gap=4
          Text.caption "NEXT CHANGE IN"
          Text "{duration(segLen-(elapsed%segLen))}" size=56 weight=700
          Text.muted "to {profile[min(length(profile)-1,segment+1)]}% incline"
      Card gap=12
        Row between
          Text.strong "Course profile"
          Text.muted "{duration(elapsed)} of {duration(total)}"
        ProfileTrack values={profile} position={elapsed/total} h=200
        Progress {elapsed/total}
      Row gap=8
        IconButton "remove" tooltip="Slower" onTap={speed=max(0.5,round(speed-0.1,1))}
        IconButton "add" tooltip="Faster" onTap={speed=min(12,round(speed+0.1,1))}
        Spacer grow
        Button.outline "Skip segment" icon="skip_next" onTap={elapsed=min(total,(segment+1)*segLen)}
        Button.outline "{running ? 'Pause' : 'Resume'}" icon={running ? "pause" : "play_arrow"} onTap={running=!running}
        Button.secondary "Start cool-down" onTap={elapsed=total-300}
  Dialog#stop "Stop the belt?" "The belt slows to a stop over three seconds."
    Row gap=8
      Button.outline grow "Keep running" onTap={close}
      Button.destructive grow "Stop now" onTap={running=false; close; emit emergencyStop; go summary}
