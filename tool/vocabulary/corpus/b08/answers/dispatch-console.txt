      Heading "Open incidents" level=1
      Row gap=8
        Input "Search incidents" @q w=220
        Button "New incident" icon="add" onTap={open newIncident}

      Banner "PRIORITY 1: {newP1.type} at {newP1.address}" tone=alert icon="notifications_active" sound="alert-p1" animate=pulse

              Row gap=8
                Badge.error "P{inc.pri}" visible={inc.pri==1}
                Badge.warning "P{inc.pri}" visible={inc.pri==2}
                Badge.info "P{inc.pri}" visible={inc.pri==3}
                Text.strong "{inc.type}"
              Text.mono "{duration(inc.age+now)}"
            Text.muted "{inc.address}"
            Row gap=8
              Chip "{inc.unit}" visible={inc.unit!=""}
              Text.caption "Unassigned" visible={inc.unit==""}
      Pane w=420 pad=16 gap=12
        each inc in filter(incidents,i=>i.id==sel)
          Heading "{inc.id}" level=3
          Input "Address" @inc.address
          Select "Type" items=["Cardiac arrest","Structure fire","Vehicle collision","Assault","Noise complaint"] @inc.type
          SegmentedButton @inc.pri items=["P1","P2","P3"]
          Input "Call notes" "Typed during the call" multiline @inc.notes
          Row gap=8
            Button "Save" onTap={emit saveIncident inc.id; toast "Saved"}
            Button.outline "Close incident" onTap={incidents=removeWhere(incidents,i=>i.id==inc.id); toast "Closed"}
        Divider
        Text.fieldLabel "Units available (drag onto an incident)"
        Row gap=8 scroll
          each u in filter(units,x=>x.state=="free")
            Chip "{u.id}" icon="local_shipping" drag=unit(u.id)
  Dialog#newIncident "New incident" "Log a call"
    Input "Address" @draftAddr
    Select "Type" items=["Cardiac arrest","Structure fire","Vehicle collision","Assault","Noise complaint"] @draftType
    Input "Notes" multiline @draftNotes
    Button "Create" onTap={incidents=append(incidents,{"id":"INC-4830","pri":2,"type":draftType,"address":draftAddr,"age":0,"unit":"","notes":draftNotes}); close; toast "Incident created"}
