### Shopping List
state items=[{"id":1,"name":"Whole milk","qty":1,"low":true},{"id":2,"name":"Eggs","qty":6,"low":true},{"id":3,"name":"Butter","qty":1,"low":false},{"id":4,"name":"Spinach","qty":1,"low":true}]
state listening=false
state heard=""
state draft=""
App
  Sidebar
    Brand "Icebox" icon="kitchen"
    NavItem "Family board" icon="dashboard" onTap={go family-board}
    NavItem "Shopping" icon="shopping_cart" badge="{length(items)}" selected={true}
    NavItem "Inside the fridge" icon="videocam" onTap={go inside-the-fridge}
    NavItem "Recipe" icon="restaurant_menu" onTap={go recipe}
  Main
    Nav
      Heading "Shopping list"
      Row gap=8
        Button.outline "Clear ticked" icon="done_all" onTap={items=removeWhere(items,i=>i.done)}
        Button "Send to phone" icon="send" onTap={emit shareList items; toast "Sent to Mum's phone"}
    Column pad=24 gap=16
      Card gap=12
        Row gap=12 middle
          VoiceInput @heard listening={listening} onStart={listening=true} onResult={items=append(items,{"id":length(items)+10,"name":result.text,"qty":1,"low":false}); heard=""; listening=false; toast "Added {result.text}"} onError={listening=false; toast "Did not catch that"}
          Column grow gap=2
            Text.strong "{listening ? 'Listening...' : 'Tap the mic and say an item'}"
            if heard!=""
              Text.muted "Heard: {heard}"
          Input "Or type" @draft w=220
          IconButton "add" tooltip="Add item" onTap={draft!="" ? {items=append(items,{"id":length(items)+10,"name":draft,"qty":1,"low":false}); draft=""} : toast "Type an item first"}
      Row gap=24 top
        Card grow gap=8
          Text.strong "To buy ({length(items)})"
          each it in items
            ListTile "{it.name}" subtitle="{it.low ? 'Nearly out' : 'Added by voice'}"
              Checkbox @it.done
              Row gap=4
                IconButton "remove" tooltip="Less" onTap={items=updateWhere(items,x=>x.id==it.id,x=>with(x,"qty",max(1,x.qty-1)))}
                Text.strong "{it.qty}"
                IconButton "add" tooltip="More" onTap={items=updateWhere(items,x=>x.id==it.id,x=>with(x,"qty",x.qty+1))}
                IconButton "close" tooltip="Remove" onTap={items=removeWhere(items,x=>x.id==it.id)}
        Column w=300 gap=12
          Card gap=8
            Text.strong "Nearly out"
            each it in filter(items,x=>x.low)
              Row between middle
                Text "{it.name}"
                Badge.warning "low"
            Progress 0.2
            Text.caption "Milk is about 20% full by weight"

### Recipe Step
state step=0
state steps=[{"title":"Season the chicken","text":"Rub 4 chicken thighs with 2 tsp paprika, salt and a little oil.","timer":0},{"title":"Sear the skin","text":"Place skin side down in a cold pan. Heat to medium and cook for 8 minutes.","timer":480},{"title":"Flip and roast","text":"Turn the thighs, add the lemon halves and roast at 200 degrees for 20 minutes.","timer":1200},{"title":"Rest and serve","text":"Rest for 5 minutes, then spoon the pan juices over the top.","timer":300}]
state left=0
state running=false
every 1s when running onTick={left=max(0,left-1); left==0 ? {running=false; toast "Timer done"} : }
App
  Sidebar
    Brand "Icebox" icon="kitchen"
    NavItem "Family board" icon="dashboard" onTap={go family-board}
    NavItem "Shopping" icon="shopping_cart" onTap={go shopping}
    NavItem "Inside the fridge" icon="videocam" onTap={go inside-the-fridge}
    NavItem "Recipe" icon="restaurant_menu" selected={true}
  Main
    Nav
      Heading "Lemon paprika chicken"
      Text.muted "Step {step+1} of {length(steps)}"
    Column pad=24 gap=24 center
      Progress {(step+1)/length(steps)}
      Text.pageTitle "{steps[step].title}" center size=48
      Text "{steps[step].text}" center size=32
      if steps[step].timer>0
        Row gap=16 center middle
          Text "{duration(left)}" size=64 weight=700
          Button "Start timer" icon="timer" onTap={left=steps[step].timer; running=true}
          Button.outline "Pause" onTap={running=false}
      Row gap=16 center
        Button.outline "Back" size=large icon="arrow_back" onTap={step=max(0,step-1); running=false; left=0}
        Button "Next step" size=large icon="arrow_forward" onTap={step<length(steps)-1 ? {step=step+1; running=false; left=0} : toast "All done, enjoy"}
