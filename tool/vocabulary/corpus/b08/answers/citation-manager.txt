state dupes=[{"id":"d1","keep":{"id":"r1","title":"Attention Is All You Need","author":"Vaswani, A.","year":2017,"tags":["nlp","transformers"],"pdf":true,"notes":1},"other":{"id":"r5","title":"Attention is all you need","author":"Vaswani, A.","year":2017,"tags":["nlp"],"pdf":false,"notes":0},"reason":"Same DOI 10.5555/3295222.3295349"}]
state picked="d1"
state keepTitle="a"
App
  Sidebar
    Brand "Footnote" icon="menu_book"
    NavItem "Library" icon="library_books" onTap={go library}
    NavItem "Bibliography" icon="format_list_numbered" onTap={go bibliography}
    NavItem "Duplicates" icon="content_copy" badge="{length(dupes)}" selected={true}

    Text.muted "Mira Okafor"
  Main
    Nav
      Heading "Duplicates" level=1
      Text.muted "{length(dupes)} possible duplicate sets"
    Column pad=24 gap=16
      each d in dupes
        Card gap=12
          Row between
            Badge.warning "{d.reason}"
            Text.caption "Merging keeps both sets of tags, the pdf and all notes"
          Grid cols=2 gap=16
            Card gap=8
              Radio "Keep this title" value="a" bind=keepTitle
              Text.strong "{d.keep.title}"
              Text "{d.keep.author}, {d.keep.year}"
              Row gap=4
                each t in d.keep.tags
                  Badge.neutral "{t}"
              Text.caption "PDF attached, {d.keep.notes} note"
            Card gap=8
              Radio "Keep this title" value="b" bind=keepTitle
              Text.strong "{d.other.title}"
              Text "{d.other.author}, {d.other.year}"
              Row gap=4
                each t in d.other.tags
                  Badge.neutral "{t}"
              Text.caption "No PDF, {d.other.notes} notes"
          Row gap=8 end
            Button.ghost "Not a duplicate" onTap={dupes=removeWhere(dupes, x => x.id==d.id); toast "Marked as different"}
            Button "Merge" icon="merge" onTap={open confirmMerge}
      if length(dupes)==0
        Text.muted "No duplicates. Your library is clean."
  Dialog#confirmMerge "Merge these references?" "The merged entry keeps tags {union of both}, the pdf and all notes."
    Row gap=8 end
      Button.ghost "Cancel" onTap={close}
      Button "Merge" onTap={dupes=[]; emit merge picked; toast "Merged into one reference"; close}
