  Main

        Input "Notes" multiline @notes
      Card gap=12
        Text.sectionTitle "Species count"
        each s in species
          Row between middle
            Text "{s.name}"
            Row gap=8 middle
              IconButton "remove" tooltip="Fewer" onTap={species=updateWhere(species,x=>x.name==s.name,x=>with(x,"n",max(0,x.n-1)))}
              Text.strong "{s.n}" w=32 center
              IconButton "add" tooltip="More" onTap={species=updateWhere(species,x=>x.name==s.name,x=>with(x,"n",x.n+1))}
        Row gap=8
          Text.muted "Total {sum(species,x=>x.n)}"
        Button.ghost "Add species" icon="add" onTap={open addSpecies}
      Card gap=12
        Row between
          Text.sectionTitle "Photos"
          Button.outline "Take photo" icon="photo_camera" onTap={emit camera}
        Row scroll gap=12
          each p in photos
            Stack
              Image "{p.url}" w=160 h=120 radius=8 onTap={at=index; open shots}
              CompassStamp "{p.dir} {p.deg}°" bearing={p.deg} bottom left
      Row gap=8

  Dialog#addSpecies "Add species"

    Button "Add" onTap={species=append(species,{"name":newSpecies,"n":0}); newSpecies=""; close}

### Sync queue
state queue=[{"id":1,"label":"Plot 14 form","size":"4 KB","state":"failed"},{"id":2,"label":"Plot 14 photo N","size":"2.1 MB","state":"pending"},{"id":3,"label":"Plot 14 photo E","size":"1.9 MB","state":"pending"},{"id":4,"label":"Plot 13 form","size":"3 KB","state":"pending"}]
state online=true
App
  Main
    Nav
      Row gap=8
        IconButton "arrow_back" tooltip="Back" onTap={go map}
        Heading "Sync queue" level=1
        Badge.info "{length(queue)}"
      Row gap=8
        Badge.success hidden={!online} "Online"
        Badge.error hidden={online} "Offline"
        Button "Retry all" icon="refresh" onTap={queue=map(queue,q=>with(q,"state","pending")); emit syncAll; toast "Retrying"}
    Column pad=24 gap=16 maxw=720
      if isEmpty(queue)
        Card gap=8
          Icon "cloud_done"
          Text.strong "Everything is uploaded"
      each q in queue
        ListTile "{q.label}" subtitle="{q.size}" icon="description"
          Row gap=8
            if q.state=="failed"
              Badge.error "Failed"
              Button.outline "Retry" onTap={queue=updateWhere(queue,x=>x.id==q.id,x=>with(x,"state","pending")); emit retry q.id}
            if q.state=="pending"
              Badge.neutral "Waiting"
            IconButton "close" tooltip="Remove" onTap={queue=removeWhere(queue,x=>x.id==q.id)}
