state pieceCount=4
every 700 when running onTick={step=step+1; step>=pieceCount ? running=false : toast ""}
App
  Main
    Nav
      Row gap=8 middle
        IconButton "arrow_back" tooltip="Back to the map" onTap={go level_map}
        Heading "Level {level}: Square Dance" level=3
      Row gap=8
        Badge.info "{pieceCount} / {maxPieces} pieces"
        Button.outline "Hint" icon="lightbulb" onTap={open hint}
    Row gap=16 pad=16 top
      Column grow gap=12
        Card gap=8
          Row between middle
            Text.strong "Goal"
            Row gap=4
              Icon "star"
              Text.muted "Reach the star"
          RobotStage grid=[6,6] robot={"x":0,"y":5,"facing":"up"} goal={"x":3,"y":2} walls=[{"x":1,"y":3},{"x":2,"y":3}] program={script} step={step} running={running} onReachGoal={running=false; open win} onCrash={running=false; toast "Bonk! Try another way."}
        Row gap=8 center
          Button "Play" icon="play_arrow" disabled={running || isEmpty(script)} onTap={step=0; running=true}
          Button.outline "Step" icon="skip_next" onTap={step=step+1; running=false}
          Button.outline "Reset" icon="replay" onTap={running=false; step=0}
          Button.ghost "Clear" icon="delete" onTap={script=[]; pieceCount=0; step=0}
      Column grow gap=12
        Card gap=8
          Text.strong "Your pieces"
          Text.caption "Drag a piece into the script. Pieces snap under each other."
          Row scroll gap=8
            each p in palette
              PuzzlePiece "{p.label}" icon="{p.icon}" color="{p.color}" drag=piece(p.cmd)
        Card gap=8 grow
          Row between middle
            Text.strong "My script"
            Text.muted "{pieceCount} / {maxPieces}"
          Column gap=4 drop=piece onDrop={pieceCount >= maxPieces ? toast "That is too many pieces! Try a loop." : script=append(script,{"id":nextId,"cmd":result.data,"n":2,"body":[]}); nextId=nextId+1; pieceCount=pieceCount+1}
            PuzzlePiece "Start" icon="flag" color="#FF607D8B" fixed
            each s in script
              PuzzlePiece "{s.cmd}" cmd={s.cmd} active={running && indexOf(script,s) == step} drag=piece(s.id) wraps={s.body} drop=piece onDrop={script=updateWhere(script,x=>x.id==s.id,x=>with(x,"body",append(x.body,{"id":nextId,"cmd":result.data})))} onTap={script=removeWhere(script,x=>x.id==s.id); pieceCount=max(0,pieceCount-1)}
                if s.cmd == "repeat"
                  Stepper @s.n min=2 max=9 label="times"
            Text.muted "Drop here to add to the end" hidden={length(script) > 0}
  Dialog#hint "Pip's hint"
    Row gap=12 middle
      Avatar "Pip the Owl" size=56
      Text "Look at the path. You walk forward twice, then turn, then forward twice again. Can one Repeat piece do the same thing with fewer pieces?"

      Button "Got it" onTap={close}
  Dialog#win "You did it!"
    Column center gap=12
