### Preflight
state warnings=[{"id":"w1","level":"error","page":2,"text":"Text frame overflows: Intro column","fixed":false},{"id":"w2","level":"warning","page":3,"text":"Image at 142 ppi, below 300 ppi","fixed":false},{"id":"w3","level":"warning","page":5,"text":"Missing font: Canela Light","fixed":false}]
state filter="all"
App
  Sidebar
    Brand "Folio" icon="menu_book"
    NavItem "Spread" icon="auto_stories" onTap={go spread}
    NavItem "Styles" icon="text_format" onTap={go styles}
    NavItem "Masters" icon="layers" onTap={go masters}
    NavItem "Preflight" icon="fact_check" badge="{count(warnings,w=>!w.fixed)}" selected={true}
    Spacer grow
  Main
    Nav
      Heading "Preflight" level=3
      Row gap=8
        Button.outline "Run again" icon="refresh" onTap={emit preflight; toast "Preflight complete"}
        Button "Export proof" icon="picture_as_pdf" onTap={count(warnings,w=>w.level=="error" && !w.fixed)>0 ? toast "Fix the errors first" : open proof}
    Column pad=24 gap=16
      Row gap=8
        ChoiceChip "All" selected={filter=="all"} onTap={filter="all"}
        ChoiceChip "Errors" selected={filter=="error"} onTap={filter="error"}
        ChoiceChip "Warnings" selected={filter=="warning"} onTap={filter="warning"}
      Card gap=8
        each w in warnings
          if !w.fixed && (filter=="all" || filter==w.level)
            ListTile "{w.text}" subtitle="Page {w.page}" icon={w.level=="error" ? "error" : "warning"}
              Badge.error "Error" visible={w.level=="error"}
              Button.outline "Go to page" onTap={emit goPage w.page; go spread}
              Button.ghost "Mark fixed" onTap={warnings=updateWhere(warnings,x=>x.id==w.id,x=>with(x,"fixed",true))}
        if count(warnings,w=>!w.fixed)==0
          Row gap=8 center
            Icon "check_circle" color=#FF2E7D32
            Text.success "No problems found"
  Dialog#proof "Export proof" "A low-resolution PDF for review"
    Toggle "Include crop marks" bind=marks
    Row end gap=8
      Button.ghost "Cancel" onTap={close}
      Button "Export" onTap={emit exportProof; toast "Proof exported"; close}
