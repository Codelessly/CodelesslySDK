### Glossary
state glossary=[{"term":"Bridgewater","note":"Client name","langs":["pt","ja","es"]},{"term":"Aiko","note":"Person","langs":["pt","ja","es"]},{"term":"Sagrada","note":"Project codename","langs":["pt","es"]}]
state newTerm=""
state newNote=""
state q=""
state sel=null
App
  Sidebar
    Brand "Bridge" icon="translate"
    NavItem "Chats" icon="chat" onTap={go chat-thread}
    NavItem "Glossary" icon="menu_book" selected={true}
    NavItem "Settings" icon="settings" onTap={go settings}
    Spacer grow
    Text.muted "Sam Patel"
  Main
    Nav
      Heading "Glossary" level=1
      Row gap=8
        Input "Search terms" @q w=220
        Button "Add term" icon="add" onTap={open addTerm}
    Column pad=24 gap=16
      Text.muted "Names and words on this list are never translated. Everyone sees them exactly as written."
      Card gap=0
        Row pad=12 border-bottom=#FFE0E0E0
          Text.strong "Term" grow
          Text.strong "Note" w=160 hide=phone
          Text.strong "Kept in" w=120
          Spacer w=40
        each g in filter(glossary,x=>contains(lower(x.term),lower(q)))
          Row pad=12 middle border-bottom=#FFF0F0F0 menu=rowMenu onLongPress={sel=g.term; open rowMenu}
            Text.strong "{g.term}" grow
            Text.muted "{g.note}" w=160 hide=phone
            Row gap=4 w=120
              each l in g.langs
                LanguageFlag "{l}"
            IconButton "more_vert" tooltip="Options" onTap={sel=g.term; open rowMenu}
        if isEmpty(filter(glossary,x=>contains(lower(x.term),lower(q))))
          Column center pad=32
            Text.muted "No terms match \"{q}\""
  Dialog#addTerm "Add a term" "It will stay untranslated in every language"
    Input "Term" @newTerm
    Input "Note" "Client, person, product" @newNote
    Row end gap=8
      Button.ghost "Cancel" onTap={close}
      Button "Add" onTap={trim(newTerm)!="" ? {glossary=append(glossary,{"term":trim(newTerm),"note":newNote,"langs":["pt","ja","es"]}); newTerm=""; newNote=""; toast "Added to glossary"; close} : toast "Enter a term"}
  Menu#rowMenu end
    MenuItem "Copy term" icon="content_copy" onTap={copy sel; toast "Copied"}
    MenuItem "Remove" icon="delete" onTap={glossary=removeWhere(glossary,x=>x.term==sel); toast "Removed"}
