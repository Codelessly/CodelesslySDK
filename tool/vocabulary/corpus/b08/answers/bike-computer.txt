### Ride
state screen="ride"
state speed=31.4
state running=true
state autoPause=true
state elapsed=3725
state distance=48.2
state cadence=88
state power=214
state hr=142
state laps=[{"n":1,"time":1204,"km":14.8},{"n":2,"time":1311,"km":15.9}]
state slots=["time","distance","cadence","power","hr"]
every 1s when running onTick={elapsed=elapsed+1; distance=round(distance+speed/3600,2)}
every 1s when autoPause && speed<2 onTick={running=false; toast "Auto paused"}
App
  Main
    Nav
      Brand "Cadence" icon="pedal_bike"
      Row gap=8
        Badge.success "GPS" hide=phone
        Badge.info "{hr} bpm"
        IconButton "tune" tooltip="Set up tiles" onTap={go setup}
        IconButton "route" tooltip="Route" onTap={go route}
    Column pad=16 gap=16 center
      Gauge value={speed} min=0 max=60 limit=45 unit="km/h"
      if !running
        Badge.warning "Paused" animate=pulse
      Grid cols=3 gap=12
        each s in slots
          Card gap=4
            Text.caption "{upper(s)}"
            Text.strong "{s=="time" ? duration(elapsed) : s=="distance" ? fixed(distance,1) + " km" : s=="cadence" ? cadence + " rpm" : s=="power" ? power + " W" : hr + " bpm"}" size=28
      Row gap=12
        Button "Lap" size=large grow onTap={laps=append(laps,{"n":length(laps)+1,"time":elapsed,"km":distance}); toast "Lap {length(laps)+1} marked"}
        Button.outline "{running ? "Pause" : "Resume"}" size=large grow onTap={running=!running}
        Button.destructive "Finish" size=large onTap={open finish}
      Toggle "Auto pause below 2 km/h" @autoPause
      Card gap=8
        Text.panelTitle "Laps"
        each l in laps
          ListTile "Lap {l.n}" subtitle="{fixed(l.km,1)} km" trailing="{duration(l.time)}"
  Dialog#finish "Finish this ride?" "{fixed(distance,1)} km in {duration(elapsed)}"
    Row gap=8
      Button.outline "Keep riding" onTap={close}
      Button "Finish and save" onTap={running=false; close; go summary}

### Setup
state slots=["time","distance","cadence","power","hr"]
state pool=["time","distance","cadence","power","hr","speed_avg","elevation","calories"]
state sensors=[{"name":"Wahoo KICKR Core","kind":"Power","bars":4,"on":true},{"name":"Garmin HRM-Pro","kind":"Heart rate","bars":3,"on":true},{"name":"Cadence sensor 7B","kind":"Cadence","bars":1,"on":false}]
App
  Main
    Nav
      IconButton "arrow_back" tooltip="Back" onTap={go ride}
      Heading "Set up" level=1
      Button "Done" onTap={emit saveLayout slots; go ride}
    Column pad=24 gap=16 scroll
      Text.sectionTitle "Your tiles"
      Text.muted "Drag a tile from the tray onto a slot to replace it."
      Grid cols=3 gap=12
        each s in slots
          Box fill=#FFEEF1F5 radius=12 pad=16 drop=tile onDrop={slots=setAt(slots,index,result.data)}
            Text.caption "Slot {index+1}"
            Text.strong "{upper(s)}"
      Text.sectionTitle "Tray"
      Row scroll gap=8
        each p in pool
          Card drag=tile(p)
            Row gap=8 middle
              Icon "drag_indicator"
              Text "{upper(p)}"
      Divider
      Text.sectionTitle "Sensors"
      each s in sensors
        ListTile "{s.name}" subtitle="{s.kind}"
          Row gap=8 middle
            SignalBars value={s.bars} max=4
            Toggle @s.on
