### Watch face
state hour=10
state minute=9
state second=30
state steps=6420
state goal=10000
state battery=78
state temp=14
state awake=true
state dim=false
state showBattery=true
state showEvent=true
state showTemp=true
state accent="#FF4FD1C5"
every 1s onTick={second=(second+1)%60; second==0 ? {minute=(minute+1)%60} : {}; second==0 && minute==0 ? {hour=(hour+1)%12} : {}}
App
  Main
    Column center middle pad=12 gap=8 onSwipeUp={go cards} onLongPress={go gallery} onRaise={dim=false; awake=true} onScrollCrown={go cards}
      Stack center middle w=300 h=300 fill=#FF000000 radius=150 animate=fade
        Ring {steps/goal} color={accent} size=300
        Clock hour={hour} minute={minute} second={second} style="analogue" ticks=12 w=240 h=240
        if showBattery
          Gauge {battery} min=0 max=100 unit="%" size=56 top left onTap={open batteryTip}
        if showEvent
          Column center gap=2 bottom onTap={go cards}
            Icon "event"
            Text.caption "Stand-up 10:30" color=#FFFFFFFF
        if showTemp
          Column center gap=2 top right
            Icon "thermostat"
            Text.caption "{temp}°C" color=#FFFFFFFF
        if dim
          Box fill=#CC000000 w=300 h=300 radius=150 onTap={dim=false}
      Text.caption "{formatDate(today(),"EEE MMM d")}  |  {steps} / {goal} steps"
      Row gap=8 center
        Button.ghost "Dim" onTap={dim=true}
        Button.ghost "Wake" onTap={dim=false; awake=true; toast "Raised to wake"}
        Button.ghost "Faces" onTap={go gallery}
  Popover#batteryTip
    Text.strong "Battery {battery}%"
    Text.muted "About {round(battery*0.3,0)} hours left"

### Customise
state color="Teal"
state palette=["Teal","Coral","Gold","Violet","White"]
state showBattery=false
state showEvent=false
state showTemp=false
state dim=false
App
  Main
    Nav
      IconButton "arrow_back" tooltip="Back" onTap={go gallery}
      Heading "Customise"
      Button.ghost "Done" onTap={emit saveFace {"color":color,"battery":showBattery,"event":showEvent,"temp":showTemp}; toast "Saved"; go watch-face}
    Column pad=16 gap=16 scroll
      Stack center middle h=180
        Ring {steps/goal} size=160
        Clock hour={hour} minute={minute} second={second} style="analogue" w=130 h=130
      Text.sectionTitle "Colour"
      Row gap=8 scroll
        each c in palette
          ChoiceChip "{c}" selected={color==c} onTap={color=c; accent=c=="Teal" ? "#FF4FD1C5" : c=="Coral" ? "#FFFF7F6B" : c=="Gold" ? "#FFFFC857" : c=="Violet" ? "#FF9B7BFF" : "#FFFFFFFF"}
      Text.sectionTitle "Dials"
      Toggle "Battery" @showBattery
      Toggle "Next event" @showEvent
      Toggle "Temperature" @showTemp
      Toggle "Always-on dim" @dim
