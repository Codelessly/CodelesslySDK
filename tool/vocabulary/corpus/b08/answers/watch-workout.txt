### Running
state elapsed=0
state running=true
state locked=false
state hr=128
state page=0
state distance=0.0
state lapCount=0
state lapBanner=false
state pulse=false
state zones=[{"name":"Easy","max":120,"color":"#FF4FC3F7"},{"name":"Fat burn","max":140,"color":"#FF66BB6A"},{"name":"Aerobic","max":160,"color":"#FFFFCA28"},{"name":"Threshold","max":175,"color":"#FFFF7043"},{"name":"Max","max":220,"color":"#FFE53935"}]
every 1s when running onTick={elapsed=elapsed+1; distance=distance+0.0031; hr=clamp(hr+(elapsed%7<4?1:-1),110,182); lapBanner=distance>=(lapCount+1)*1.0 ? true : lapBanner; pulse=distance>=(lapCount+1)*1.0 ? true : pulse}
every 3s when lapBanner onTick={lapBanner=false; pulse=false; lapCount=floor(distance)}
App
  Main
    Stack center middle
      Pages bind=page h=fill
        Column center middle gap=4
          Text.caption "ELAPSED"
          Text "{duration(elapsed)}" size=48 weight=700
          Text.muted "Outdoor Run"
        Column center middle gap=4
          Stack center middle
            Gauge bind=hr min=60 max=200 limit=175 unit="bpm"
            Ring {clamp((hr-60)/140,0,1)}
          Text.strong "{find(zones,z=>hr<=z.max).name}" color={find(zones,z=>hr<=z.max).color}
          Text.caption "Zone {indexOf(zones,find(zones,z=>hr<=z.max))+1} of 5"
        Column center middle gap=6
          Row center gap=16
            Column center
              Text.caption "PACE"
              Text "{fixed(distance>0 ? (elapsed/60)/distance : 0,2)}" size=32 weight=700
              Text.caption "min/km"
            Column center
              Text.caption "DIST"
              Text "{fixed(distance,2)}" size=32 weight=700
              Text.caption "km"
          Sparkline values=[5.8,5.6,5.5,5.7,5.4,5.3,5.4] h=32
        Column center middle gap=8
          Text.caption "LAPS"
          Text "{lapCount}" size=48 weight=700
          Text.muted "Last lap 5:41"
      Dots bind=page count=4
      if lapBanner
        Box fill=#FF1B5E20 radius=16 pad=12 animate=slide-up
          Column center gap=2
            Row center gap=6
              Icon "flag"
              Text.strong "Lap {lapCount+1}" color=#FFFFFFFF
            Text.small "{fixed(distance,2)} km, auto-lap" color=#FFFFFFFF
      if pulse
        Ring 1.0 animate=pulse duration=300
        Text.caption "Buzz buzz" animate=fade
      if locked
        Box fill=#CC000000 radius=999 animate=fade
          Column center middle gap=6
            Icon "lock"
            Text.strong "Water lock" color=#FFFFFFFF
            Text.small "Hold crown to unlock" color=#FFFFFFFF
    Footer
      Row center gap=12
        IconButton "pause" tooltip="Pause" onTap={running=false; go paused}
        IconButton "water_drop" tooltip="Water lock" onTap={locked=true; toast "Touch locked"}
        IconButton "lock_open" tooltip="Unlock" hidden={!locked} onTap={locked=false}
