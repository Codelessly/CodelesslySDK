### Play
state score=0
state combo=0
state best=0
state life=1
state judge=""
state running=true
state notes=[{"id":1,"lane":0,"t":1200,"len":0},{"id":2,"lane":2,"t":1800,"len":900},{"id":3,"lane":1,"t":2400,"len":0},{"id":4,"lane":3,"t":3000,"len":0}]
state holding=-1
every 1s when running onTick={emit tick}
App
  Main
    Nav
      Row gap=12
        IconButton "pause" tooltip="Pause" onTap={running=false; open pause}
        Text.strong "Neon Tide"
        Badge.neutral "Normal"
      Row gap=16
        Text.heading "{score}"
        Text.muted "Life"
        Progress {life} w=160
    Stack center
      NoteHighway lanes=4 notes={notes} speed=1.4 hitLine=0.85 holding={holding} h=520 maxw=480
        onLaneDown={emit laneDown result; holding=result}
        onLaneUp={emit laneUp result; holding=-1}
        onHit={judge=result.grade; combo=result.grade=="miss" ? 0 : combo+1; best=max(best,combo); score=score+result.points; life=clamp(life+result.life,0,1)}
        onMiss={judge="miss"; combo=0; life=clamp(life-0.08,0,1)}
        onHoldEnd={judge="perfect"; combo=combo+1; score=score+300}
      if judge!=""
        Text.heading1 "{upper(judge)}" animate=pop duration=300 color={judge=="perfect" ? "#FFFFD54F" : judge=="good" ? "#FF4FC3F7" : "#FFEF5350"}
      if combo>1
        Column center gap=0
          Text "{combo}" size=72 weight=700 animate=pulse
          Text.muted "COMBO"
    Footer
      Row between
        Text.muted "Best combo {best}"
        Row gap=8
          Button.outline "Retry" onTap={score=0; combo=0; life=1; emit restart}
          Button.destructive "Quit" onTap={go songs}
  Dialog#pause "Paused" "The song is stopped"
    Button "Resume" onTap={running=true; close}
    Button.outline "Calibrate" onTap={close; go calibration}
    Button.destructive "Quit song" onTap={close; go songs}

### Calibration
state latency=0
state testing=false
state taps=[]
App
  Main
    Nav
      Heading "Calibration" level=1
      Button.outline "Back" onTap={go songs}
    Column pad=24 gap=16 maxw=560
      Card gap=12
        Text.strong "Audio latency"
        Text.muted "Tap the pad on every beat of the metronome. Then fine tune with the slider."
        Slider @latency min=-200 max=200 step=1
        Row between
          Text.muted "-200 ms"
          Text.strong "{latency} ms"
          Text.muted "+200 ms"
      Card gap=12
        Metronome bpm=120 running={testing} onBeat={emit beat}
        Box fill=#FF263238 radius=12 pad=32 onTap={taps=append(taps,now()); emit calibrationTap}
          Text "Tap here on the beat" center color=#FFFFFFFF
        Row gap=8
          Button "{testing ? 'Stop' : 'Start metronome'}" onTap={testing=!testing}
          Button.outline "Use measured value" onTap={latency=round(average(taps),0); toast "Latency set"}
      Row end gap=8
        Button.ghost "Reset" onTap={latency=0; taps=[]}
        Button "Save" onTap={emit saveLatency latency; toast "Saved"; go songs}
