### Tracker
state tracking=false
state secs=0
state runsDone=[{"name":"Aspen Glide","vert":412,"top":58,"time":"6:12"},{"name":"Ridgeline","vert":538,"top":71,"time":"5:20"},{"name":"Cub Alley","vert":190,"top":34,"time":"3:05"}]
state speed=0
every 1s when tracking onTick={secs=secs+1; speed=clamp(speed+((secs%7)-3)*4,0,82)}
App
  Sidebar
    Brand "Powder Day" icon="downhill_skiing"
    NavItem "Trail map" icon="map" onTap={go trail-map}
    NavItem "Tracker" icon="speed" selected onTap={go tracker}
    NavItem "Weather" icon="cloud" onTap={go weather}
    NavItem "Buddies" icon="group" badge="3" onTap={go buddies}
    NavItem "Snow report" icon="ac_unit" onTap={go snow-report}
    Spacer grow
    Text.muted "Season pass: Alex Moreau"
  Main
    Nav
      Heading "Live tracker"
      Row gap=8
        if tracking
          Badge.danger "Recording"
        Button.outline "Save day" onTap={emit saveDay runsDone; toast "Day saved"}
    Column pad=24 gap=16
      Row gap=16
        Card grow gap=12
          Text.caption "Current speed"
          Gauge bind=speed min=0 max=100 limit=80 unit="km/h"
          Text.muted "Time on mountain {duration(secs)}"
        Card grow gap=12
          Text.caption "Today"
          Grid cols=2 gap=12
            Column gap=2
              Text.caption "Runs"
              Text "{length(runsDone)}" size=36 weight=700
            Column gap=2
              Text.caption "Vertical"
              Text "{sum(runsDone,r=>r.vert)} m" size=36 weight=700
            Column gap=2
              Text.caption "Top speed"
              Text "{max(71,speed)} km/h" size=36 weight=700
            Column gap=2
              Text.caption "Distance"
              Text "{round(length(runsDone)*2.1,1)} km" size=36 weight=700
      Row gap=8
        if !tracking
          Button "Start tracking" onTap={tracking=true; emit trackStart}
        if tracking
          Button.destructive "Finish run" onTap={runsDone=append(runsDone,{"name":"Free ski","vert":280+secs,"top":max(40,speed),"time":duration(secs)}); secs=0; speed=0; tracking=false; toast "Run recorded"}
      Card gap=8
        Text.strong "Vertical per run"
        BarChart values={map(runsDone,r=>r.vert)} labels={map(runsDone,r=>r.name)} h=180
      Card gap=8
        Text.strong "Runs"
        each r in runsDone
          ListTile "{r.name}" subtitle="{r.vert} m vertical, top {r.top} km/h" trailing="{r.time}" leading="downhill_skiing"
            IconButton "delete" tooltip="Remove run" onTap={runsDone=removeWhere(runsDone,x=>x.name==r.name && x.time==r.time)}
