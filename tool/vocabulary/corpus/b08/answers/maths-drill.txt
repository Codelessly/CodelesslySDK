  Sidebar

    NavItem "Times grid" icon="grid_on" onTap={go grid}

      Button "Start" onTap={emit start; go drill}

        Text.strong "What to practise"
        SegmentedButton @mode items=["tables","mixed"]

          Text.muted "Pick the times tables to include"
          Grid cols=3 gap=8
            each t in tables
              Checkbox "{t.n} times table" @t.on
          Row gap=8
            Button.ghost "Select all" onTap={tables=map(tables,t=>with(t,"on",true))}
            Button.ghost "Clear" onTap={tables=map(tables,t=>with(t,"on",false))}
        if mode=="mixed"
          Text.muted "Operations to mix"
          Row gap=8
            FilterChip "Add" selected={contains(ops,"+")} onTap={ops=contains(ops,"+") ? without(ops,"+") : append(ops,"+")}
            FilterChip "Subtract" selected={contains(ops,"-")} onTap={ops=contains(ops,"-") ? without(ops,"-") : append(ops,"-")}
            FilterChip "Multiply" selected={contains(ops,"×")} onTap={ops=contains(ops,"×") ? without(ops,"×") : append(ops,"×")}
            FilterChip "Divide" selected={contains(ops,"÷")} onTap={ops=contains(ops,"÷") ? without(ops,"÷") : append(ops,"÷")}
      Card gap=12
        Text.strong "Pace"
        Row between
          Text "Seconds per question"
          Text.strong "{limit}s"
        Slider @limit min=3 max=30
        Row between
          Text "Questions"
          Text.strong "{total}"
        Slider @total min=10 max=50

### Retry round
state wrong=[{"a":7,"b":8,"op":"×","ans":56},{"a":9,"b":6,"op":"×","ans":54},{"a":12,"b":4,"op":"×","ans":48}]
state idx=0
state typed=""
state fixedCount=0
App
  Sidebar
    Brand "Sum Sprint" icon="bolt"
    NavItem "Drill" icon="timer" onTap={go drill}
    NavItem "Setup" icon="tune" onTap={go setup}
    NavItem "Retry round" icon="replay" badge="{length(wrong)}" selected={true}
    NavItem "Times grid" icon="grid_on" onTap={go grid}
  Main
    Nav
      Heading "Retry the ones you missed" level=1
      Text.muted "{fixedCount} fixed"
    Column pad=24 gap=16
      if isEmpty(wrong)
        Card gap=12 center
          Icon "emoji_events" size=48
          Text.heading "Nothing left to retry"
          Button "Back to drill" onTap={go drill}
      if !isEmpty(wrong)
        Card gap=16 center
          Text.muted "{idx+1} of {length(wrong)}"
          Text "{wrong[idx].a} {wrong[idx].op} {wrong[idx].b}" size=72 weight=700
          Input "Your answer" type=number @typed w=200
          Row gap=8
            Button "Check" onTap={typed==fixed(wrong[idx].ans,0) ? {wrong=removeAt(wrong,idx); fixedCount=fixedCount+1; idx=0; typed=""; toast "Right"} : {toast "Not quite, the answer is {wrong[idx].ans}"; idx=(idx+1)%length(wrong); typed=""}}
            Button.outline "Skip" onTap={idx=(idx+1)%length(wrong); typed=""}
        Card gap=8
          Text.strong "Still to practise"
          each w in wrong
            ListTile "{w.a} {w.op} {w.b}" icon="close"
