      Row gap=12 center
        Button.ghost "Drive" onTap={go drive}
        Button.ghost "Media" onTap={go media}

      Button.outline "Keep going" onTap={close}
      Button.destructive "End route" onTap={emit endRoute; close; go drive}

### Media
state mode="normal"
state night=false
state playing=true
state pos=84
state trackIndex=0
state tracks=[{"title":"Midnight City","artist":"M83","len":243},{"title":"Nightcall","artist":"Kavinsky","len":258},{"title":"Resonance","artist":"HOME","len":212}]
state volume=40
every 1s when playing onTick={pos=pos+1}
every 1s when pos>=tracks[trackIndex].len onTick={trackIndex=(trackIndex+1)%length(tracks); pos=0}
App
  Main
    Nav fill={night ? "#FF05070A" : "#FF0B1426"}
      Brand "Roadwise" icon="music_note"
      Row gap=8
        Text.muted "{upper(mode)}"
        IconButton "dark_mode" tooltip="Night mode" onTap={night=!night}
    Column pad=24 gap=16
      Row gap=16 stretch
        Box grow h=240 radius=16 fill={night ? "#FF10141A" : "#FF152038"} center middle onTap={trackIndex=(trackIndex+length(tracks)-1)%length(tracks); pos=0} onSwipeRight={trackIndex=(trackIndex+length(tracks)-1)%length(tracks); pos=0}
          Icon "skip_previous" size=64
          Text.muted "Previous"
        Column grow gap=8 center middle
          Image "https://images.example.com/covers/{tracks[trackIndex].title}.jpg" w=160 h=160 radius=12
          Text.heading "{tracks[trackIndex].title}" center
          Text.muted "{tracks[trackIndex].artist}" center
        Box grow h=240 radius=16 fill={night ? "#FF10141A" : "#FF152038"} center middle onTap={trackIndex=(trackIndex+1)%length(tracks); pos=0} onSwipeLeft={trackIndex=(trackIndex+1)%length(tracks); pos=0}
          Icon "skip_next" size=64
          Text.muted "Next"
      Column gap=4
        Slider bind=pos min=0 max={tracks[trackIndex].len}
        Row between
          Text.muted "{duration(pos)}"
          Text.muted "{duration(tracks[trackIndex].len)}"
      Row gap=24 center middle
        IconButton "volume_down" tooltip="Volume down" onTap={volume=max(0,volume-10)}
        IconButton "{playing ? "pause" : "play_arrow"}" tooltip="Play or pause" onTap={playing=!playing}
        IconButton "volume_up" tooltip="Volume up" onTap={volume=min(100,volume+10)}
      Slider bind=volume min=0 max=100
      Row gap=12 center
        Button.ghost "Drive" onTap={go drive}
        Button.ghost "Map" onTap={go map}
        Button.ghost "Camera" onTap={go reverse}

### Reverse
state night=false
state gear="R"
state distance=1.8
state sensorLeft=0.9
state sensorRight=0.4
state blink=true
every 400 onTick={blink=!blink}
every 1s onTick={distance=max(0.2,distance-0.1)}
App
  Main
    Nav fill="#FF000000"
      Brand "Reverse camera" icon="videocam"
      Row gap=8
        Badge.error "R"
        Text.strong "{fixed(distance,1)} m"
    Stack
      Image "camera://rear" h=420 animate=fade
      Column center bottom pad=24 gap=8
        Row gap=0 center
          Box w=4 h=200 fill="#FFFF4D4D" visible={distance<0.6}
          Box w=4 h=200 fill="#FFFFB020" visible={distance>=0.6 && distance<1.2}
          Box w=4 h=200 fill="#FF4DFF9A" visible={distance>=1.2}
        Text "Guide lines" color="#FFFFFFFF" size=12
      Row between bottom pad=16
        Icon "warning" size=48 color={sensorLeft<0.5 && blink ? "#FFFF4D4D" : "#FF4DFF9A"}
        Icon "warning" size=48 color={sensorRight<0.5 && blink ? "#FFFF4D4D" : "#FF4DFF9A"}
    Row gap=12 pad=16 center
      Switch "Guide lines" bind=guides
      Button.outline "Zoom" onTap={emit zoomCamera}
      Button "Done" onTap={gear="D"; go drive}
