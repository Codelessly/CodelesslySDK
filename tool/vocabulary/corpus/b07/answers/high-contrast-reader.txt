### Library
state tab="library"
state query=""
state books=[{"id":"b1","title":"The Lighthouse Keeper's Year","author":"Mara Ellison","pages":212,"progress":0.64},{"id":"b2","title":"Letters from the Orchard","author":"Tomas Reyes","pages":156,"progress":0.18},{"id":"b3","title":"A Short History of Rain","author":"Priya Nair","pages":288,"progress":1},{"id":"b4","title":"Salt and Iron","author":"Jonas Weber","pages":344,"progress":0},{"id":"b5","title":"The Quiet Garden","author":"Helen Okafor","pages":120,"progress":0.4}]
state current="b1"
App
  Sidebar
    Brand "Bold Page" icon="menu_book"
    NavItem "Library" icon="local_library" selected={tab=="library"} onTap={tab="library"}
    NavItem "Reading" icon="chrome_reader_mode" onTap={go reader}
    NavItem "Text settings" icon="format_size" onTap={go text-settings}
    Spacer grow
    Text.muted "Large print on"
  Main
    Nav
      Heading "My books" level=1
      Row gap=8
        Input "Search books" @query w=220
        IconButton "mic" tooltip="Search by voice" onTap={emit voiceSearch}
    Column pad=24 gap=16 scroll
      Card gap=8
        Text.muted "Continue reading"
        Text.heading "{find(books,b=>b.id==current).title}" size=32 weight=700
        Progress {find(books,b=>b.id==current).progress}
        Button "Open" size=large onTap={go reader}
      each book in filter(books,b=>contains(lower(b.title),lower(query)) || contains(lower(b.author),lower(query)))
        ListTile onTap={current=book.id; go reader} menu=bookMenu
          Column gap=4
            Text.heading "{book.title}" size=28 weight=700
            Text.muted "{book.author} - {book.pages} pages" size=20
            Progress {book.progress}
          IconButton "more_vert" tooltip="More" onTap={current=book.id; open bookMenu}
      if length(filter(books,b=>contains(lower(b.title),lower(query))))==0
        Text.muted "No books match \"{query}\"" size=24
  Menu#bookMenu end
    MenuItem "Read now" icon="menu_book" onTap={go reader}
    MenuItem "Mark as finished" icon="check" onTap={books=updateWhere(books,b=>b.id==current,b=>with(b,"progress",1)); toast "Marked as finished"}
    MenuItem "Remove" icon="delete" onTap={books=removeWhere(books,b=>b.id==current); toast "Removed"}
