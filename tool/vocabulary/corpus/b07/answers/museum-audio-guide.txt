            Text.caption "About {(d.total-d.seen)*4} minutes left"

### Favourites
state email=""
state note=""
state sent=false
App
  Sidebar
    Brand "Wanderlust Wing" icon="headphones"
    NavItem "Plan" icon="map" onTap={go floor_plan}
    NavItem "Tour" icon="route" onTap={go tour}
    NavItem "Favourites" icon="favorite" badge="{length(favs)}" selected={true}
  Main
    Nav
      Heading "Favourites bag" level=1
      Button "Email me my bag" icon="mail" onTap={open email}
    Column pad=24 gap=12 scroll
      if length(favs)==0
        Card center gap=8
          Icon "favorite_border"
          Text.strong "Nothing saved yet"
          Text.muted "Tap the heart on any stop and it will wait here."
          Button.outline "Back to the plan" onTap={go floor_plan}
      each f in favs
        ListTile "Stop {f}" icon="headphones" subtitle="Saved today"
          Row gap=8
            IconButton "play_arrow" tooltip="Play" onTap={stop=f; go stop}
            IconButton "delete" tooltip="Remove" onTap={favs=without(favs,f)}
      if sent
        Text.success "Sent to {email}. Enjoy the rest of your visit."
  Dialog#email "Email my favourites" "We send one email and keep no list"
    Input "Your email" "you@example.com" bind=email type=email
    Input "Add a note" bind=note multiline
    Checkbox "Include transcripts" bind=withText
    Button "Send" onTap={contains(email,"@") ? {emit sendBag {"email":email,"note":note,"stops":favs}; sent=true; close; toast "Bag on its way"} : toast "Enter a valid email"}
