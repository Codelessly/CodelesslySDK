### Duel
state start="COLD"
state end="WARM"
state rungs=["COLD"]
state guess=""
state bad=false
state left=90
state running=true
state opp=["COLD","CORD","CARD"]
state shortest=4
state links=[{"a":"COLD","b":"CORD"},{"a":"COLD","b":"CORE"},{"a":"COLD","b":"GOLD"},{"a":"CORD","b":"CARD"},{"a":"CORD","b":"WORD"},{"a":"CORE","b":"CARE"},{"a":"CARE","b":"CARD"},{"a":"CARE","b":"WARE"},{"a":"CARD","b":"WARD"},{"a":"WORD","b":"WARD"},{"a":"WORD","b":"WORM"},{"a":"WARD","b":"WARM"},{"a":"WARE","b":"WARM"},{"a":"WORM","b":"WARM"}]
every 1s when running onTick={left<=1 ? {left=0; running=false; go results} : left=left-1}
App
  Main
    Nav
      Heading "Rungs" level=3
      Row gap=12 middle
        Text.mono "{duration(left)}"
        IconButton "flag" tooltip="Give up" onTap={running=false; go results}
    Column pad=24 gap=16
      Progress {left/90}
      Row gap=24 top
        Column grow gap=8
          Row middle gap=8
            Badge.primary "START"
            Text.mono "{start}" size=28 weight=700
          each w in rungs
            if index > 0
              Row middle gap=8 animate=slide-up duration=300
                Badge.neutral "{index}"
                Text.mono "{w}" size=28
          if bad
            Text.danger "Not a word one letter away" animate=shake duration=400
          Row gap=8 middle
            Input "Next word" "one letter changed" @guess maxlength=4 autofocus onSubmit={any(links,l=>(l.a==last(rungs) && l.b==upper(guess)) || (l.b==last(rungs) && l.a==upper(guess))) && !contains(rungs,upper(guess)) ? {bad=false; rungs=append(rungs,upper(guess)); guess=""; upper(guess)==end ? {running=false; go results} : {}} : {bad=true; guess=""}}
            Button "Climb" icon="arrow_upward" onTap={any(links,l=>(l.a==last(rungs) && l.b==upper(guess)) || (l.b==last(rungs) && l.a==upper(guess))) && !contains(rungs,upper(guess)) ? {bad=false; rungs=append(rungs,upper(guess)); guess=""; upper(guess)==end ? {running=false; go results} : {}} : {bad=true; guess=""}}
          Row middle gap=8
            Badge.success "GOAL"
            Text.mono "{end}" size=28 weight=700
        Pane w=140 pad=12 gap=8
          Text.caption "NOOR"
          each o in opp
            Text.mono "{o}"
          Progress {(length(opp)-1)/shortest}
          Text.muted "{length(opp)-1} steps"
    Footer
      Row between middle
        Text.muted "Steps: {length(rungs)-1}"
        Button.ghost "Undo step" icon="undo" onTap={length(rungs)>1 ? rungs=removeAt(rungs,length(rungs)-1) : toast "Nothing to undo"}
