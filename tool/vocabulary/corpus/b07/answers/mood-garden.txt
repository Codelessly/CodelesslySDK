### Settings
state locked=true
state lockType="PIN"
state remindersOn=true
state reminders=[{"id":"r1","time":"08:30","label":"Morning check-in","on":true},{"id":"r2","time":"21:00","label":"Evening gratitude","on":true}]
state newTime="12:00"
state newLabel=""
state pin=""
App
  Sidebar
    Brand "Soil" icon="spa"
    NavItem "Garden" icon="yard" onTap={go garden}
    NavItem "Today" icon="edit_note" onTap={go today}
    NavItem "Seasons" icon="insights" onTap={go seasons}
    NavItem "Settings" icon="settings" selected={true} onTap={go settings}
  Main
    Nav
      Heading "Settings" level=1
    Column pad=24 gap=16 scroll
      Card gap=12
        Text.strong "Lock"
        Toggle "Lock Soil when I leave" bind=locked
        if locked
          SegmentedButton bind=lockType items=["PIN","Biometric"]
          Button.outline "Set PIN" onTap={open pinpad}
      Card gap=12
        Row between
          Text.strong "Reminders"
          Toggle bind=remindersOn
        if remindersOn
          each r in reminders
            ListTile "{r.label}" subtitle="{r.time}"
              Toggle bind=r.on
              IconButton "delete" tooltip="Remove" onTap={reminders=removeWhere(reminders,x=>x.id==r.id)}
          Row gap=8 middle
            TimePicker "Time" bind=newTime
            Input "Label" "Take a breath" bind=newLabel grow
            Button "Add" onTap={newLabel=="" ? toast "Give it a label" : {reminders=append(reminders,{"id":newLabel,"time":newTime,"label":newLabel,"on":true}); newLabel=""}}
  Dialog#pinpad "Set a PIN" "Four digits"
    PinPad bind=pin length=4 onComplete={locked=true; lockType="PIN"; toast "PIN saved"; close}
    Button.ghost "Cancel" onTap={pin=""; close}
