### Wind down
state running=false
state left=1800
state dim=0.6
state warmth=2700
state minutes=30
every 1s when running onTick={left=max(0,left-1); left==0 ? {running=false; toast "Time for bed"} : }
Page
  Nav
    Brand "Nightshade" icon="bedtime"
    Link "Last night" onTap={go last-night}
  Section center narrow
    Column gap=20 center
      Heading "Wind down" level=1
      Text.muted "Dim the room and let your body slow down."
      Stack center middle
        Ring {left/(minutes*60)} size=220
        Column center
          Text "{duration(left)}" size=48 weight=700
          Text.muted "until lights out"
      Card gap=12
        Text.strong "Screen colour"
        ColorTemperature @warmth min=1800 max=4000 unit="K"
        Text.fieldLabel "Brightness"
        Slider @dim min=0 max=1
        Row gap=8
          Text.muted "Warm"
          Spacer grow
          Text.muted "{round(dim*100,0)} percent"
      Card gap=8
        Text.fieldLabel "Wind-down length"
        Row gap=8
          ChoiceChip "15 min" selected={minutes==15} onTap={minutes=15; left=900}
          ChoiceChip "30 min" selected={minutes==30} onTap={minutes=30; left=1800}
          ChoiceChip "45 min" selected={minutes==45} onTap={minutes=45; left=2700}
      Row gap=12
        Button running ? "Pause" : "Start wind-down" size=large onTap={running=!running; emit windDown {dim: dim, warmth: warmth}}
        Button.outline "Reset" size=large onTap={running=false; left=minutes*60}
      if !running && left==0
        Badge.success "Lights out. Sleep well."
  Footer
    Text.muted "Screens off by 22:30 helps most."

### Alarm
state wakeFrom="06:30"
state wakeTo="07:00"
state enabled=true
state sound="Morning tide"
state smart=true
state days=["Mon","Tue","Wed","Thu","Fri"]
state volume=0
App
  Sidebar
    Brand "Nightshade" icon="bedtime"
    NavItem "Last night" icon="nights_stay" onTap={go last-night}
    NavItem "Week" icon="show_chart" onTap={go week}
    NavItem "Wind down" icon="dark_mode" onTap={go wind-down}
    NavItem "Alarm" icon="alarm" selected={true}
  Main
    Nav
      Heading "Alarm" level=1
      Toggle "On" @enabled
    Column pad=24 gap=16 maxw=720
      Card gap=12
        Text.strong "Gentle wake window"
        Text.muted "We wake you in the lightest sleep between these times."
        Row gap=16
          TimePicker "Earliest" @wakeFrom
          TimePicker "Latest" @wakeTo
        Text.caption "Window: {wakeFrom} to {wakeTo}"
        Toggle "Wake in light sleep" @smart
      Card gap=12
        Text.strong "Repeat"
        Row gap=8
          each d in ["Mon","Tue","Wed","Thu","Fri","Sat","Sun"]
            FilterChip "{d}" selected={contains(days,d)} onTap={days=contains(days,d) ? without(days,d) : append(days,d)}
      Card gap=12
        Text.strong "Sound"
        Select "Tone" items=["Morning tide","Soft chime","Birdsong","Low hum"] @sound
        Slider @volume min=0 max=100
        Button.outline "Preview" icon="play_arrow" onTap={emit previewAlarm sound}
      Row gap=12
        Button "Save alarm" onTap={emit saveAlarm; toast "Alarm set for {wakeFrom} to {wakeTo}"}
        Button.ghost "Remove" onTap={open confirmRemove}
  Dialog#confirmRemove "Remove this alarm?" "You will not be woken tomorrow."
    Row gap=8 end
      Button.outline "Keep it" onTap={close}
      Button.destructive "Remove" onTap={enabled=false; emit removeAlarm; close}
