### Daily calendar
state month="November 2026"
state days=[1,2,3,5,6,8,9,10,12,13,14,15]
state picked=16
App
  Sidebar
    Brand "Hidden Harbour" icon="sailing"
    NavItem "Today's puzzle" icon="grid_view" onTap={go puzzle}
    NavItem "Daily calendar" icon="calendar_month" selected onTap={go calendar}
    NavItem "Themes" icon="palette" onTap={go themes}
    Spacer grow
  Main
    Nav
      Heading "Daily puzzles" level=1
      Row gap=8
        IconButton "chevron_left" tooltip="Previous month" onTap={emit prevMonth}
        Text.strong "{month}"
        IconButton "chevron_right" tooltip="Next month" onTap={emit nextMonth}
    Column pad=24 gap=16
      Row gap=24 top
        Card grow gap=12
          Grid cols=7 gap=8
            each d in ["Mon","Tue","Wed","Thu","Fri","Sat","Sun"]
              Text.caption "{d}" center
          Grid cols=7 gap=8
            each n in range(30)
              Box pad=8 radius=8 gap=4 center border-bottom=#FFE0E0E0 onTap={picked=n+1; open day}
                Text "{n+1}" center weight={picked==n+1 ? 700 : 400}
                if contains(days,n+1)
                  Icon "sailing" color=#FF2A9D8F
        Card w=280 gap=8
          Heading "This month" level=3
          Text.strong "{length(days)} of 15 sails"
          Progress {length(days)/15}
          Text.muted "Longest streak 6 days"
  Dialog#day "{picked} November"
    Column gap=12
      if contains(days,picked)
        Row gap=8
          Icon "sailing"
          Text "Completed"
      if !contains(days,picked)
        Text.muted "Not played yet."
      Button "Open puzzle" onTap={close; go puzzle}
