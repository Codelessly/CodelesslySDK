### Listening
state heard=""
state words=["How","many","tablespoons","are","in","a","third","of","a","cup"]
state shown=0
state phase="listening"
state step=0
state answers=[{"kind":"convert","title":"Unit conversion","big":"5 1/3 tbsp","line":"1/3 cup equals 5 tablespoons and 1 teaspoon"},{"kind":"timer","title":"Timer set","big":"10:00","line":"Rice, 10 minutes"},{"kind":"recipe","title":"Shakshuka, step 3 of 6","big":"Simmer 8 min","line":"Crack in the eggs, cover, and cook until the whites are set."}]
every 400 when phase=="listening" onTick={shown=min(length(words),shown+1); heard=join(take(words,shown)," ")}
every 400 when phase=="listening" && shown>=length(words) onTick={phase="answer"}
App
  Main
    Nav
      Brand "Hearth Hub" icon="speaker"
      IconButton "close" tooltip="Dismiss" onTap={go idle}
    Column pad=24 gap=20
      if phase=="listening"
        Column gap=16 center middle
          Text.muted "Listening..." center animate=pulse
          Waveform active=true colors=["#FF6C5CE7","#FF00B8D9","#FFFF7AB6"] h=120 animate=pulse
          Text "{heard}" size=36 weight=500 center
      if phase=="answer"
        Text.muted "\"How many tablespoons are in a third of a cup\"" center
        Pages @step items=answers h=320 peek=0.9
          Card gap=12 pad=24
            Row between
              Text.sectionTitle "{item.title}"
              Badge.info "{item.kind}"
            Text "{item.big}" size=72 weight=700
            Text.muted "{item.line}" size=22
        Dots @step count=3
        Row gap=8 center
          Button.outline "Ask another" onTap={heard=""; shown=0; phase="listening"}
          Button.ghost "Read it aloud" onTap={emit speak step}
          Button "Done" onTap={go idle}

### Timers and Playing
state timers=[{"id":1,"name":"Pasta","left":492,"total":600,"running":true},{"id":2,"name":"Garlic bread","left":205,"total":420,"running":true},{"id":3,"name":"Laundry","left":2710,"total":3600,"running":false}]
state volume=45
state playing=true
state elapsed=83
state track={"title":"Golden Hour","artist":"Mira Okafor","album":"Slow Mornings","length":214}
state newName=""
every 1s onTick={timers=map(timers,t=>t.running && t.left>0 ? with(t,"left",t.left-1) : t); elapsed=playing ? min(track.length,elapsed+1) : elapsed}
App
  Main
    Nav
      Brand "Hearth Hub" icon="speaker"
      Row gap=8
        IconButton "add_alarm" tooltip="New timer" onTap={open newTimer}
        IconButton "home" tooltip="Home" onTap={go idle}
    Column pad=24 gap=20
      Text.sectionTitle "Timers"
      Grid cols=3 gap=16
        each t in timers
          Card gap=8
            Text.strong "{t.name}"
            Stack center middle
              Ring {1 - t.left/t.total} size=140
              Text "{duration(t.left)}" size=28 weight=700
            Row gap=8 center
              IconButton "{t.running ? 'pause' : 'play_arrow'}" tooltip="Pause or resume" onTap={timers=updateWhere(timers,x=>x.id==t.id,x=>with(x,"running",!x.running))}
              IconButton "add" tooltip="Add one minute" onTap={timers=updateWhere(timers,x=>x.id==t.id,x=>with(x,"left",x.left+60))}
              IconButton "close" tooltip="Cancel timer" onTap={timers=removeWhere(timers,x=>x.id==t.id)}
      Card gap=16 pad=20
        Row gap=16 middle
          Image "https://images.hearth.example/album/slow-mornings.jpg" w=96 h=96 radius=12
          Column grow gap=4
            Text.caption "Now playing"
            Text.title "{track.title}"
            Text.muted "{track.artist}, {track.album}"
          IconButton "skip_previous" tooltip="Previous" onTap={emit previous}
          IconButton "{playing ? 'pause_circle' : 'play_circle'}" tooltip="Play or pause" size=48 onTap={playing=!playing}
          IconButton "skip_next" tooltip="Next" onTap={emit next}
        Row gap=8 middle
          Text.caption "{duration(elapsed)}"
          Progress {elapsed/track.length} grow
          Text.caption "{duration(track.length)}"
        Row gap=12 middle
          Icon "volume_down"
          Slider @volume min=0 max=100 h=48 grow
          Icon "volume_up"
          Text.strong "{volume}" w=40
  Dialog#newTimer "New timer"
    Input "Name" "Tea" @newName
    Row gap=8
      Button.outline "5 min" onTap={timers=append(timers,{"id":length(timers)+10,"name":coalesce(newName,"Timer"),"left":300,"total":300,"running":true}); close}
      Button.outline "10 min" onTap={timers=append(timers,{"id":length(timers)+10,"name":coalesce(newName,"Timer"),"left":600,"total":600,"running":true}); close}
      Button.outline "20 min" onTap={timers=append(timers,{"id":length(timers)+10,"name":coalesce(newName,"Timer"),"left":1200,"total":1200,"running":true}); close}

### Night Clock
state now="2:07"
state dim=15
state ringing=false
state nextAlarm="6:30 am"
every 1s when ringing==false onTick={emit tick}
App
  Main fill=#FF000000
    Column pad=24 gap=16 center middle h=fill onTap={ringing ? (ringing=false; toast "Alarm stopped") : (dim=dim<50 ? 60 : 15)}
      Text "{now}" size=160 weight=200 color=#FF8A3B2E center
      Row gap=8 middle
        Icon "alarm" color=#FF5C2A22
        Text "{nextAlarm}" size=24 color=#FF5C2A22
      if ringing
        Text "Tap anywhere to stop" size=32 color=#FFB04A3A animate=pulse
      Row gap=8 hide=phone
        Text "Dimming to {dim}%" size=16 color=#FF4A2019
        Slider @dim min=5 max=100 w=200
      Row gap=12
        Button.ghost "Wake screen" onTap={dim=60; go idle}
        Button.ghost "Alarm off" onTap={nextAlarm="Off"; toast "Alarm off"}
