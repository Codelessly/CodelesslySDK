              IconButton "more_vert" tooltip="More" onTap={sel=h.id; open heldMenu}
  Menu#heldMenu end
    MenuItem "Move to Table 1" icon="table_restaurant" onTap={held=updateWhere(held,x=>x.id==sel,x=>with(x,"table","Table 1"))}

    MenuItem "Void ticket" icon="delete" onTap={held=removeWhere(held,x=>x.id==sel); toast "Ticket voided"}
