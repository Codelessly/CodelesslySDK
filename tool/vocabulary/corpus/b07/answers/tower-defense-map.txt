### Battle
state gold=250
state lives=20
state wave=1
state totalWaves=10
state speed=1
state paused=false
state running=true
state over=""
state sel=""
state plots=[{"id":"p1","x":0.18,"y":0.30,"tower":null},{"id":"p2","x":0.34,"y":0.52,"tower":null},{"id":"p3","x":0.50,"y":0.28,"tower":null},{"id":"p4","x":0.62,"y":0.62,"tower":null},{"id":"p5","x":0.78,"y":0.38,"tower":null},{"id":"p6","x":0.86,"y":0.70,"tower":null}]
state towerTypes=[{"id":"archer","name":"Archer","icon":"gps_fixed","cost":60,"range":0.14},{"id":"cannon","name":"Cannon","icon":"circle","cost":120,"range":0.17},{"id":"frost","name":"Frost","icon":"ac_unit","cost":100,"range":0.15},{"id":"mage","name":"Mage","icon":"auto_fix_high","cost":150,"range":0.16}]
state enemies=[]
state waveActive=false
state countdown=30
state pick=""
state sfx=false
state volume=0
every 1s when running && !paused && !waveActive onTick={countdown=max(0,countdown-1)}
every 1s when running && !paused && countdown==0 && !waveActive onTick={waveActive=true; emit spawnWave wave}
every 1s when running && lives<=0 onTick={running=false; over="defeat"; open defeat}
every 1s when running && wave>totalWaves onTick={running=false; over="victory"; open victory}
App
  Main
    Nav
      Row gap=16
        Row gap=4
          Icon "paid"
          Text.strong "{gold}"
        Row gap=4
          Icon "favorite"
          Text.strong "{lives}" color={lives<=5 ? #FFD32F2F : #FF000000}
        Row gap=4
          Text.muted "Wave"
          Text.strong "{min(wave,totalWaves)} / {totalWaves}"
      Row gap=8
        Button.outline "{speed}x" icon="fast_forward" onTap={speed=speed==3 ? 1 : speed+1; emit setSpeed speed}
        IconButton "pause" tooltip="Pause" onTap={paused=true; open pauseMenu}
    Stack
      GameMap src="https://img.bastionline.app/maps/1.jpg" onTap={sel=""}
        Path points=[[0,0.45],[0.25,0.45],[0.25,0.75],[0.55,0.75],[0.55,0.45],[0.72,0.45],[0.72,0.20],[1,0.20]] width=0.05 color=#FFC9A66B
        Marker "Spawn" x=0 y=0.45 icon="flag"
        Marker "Keep" x=1 y=0.20 icon="castle"
        each e in enemies
          Enemy kind={e.kind} progress={e.progress} hp={e.hp} maxHp={e.maxHp}
        each plot in plots
          if plot.tower==null
            Plot x={plot.x} y={plot.y} selected={sel==plot.id} onTap={sel=plot.id; open towerRing}
          if plot.tower!=null
            Tower kind={plot.tower.kind} level={plot.tower.level} x={plot.x} y={plot.y} selected={sel==plot.id} onTap={sel=plot.id; open towerPanel}
            if sel==plot.id
              RangeCircle x={plot.x} y={plot.y} radius={plot.tower.range} animate=fade
      Column top end pad=12
        if !waveActive && wave<=totalWaves
          Button "Next wave +{10+wave*2} gold" icon="skip_next" onTap={gold=gold+10+wave*2; countdown=0; emit earlyWave wave}
    Footer
      Row between pad=12
        Text.muted "Next wave in {countdown}s" hide=phone
        Progress {waveActive ? 0.5 : 1-countdown/30} w=240
        Text.muted "{length(enemies)} enemies"
  Popover#towerRing "Build"
    RadialMenu @pick
      each t in towerTypes
        RadialItem icon={t.icon} label="{t.name} {t.cost}" disabled={gold<t.cost} onTap={gold=gold-t.cost; plots=updateWhere(plots,p=>p.id==sel,p=>with(p,"tower",{"kind":t.id,"level":1,"range":t.range,"spent":t.cost})); emit build t.id; close}
  Sheet#towerPanel "Tower"
    Column gap=12
      each p in filter(plots,x=>x.id==sel)
        Row between
          Heading "{upper(p.tower.kind)} level {p.tower.level}" level=3
          Badge.info "Range {round(p.tower.range*100,0)}"
        Row gap=8
          Button "Upgrade {p.tower.spent}" icon="arrow_upward" disabled={gold<p.tower.spent || p.tower.level>=3} onTap={gold=gold-p.tower.spent; plots=updateWhere(plots,x=>x.id==sel,x=>with(x,"tower",with(with(x.tower,"level",x.tower.level+1),"range",x.tower.range+0.02))); emit upgrade sel}
          Button.destructive "Sell +{floor(p.tower.spent*0.7)}" icon="sell" onTap={gold=gold+floor(p.tower.spent*0.7); plots=updateWhere(plots,x=>x.id==sel,x=>with(x,"tower",null)); sel=""; emit sell sel; close}
  Dialog#pauseMenu "Paused"
    Column gap=12
      Button "Resume" icon="play_arrow" onTap={paused=false; close}
      Button.outline "Restart level" icon="replay" onTap={emit restart; paused=false; close}
      Toggle "Sound effects" @sfx
      Slider @volume min=0 max=100
      Button.destructive "Quit to map" icon="logout" onTap={go level-select}
  Dialog#victory "Victory"
    Column gap=12 center
      Icon "emoji_events" size=48
      Heading "The line held" level=3
      Rating value={lives>=20 ? 3 : lives>=10 ? 2 : 1} max=3 readonly animate=pop
      Text.muted "{lives} lives left, {gold} gold unspent"
      Row gap=8
        Button.outline "Replay" onTap={emit restart; close}
        Button "Next level" onTap={go battle}
  Dialog#defeat "Defeat"
    Column gap=12 center
      Icon "heart_broken" size=48
      Heading "The keep has fallen" level=3
      Text.muted "You reached wave {wave} of {totalWaves}."
      Row gap=8
        Button.outline "Quit" onTap={go level-select}
        Button "Try again" onTap={emit restart; close}
