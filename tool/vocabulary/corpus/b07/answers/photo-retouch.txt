### Layers and history
state layers=[{"id":1,"name":"Skin smoothing","visible":true,"opacity":60},{"id":2,"name":"Eye brighten","visible":true,"opacity":45},{"id":3,"name":"Warm grade","visible":false,"opacity":80},{"id":4,"name":"Background","visible":true,"opacity":100}]
state history=[{"id":1,"label":"Open Portrait_0412.jpg","time":"10:02"},{"id":2,"label":"Exposure +18","time":"10:04"},{"id":3,"label":"Heal blemish","time":"10:06"},{"id":4,"label":"Brush: eye brighten","time":"10:09"},{"id":5,"label":"Warmth +12","time":"10:11"}]
state at=5
state sel=0
App
  Sidebar w=72
    IconButton "arrow_back" tooltip="Back to editor" onTap={go editor}
  Main
    Nav
      Heading "Layers and history" level=3
      Button.outline "Add layer" icon="add" onTap={layers=append(layers,{"id":length(layers)+10,"name":"New layer","visible":true,"opacity":100})}
    Column pad=24 gap=16
      Card gap=0 pad=0
        each l in layers
          Row between pad=12 border-bottom=#1F000000
            Row gap=12
              IconButton "{l.visible ? 'visibility' : 'visibility_off'}" tooltip="Show or hide" onTap={layers=updateWhere(layers,x=>x.id==l.id,x=>with(x,"visible",!x.visible))}
              Image "https://images.example.com/retouch/thumb_{l.id}.jpg" w=48 h=48 radius=6
              Text.strong "{l.name}"
            Row gap=12
              Slider @l.opacity min=0 max=100 w=140
              Text.muted "{l.opacity}%" w=44
              IconButton "more_vert" tooltip="Layer menu" onTap={sel=l.id; open layerMenu}
  Pane w=320 pad=16 gap=12
    Heading "History" level=3
    each h in history
      ListTile "{h.label}" subtitle="{h.time}" selected={h.id==at} onTap={at=h.id; emit historyJump h.id; toast "Back to: {h.label}"}
    Button.outline "Clear history" onTap={history=take(history,1); at=1}
  Menu#layerMenu end
    MenuItem "Duplicate" icon="content_copy" onTap={layers=append(layers,with(find(layers,x=>x.id==sel),"id",sel+100))}
    MenuItem "Merge down" icon="merge_type" onTap={emit mergeDown sel}
    MenuItem "Delete" icon="delete" onTap={layers=removeWhere(layers,x=>x.id==sel)}
