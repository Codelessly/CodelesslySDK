### Proof
App
  Sidebar
    Brand "Dropoff" icon="local_shipping"
    NavItem "Route" icon="route" onTap={go route}
    NavItem "Proof" icon="verified" selected={true}
    NavItem "Summary" icon="summarize" onTap={go summary}
  Main
    Nav
      Heading "Proof of delivery" level=1
      Text.muted "{find(stops,s=>s.id==sel).name} · {find(stops,s=>s.id==sel).addr}"
    maxw=720
    Column pad=24 gap=16
      SegmentedButton bind=proof items=["photo","signature","code"]
      if proof=="photo"
        Card gap=12
          Text.strong "Photograph the parcel at the door"
          CameraCapture bind=photo h=280 facing="back"
          if photo!=""
            Button.outline "Retake" icon="refresh" onTap={photo=""}
      if proof=="signature"
        Card gap=12
          Text.strong "Recipient signs below"
          SignaturePad bind=sig h=200
          Row between
            Text.muted "Signed by recipient"
            Button.ghost "Clear" onTap={sig=""}
      if proof=="code"
        Card gap=12
          Text.strong "Ask the recipient for their 6-digit code"
          Input "Delivery code" "000000" bind=code type=number maxlength=6
          Text.danger "That code does not match" visible={length(code)==6 && code!="482913"}
      Row gap=8
        Button.outline "Back" grow onTap={go route}
        Button "Confirm delivery" grow icon="check" onTap={(proof=="photo" && photo=="") || (proof=="signature" && sig=="") || (proof=="code" && code!="482913") ? toast "Proof is missing" : {stops=updateWhere(stops,s=>s.id==sel,s=>with(s,"status","done")); emit delivered sel; toast "Delivered"; photo=""; sig=""; code=""; go route}}
