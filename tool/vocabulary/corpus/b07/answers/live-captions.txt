### Live captions
state listening=true
state paused=false
state silence=0
state spoken="English"
state shown="English"
state fontSize=44
state fontFamily="Sans"
state theme="Dark"
state keyword="invoice"
state lines=[{"id":1,"speaker":"Maya","color":"#FF4FC3F7","text":"Thanks everyone for joining, let's start with the invoice from last week.","tentative":""},{"id":2,"speaker":"Daniel","color":"#FFFFB74D","text":"I checked it this morning and the totals match the purchase order","tentative":" except for the shipping"},{"id":3,"speaker":"Maya","color":"#FF4FC3F7","text":"Good, can you send the corrected invoice before","tentative":" noon today"}]
state liveCount=3
every 1s when listening onTick={silence=silence+1}
App
  Sidebar
    Brand "Subtitler" icon="closed_caption"
    NavItem "Live" icon="subtitles" selected={true}
    NavItem "History" icon="history" onTap={go history}
    NavItem "Appearance" icon="text_fields" onTap={go appearance}
    NavItem "Languages" icon="translate" onTap={go languages}
    Spacer grow
    Text.muted "Room: Studio B"
  Main
    Nav
      Heading "Live captions" level=1
      Row gap=8
        Badge.success "Listening" visible={listening && silence<10}
        Badge.warning "Paused" visible={paused}
        Button.outline "Save transcript" icon="download" onTap={emit saveTranscript lines; toast "Transcript saved"}
        Button.outline "Pause" onTap={paused=!paused; toast paused ? "Resumed" : "Paused for reading"}
        Button.destructive "Stop" onTap={listening=false; silence=0}
    Column pad=24 gap=16
      if silence>=10 && listening
        Card fill=#FFFFF3CD gap=8
          Row gap=8 middle
            Icon "volume_off" color=#FF8A6D00
            Text.strong "The room is very quiet" color=#FF5C4600
          Text "No speech for {silence} seconds. Move the microphone closer or check that it is not muted." color=#FF5C4600
          Row gap=8
            Button.outline "Check microphone" onTap={open micCheck}
            Button.ghost "Dismiss" onTap={silence=0}
      Row gap=12 middle
        Input "Highlight words" "invoice, deadline" @keyword w=240
        Select @spoken items=["English","Spanish","French","German","Japanese"] label="Spoken"
        Icon "arrow_forward"
        Select @shown items=["English","Spanish","French","German","Japanese"] label="Shown"
        Spacer grow
        Waveform active={listening && !paused} onVoice={silence=0}
      Box fill={theme=="Dark" ? "#FF111418" : theme=="Light" ? "#FFFFFFFF" : "#FF000000"} radius=16 pad=24 gap=20 h=420
        Column scroll gap=20 autoscroll={!paused}
          each line in lines
            Column gap=4 animate=fade
              Text.caption "{line.speaker}" color={line.color} weight=700
              Text "{line.text}{line.tentative}" size={fontSize} font={fontFamily} color={theme=="Light" ? "#FF111418" : "#FFFFFFFF"} highlight={keyword} highlightColor=#FFFFD54F
                Span "{line.tentative}" opacity=0.5
      Row between middle
        Text.muted "{length(lines)} segments, {sum(lines,l=>length(l.text))} characters"
        Row gap=8
          Button.outline "Add test speech" onTap={lines=append(lines,{"id":length(lines)+1,"speaker":"Daniel","color":"#FFFFB74D","text":"This is a sample line of speech.","tentative":" still arriving"}); silence=0}
          Button.ghost "Clear" onTap={lines=[]}
  Dialog#micCheck "Microphone check" "Speak normally for a few seconds."
    Waveform active={true}
    Progress value={micLevel}
    Text.muted "Input level: {round(micLevel*100,0)}%"
    Button "Done" onTap={silence=0; close}

              ChoiceChip "Light" selected={theme=="Light"} onTap={theme="Light"}

              Text.caption "Daniel" color=#FFFFB74D weight=700
            Text "Totals match." size={fontSize} font={fontFamily} weight={weight} lineHeight={lineSpacing} color={theme=="Light" ? "#FF111418" : "#FFFFFFFF"}
          Text.muted "The lighter words are still being recognised and settle once confirmed."
      Row end gap=8
        Button.outline "Cancel" onTap={go live-captions}
        Button "Apply" onTap={emit saveAppearance {"size":fontSize,"font":fontFamily,"theme":theme}; toast "Appearance saved"; go live-captions}

state query=""

      Row gap=12 middle
        Input "Search transcript" @query w=260
        Input "Highlight" @keyword w=180
        SegmentedButton @speakerFilter items=["All","Maya","Daniel","Priya"]
        Spacer grow
        Text.muted "{count(history,h=>speakerFilter=="All" || h.speaker==speakerFilter)} segments"
      Card gap=0
        Column scroll h=460 gap=16
          each h in history
            if (speakerFilter=="All" || h.speaker==speakerFilter) && contains(lower(h.text),lower(query))
              Row gap=12 top
                Text.muted "{h.time}" w=56
                Avatar "{h.speaker}" size=32
                Column grow gap=4
                  Text.strong "{h.speaker}" color={h.color}
                  Text "{h.text}" highlight={keyword} highlightColor=#FFFFD54F
                IconButton "content_copy" tooltip="Copy line" onTap={copy h.text; toast "Copied"}
      Row between middle
        Text.muted "Scrolling back pauses the live captions."
        Button.ghost "Jump to latest" icon="arrow_downward" onTap={scroll latest}
  Dialog#saveFormat "Save transcript" "Choose how to export the conversation."
    Select "Format" @format items=["Plain text (.txt)","Subtitles (.srt)","Document (.docx)"]
    Toggle "Include timestamps" @stamps
    Toggle "Include speaker names" @names
    Row end gap=8
      Button.outline "Cancel" onTap={close}
      Button "Save" onTap={emit saveTranscript {"format":format,"timestamps":stamps,"names":names}; toast "Transcript saved"; close}

### Languages
state spoken="English"
state shown="Spanish"
state translate=true
state autoDetect=false
state langs=["English","Spanish","French","German","Japanese","Portuguese"]
App
  Sidebar
    Brand "Subtitler" icon="closed_caption"
    NavItem "Live" icon="subtitles" onTap={go live-captions}
    NavItem "History" icon="history" onTap={go history}
    NavItem "Appearance" icon="text_fields" onTap={go appearance}
    NavItem "Languages" icon="translate" selected={true}
  Main
    Nav
      Heading "Languages" level=1
      Button.outline "Swap" icon="swap_horiz" onTap={spoken=shown; shown=spoken}
    Column pad=24 gap=16
      Grid cols=2 gap=24
        Card gap=12
          Heading "Spoken language" level=3
          Text.muted "What people in the room are speaking."
          Select @spoken items={langs}
          Toggle "Detect the language automatically" @autoDetect
        Card gap=12
          Heading "Shown language" level=3
          Text.muted "What appears on the caption stripe."
          Select @shown items={langs}
          Toggle "Translate the captions" @translate
      Card gap=8
        Text.strong "Preview"
        Row gap=8 middle
          Badge.info "{spoken}"
          Icon "arrow_forward"
          Badge.primary "{translate ? shown : spoken}"
        Text.muted "Hello everyone, shall we begin?"
        Text size=28 "{translate && shown=="Spanish" ? "Hola a todos, ¿empezamos?" : translate && shown=="French" ? "Bonjour à tous, on commence ?" : "Hello everyone, shall we begin?"}"
      Row end gap=8
        Button.outline "Cancel" onTap={go live-captions}
        Button "Apply" onTap={emit setLanguages {"spoken":spoken,"shown":shown,"translate":translate}; toast "Languages updated"; go live-captions}
