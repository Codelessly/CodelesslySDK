            Text "{money(r.principal,'$',0)}" w=110

        Card grow gap=4
          Text.caption "Interest in year 1"
          Text.heading "{money(first(amortize(price-deposit,rate,term,showExtra?extra:0)).interest,'$',0)}"

          Text.caption "Balance after year 10"
          Text.heading "{money(amortize(price-deposit,rate,term,showExtra?extra:0)[9].balance,'$',0)}"

### Scenarios
state tab="scenarios"
state scenarios=[{"id":1,"name":"Baseline","price":385000,"deposit":57750,"term":25,"rate":4.6,"extra":0,"payment":1819,"interest":218700,"years":25},{"id":2,"name":"Bigger deposit","price":385000,"deposit":90000,"term":25,"rate":4.4,"extra":0,"payment":1620,"interest":191000,"years":25},{"id":3,"name":"Overpay 300","price":385000,"deposit":57750,"term":25,"rate":4.6,"extra":300,"payment":1819,"interest":161300,"years":21}]
state picked=[1,2]
state sel=0
App
  Sidebar
    Brand "Hearthloan" icon="home"
    NavItem "Planner" icon="calculate" selected={tab=="planner"} onTap={go planner}
    NavItem "Repayment table" icon="table_chart" selected={tab=="table"} onTap={go repayment-table}
    NavItem "Scenarios" icon="compare" badge="{length(scenarios)}" selected={tab=="scenarios"} onTap={go scenarios}
    Spacer grow
    Text.muted "Estimates only. Not a mortgage offer."
  Main
    Nav
      Heading "Saved scenarios" level=1
      Button "New scenario" icon="add" onTap={go planner}
    Column pad=24 gap=16 scroll
      Text.muted "Tick two or more to compare them side by side."
      Row gap=8
        each s in scenarios
          FilterChip "{s.name}" selected={contains(picked,s.id)} onTap={picked=contains(picked,s.id)?without(picked,s.id):append(picked,s.id)}
      Row gap=16 top scroll
        each s in filter(scenarios,x=>contains(picked,x.id))
          Card w=300 gap=12
            Row between
              Heading "{s.name}" level=3
              IconButton "more_vert" tooltip="Options" onTap={sel=s.id; open scenarioMenu}
            Text "{money(s.payment,'$',0)}" size=40 weight=700
            Text.caption "per month"
            Divider
            Row between
              Text.muted "Price"
              Text.strong "{money(s.price,'$',0)}"
            Row between
              Text.muted "Deposit"
              Text.strong "{money(s.deposit,'$',0)}"
            Row between
              Text.muted "Rate and term"
              Text.strong "{s.rate}% / {s.term} yrs"
            Row between
              Text.muted "Overpayment"
              Text.strong "{money(s.extra,'$',0)}"
            Row between
              Text.muted "Total interest"
              Text.strong "{money(s.interest,'$',0)}"
            Row between
              Text.muted "Mortgage-free in"
              Text.strong "{s.years} yrs"
            if s.interest==min(s.interest,min(first(scenarios).interest,last(scenarios).interest))
              Badge.success "Lowest interest"
      if length(picked)>=2
        Card gap=12
          Heading "Interest compared" level=3
          BarChart values={map(filter(scenarios,x=>contains(picked,x.id)),x=>x.interest)} labels={map(filter(scenarios,x=>contains(picked,x.id)),x=>x.name)} h=180
  Menu#scenarioMenu
    MenuItem "Load into planner" icon="open_in_new" onTap={emit loadScenario sel; go planner}
    MenuItem "Delete" icon="delete" onTap={scenarios=removeWhere(scenarios,x=>x.id==sel); picked=without(picked,sel); toast "Scenario deleted"}
