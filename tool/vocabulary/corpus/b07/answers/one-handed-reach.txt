### Note
state hand="right"
state text="Oat milk, lemons, basil, fresh pasta"
state title="Groceries for Sunday"
state saved=false
App onEdgeSwipeRight={go notes} onEdgeSwipeLeft={go reach}
  Main
    Column pad=16 gap=12 scroll
      Text.caption "Edited today, 9:41"
      Input "Title" bind=title
      Input "Note" multiline bind=text h=360
    FloatingActions side={hand} bottom
      IconButton "check" tooltip="Save" onTap={saved=true; toast "Saved"}
      IconButton "format_list_bulleted" tooltip="Checklist" onTap={text=text + "\n- "}
      IconButton "push_pin" tooltip="Pin" onTap={toast "Pinned"}
      IconButton "delete" tooltip="Delete" onTap={open confirmDelete}
      IconButton "swap_horiz" tooltip="Move buttons" onTap={hand = hand=="right" ? "left" : "right"}
    Footer
      Row between pad=8
        IconButton "arrow_back" tooltip="Back" onTap={go notes}
        Segmented bind=hand items=["left","right"]
        IconButton "arrow_forward" tooltip="Forward" onTap={go reach}
  Dialog#confirmDelete "Delete this note?" "It cannot be recovered."
    Row end gap=8
      Button.outline "Cancel" onTap={close}
      Button.destructive "Delete" onTap={toast "Deleted"; close; go notes}
