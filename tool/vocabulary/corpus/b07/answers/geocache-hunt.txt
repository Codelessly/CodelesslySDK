### Compass
state sel="c1"
state dist=340
state heading=0
state bearing=48
state prev=340
state hunting=true
every 1s when hunting onTick={prev=dist; dist=max(0,dist-12); heading=(heading+7)%360; bearing=(bearing+2)%360}
Page
  Nav
    Row gap=8 middle
      IconButton "arrow_back" tooltip="Back to map" onTap={go map}
      Heading "Old Mill Stash" level=3
    Button.ghost "I found it" onTap={go log}
  Section center
    Column center gap=20 maxw=420
      Compass heading={heading} bearing={bearing} size=280
      Text "{dist} m" size=64 weight=700
      Text.muted "{dist < prev ? 'Getting closer' : 'Moving away'}"
      if dist > 200
        Badge.info "Cold: keep walking toward the north-east"
      if dist <= 200 && dist > 60
        Badge.warning "Warm: look around the old mill"
      if dist <= 60
        Badge.success "Hot: it is within a few steps"
        animate=pulse
      Progress {clamp(1-dist/340,0,1)}
      Row gap=12
        Button.outline "Pause" visible={hunting} onTap={hunting=false}
        Button "Resume" visible={!hunting} onTap={hunting=true}
        Button.outline "Cache details" onTap={go cache}

### Log
state result="found"
state note=""
state photos=[]
state date=today()
state logs=[]
state moved=false
state tb=""
Page
  Nav
    Row gap=8 middle
      IconButton "close" tooltip="Cancel" onTap={go cache}
      Heading "Log Old Mill Stash" level=3
  Section narrow
    Column gap=16
      SegmentedButton items=["Found it","Did not find it"] @result
      DatePicker "Date" @date
      Input "Your log" "What did you see out there?" multiline @note
      Toggle "Moved a trackable" @moved
      if moved
        Select "Trackable" items=["Copper Fox","Paper Boat","Lucky Dice"] @tb
      Row gap=8
        each p in photos
          Image "{p}" h=80 w=80
        Button.outline "Add photo" icon="add_a_photo" onTap={pick photo}
      Row gap=12
        Button.outline "Cancel" onTap={go cache}
        Button "Submit log" onTap={note=="" ? toast "Write a line first" : logs=append(logs,{"user":"Maya Reyes","result":result,"text":note,"date":date}); emit logCache; toast "Log posted"; open done}
  Dialog#done "Log posted" "Thanks, Maya."
    Button "Back to map" onTap={close; go map}
