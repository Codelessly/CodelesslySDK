### Session
state phase="hold"
state secs=10
state reps=0
state set=1
state rest=30
state running=true
state pain=0
every 1s when running && phase=="hold" onTick={secs=max(0,secs-1)}
every 1s when running && phase=="rest" onTick={rest=max(0,rest-1)}
App
  Main
    Nav
      Heading "Quad sets" level=1
      Row gap=8
        Text.muted "Set {set} of 3"
        Button.ghost "End" onTap={open rate}
    Column pad=24 gap=16 center
      if phase=="hold"
        Text "{secs}" size=96 weight=700 center
        Text.muted "Hold the squeeze" center
        Card gap=4
          Row gap=8 middle
            Icon "record_voice_over"
            Text.muted "Voice cue: Keep breathing. Hold steady, you are doing well."
        Text.strong "Reps {reps} of 10"
        Button "Rep done" size=large onTap={reps=reps+1; secs=10; reps>=10 ? {phase="rest"; rest=30} : toast "Rep counted"}
      if phase=="rest"
        Text "{rest}" size=96 weight=700 center
        Text.muted "Rest" center
        Row gap=8
          Button.outline "Skip rest" onTap={phase="hold"; reps=0; set=set+1; secs=10; set>=3 ? open rate : toast "Next set"}
          Button.ghost "+10 s" onTap={rest=rest+10}
      Row gap=8
        Button.outline "Pause" onTap={running=!running}
  Sheet#rate "Pain after"
    Text.muted "How did that feel?"
    Row gap=8
      ChoiceChip "0 none" selected={pain==0} onTap={pain=0}
      ChoiceChip "3 mild" selected={pain==3} onTap={pain=3}
      ChoiceChip "6 strong" selected={pain==6} onTap={pain=6}
      ChoiceChip "9 severe" selected={pain==9} onTap={pain=9}
    Slider @pain min=0 max=10
    Button "Save" onTap={emit painRated pain; toast "Logged"; close; go progress}
