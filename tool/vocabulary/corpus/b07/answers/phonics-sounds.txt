          Text.heading "{upper(letters[at].l)} is for {letters[at].word}"

            SoundWave idle bars=24 h=60 color=#FFB8C4E0

        IconButton "chevron_right" tooltip="Next letter" onTap={at=min(length(letters)-1,at+1)}

    state b=6
    state guess=""
    Text.heading "{a} + {b} = ?"

### Catch the sound
state target={"l":"s","word":"sun"}
state options=[{"id":1,"word":"sun","pic":"https://img.soundbox.app/sun.png","starts":"s"},{"id":2,"word":"moon","pic":"https://img.soundbox.app/moon.png","starts":"m"},{"id":3,"word":"snake","pic":"https://img.soundbox.app/snake.png","starts":"s"}]
state round=1
state caught=0
state stars=7
state miss=false
App
  Sidebar
    Brand "Soundbox" icon="graphic_eq"
    NavItem "Letters" icon="abc" onTap={go letters}
    NavItem "Catch the sound" icon="sports_esports" selected onTap={go game}
    NavItem "Blend a word" icon="join_inner" onTap={go blend}
    Spacer grow
    NavItem "Parent corner" icon="lock" onTap={open gate}
  Main
    Nav
      Heading "Catch the sound" level=1
      Row gap=8
        Text.muted "Round {round} of 5"
        Progress {round/5} w=120
    Column pad=24 gap=20 center
      Row gap=16 middle
        Text.heading "Which one starts with"
        Text "{target.l}" size=72 weight=800 color=#FF3B6FE0
        IconButton "volume_up" tooltip="Hear the sound" onTap={emit speak target.l}
      Row gap=32 center h=260
        each o in options
          Card drag=pic(o.id) animate=slide-up stagger=300 gap=8 center
            Image "{o.pic}" w=140 h=140 radius=16
            Text.strong "{o.word}"
      Box#basket drop=pic fill=#FFFFF3D6 radius=32 pad=24 center h=140 w=320 onDrop={
        find(options,x=>x.id==result.data).starts==target.l ? {caught=caught+1; stars=stars+1; round=round+1; miss=false; toast "Yes! {target.word}"; emit correct} : {miss=true; emit tryAgain}
      }
        Icon "shopping_basket" size=56 color=#FFB26A00
        Text.muted "Drop it in the basket"
        if caught>0
          Text.success "{caught} caught"
      if miss
        Text.danger "Listen again: {target.l}" animate=shake
  Dialog#gate "Grown-ups only" "Answer the sum to open the parent corner"
    state guess=""
    Text.heading "7 + 6 = ?"
    Input "Your answer" @guess type=number
    Button "Open" onTap={guess==13 ? go parent : toast "Not quite, try again"}

### Blend a word
state tiles=["c","a","t"]
state merged=false
state stars=7
state shelf=[{"w":"sun","got":true},{"w":"hat","got":true},{"w":"dog","got":true},{"w":"pig","got":true},{"w":"cup","got":true},{"w":"bus","got":true},{"w":"cat","got":false},{"w":"map","got":false}]
state spread=0
App
  Sidebar
    Brand "Soundbox" icon="graphic_eq"
    NavItem "Letters" icon="abc" onTap={go letters}
    NavItem "Catch the sound" icon="sports_esports" onTap={go game}
    NavItem "Blend a word" icon="join_inner" selected onTap={go blend}
    Spacer grow
    NavItem "Parent corner" icon="lock" onTap={open gate}
  Main
    Nav
      Heading "Blend a word" level=1
      Text.muted "{stars} stars"
    Column pad=24 gap=24 center
      Text.heading "Slide the sounds together"
      if !merged
        Row gap=0 center
          each t in tiles
            Box fill=#FF3B6FE0 radius=20 pad=24 w=120 h=140 center middle animate=scale
              Text "{t}" size=80 weight=800 color=#FFFFFFFF
      if merged
        Box fill=#FF2E9E5B radius=28 pad=24 center animate=pop
          Text "{join(tiles,"")}" size=96 weight=800 color=#FFFFFFFF
      Slider @spread min=0 max=100 onChange={result>=90 ? {merged=true; emit speak join(tiles,""); stars=stars+1; toast "cat!"} : {merged=false}}
      Text.muted "Slide to push the tiles together"
      Row gap=12
        Button.outline "Hear each sound" icon="hearing" onTap={emit speakEach tiles}
        Button "Again" icon="replay" onTap={merged=false; spread=0}
      Divider
      Text.heading "My star shelf"
      Card gap=12
        Grid cols=4 gap=12
          each s in shelf
            Column center gap=4
              Icon "star" size=48 color={s.got ? #FFF5B301 : #FFD5D5D5}
              Text.caption "{s.w}"
  Dialog#gate "Grown-ups only" "Answer the sum to open the parent corner"
    state guess=""
    Text.heading "9 + 4 = ?"
    Input "Your answer" @guess type=number
    Button "Open" onTap={guess==13 ? go parent : toast "Not quite, try again"}
