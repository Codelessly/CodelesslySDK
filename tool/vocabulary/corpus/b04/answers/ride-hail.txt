### Rate and tip
state rating=0
state tip=2
state note=""
state custom=""
Page
  Section narrow center
    Column gap=16 center
      Avatar "Marcus Bell" size=72
      Heading "How was your ride with Marcus?" level=1
      Text.muted "Trip total {money(16.90,'$',2)}"
      Rating bind=rating max=5 size=40
      if rating>0 && rating<4
        Input "What went wrong?" multiline bind=note
      if rating>=4
        Row gap=8
          ChoiceChip "Great conversation" onTap={emit tag "conversation"}
          ChoiceChip "Clean car" onTap={emit tag "clean"}
          ChoiceChip "Smooth driving" onTap={emit tag "smooth"}
      Text.sectionTitle "Add a tip"
      Row gap=8 center
        ChoiceChip "No tip" selected={tip==0} onTap={tip=0}
        ChoiceChip "$2" selected={tip==2} onTap={tip=2}
        ChoiceChip "$4" selected={tip==4} onTap={tip=4}
        ChoiceChip "$6" selected={tip==6} onTap={tip=6}
        ChoiceChip "Other" selected={tip==-1} onTap={tip=-1}
      if tip==-1
        Input "Custom tip" type=number bind=custom
      Text.strong "Total with tip: {money(16.90+(tip==-1 ? coalesce(custom,0) : tip),'$',2)}"
      Button "Submit" size=large disabled={rating==0} onTap={emit rated {"rating":rating,"tip":tip,"note":note}; toast "Thanks for riding"; go where-to}
      Button.ghost "Skip" onTap={go where-to}
