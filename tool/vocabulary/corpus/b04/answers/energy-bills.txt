        DigitInput @digits count=5 type=number

      Card gap=12
        Text.panelTitle "Add a photo (optional)"
        if photo == ""
          Button.outline "Take or upload a photo" icon="photo_camera" onTap={photo=pickImage()}

        if photo != ""
          Image "{photo}" h=160
          Button.ghost "Remove photo" onTap={photo=""}
      Row end gap=8
        Button.ghost "Clear" onTap={digits=["","","","",""]; photo=""}
        Button "Submit reading" onTap={all(digits,d=>d!="") ? {sent=true; emit submitReading; toast "Reading sent"} : toast "Enter all five digits"}
      if sent
        Card gap=8
          Badge.success "Received"
          Text "Thank you. Your next bill will use this reading."
