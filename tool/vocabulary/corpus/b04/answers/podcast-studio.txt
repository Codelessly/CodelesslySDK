        Button "Record" icon="fiber_manual_record" size=large hidden={recording} onTap={recording=true; elapsed=0; markers=[]}

      Card gap=8

        if isEmpty(markers)

  Sidebar
    Brand "Take Two" icon="mic"
    NavItem "Studio" icon="graphic_eq" onTap={go studio}
    NavItem "Tracks" icon="library_music" badge="4" selected={true} onTap={go tracks}
    NavItem "Soundboard" icon="grid_view" onTap={go soundboard}
    Spacer grow
    Text.muted "Episode 42: The Long Tail"
  Main
    Nav
      Heading "Tracks" level=1
      Row gap=8
        Button.outline "Upload all" icon="cloud_upload" onTap={tracks=map(tracks,t=>with(t,"up",1.0)); toast "Uploading 4 tracks"}
        Button "Export mix" icon="download" onTap={emit exportMix}
    Column pad=24 gap=16 scroll
      SegmentedButton @tab items=["all","uploading","done"]
      each t in tracks
        if tab=="all" || (tab=="done" && t.up==1.0) || (tab=="uploading" && t.up<1.0)
          Card gap=12
            Row between middle
              Row gap=12 middle
                IconButton {playing==t.id ? "pause" : "play_arrow"} tooltip="Play track" onTap={playing=(playing==t.id ? 0 : t.id)}
                Column
                  Text.strong "{t.name}"
                  Text.muted "{t.size}"
              Badge.success "Uploaded" hidden={t.up<1.0}
              Badge.info "{round(t.up*100,0)}%" hidden={t.up>=1.0}
            Waveform values={t.wave} h=56 progress={playing==t.id ? 0.4 : 0}
            Row gap=12 middle
              Progress {t.up} grow
              Button.ghost "Retry" icon="refresh" hidden={t.up>=0.3 || t.up==0} onTap={tracks=updateWhere(tracks,x=>x.id==t.id,x=>with(x,"up",0.5))}
              Button.ghost "Upload" hidden={t.up!=0} onTap={tracks=updateWhere(tracks,x=>x.id==t.id,x=>with(x,"up",0.1))}
