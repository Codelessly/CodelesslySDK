### Call
state name="Priya Nair"
state mic=true
state cam=true
state sharing=false
state handUp=false
state panel="chat"
state speaker=2
state idle=0
state draft=""
state reactions=[]
state participants=[{"id":1,"name":"Priya Nair","muted":false,"hand":false,"you":true},{"id":2,"name":"Daniel Okafor","muted":false,"hand":false,"you":false},{"id":3,"name":"Mei Lin","muted":true,"hand":false,"you":false},{"id":4,"name":"Tomas Berg","muted":false,"hand":true,"you":false},{"id":5,"name":"Aisha Rahman","muted":true,"hand":false,"you":false}]
state messages=[{"from":"Daniel Okafor","text":"Sharing the new onboarding flow in a sec","time":"10:32"},{"from":"Mei Lin","text":"Can you zoom in on step three?","time":"10:34"},{"from":"Tomas Berg","text":"Link to the Figma file: roundtable.example/f/ob3","time":"10:35"}]
every 1s onTick={idle=idle+1}
every 1s onTick={reactions=filter(reactions,r=>r.age<3)}
App
  Main
    Nav
      Row gap=8
        Heading "Weekly design review" level=3
        Badge.error "Recording"
      Row gap=8
        Text.muted "{duration(1245)}"
        IconButton "chat" tooltip="Chat" onTap={panel="chat"}
        IconButton "group" tooltip="People" onTap={panel="people"}
    Column pad=16 gap=12 grow onPointerMove={idle=0}
      if !sharing
        Grid cols={length(participants)<=1 ? 1 : length(participants)<=4 ? 2 : 3} gap=12 grow
          each p in participants
            ParticipantTile name={p.name} muted={p.muted} hand={p.hand} speaking={p.id==speaker} video={p.id!=3} onTap={speaker=p.id} menu=tileMenu
      if sharing
        ScreenShare "Onboarding flow v3.fig" presenter="Daniel Okafor" grow
        Row scroll gap=8 h=110
          each p in participants
            ParticipantTile name={p.name} muted={p.muted} hand={p.hand} speaking={p.id==speaker} video={p.id!=3} w=160
      Stack
        each r in reactions
          Text "{r.emoji}" size=40 animate=slide-up duration=2500 right={r.x}
    Footer visible={idle<5}
      Row center middle gap=12 pad=12
        IconButton {mic ? "mic" : "mic_off"} tooltip={mic ? "Mute" : "Unmute"} selected={!mic} onTap={mic=!mic; participants=updateWhere(participants,p=>p.you,p=>with(p,"muted",mic))}
        IconButton {cam ? "videocam" : "videocam_off"} tooltip={cam ? "Stop video" : "Start video"} selected={!cam} onTap={cam=!cam}
        IconButton "present_to_all" tooltip={sharing ? "Stop sharing" : "Share screen"} selected={sharing} onTap={sharing=!sharing; sharing ? toast "You are sharing your screen" : toast "Sharing stopped"}
        IconButton "front_hand" tooltip={handUp ? "Lower hand" : "Raise hand"} selected={handUp} onTap={handUp=!handUp; participants=updateWhere(participants,p=>p.you,p=>with(p,"hand",handUp))}
        IconButton "add_reaction" tooltip="Reactions" onTap={open reactionPicker}
        IconButton "more_vert" tooltip="More" onTap={open moreMenu}
        Button.destructive "Leave" icon="call_end" onTap={open leave}
  Pane w=320 pad=16 gap=12
    SegmentedButton bind=panel items=["chat","people"]
    if panel=="chat"
      Column grow gap=8 scroll
        each m in messages
          ChatBubble from={m.from} text={m.text} time={m.time} own={m.from==name}
      Row gap=8
        Input "Message" bind=draft grow
        IconButton "send" tooltip="Send" onTap={draft!="" ? {messages=append(messages,{"from":name,"text":draft,"time":"10:41"}); draft=""} : toast "Type a message first"}
    if panel=="people"
      Text.caption "{length(participants)} in this call"
      Column grow gap=4 scroll
        each p in participants
          ListTile "{p.name}{p.you ? ' (you)' : ''}" subtitle={p.hand ? "Hand raised" : ""}
            Avatar "{p.name}" size=32
            Icon {p.muted ? "mic_off" : "mic"}
      Button.outline "Invite people" icon="person_add" onTap={open invite}
  Popover#reactionPicker
    Row gap=8
      Button.ghost "👍" onTap={reactions=append(reactions,{"emoji":"👍","x":random(200),"age":0}); close}
      Button.ghost "👏" onTap={reactions=append(reactions,{"emoji":"👏","x":random(200),"age":0}); close}
      Button.ghost "❤️" onTap={reactions=append(reactions,{"emoji":"❤️","x":random(200),"age":0}); close}
      Button.ghost "😂" onTap={reactions=append(reactions,{"emoji":"😂","x":random(200),"age":0}); close}
      Button.ghost "🎉" onTap={reactions=append(reactions,{"emoji":"🎉","x":random(200),"age":0}); close}
  Menu#moreMenu above
    MenuItem "Copy meeting link" icon="link" onTap={copy "https://roundtable.example/j/wdr-4821"; toast "Link copied"}
    MenuItem "Start recording" icon="fiber_manual_record" onTap={toast "Recording started"}
    MenuItem "Settings" icon="settings" onTap={open settings}
  Menu#tileMenu
    MenuItem "Pin" icon="push_pin" onTap={speaker=speaker}
    MenuItem "Mute for everyone" icon="mic_off" onTap={emit muteParticipant}
  Dialog#settings "Audio and video"
    Select "Microphone" bind=micDevice items=["MacBook Pro Microphone","AirPods Pro"]
    Select "Camera" bind=camDevice items=["FaceTime HD Camera","Logitech C920"]
    Toggle "Noise suppression" bind=noise
    Button "Done" onTap={close}
  Dialog#invite "Invite people"
    Input "Email addresses" "name@company.com" bind=inviteEmail
    Row gap=8
      Button.outline "Copy link" onTap={copy "https://roundtable.example/j/wdr-4821"; toast "Link copied"}
      Button "Send invite" onTap={emit invite inviteEmail; toast "Invite sent"; close}
  Dialog#leave "Leave the call?" "The others will stay in the call."
    Row gap=8 end
      Button.outline "Stay" onTap={close}
      Button.destructive "Leave" onTap={emit leave; go prejoin}
