  Sidebar
    Brand "Shelf" icon="folder_open"

    NavItem "Trash" icon="delete" badge="{length(trash)}" selected
    Spacer grow

      Text.muted "Items in trash are deleted forever after 30 days."
      if isEmpty(trash)
        Column center middle gap=8 pad=48
          Icon "delete_outline" size=48
          Text.muted "Trash is empty"
      each t in trash
        Row middle gap=12 pad=8 border-bottom=#FFE0E0E0
          Icon "description"
          Text grow "{t.name}"
          Text.muted "{t.size} KB"
          Button.ghost "Restore" onTap={files=append(files,t); trash=without(trash,t)}
          IconButton "delete_forever" tooltip="Delete forever" onTap={trash=without(trash,t)}
  Dialog#emptyConfirm "Empty trash?" "This cannot be undone."
    Row end gap=8
      Button.outline "Cancel" onTap={close}
      Button.destructive "Delete forever" onTap={trash=[]; close; toast "Trash emptied"}

### Search
state q=""
state kind="all"
state when="any"
state hits=[]
state after=""
App
  Sidebar
    Brand "Shelf" icon="folder_open"
    NavItem "My files" icon="folder" onTap={go files}
    NavItem "Search" icon="search" selected
    NavItem "Trash" icon="delete" onTap={go trash}
  Main
    Nav
      Input "Search files" "Search all of Shelf" @q icon="search" w=420 onChange={q=result}
      Button.ghost "Clear" onTap={q=""; kind="all"; when="any"}
    Column pad=24 gap=16
      Row gap=8 wrap
        ChoiceChip "All" selected={kind=="all"} onTap={kind="all"}
        ChoiceChip "Images" selected={kind=="image"} onTap={kind="image"}
        ChoiceChip "PDFs" selected={kind=="pdf"} onTap={kind="pdf"}
        ChoiceChip "Sheets" selected={kind=="sheet"} onTap={kind="sheet"}
        ChoiceChip "Slides" selected={kind=="slides"} onTap={kind="slides"}
        Select "Modified" items=["any","today","last 7 days","last 30 days"] @when
        DatePicker "After" @after
      Text.muted "{length(filter(files, x=>(kind=='all' or x.type==kind) and contains(lower(x.name), lower(q))))} results for \"{q}\""
      each f in filter(files, x=>(kind=="all" or x.type==kind) and contains(lower(x.name), lower(q)) and (when=="any" or x.modified >= (when=="today" ? today() : when=="last 7 days" ? formatDate(today()-7,"yyyy-MM-dd") : formatDate(today()-30,"yyyy-MM-dd"))))
        ListTile "{f.name}" subtitle="in {f.folder} · {formatDate(f.modified,'MMM d')}" icon="description" onTap={folder=f.folder; sel=[f.id]; go files}
