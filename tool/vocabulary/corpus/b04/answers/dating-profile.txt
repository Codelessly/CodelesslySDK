### Prompts
state prompts=["A perfect Sunday looks like","I'm weirdly good at","Two truths and a lie","The way to win me over is","My most controversial opinion"]
state answers=[{"q":"A perfect Sunday looks like","a":"Farmers market, a long run, and cooking too much pasta."}]
state pick=""
state draft=""
App
  Sidebar
    Brand "Kindred" icon="favorite"
    NavItem "Photos and basics" icon="photo_library" onTap={go photos-and-basics}
    NavItem "Prompts" icon="chat_bubble" selected={true}
    NavItem "Interests" icon="interests" onTap={go interests}
    NavItem "Preview" icon="visibility" onTap={go preview}
  Main
    Nav
      Heading "Prompts" level=1
      Button "Add prompt" disabled={length(answers)>=3} onTap={open pickPrompt}
    Column pad=24 gap=16 scroll
      Text.muted "Choose up to three prompts and answer each in 140 characters or fewer."
      each a in answers
        Card gap=8
          Row between
            Text.strong "{a.q}"
            IconButton "more_vert" tooltip="Options" onTap={pick=a.q; open promptMenu}
          Text "{a.a}"
          Text.caption "{length(a.a)} / 140"
      if isEmpty(answers)
        Card gap=8
          Text.muted "No prompts yet. Profiles with prompts get more first messages."
  Sheet#pickPrompt "Pick a prompt"
    each p in prompts
      ListTile "{p}" icon="chevron_right" onTap={pick=p; draft=""; close; open answerSheet}
  Sheet#answerSheet "{pick}"
    Input "Your answer" bind=draft multiline maxLength=140
    Row between
      Text.caption "{length(draft)} / 140"
      Button "Save answer" disabled={trim(draft)=="" || length(draft)>140} onTap={answers=append(removeWhere(answers,x=>x.q==pick),{"q":pick,"a":draft}); close; toast "Prompt saved"}
  Menu#promptMenu end
    MenuItem "Edit answer" icon="edit" onTap={draft=find(answers,x=>x.q==pick).a; open answerSheet}
    MenuItem "Remove" icon="delete" onTap={answers=removeWhere(answers,x=>x.q==pick)}

### Interests
state all=["Hiking","Coffee","Climbing","Board games","Film photography","Salsa","Vinyl","Cooking","Yoga","Poetry","Trail running","Pottery","Karaoke","Travel","Dogs","Cycling","Astronomy","Baking","Museums","Surfing"]
state interests=["Hiking","Coffee"]
state q=""
App
  Sidebar
    Brand "Kindred" icon="favorite"
    NavItem "Photos and basics" icon="photo_library" onTap={go photos-and-basics}
    NavItem "Prompts" icon="chat_bubble" onTap={go prompts}
    NavItem "Interests" icon="interests" selected={true}
    NavItem "Preview" icon="visibility" onTap={go preview}
  Main
    Nav
      Heading "Interests" level=1
      Text.muted "{length(interests)} of 5 chosen"
    Column pad=24 gap=16 scroll
      Card gap=12
        Text.strong "Your picks"
        Row gap=8
          each i in interests
            Chip "{i}" icon="close" onTap={interests=without(interests,i)}
          if isEmpty(interests)
            Text.muted "Nothing yet"
      Card gap=12
        Input "Search interests" bind=q
        Row gap=8 wrap
          each i in filter(all,x=>contains(lower(x),lower(q)))
            FilterChip "{i}" selected={contains(interests,i)} onTap={contains(interests,i) ? interests=without(interests,i) : (length(interests)<5 ? interests=append(interests,i) : toast "Up to five interests")}
        if length(filter(all,x=>contains(lower(x),lower(q))))==0
          Text.muted "No interests match {q}"
