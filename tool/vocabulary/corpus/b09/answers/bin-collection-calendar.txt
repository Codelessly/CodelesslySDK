### Report a missed bin
state binType="Recycling"
state when=""
state note=""
state photo=""
state sent=false
App
  Sidebar
    Brand "Binday" icon="delete"
    NavItem "Calendar" icon="calendar_month" onTap={go home}
    NavItem "What goes where" icon="recycling" onTap={go guide}
    NavItem "Report a missed bin" icon="report" selected={true} onTap={go report}
    NavItem "Settings" icon="settings" onTap={go settings}
  Main
    Nav
      Heading "Report a missed bin" level=1
    Column pad=24 gap=16 maxw=640 scroll
      if sent
        Card gap=8
          Icon "check_circle" color=#FF2E7D32 size=40
          Text.heading "Report sent"
          Text.muted "Reference BD-20418. The crew returns within two working days."
          Button.outline "Back to calendar" onTap={go home}
      if !sent
        Card gap=12
          Text.fieldLabel "Which bin?"
          Row gap=8
            ChoiceChip "General" selected={binType=="General"} onTap={binType="General"}
            ChoiceChip "Recycling" selected={binType=="Recycling"} onTap={binType="Recycling"}
            ChoiceChip "Garden" selected={binType=="Garden"} onTap={binType="Garden"}
            ChoiceChip "Glass" selected={binType=="Glass"} onTap={binType="Glass"}
          DatePicker "Collection day" bind=when
          Input "What happened?" "Bin left out by 7 am, not emptied" bind=note multiline
          Row gap=12 middle
            Button.outline "Add a photo" icon="photo_camera" onTap={emit pickPhoto}
            if photo!=""
              Image "{photo}" w=64 h=64 radius=8
          Button "Send report" onTap={when!="" ? {emit report binType; sent=true; toast "Report sent"} : toast "Choose the collection day"}
