### Handbrake and lift test
state secs=0
state running=false
state handbrake=""
state lift=""
state tab="test"
every 1s when running onTick={secs=secs+1; secs>=30 ? running=false : }
App
  Sidebar
    Brand "Prestart" icon="forklift"
    NavItem "Walk-round" icon="fact_check" onTap={go walk-round}
    NavItem "Hours and licence" icon="badge" onTap={go hours-licence}
    NavItem "Handbrake and lift test" icon="timer" selected={tab=="test"} onTap={tab="test"}
    NavItem "Signed checks" icon="history" onTap={go signed-checks}
  Main
    Nav
      Heading "Handbrake and lift test" level=1
    Column pad=24 gap=16
      Grid cols=2 gap=24
        Card gap=12 center
          Text.strong "Hold test, 30 seconds"
          Text.muted "Raise the forks 1 m with a load. Start the timer and watch for any drop or creep."
          Ring {secs/30} size=160
          Text "{duration(secs)}" size=40 weight=700
          Row gap=8
            Button "{running ? 'Pause' : 'Start timer'}" onTap={running=!running}
            Button.outline "Reset" onTap={running=false; secs=0}
        Card gap=16
          Text.strong "Results"
          SegmentedButton "Handbrake holds on the ramp" items=["Pass","Fail"] bind=handbrake
          SegmentedButton "Forks hold with no drift" items=["Pass","Fail"] bind=lift
          Text.success "Both tests passed" hidden={handbrake!="Pass" || lift!="Pass"}
          Text.danger "A failed test tags the truck do not use" hidden={handbrake!="Fail" && lift!="Fail"}
      Row gap=8
        Spacer grow
        Button "Sign and save check" onTap={isEmpty(handbrake) || isEmpty(lift) ? toast "Record both results" : (handbrake=="Fail" || lift=="Fail" ? go lockout : open sign)}
  Dialog#sign "Sign this check" "Operator Dale Whitcombe, FLT-07"
    SignaturePad "Sign here" bind=signature
    Row gap=8
      Button.ghost "Cancel" onTap={close}
      Spacer grow
      Button "Save" onTap={isEmpty(signature) ? toast "Please sign" : {emit saveCheck signature; toast "Check signed and saved"; close; go signed-checks}}
