### Run
state env="staging"
state running=false
state done=0
state total=41200
state status="idle"
state errorText=""
state log=[]
every 500 when running onTick={done=min(total,done+3400); log=append(log,"batch committed, " + done + " rows"); status=done>=total ? "applied" : (done>=27000 && env=="staging" ? "failed" : "running"); running=status=="running"; errorText=status=="failed" ? "ERROR 23502: column \"status\" of relation \"orders\" contains null values (batch 9, row id 27114)" : ""}
App
  Sidebar
    Brand "Shiftlog" icon="storage"
    NavItem "Migrations" icon="list" onTap={go migrations}
    NavItem "Plan" icon="rule" onTap={go plan}
    NavItem "Run" icon="play_circle" selected={true}
    NavItem "Schema diff" icon="difference" onTap={go schema-diff}
  Main
    Nav
      Heading "Run 0043" level=1
      Row gap=8
        Button "Start" hide=phone onTap={done=0; log=[]; errorText=""; status="running"; running=true}
        Button.outline "Pause" onTap={running=false; status="paused"}
        Button.destructive "Cancel" onTap={running=false; status="idle"; done=0}
    Column pad=24 gap=16
      Card gap=12
        Row between middle
          Text.heading3 "add_not_null_to_orders_status on {env}"
          if status=="running"
            Badge.info "running"
          if status=="applied"
            Badge.success "applied"
          if status=="failed"
            Badge.error "failed"
          if status=="paused"
            Badge.warning "paused"
          if status=="idle"
            Badge.neutral "ready"
        Progress {done/total}
        Row between
          Text.strong "{done} of {total} rows"
          Text.muted "{round(done/total*100,0)}%"
      if status=="failed"
        Card gap=12 fill=#FFFDECEA
          Row gap=8 middle
            Icon "error"
            Text.strong "Migration failed at step 2"
          Text.mono "{errorText}"
          Text.muted "The transaction for batch 9 was rolled back. Batches 1 to 8 are committed."
          Row gap=8
            Button.destructive "Roll back to 0042" onTap={done=0; status="rolled back"; log=append(log,"rollback: restored 0043 changes"); toast "Rolled back 0043"}
            Button.outline "Retry from batch 9" onTap={status="running"; running=true; errorText=""}
            Button.ghost "Copy error" onTap={copy errorText; toast "Copied"}
      if status=="rolled back"
        Card gap=8 fill=#FFFFF4E5
          Text.strong "Rolled back"
          Text.muted "orders is back to its state before 0043."
      Card gap=8
        Text.heading3 "Log"
        if isEmpty(log)
          Text.muted "Nothing yet."
        each line in log
          Text.mono "{line}"
