          Card gap=12
            Text.panelTitle "New reading"

              Input "Start volume (mL)" bind=startV type=number grow

              Text.heading "{fixed(max(0,endV-startV),2)} mL"
              Spacer grow

          Card gap=12
            Text.panelTitle "Indicator"

              ChoiceChip "Bromothymol blue" selected={indicator=="Bromothymol blue"} onTap={indicator="Bromothymol blue"; endpointColor=#FF1E88E5}
            Row gap=12 middle
              Swatch color={endpointColor} size=40 radius=20
              Text.body "Endpoint colour of {indicator}"
      Card gap=12
        Row between middle
          Text.panelTitle "Repeated trials"
          Row gap=8 middle
            Text.fieldLabel "Spread limit (mL)"

        Row gap=8 border-bottom=#FFCCCCCC
          Text.fieldLabel "#" w=40
          Text.fieldLabel "Start" w=80
          Text.fieldLabel "End" w=80
          Text.fieldLabel "Used" grow
          Text.fieldLabel "" w=48
        each t in trials
          Row gap=8 middle
            Text "{t.id}" w=40
            Text "{fixed(t.start,2)}" w=80
            Text "{fixed(t.end,2)}" w=80
            Text.strong "{fixed(t.end-t.start,2)} mL" grow
            IconButton "delete" tooltip="Remove trial" onTap={trials=removeWhere(trials,x=>x.id==t.id)}
        Divider
        Row between middle
          Text.body "Spread (highest minus lowest)"
          Row gap=8 middle
            Text.strong "{fixed(last(sortBy(map(trials,t=>t.end-t.start),v=>v))-first(sortBy(map(trials,t=>t.end-t.start),v=>v)),2)} mL"
            if last(sortBy(map(trials,t=>t.end-t.start),v=>v))-first(sortBy(map(trials,t=>t.end-t.start),v=>v))>limit
              Badge.error "Outside limit: repeat a run"
            if last(sortBy(map(trials,t=>t.end-t.start),v=>v))-first(sortBy(map(trials,t=>t.end-t.start),v=>v))<=limit
              Badge.success "Within limit"
  Dialog#addTrial "Add trial" "Enter the burette readings"
    Input "Start volume (mL)" bind=startV type=number
    Input "End volume (mL)" bind=endV type=number
    Text.muted "Used: {fixed(max(0,endV-startV),2)} mL"
    Button "Add" onTap={endV>startV ? {trials=append(trials,{"id":nextId,"start":startV,"end":endV}); nextId=nextId+1; close} : toast "End volume must exceed start"}
  Dialog#confirmClear "Clear all trials?" "This removes every logged run."
    Row gap=8 end
      Button.ghost "Cancel" onTap={close}
      Button.destructive "Clear" onTap={trials=[]; close}

            Input "Volume (mL)" bind=newV type=number

          Card gap=8
            Text.panelTitle "Readings"
            each p in phPoints
              Row between
                Text "{fixed(p.v,2)} mL"
