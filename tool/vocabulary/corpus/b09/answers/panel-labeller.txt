App
  Sidebar
    Brand "Breakerbook" icon="electrical_services"
    NavItem "Panel" icon="developer_board" selected={tab=="panel"} onTap={tab="panel"}
    NavItem "Circuit warnings" icon="warning" badge="{count(breakers,b=>b.amps>find(wireMax,w=>w.gauge==b.gauge).max)}" onTap={open warnings}
    NavItem "Print directory" icon="print" onTap={go print-directory}
    NavItem "Panel photo" icon="photo_camera" onTap={go panel-photo}
    Spacer grow

  Main
    Nav
      Heading "Main panel" level=1
      Row gap=8
        SegmentedButton @view items=["drawing","list"]

    Column pad=24 gap=16 scroll
      Row gap=16 top
        Card grow gap=12
          Text.strong "Distribution board"
          PanelDiagram columns=2 count=12 @sel
            Grid cols=2 gap=8
              each b in breakers
                Box fill=#FFF4F4F2 radius=8 pad=8 gap=4 onTap={sel=b.id} menu=breakerMenu
                  Row between
                    Badge.neutral "{b.id}"
                    Badge.primary "{b.amps} A"
                  Text.strong "{b.spare ? 'SPARE' : b.name}" strike={false}
                  Text.muted "{b.room} / phase {b.phase}"
                  if b.amps>find(wireMax,w=>w.gauge==b.gauge).max
                    Row gap=4
                      Icon "warning" color=#FFD32F2F
                      Text.danger "Wire too small"
        Card w=320 gap=8
          Heading "Load per phase" level=3
          Text.caption "Phase A  {sum(filter(breakers,b=>b.phase=='A'&&!b.spare),b=>b.amps)} A"

          Text.caption "Phase B  {sum(filter(breakers,b=>b.phase=='B'&&!b.spare),b=>b.amps)} A"
          Progress {sum(filter(breakers,b=>b.phase=="B"&&!b.spare),b=>b.amps)/60}
          Text.caption "Phase C  {sum(filter(breakers,b=>b.phase=='C'&&!b.spare),b=>b.amps)} A"
          Progress {sum(filter(breakers,b=>b.phase=="C"&&!b.spare),b=>b.amps)/60}
          Text.muted "Aim for the three bars within 10 A of each other."
    Footer
      Row between
        Text.muted "Breaker {sel} selected"
        Button "Edit breaker {sel}" onTap={open edit}
  Pane w=320 pad=16 gap=12
    Heading "Breaker {sel}" level=3
    Text.muted "{find(breakers,b=>b.id==sel).spare ? 'Spare' : find(breakers,b=>b.id==sel).name}"
    Image "https://example.com/panel/breaker-{sel}.jpg" h=120
    Button.outline "Open photo of panel" onTap={go panel-photo}
Menu#breakerMenu
  MenuItem "Edit" icon="edit" onTap={open edit}
  MenuItem "Mark as spare" icon="bookmark" onTap={breakers=updateWhere(breakers,b=>b.id==sel,b=>with(b,"spare",true))}
  MenuItem "Clear label" icon="backspace" onTap={breakers=updateWhere(breakers,b=>b.id==sel,b=>with(with(b,"name",""),"room",""))}
Dialog#edit "Edit breaker" "Set the rating, circuit name and room"
  Select "Amperage" @editAmps items=[15,20,30,40,50]
  Input "Circuit name" "e.g. Kitchen counters" @editName
  Input "Room" @editRoom
  Select "Wire gauge" @editGauge items=["14 AWG","12 AWG","10 AWG","8 AWG","6 AWG"]
  Select "Phase" @editPhase items=["A","B","C"]
  Toggle "Spare" @editSpare
  Row end gap=8
    Button.ghost "Cancel" onTap={close}
    Button "Save" onTap={breakers=updateWhere(breakers,b=>b.id==sel,b=>with(with(with(with(with(with(b,"amps",editAmps),"name",editName),"room",editRoom),"gauge",editGauge),"phase",editPhase),"spare",editSpare)); toast "Breaker saved"; close}
Sheet#warnings "Circuit warnings"
  each b in filter(breakers,b=>b.amps>find(wireMax,w=>w.gauge==b.gauge).max)
    ListTile "Breaker {b.id}: {b.name}" icon="warning" onTap={sel=b.id; close}
      Text.muted "{b.amps} A breaker on {b.gauge} wire. Use a breaker of {find(wireMax,w=>w.gauge==b.gauge).max} A or less, or a heavier wire."
  if count(breakers,b=>b.amps>find(wireMax,w=>w.gauge==b.gauge).max)==0
    Text.success "Every wire suits its breaker."
