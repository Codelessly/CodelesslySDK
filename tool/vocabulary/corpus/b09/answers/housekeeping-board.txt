state inspRoom="303"
state failNote=""
state result=""
state log=[]
App
  Sidebar
    Brand "Turndown" icon="bed"
    NavItem "Floor board" icon="grid_view" onTap={go floor-board}
    NavItem "Inspections" icon="fact_check" selected={true}
    Spacer grow
    Text.muted "Inspector: Dana Whitfield"
  Main
    Nav
      Heading "Inspect room {inspRoom}" level=1
      Select bind=inspRoom items=["303","308","311"] w=140
    Column pad=24 gap=16 scroll
      Card gap=8
        Row between
          Text.strong "Checklist"
          Text.muted "{count(checks,c=>c.ok)} of {length(checks)} ticked"
        Progress {count(checks,c=>c.ok)/length(checks)}
        each c in checks
          Checkbox "{c.label}" bind=c.ok
      Card gap=8
        Text.strong "If it fails"
        Input "What needs doing" multiline bind=failNote
      Row gap=12
        Button "Pass" icon="check_circle" onTap={all(checks,c=>c.ok) ? (log=append(log,{"room":inspRoom,"result":"pass"}); emit inspect {room: inspRoom, result: "pass"}; toast "Room {inspRoom} inspected"; checks=map(checks,c=>with(c,"ok",false))) : toast "Tick every item before passing"}
        Button.destructive "Fail" icon="cancel" onTap={trim(failNote)=="" ? toast "Say what needs doing" : (log=append(log,{"room":inspRoom,"result":"fail","note":failNote}); emit inspect {room: inspRoom, result: "fail", note: failNote}; toast "Room {inspRoom} sent back to dirty"; failNote="")}
      Card gap=8
        Text.strong "Today's results"
        each l in log
          ListTile "Room {l.room}" subtitle="{l.note}" icon="{l.result=='pass' ? 'check_circle' : 'cancel'}"
            Badge.success "Pass" visible={l.result=="pass"}
            Badge.error "Fail" visible={l.result=="fail"}
        if length(log)==0
          Text.muted "Nothing inspected yet."
