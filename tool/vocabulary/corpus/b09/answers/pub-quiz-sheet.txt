### Question reader
state q=0
state shown=false
state secs=0
state playing=false
state hidden=true
state questions=[{"text":"Which element has the chemical symbol Fe?","answer":"Iron"},{"text":"What is the capital of Australia?","answer":"Canberra"},{"text":"Who painted The Night Watch?","answer":"Rembrandt"}]
every 1s when secs>0 onTick={secs=secs-1; secs==1 ? shown=true : toast ""}
App
  Sidebar
    Brand "Round Table" icon="sports_bar"
    NavItem "Scoresheet" icon="grid_on" onTap={go scoresheet}
    NavItem "Question reader" icon="quiz" selected={true}
    NavItem "Leaderboard" icon="leaderboard" onTap={go leaderboard}
  Main
    Nav
      Heading "Round 2: Geography" level=1
      Text.muted "Question {q+1} of {length(questions)}"
    Column pad=24 gap=24 center
      Text "{questions[q].text}" size=44 weight=700 center
      if secs>0
        Text "{secs}" size=96 weight=700 center animate=pulse
      if shown
        Card gap=8 animate=scale
          Text.muted "Answer" center
          Text "{questions[q].answer}" size=40 weight=700 center
      Row gap=12
        Button.outline "Previous" disabled={q==0} onTap={q=q-1; shown=false; secs=0}
        Button "Reveal in 5 seconds" icon="hourglass_top" onTap={shown=false; secs=5}
        Button.outline "Next" disabled={q==length(questions)-1} onTap={q=q+1; shown=false; secs=0}
      Divider
      Card gap=12
        Text.strong "Music round"
        Row gap=12 middle
          IconButton "play_arrow" tooltip="Play clip" hidden={playing} onTap={playing=true; emit playClip}
          IconButton "pause" tooltip="Pause" hidden={!playing} onTap={playing=false; emit pauseClip}
          Waveform playing={playing} grow h=48
          Text.mono "{hidden ? '???' : 'Billie Jean - Michael Jackson'}"
        Row gap=8
          Button.outline "Reveal title" onTap={hidden=false}
          Button.ghost "Hide title" onTap={hidden=true}
