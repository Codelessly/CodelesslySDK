### Measure
state curA=2
state curB=6
state tdiv=1
state v1=1
state yA=1.6
state yB=-0.8
Page
  Nav
    Brand "Wavepost" icon="graphic_eq"
    Row gap=4
      Link "Scope" onTap={go scope}
      Link "Measure"
      Link "Saved traces" onTap={go saved}
    Button "Back to scope" onTap={go scope}
  Section
    Column gap=16
      Heading "Cursor measurements" level=1
      Text.muted "Drag cursors A and B across the frozen trace to read the time and voltage between them."
      Card gap=8
        Scope#measureScreen h=320 divisions=[10,8] frozen timePerDiv={tdiv}
          Trace "CH1" channel=1 color=#FFFFD600 volts={v1} source="sine" freq=1200 amp=3.2
          Cursor#cursorA bind=curA axis="x" label="A"
          Cursor#cursorB bind=curB axis="x" label="B"
          Cursor#levelA bind=yA axis="y" label="1"
          Cursor#levelB bind=yB axis="y" label="2"
      Grid cols=4 gap=16
        Card gap=4
          Text.caption "A to B (time)"
          Text.strong "{fixed(abs(curB-curA)*tdiv,2)} ms" size=28
        Card gap=4
          Text.caption "Frequency from period"
          Text.strong "{fixed(1000/max(0.001,abs(curB-curA)*tdiv),1)} Hz" size=28
        Card gap=4
          Text.caption "Delta voltage"
          Text.strong "{fixed(abs(yA-yB)*v1,2)} V" size=28
        Card gap=4
          Text.caption "Cursor positions"
          Text.mono "A {fixed(curA,2)} div, B {fixed(curB,2)} div"
      Row gap=8
        Button.outline "Reset cursors" onTap={curA=2; curB=6; yA=1.6; yB=-0.8}
        Button.secondary "Snap to peaks" onTap={curA=1.25; curB=5.0}
        Button "Copy readings" icon="content_copy" onTap={copy "dt={fixed(abs(curB-curA)*tdiv,2)} ms, dV={fixed(abs(yA-yB)*v1,2)} V"; toast "Copied"}
