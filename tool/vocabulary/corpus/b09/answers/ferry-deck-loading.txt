### Load Order
App
  Sidebar
    Brand "Roll On" icon="directions_boat"
    NavItem "Deck plan" icon="map" onTap={go deck-plan}
    NavItem "Load order" icon="format_list_numbered" badge="{length(queue)}" selected={true}
    NavItem "Sailing" icon="schedule" onTap={go sailing}
    Spacer grow
    Text.muted "MV Aran Spirit"
  Main
    Nav
      Heading "Load order" level=1
      Row gap=8
        Badge.info "Departs in {duration(seconds)}"
        Button.outline "Print for the deck crew" icon="print" onTap={emit printOrder}
    Column pad=24 gap=16 scroll
      Text.muted "The list follows the plan. Lanes load front to back, lower deck first, heaviest vehicles first within a lane."
      each lane in sortBy(filter(lanes,l=>!isEmpty(l.vehicles)),l=>l.deck=="lower"?0:1)
        Card gap=8
          Row between
            Column
              Text.strong "{lane.name}"
              Text.caption "{upper(lane.deck)} DECK"
            Row gap=8
              if sum(lane.vehicles,v=>v.len)>=lane.cap-2
                Badge.error "Full"
              Badge.neutral "{length(lane.vehicles)} vehicles"
          Divider
          each v in sortByDesc(lane.vehicles,v=>v.weight)
            ListTile "{index+1}. {v.plate}" subtitle="{v.type}, {v.len} m, {v.weight} t" leading="directions_car"
              Row gap=8
                IconButton "map" tooltip="Show on plan" onTap={deck=lane.deck;go deck-plan}
                IconButton "undo" tooltip="Back to queue" onTap={sel=v.id;queue=append(queue,v);lanes=map(lanes,l=>with(l,"vehicles",removeWhere(l.vehicles,x=>x.id==sel)))}
      if sum(lanes,l=>length(l.vehicles))==0
        Card gap=8
          Text.strong "Nothing placed yet"
          Text.muted "Drag vehicles from the waiting list onto a lane."
          Button "Go to the deck plan" onTap={go deck-plan}
      Card gap=8
        Text.strong "Still waiting ({length(queue)})"
        each q in queue
          Row between
            Text "{q.plate} {q.type}"
            Text.muted "{q.len} m, {q.weight} t"
