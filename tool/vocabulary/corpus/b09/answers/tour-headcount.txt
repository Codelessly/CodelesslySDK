          Row gap=8 middle

          Button.destructive "Someone is missing" onTap={go lost-guest}

        Text.muted "People on the bus"
        Text "{count}" size=96 weight=700 center
        Text.muted "of {expected} expected"
        Progress {count / expected}
        Row gap=16 center
          IconButton "remove" tooltip="One fewer" onTap={count=max(0,count-1)}
          IconButton "add" tooltip="One more" onTap={count=min(expected,count+1)}
        Button "Count again at this stop" onTap={recounts=append(recounts,{"stop":stop,"n":count}); count==expected ? toast "All {expected} on board" : toast "Count is off by {abs(expected-count)}"}
      Card gap=8
        Row between
          Text.strong "Boarding list"
          Text.muted "{count(guests,g=>g.boarded)} of {length(guests)} ticked"
        each g in guests
          Row between middle
            Checkbox "{g.name}" bind=g.boarded onChange={count=count(guests,x=>x.boarded)}
            if !g.boarded
              Badge.warning "Not yet"
      Card gap=8
        Text.strong "Counts today"
        each r in recounts
          Row between
            Text "{r.stop}"
            Text.strong "{r.n}" color={r.n==expected ? #FF2E7D32 : #FFC62828}
