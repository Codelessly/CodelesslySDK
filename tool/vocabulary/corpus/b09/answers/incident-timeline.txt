### Status update
state status="investigating"
state elapsed=2535
state draft="We are investigating elevated errors at checkout. A rollback is in progress and we will update in 30 minutes."
state audience="Customers"
state preview=false
state posted=[{"t":"14:20","who":"Dana Ortiz","to":"Customers","text":"We are aware of failures at checkout and are investigating."}]
every 1s when status!="resolved" onTick={elapsed=elapsed+1}
App
  Sidebar
    Brand "Postmortem" icon="local_fire_department"
    NavItem "Timeline" icon="timeline" onTap={go timeline}
    NavItem "Status update" icon="campaign" selected={true}
    NavItem "Follow-ups" icon="checklist" onTap={go followups}
  Main
    Nav
      Heading "Status update" level=1
      Text.mono "{duration(elapsed)} open"
    Column pad=24 gap=16 scroll
      Row gap=16 top
        Card grow gap=12
          Text.strong "Draft"
          SegmentedButton @audience items=["Customers","Internal","Executives"]
          Input "Message" @draft multiline
          Text.caption "{length(draft)} of 280"
          Row gap=8
            Button.outline "Preview" onTap={preview=!preview}
            Button "Post update" disabled={draft=="" || length(draft)>280} onTap={posted=append(posted,{"t":"14:40","who":"Dana Ortiz","to":audience,"text":draft}); draft=""; preview=false; toast "Update posted"}
        if preview
          Card grow gap=8
            Text.muted "Preview for {lower(audience)}"
            Row gap=8
              Avatar "Dana Ortiz" size=32
              Text.strong "Postmortem status"
              Badge.warning "{status}"
            Text "{draft}"
      Card gap=12
        Text.strong "Posted"
        each p in reverse(posted)
          ListTile "{p.text}" subtitle="{p.t} to {p.to}, by {p.who}" icon="campaign"
