### Finish Board
state finished=[{"name":"Marta Koval","time":1512},{"name":"Elin Sjoberg","time":1578},{"name":"Jonas Berg","time":1633}]
Page
  Nav sticky
    Brand "Control Point" icon="explore"
    Row gap=4
      Link "Splits" onTap={go course-and-splits}
      Link "Start list" onTap={go start-list}
    Badge.success "Live"
  Section
    Heading "Finish board" level=1
    Text.muted "Runners are ranked as they punch the finish."
    LiveFeed bind=finished source="finish" onPunch={finished=sortBy(append(finished,result),r=>r.time)}
    Card gap=8
      each r in finished
        Row between animate=slide-up
          Row gap=12
            Text.strong "{index+1}"
            Avatar "{r.name}" size=32
            Text "{r.name}"
          Text.mono "{duration(r.time)}"
      if isEmpty(finished)
        Text.muted "No finishers yet."

### Start List and Compare
state starts=[{"name":"Aiko Tanaka","in":95},{"name":"Jonas Berg","in":215},{"name":"Priya Nair","in":335}]
state a="Elin Sjoberg"
state b="Marta Koval"
every 1s onTick={starts=map(starts,s=>with(s,"in",max(0,s.in-1)))}
Page
  Nav
    Brand "Control Point" icon="explore"
    Link "Splits" onTap={go course-and-splits}
  Section
    Heading "Start list" level=1
    Card gap=8
      each s in starts
        Row between
          Text "{s.name}"
          if s.in>0
            Badge.info "Starts in {duration(s.in)}"
          if s.in==0
            Badge.success "Started"
  Section tint
    Heading "Compare two runners" level=1
    Row gap=12
      Select bind=a items=["Elin Sjoberg","Marta Koval","Jonas Berg"]
      Select bind=b items=["Elin Sjoberg","Marta Koval","Jonas Berg"]
    Card
      CourseMap controls={controls} routes=[routes[a],routes[b]] h=320
    Card
      LineChart labels=["1","2","3","4","5"] series=[{"name":"{a}","values":cumulative[a]},{"name":"{b}","values":cumulative[b]}] h=200
