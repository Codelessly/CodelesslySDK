### Review
App
  Main
    Nav
      Brand "Cast" icon="how_to_vote"
      Text.muted "Review your ballot"
    Column pad=24 gap=16 maxw=720
      Heading "Check your choices" level=1
      Text.muted "Nothing is cast until you confirm."
      Card gap=12
        Row between
          Column gap=2
            Text.caption "Mayor"
            Text.strong "{isEmpty(mayor) ? 'No selection' : find(mayors,c=>c.id==mayor[0]).name}"
          Link "Change" onTap={step=0; go ballot}
        Divider
        Row between
          Column gap=2
            Text.caption "City Council"
            Text.strong "{isEmpty(council) ? 'No selection' : join(map(filter(councilors,c=>contains(council,c.id)),c=>c.name),', ')}"
            if length(council)>3
              Text.danger "Too many selected: this contest will not count."
          Link "Change" onTap={step=1; go ballot}
        Divider
        Row between
          Column gap=2
            Text.caption "Park Commissioner, ranked"
            each slot in ranked
              Text "{index+1}. {slot=='' ? 'Not ranked' : find(parks,p=>p.id==slot).name}"
          Link "Change" onTap={step=2; go ballot}
        Divider
        Row between
          Column gap=2
            Text.caption "Measure B: Library Bond"
            Text.strong "{measure=='' ? 'No selection' : upper(measure)}"
          Link "Change" onTap={step=3; go ballot}
      Row between
        Button.outline "Back to ballot" onTap={step=0; go ballot}
        Button "Cast my ballot" size=large onTap={open confirm}
  Dialog#confirm "Cast your ballot?" "You cannot change it after it is printed."
    if isEmpty(mayor) || measure=="" || length(council)>3
      Text.danger "Some contests are blank or over-voted. They will not be counted."
    Row gap=8 end
      Button.outline "Go back" onTap={close}
      Button "Yes, print my ballot" onTap={close; go printing}

### Printing
state pct=0
state done=false
state left=15
every 200 when !done onTick={pct=min(1,pct+0.04); done=pct>=1}
every 1s when done onTick={left=left-1; left<=0 ? (mayor=[]; council=[]; ranked=["","",""]; measure=""; step=0; go ballot) : toast ""}
Page
  Section center
    if !done
      Column center gap=16 animate=fade
        Icon "print" size=64
        Heading "Printing your ballot" level=1
        Text.muted "Please wait. Do not leave the booth."
        Progress bind=pct
        Text.caption "{round(pct*100,0)}%"
    if done
      Column center gap=16 animate=pop
        Icon "check_circle" size=72 color=#FF2E8B57
        Heading "Thank you for voting" level=1
        Text.muted "Take your printed ballot to the scanner by the exit."
        Button.outline "Finish now" onTap={mayor=[]; council=[]; ranked=["","",""]; measure=""; step=0; go ballot}
        Text.caption "This screen resets in {left} seconds."
