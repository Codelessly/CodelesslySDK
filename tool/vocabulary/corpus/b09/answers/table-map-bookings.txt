### Reservations
state q=""
state when="all"
state bookings=[{"id":"b1","name":"Patel","party":4,"time":19.5,"phone":"555-0142","placed":false},{"id":"b2","name":"Nguyen","party":6,"time":20,"phone":"555-0177","placed":false},{"id":"b3","name":"Brooks","party":2,"time":20.5,"phone":"555-0119","placed":false},{"id":"b4","name":"Rossi","party":8,"time":21,"phone":"555-0103","placed":false}]

App
  Sidebar
    Brand "Covers" icon="restaurant"
    NavItem "Floor" icon="table_restaurant" onTap={go floor}
    NavItem "Reservations" icon="event_seat" selected={true}
    NavItem "Waitlist" icon="hourglass_top" onTap={go waitlist}
  Main
    Nav
      Heading "Reservations"
      Row gap=8
        Input "Search guests" @q w=220
        Button "New booking" icon="add" onTap={open newbooking}
    Column pad=24 gap=16
      Row gap=8
        FilterChip "All" selected={when=="all"} onTap={when="all"}
        FilterChip "Early" selected={when=="early"} onTap={when="early"}
        FilterChip "Late" selected={when=="late"} onTap={when="late"}
      Card gap=0
        Row gap=8 pad=8 border-bottom=#FFDDDDDD
          Text.muted "Guest" w=140
          Text.muted "Party" w=70
          Text.muted "Time" w=80
          Text.muted "Phone" grow
          Text.muted "Status" w=100
        each b in filter(bookings,x=>contains(lower(x.name),lower(q)) && (when=="all" || (when=="early" ? x.time<20.5 : x.time>=20.5)))
          Row gap=8 pad=8 middle
            Text.strong "{b.name}" w=140
            Text "{b.party}" w=70
            Text "{b.time}:00" w=80
            Text.muted "{b.phone}" grow
            Badge.success "Seated" w=100 visible={b.placed}
            Badge.neutral "Booked" w=100 visible={!b.placed}
            IconButton "more_vert" tooltip="Actions" onTap={sel=b.id; open rowMenu}
  Menu#rowMenu end
    MenuItem "Text the guest" icon="sms" onTap={toast "Text sent to {find(bookings,b=>b.id==sel).name}: your table is ready"}
    MenuItem "Cancel booking" icon="delete" onTap={bookings=removeWhere(bookings,b=>b.id==sel); toast "Cancelled"}
  Dialog#newbooking "New booking"
    Input "Guest name" @newName
    Input "Party size" @newParty type=number
    TimePicker "Time" @newTime

### Waitlist
state waitlist=[{"id":"w1","name":"Kim","party":2,"wait":600,"texted":false},{"id":"w2","name":"Dubois","party":3,"wait":1200,"texted":false},{"id":"w3","name":"Haddad","party":5,"wait":1800,"texted":true}]
every 1s onTick={waitlist=map(waitlist,w=>with(w,"wait",max(0,w.wait-1)))}
App
  Sidebar
    Brand "Covers" icon="restaurant"
    NavItem "Floor" icon="table_restaurant" onTap={go floor}
    NavItem "Reservations" icon="event_seat" onTap={go reservations}
    NavItem "Waitlist" icon="hourglass_top" badge="{length(waitlist)}" selected={true}
  Main
    Nav
      Heading "Waitlist"
      Text.muted "{sum(waitlist,w=>w.party)} guests waiting"
    Column pad=24 gap=12
      if isEmpty(waitlist)
        Card
          Text.muted "Nobody is waiting."
      each w in waitlist
        Card
          Row between middle
            Column gap=4
              Text.strong "{w.name}, party of {w.party}"
              Text.muted "Quoted wait left: {duration(w.wait)}"
              Progress {1 - w.wait/1800}
            Row gap=8
              Button.outline "Text a guest" icon="sms" disabled={w.texted} onTap={waitlist=updateWhere(waitlist,x=>x.id==w.id,x=>with(x,"texted",true)); toast "Texted {w.name}: your table is almost ready"}
              Button "Seat now" icon="chair" onTap={waitlist=removeWhere(waitlist,x=>x.id==w.id); emit seat w.id; toast "{w.name} seated"}
              IconButton "close" tooltip="Remove" onTap={waitlist=removeWhere(waitlist,x=>x.id==w.id)}
