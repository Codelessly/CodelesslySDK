### Alerts
state subs=[{"id":1,"station":"Millford","level":"Warning 3.5 m","sms":true,"mail":true,"on":true},{"id":2,"station":"Ashby Bridge","level":"Watch 2.5 m","sms":false,"mail":true,"on":true},{"id":3,"station":"Dunmore Lock","level":"Severe 4.5 m","sms":true,"mail":false,"on":false}]
state sel=0
state newStation="Harlow Weir"
state newLevel="Watch 2.5 m"
App
  Sidebar
    Brand "Streamline" icon="water"
    NavItem "River map" icon="map" onTap={go river-map}
    NavItem "Stations" icon="sensors" onTap={go station}
    NavItem "Alerts" icon="notifications_active" badge="2" selected={true}
    Spacer grow
    Text.muted "River Tarn, 5 gauges"
  Main
    Nav
      Heading "Alerts" level=1
      Button "New alert" onTap={open newsub}
    Column pad=24 gap=16 scroll maxw=720
      Text.muted "You hear from us when a gauge crosses the level you choose."
      if isEmpty(subs)
        Card gap=8 center
          Icon "notifications_off"
          Text.strong "No alerts yet"
          Button "New alert" onTap={open newsub}
      each s in subs
        Card gap=8
          Row between middle
            Column grow gap=2
              Text.strong "{s.station}"
              Text.muted "{s.level}"
            Toggle bind=s.on
            IconButton "more_vert" tooltip="More" onTap={sel=s.id; open subMenu}
          Row gap=16
            Checkbox "Text message" bind=s.sms
            Checkbox "Email" bind=s.mail
  Menu#subMenu end
    MenuItem "Send a test" icon="send" onTap={emit testAlert sel; toast "Test sent"}
    MenuItem "Delete" icon="delete" onTap={subs=removeWhere(subs,x=>x.id==sel); toast "Alert deleted"}
  Dialog#newsub "New alert" "Pick a gauge and a level"
    Select "Station" bind=newStation items=["Harlow Weir","Ashby Bridge","Millford","Tern Crossing","Dunmore Lock"]
    Select "Level" bind=newLevel items=["Watch 2.5 m","Warning 3.5 m","Severe 4.5 m"]
    Button "Add alert" onTap={subs=append(subs,{"id":length(subs)+10,"station":newStation,"level":newLevel,"sms":true,"mail":true,"on":true}); toast "Alert added"; close}
