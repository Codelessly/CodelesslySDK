### Chores and wishes
state pinEntry=""
state locked=true
state save=48.00
state spend=12.50
state chores=[{"id":1,"name":"Feed the cat","pay":1.00,"done":true,"approved":false},{"id":2,"name":"Take out recycling","pay":1.50,"done":true,"approved":true},{"id":3,"name":"Wash dishes","pay":2.00,"done":false,"approved":false},{"id":4,"name":"Tidy bedroom","pay":2.50,"done":true,"approved":false}]
state wishes=[{"id":1,"name":"LEGO space shuttle","price":34.99,"icon":"rocket"},{"id":2,"name":"Soccer cleats","price":29.00,"icon":"sports_soccer"},{"id":3,"name":"Sticker book","price":6.50,"icon":"menu_book"}]
state picked=0
App
  Sidebar
    Brand "Three Jars" icon="savings"
    NavItem "My jars" icon="local_drink" onTap={go jars}
    NavItem "Chores" icon="checklist" selected={true}
    NavItem "Wish list" icon="favorite" onTap={go wishes}
    NavItem "Bank log" icon="receipt_long" onTap={go bank}
    Spacer grow
    if locked
      Button.outline "Parent mode" icon="lock" onTap={open pin}
    if !locked
      Button.outline "Lock again" icon="lock_open" onTap={locked=true; toast "Locked"}
  Main
    Nav
      Heading "Chores and wishes" level=1
      Badge.warning "{count(chores,c=>c.done && !c.approved)} waiting for a parent"
    Column pad=24 gap=16 scroll
      Row gap=16 top
        Card grow gap=8
          Heading "This week's chores" level=3
          each c in chores
            ListTile "{c.name}" subtitle="{money(c.pay,"$",2)}" leading="task_alt"
              Row gap=8 middle
                Toggle bind=c.done
                if c.approved
                  Badge.success "Approved"
                if !c.approved && c.done
                  Button "Approve" size=small onTap={locked ? open pin : chores=updateWhere(chores,x=>x.id==c.id,x=>with(x,"approved",true)); save=save+c.pay; toast "Approved {money(c.pay,"$",2)}"}
          Divider
          Row between
            Text.strong "Earned so far"
            Text.strong "{money(sum(filter(chores,c=>c.approved),c=>c.pay),"$",2)}"
        Card grow gap=8
          Heading "Wish list" level=3
          each w in wishes
            Column gap=4
              Row gap=8 middle
                IconTile "{w.icon}"
                Column grow
                  Text.strong "{w.name}"
                  Text.muted "{money(w.price,"$",2)}"
                IconButton "delete" tooltip="Remove" onTap={wishes=removeWhere(wishes,x=>x.id==w.id)}
              Progress {clamp(save/w.price,0,1)}
              if save >= w.price
                Badge.success "You can afford it"
              if save < w.price
                Text.muted "{money(w.price-save,"$",2)} more to go"
          Button.outline "Add a wish" icon="add" onTap={open newWish}
  Dialog#pin "Parent lock" "Enter your PIN"
    PinInput bind=pinEntry length=4
    Button "Unlock" onTap={pinEntry=="2468" ? locked=false; pinEntry=""; close : toast "Wrong PIN"}
  Dialog#newWish "New wish"
    Input "What do you want?" bind=wishName
    Input "Price" bind=wishPrice type=number
    Button "Add" onTap={wishes=append(wishes,{"id":length(wishes)+10,"name":wishName,"price":wishPrice,"icon":"star"}); close}

### Coin game
state coins=[1,5,10,25,5,10,1]
state guess=""
state score=0
state round=1
state msg=""
Page
  Nav
    Brand "Three Jars" icon="savings"
    Row gap=4
      Link "My jars" onTap={go jars}
      Link "Bank log" onTap={go bank}
    Button.outline "Back" onTap={go jars}
  Section center
    Column gap=16 center
      Heading "Count the coins" level=1
      Text.muted "Round {round}: how many cents is this pile worth?"
      Row gap=8 center
        each c in coins
          Coin value={c} animate=pop stagger=80
      Input "Your answer in cents" bind=guess type=number w=240
      Row gap=8
        Button "Check" onTap={guess==sum(coins,c=>c) ? score=score+1; msg="Right! {sum(coins,c=>c)} cents"; round=round+1; coins=map(range(6),i=>[1,5,10,25][i%4]) : msg="Not quite, try again"; animate=shake}
        Button.outline "Hint" onTap={toast "Count the quarters first"}
      if msg!=""
        Text.strong "{msg}"
      Row gap=8 middle
        Icon "star"
        Text.strong "Score {score}"
