App
  Sidebar
    Brand "Handoff" icon="swap_horiz"
    NavItem "Calendar" icon="calendar_month" onTap={go calendar}
    NavItem "Requests" icon="mark_email_unread" selected badge="{count(requests,r=>r.status=="pending")}"
    NavItem "Expenses" icon="receipt_long" onTap={go expenses}
  Main
    Nav
      Heading "Change requests" level=1
      SegmentedButton items=["all","pending","approved","declined"] bind=filter

      each r in filter=="all" ? requests : filter(requests,x=>x.status==filter)
        Card gap=8
          Row between middle
            Text.strong "{r.title}"
            Badge.warning "Pending" visible={r.status=="pending"}
            Badge.success "Approved" visible={r.status=="approved"}
            Badge.error "Declined" visible={r.status=="declined"}
          Text.muted "From {r.from}"
          Text "Reason: {r.reason}"
          if r.status=="pending"&&r.from!="Maya"
            Row gap=8
              Button "Approve" icon="check" onTap={requests=updateWhere(requests,x=>x.id==r.id,x=>with(x,"status","approved")); emit approve r.id; toast "Approved"}
              Button.outline "Decline" icon="close" onTap={sel=r.id; open decline}
      if length(requests)==0
        Text.muted "No requests yet."

  Input "Note" "Optional" bind=note multiline
  Button.destructive "Decline" onTap={requests=updateWhere(requests,x=>x.id==sel,x=>with(x,"status","declined")); emit decline sel; note=""; toast "Declined"; close}
