state selected=0

App
  Sidebar
    Brand "Inkplace" icon="draw"
    NavItem "Design" icon="brush" onTap={go design}
    NavItem "Placements" icon="bookmarks" badge="{length(placements)}" selected={true}
    NavItem "Book a session" icon="event" onTap={go booking}
    Spacer grow
    Text.muted "Mara Ellison"
  Main
    Nav
      Heading "Saved placements" level=1
      Button "Book with a placement" icon="event" onTap={go booking}
    Column pad=24 gap=16
      Row gap=24 top
        Column grow gap=12
          if isEmpty(placements)
            Card gap=8
              Text.strong "Nothing saved yet"
              Text.muted "Drag the design onto the outline and save it to compare options."
          each p in placements
            Card gap=8 onTap={selected=p.id}
              Row between
                Column gap=4
                  Text.strong "{p.name}"
                  Text.muted "{upper(p.side)} side, {p.cm} cm wide, turned {p.angle} degrees"
                Row gap=4
                  IconButton "open_in_new" tooltip="Open in editor" onTap={go design}
                  IconButton "content_copy" tooltip="Duplicate" onTap={placements=append(placements,with(with(p,"id",length(placements)+1),"name",concat(p.name," copy")))}
                  IconButton "delete" tooltip="Delete" onTap={placements=removeWhere(placements,x=>x.id==p.id); toast "Removed"}
        Column w=320 gap=12
          Card gap=8
            Text.strong "Preview"
            if selected!=0
              BodyOutline view={find(placements,x=>x.id==selected).side} h=320
                DesignOverlay src="https://cdn.inkplace.app/designs/koi-waves.png" x={find(placements,x=>x.id==selected).px} y={find(placements,x=>x.id==selected).py} scale={find(placements,x=>x.id==selected).scale} angle={find(placements,x=>x.id==selected).angle} readonly
            if selected==0
              Text.muted "Select a placement to preview it."

### Booking
state step=0
state artist="Jonah Reyes"
state sessionHours=3
state sessionDate=""
state deposit=true
state notes=""
state artists=[{"name":"Jonah Reyes","style":"Fine line, 9 years","rate":180},{"name":"Priya Nair","style":"Japanese, 12 years","rate":220},{"name":"Theo Marsh","style":"Blackwork, 6 years","rate":160}]
App
  Sidebar
    Brand "Inkplace" icon="draw"
    NavItem "Design" icon="brush" onTap={go design}
    NavItem "Placements" icon="bookmarks" onTap={go placements}
    NavItem "Book a session" icon="event" selected={true}
    Spacer grow
    Text.muted "Mara Ellison"
  Main
    Nav
      Heading "Book your session" level=1
      Text.muted "Step {step+1} of 3"
    Column pad=24 gap=16 maxw=720
      Progress {(step+1)/3}
      if step==0
        Card gap=12
          Text.strong "Choose your artist"
          each a in artists
            ListTile "{a.name}" "{a.style}, ${a.rate} per hour" selected={artist==a.name} onTap={artist=a.name}
          Button "Next" onTap={step=1}
      if step==1
        Card gap=12
          Text.strong "Session length"
          Slider bind=sessionHours min=1 max=8
          Text "{sessionHours} hours with {artist}"
          DatePicker "Preferred date" bind=sessionDate
          Input "Notes for the artist" bind=notes multiline
          Row gap=8
            Button.outline "Back" onTap={step=0}
            Button "Next" onTap={sessionDate=="" ? toast "Pick a date" : step=2}
      if step==2
        Card gap=12
          Text.strong "Summary"
          Row between
            Text "Artist"
            Text.strong "{artist}"
          Row between
            Text "Length"
            Text.strong "{sessionHours} hours"
          Row between
            Text "Estimate"
            Text.strong "{money(sessionHours*find(artists,a=>a.name==artist).rate,"$",2)}"
          Row between
            Text "Deposit due now"
            Text.strong "{money(sessionHours*find(artists,a=>a.name==artist).rate*0.2,"$",2)}"
          Text.muted "The deposit is 20% of the estimate. It is taken off your final bill and is non-refundable within 48 hours of the session."
          Checkbox "I understand the deposit terms" bind=deposit
          Row gap=8
            Button.outline "Back" onTap={step=1}
            Button "Confirm booking" onTap={deposit ? open booked : toast "Accept the deposit terms first"}
  Dialog#booked "Booking requested" "{artist} will confirm within a day"
    Text "Session on {sessionDate}, {sessionHours} hours."
    Button "Done" onTap={emit booked; step=0; close; go design}
