### Table
state level=3
state target=60
state score=24
state moves=0
state hints=2
state selected="g1"
state guests=[{"id":"g1","name":"Ada Lindqvist","likes":["jazz","sailing"],"dislikes":["gossip"],"feud":"g4"},{"id":"g2","name":"Bram Coetzee","likes":["chess","wine"],"dislikes":["loud music"],"feud":""},{"id":"g3","name":"Chloe Marchetti","likes":["opera","travel"],"dislikes":["chess"],"feud":""},{"id":"g4","name":"Dev Raman","likes":["gossip","sailing"],"dislikes":["opera"],"feud":"g1"},{"id":"g5","name":"Elin Haugen","likes":["wine","jazz"],"dislikes":["travel"],"feud":""},{"id":"g6","name":"Farid Nasser","likes":["travel","chess"],"dislikes":["gossip"],"feud":""}]
state seats=["g1","g4","g2","","g5","g6"]
state ties=[{"a":"g1","b":"g4","w":-12},{"a":"g1","b":"g5","w":8},{"a":"g2","b":"g6","w":6},{"a":"g3","b":"g6","w":7},{"a":"g3","b":"g4","w":-5}]
state log=[]
state finished=false
state showTies=true
every 1s when !finished onTick={elapsed=elapsed+1}
state elapsed=0
App
  Sidebar
    Brand "Place Cards" icon="table_restaurant"
    NavItem "Levels" icon="grid_view" onTap={go levels}
    NavItem "Table" icon="restaurant" selected={true}
    NavItem "Daily table" icon="today" onTap={go daily-table}
    Spacer grow
    Text.muted "Level {level}"
    Text.caption "Time {duration(elapsed)}"
  Main
    Nav
      Heading "Level {level}: the Okafor dinner"
      Row gap=8
        Toggle "Show lines" @showTies
        Button.outline "Undo" disabled={isEmpty(log)} onTap={seats=last(log); log=removeAt(log,length(log)-1); moves=moves+1}
        Button.outline "Hint ({hints})" disabled={hints==0} onTap={hints=hints-1; emit hint seats; toast "Swapped one pair"}
        Button "Done" disabled={score < target} onTap={finished=true; open result}
    Column pad=24 gap=16
      Card gap=8
        Row between
          Text.strong "Table score"
          Text.strong "{score} / {target}" color={score >= target ? "#FF2E7D32" : "#FFC62828"}
        Progress {clamp(score/target,0,1)} animate=fade
        Row gap=16
          Badge.success "Happy pairs {count(ties, t=>t.w>0)}"
          Badge.error "Upsets {count(ties, t=>t.w<0)}"
          Badge.neutral "Moves {moves}"
      Row gap=16 top
        Card grow gap=12
          Text.strong "The table"
          SeatMap.round seats={seats} guests={guests} ties={ties} showTies={showTies} positive=#FF2E7D32 negative=#FFC62828 drop=card onDrop={log=append(log,seats); seats=result.seats; score=result.score; moves=moves+1} onSeatTap={selected=result.guest} h=420
          Text.caption "Drag a card onto a seat. Dropping on a taken seat swaps the two guests."
        Column w=300 gap=12
          Text.strong "Cards to seat"
          each g in guests
            if !contains(seats,g.id)
              Card gap=4 drag=card(g.id) onTap={selected=g.id}
                Text.strong "{g.name}"
                Text.caption "Likes {join(g.likes,', ')}"
                Text.caption "Dislikes {join(g.dislikes,', ')}"
          if all(guests,g=>contains(seats,g.id))
            Text.muted "Every guest is seated."
  Pane w=320
    Heading "Guest" level=3
    each g in filter(guests,x=>x.id==selected)
      Column gap=12
        Row gap=12 middle
          Avatar "{g.name}" size=48
          Text.strong "{g.name}"
        Text.fieldLabel "Likes"
        Row gap=8
          each l in g.likes
            Chip "{l}"
        Text.fieldLabel "Dislikes"
        Row gap=8
          each d in g.dislikes
            Chip "{d}"
        if g.feud != ""
          Card gap=4
            Row gap=8
              Icon "warning" color=#FFC62828
              Text.danger "A feud with {find(guests,x=>x.id==g.feud).name}"
            Text.caption "Keep them out of neighbouring seats."
    Divider
    Text.fieldLabel "Ties on the table"
    each t in ties
      ListTile "{find(guests,x=>x.id==t.a).name} and {find(guests,x=>x.id==t.b).name}" subtitle="{t.w > 0 ? 'Happy' : 'Upset'}" trailing="{t.w}"
  Dialog#result "Dinner is served" "Your table scored {score}"
    Column gap=12 center
      Row gap=8
        each s in [1,2,3]
          Icon "star" size=48 animate=pop delay={s*300} color={s <= (score >= target+20 ? 3 : score >= target+8 ? 2 : 1) ? "#FFF5B301" : "#FFBDBDBD"}
      Text.muted "Target {target}, {moves} moves, {2-hints} hints used"
      Row gap=8
        Button.outline "Replay" onTap={close; seats=["g1","g4","g2","","g5","g6"]; moves=0; hints=2; finished=false; elapsed=0}
        Button "Next level" onTap={close; level=level+1; emit nextLevel level; go levels}
