### Sign-off
state signed=false
state signName=""
App
  Sidebar
    Brand "Changeover" icon="engineering"
    NavItem "Map and notes" icon="map" onTap={go map}
    NavItem "Equipment" icon="precision_manufacturing" onTap={go equipment}
    NavItem "Crew" icon="groups" onTap={go crew}
    NavItem "Sign-off" icon="draw" selected={true}
  Main
    Nav
      Heading "Acknowledge and sign" level=1
      Badge.warning "{length(notes)-length(acked)} unread"
    Column pad=24 gap=16 maxw=720
      Card gap=8
        Text.strong "Notes still to acknowledge"
        each n in filter(notes, x=>!contains(acked,x.id))
          ListTile "{n.face}: {n.tag}" "{n.text}" onSwipeLeft={acked=append(acked,n.id); toast "Acknowledged"} onSwipeRight={acked=append(acked,n.id); toast "Acknowledged"}
            Icon "swipe_left"
        if length(acked)==length(notes)
          Text.success "All notes acknowledged."
      Card gap=8
        Text.strong "Open equipment faults"
        Text.body "{count(eq,e=>e.state=="faulty")} units tagged out: {join(map(filter(eq,e=>e.state=="faulty"),e=>e.name),", ")}"
        Text.body "Underground not yet out: {length(crewOut)-length(left)}"
      Card gap=12
        Text.strong "Incoming supervisor signature"
        Input "Full name" bind=signName
        SignaturePad bind=signature h=140
        Row gap=8
          Button.outline "Clear" onTap={signature=""}
          Button "Sign handover" onTap={length(acked)==length(notes) && trim(signName)!="" && signature!="" ? signed=true; open done : toast "Acknowledge every note, then name and sign"}
  Dialog#done "Handover signed" "Changeover is complete"
    Text.body "Signed by {signName} at {formatDate(today(),"MMM d")}."
    Button "Finish" onTap={emit handoverSigned; close; go map}
