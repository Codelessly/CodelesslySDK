### Symbols
state picked=""
App
  Main
    Nav
      Row gap=8
        IconButton "arrow_back" tooltip="Back" onTap={go compose}
        Heading "Symbols"
      Text.muted "{length(symbols)} icons"
    Column pad=16 gap=12
      Input "Search symbols" bind=q
      GridView cols=6 gap=8
        each s in filter(symbols, x=>contains(x, lower(coalesce(q,""))))
          Box radius=8 pad=8 fill={picked==s ? #FFE3F2FD : #FFF5F5F5} onTap={picked=s}
            Column center gap=4
              Icon "{s}"
      if picked!=""
        Card gap=8
          Row gap=12 middle
            Icon "{picked}" size=32
            Text.strong "{picked}"
            Spacer grow
            Button "Add to label" onTap={text=text+"["+picked+"]"; toast "Symbol added"; go compose}

### Print
state printing=false
state feed=0
state cut="Full cut"
state printed=false
every 100 when printing onTick={feed=min(1,feed+0.04); feed>=1 ? {printing=false; printed=true; toast "Printed"} : }
Page
  Nav
    Row gap=8
      IconButton "arrow_back" tooltip="Back" onTap={go compose}
      Heading "Preview"
    Row gap=4
      if battery<20
        Icon "battery_alert" color=#FFD32F2F
        Text.danger "{battery}%"
  Section
    Column gap=16
      Text.fieldLabel "Preview strip"
      Row scroll gap=12
        each n in range(copies)
          Box fill=#FFFFFFFF radius=4 pad=[16,12,16,12] border=1
            Text "{text} {start+n*step}" size=28 weight={bold ? 800 : 400} color=#FF111111
      Card gap=12
        Text.fieldLabel "After printing"
        Select "Cut" bind=cut items=["Full cut","Half cut","No cut (feed only)"]
        Row between
          Text "Tape"
          Text.strong "{tape} mm"
        Row between
          Text "Labels"
          Text.strong "{copies}"
      if printing
        Column gap=6
          Text.muted "Feeding tape, label {min(copies,ceil(feed*copies))} of {copies}"
          Progress bind=feed
      if printed
        Badge.success "Done, {cut=="No cut (feed only)" ? "tear off by hand" : "cut after the last label"}"
      Row gap=8
        Button.outline "Cancel" disabled={!printing} onTap={printing=false; feed=0; toast "Stopped"}
        Button "Print {copies} labels" grow icon="print" disabled={printing || battery<5 || tape==0} onTap={feed=0; printed=false; printing=true}
        Button.outline "Save" icon="bookmark" onTap={open slot}
  Sheet#slot "Save to memory"
    Text.muted "Choose one of ten slots"
    Grid cols=5 gap=8
      each i in range(10)
        Button.outline "{i+1}" onTap={slots=setAt(slots,i,{"text":text,"size":size,"frame":frame,"bold":bold}); toast "Saved in slot {i+1}"; close}
