### Group and draw
state tab="group"
state me="Maya Reyes"
state people=["Maya Reyes","Daniel Ortiz","Priya Nair","Tom Becker","Lena Fischer","Sam Okafor"]
state newName=""
state couples=[["Maya Reyes","Daniel Ortiz"],["Priya Nair","Tom Becker"]]
state pairA=""
state pairB=""
state drawn=false
state matches={}
state revealed=false
state limit=30
state giftDay="2026-12-24"
App
  Sidebar
    Brand "Hush" icon="redeem"
    NavItem "Group and draw" icon="group" selected={tab=="group"} onTap={tab="group"}
    NavItem "My giftee" icon="card_giftcard" selected={tab=="giftee"} onTap={go my-giftee}
    NavItem "Group chat" icon="chat" badge="3" onTap={go group-chat}
    NavItem "Reveal party" icon="celebration" onTap={go reveal-party}
    Spacer grow
    Row gap=8 middle
      Avatar "{me}" size=32
      Text.muted "{me}"
  Main
    Nav
      Heading "Office Elves 2026" level=1
      Row gap=8
        Badge.info "Gift day in {max(0, ceil((dateDiff(giftDay, today())) ))} days"
        Button "Draw names" disabled={length(people)<3 || drawn} onTap={open confirmDraw}
    Column pad=24 gap=16 scroll
      Grid cols=2 gap=16
        Card gap=12
          Heading "Who is in" level=3
          Row gap=8
            Input "Add a name" "First and last name" @newName grow
            Button "Add" onTap={trim(newName)=="" ? toast "Type a name first" : {people=append(people,trim(newName)); newName=""}}
          each p in people
            ListTile "{p}" icon="person"
              IconButton "close" tooltip="Remove {p}" onTap={people=without(people,p); couples=removeWhere(couples,c=>contains(c,p))}
        Card gap=12
          Heading "Rules" level=3
          Text.muted "Pairs that must never be matched, such as couples."
          Row gap=8
            Select "Person" @pairA items={people} grow
            Select "Partner" @pairB items={people} grow
            Button.outline "Add pair" onTap={pairA=="" || pairB=="" || pairA==pairB ? toast "Pick two different people" : {couples=append(couples,[pairA,pairB]); pairA=""; pairB=""}}
          each c in couples
            Row between middle
              Text "{c[0]} and {c[1]}"
              IconButton "delete" tooltip="Remove rule" onTap={couples=without(couples,c)}
          Divider
          Text.fieldLabel "Price limit"
          Slider @limit min=10 max=100 step=5
          Text.strong "{money(limit,"$",0)} per gift"
      Card gap=12
        Heading "Your match" level=3
        if !drawn
          Text.muted "The draw has not been made yet. Once it is, only you can see your match."
        if drawn
          Text.muted "Press and hold the cover to see who you are giving to. Let go and it hides again."
          HoldToReveal "Hold to reveal" onReveal={revealed=true} onRelease={revealed=false}
            Column center gap=8 pad=24
              Avatar "{matches[me]}" size=64
              Text.heading "{matches[me]}"
              Text.muted "Keep it secret until gift day."
          Row gap=8
            Button "Open my giftee" onTap={go my-giftee}
            Button.ghost "Redraw is locked" disabled
  Dialog#confirmDraw "Draw the names?" "Nobody can see anyone else's match."
    Text "{length(people)} people, {length(couples)} pairs kept apart."
    Row end gap=8
      Button.ghost "Cancel" onTap={close}
      Button "Draw" onTap={matches=draw(people,couples); drawn=true; emit drawMade; toast "Names drawn"; close}

      Heading "Your giftee: {giftee}" level=1
      Row gap=8
        Chip "Limit {money(limit,"$",0)}"
        Toggle "Mark as bought" @bought onChange={result ? toast "Marked as bought" : toast "Back on your list"}
    Column pad=24 gap=16 scroll
      Card gap=8
        Row between middle
          Column gap=4
            Text.fieldLabel "Gift day"
            Text.heading "{formatDate(giftDay,"EEEE, MMM d")}"
          Countdown to={giftDay} units=["days","hours","minutes"]
        Progress {bought ? 1 : 0.5}
        Text.muted "{bought ? "Bought. Wrap it and hide it well." : "Not bought yet."}"
      Row gap=16 top
        Card grow gap=12
          Heading "{giftee}'s wish list" level=3
          each w in sortBy(wishes, x=>x.price)
            ListTile "{w.item}" "{money(w.price,"$",0)}{w.note=="" ? "" : " - " + w.note}"
              if w.price > limit
                Badge.warning "Over limit"
              Link "Open" href="{w.url}"
              IconButton "content_copy" tooltip="Copy link" onTap={copy w.url; toast "Link copied"}
        Card w=320 gap=12
          Heading "Anonymous messages" level=3
          Column gap=8 scroll h=220
            each m in thread
              ChatBubble "{m.text}" side={m.from=="you" ? "right" : "left"}
          Input "Ask {giftee} something" "No names, ever" @outbox multiline
          Button "Send anonymously" onTap={trim(outbox)=="" ? toast "Write something first" : {thread=append(thread,{"from":"you","text":trim(outbox)}); outbox=""; emit anonMessage; toast "Sent without your name"}}
          Text.caption "Replies come back with no name too."

### Reveal party
state me="Maya Reyes"
state pairs=[{"giver":"Maya Reyes","getter":"Priya Nair","open":false},{"giver":"Daniel Ortiz","getter":"Sam Okafor","open":false},{"giver":"Priya Nair","getter":"Lena Fischer","open":false},{"giver":"Tom Becker","getter":"Maya Reyes","open":false},{"giver":"Lena Fischer","getter":"Daniel Ortiz","open":false},{"giver":"Sam Okafor","getter":"Tom Becker","open":false}]
state step=0
Page
  Nav
    Brand "Hush" icon="redeem"
    Row gap=4
      Link "Group and draw" onTap={go group-and-draw}
      Link "My giftee" onTap={go my-giftee}
      Link "Group chat" onTap={go group-chat}
    Button.outline "Back to the group" onTap={go group-and-draw}
  Section center
    Column gap=12 center
      Heading "Reveal party" level=1
      Text.muted "Turn the cards one by one. {count(pairs,p=>p.open)} of {length(pairs)} revealed."
      Progress {count(pairs,p=>p.open)/length(pairs)}
      Row gap=8
        Button "Turn next card" disabled={step>=length(pairs)} onTap={pairs=updateWhere(pairs,p=>indexOf(pairs,p)==step,p=>with(p,"open",true)); step=step+1; emit cardTurned}
        Button.outline "Turn all" onTap={pairs=map(pairs,p=>with(p,"open",true)); step=length(pairs)}
        Button.ghost "Hide all" onTap={pairs=map(pairs,p=>with(p,"open",false)); step=0}
  Section tint
    Grid cols=3 gap=24
      each p in pairs
        FlipCard flipped={p.open} onTap={pairs=updateWhere(pairs,q=>q.giver==p.giver,q=>with(q,"open",!q.open))} stagger=80
          Front
            Card center gap=8
              Icon "redeem" size=48
              Text.heading "Who gave to {p.getter}?"
              Text.caption "Tap to turn"
          Back
            Card center gap=8
              Avatar "{p.giver}" size=56
              Text.heading "{p.giver}"
              Text.muted "gave to {p.getter}"
  Footer
    Row between
      Text.strong "Hush"
      Text.muted "Happy gift day"
