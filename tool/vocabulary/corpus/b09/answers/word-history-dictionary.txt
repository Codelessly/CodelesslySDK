### Search
state q=""
state recent=["serendipity","petrichor","quixotic","nefarious"]
state saved=["petrichor"]
state words=[{"w":"serendipity","pos":"noun","gloss":"finding something good by chance"},{"w":"serene","pos":"adjective","gloss":"calm, clear and untroubled"},{"w":"serpent","pos":"noun","gloss":"a large snake"},{"w":"sermon","pos":"noun","gloss":"a talk on a moral or religious subject"},{"w":"seraph","pos":"noun","gloss":"a six-winged angelic being"}]
App
  Sidebar
    Brand "Etymon" icon="menu_book"
    NavItem "Search" icon="search" selected={true} onTap={go search}
    NavItem "Word of the day" icon="today" onTap={go word}
    NavItem "My lists" icon="bookmarks" badge="{length(saved)}" onTap={go lists}
    Spacer grow
    Text.muted "Recent searches"
    each r in recent
      NavItem "{r}" icon="history" onTap={q=r; go word}
    Button.ghost "Clear recent" icon="delete_sweep" onTap={recent=[]; toast "Recent searches cleared"}
  Main
    Nav
      Heading "Search" level=1
      Row gap=8
        IconButton "dark_mode" tooltip="Toggle theme" onTap={emit toggleTheme}
    Column pad=24 gap=20 maxw=720
      Input "Look up a word" "serendipity, petrichor, quixotic" @q icon="search" onSubmit={recent=unique(concat([q],recent)); go word}
      if q!=""
        Card gap=4
          Text.muted "Suggestions for {q}"
          each s in filter(words, x=>startsWith(x.w,lower(q)))
            ListTile "{s.w}" "{s.pos}: {s.gloss}" icon="north_west" onTap={recent=unique(concat([s.w],recent)); q=""; go word}
          if count(words, x=>startsWith(x.w,lower(q)))==0
            Text.muted "No word begins with \"{q}\". Check the spelling."
      Card gap=12
        Row between
          Text.muted "Word of the day"
          Badge.info "Fri 17 Nov"
        Text.heading1 "petrichor"
        Row gap=8
          Text.mono "/ˈpɛtrɪkɔːr/"
          IconButton "volume_up" tooltip="Hear it" onTap={play pronunciation}
        Text "The pleasant, earthy smell that comes after rain falls on dry ground."
        Text.muted "From Greek petra, stone, and ichor, the fluid in the veins of the gods. Coined in 1964 by two Australian scientists."
        Row gap=8
          Button "Read the history" onTap={q="petrichor"; go word}
          Button.outline "Save" icon="bookmark_add" onTap={saved=contains(saved,"petrichor") ? saved : append(saved,"petrichor"); toast "Saved to My lists"}
      Card gap=8
        Row between
          Text.strong "Recent searches"
          Button.ghost "Clear" onTap={recent=[]}
        if isEmpty(recent)
          Text.muted "Nothing yet. Words you look up appear here."
        Row gap=8
          each r in recent
            Chip "{r}" onTap={q=r; go word} onDelete={recent=without(recent,r)}

### Lists
state saved=["petrichor","serendipity","quixotic","nefarious","ephemeral"]
state lists=[{"name":"Favourites","words":["petrichor","serendipity"]},{"name":"Weird words","words":["quixotic","nefarious"]},{"name":"Exam prep","words":["ephemeral"]}]
state activeList="Favourites"
state sortBy="A to Z"
state filterQ=""
state sel=""
state newName=""
Page
  Nav sticky
    Brand "Etymon" icon="menu_book"
    Row gap=4
      Link "Search" onTap={go search}
      Link "Word of the day" onTap={go word}
      Link "My lists" onTap={go lists}
    Button "New list" icon="add" onTap={open newListDialog}
  Section
    Column gap=16
      Heading "My lists" level=1
      Row gap=12 wrap
        each l in lists
          ChoiceChip "{l.name} ({length(l.words)})" selected={activeList==l.name} onTap={activeList=l.name} menu=listMenu
        Spacer grow
        Input "Filter words" @filterQ icon="filter_list" w=220
        Select @sortBy items=["A to Z","Z to A","Newest"] w=140
      Grid cols=3 gap=16
        each w in filter(sortBy=="Z to A" ? reverse(find(lists,x=>x.name==activeList).words) : find(lists,x=>x.name==activeList).words, x=>contains(x,lower(filterQ)))
          Card gap=8 menu=wordRow
            Row between
              Text.heading3 "{w}"
              IconButton "more_vert" tooltip="Options" onTap={sel=w; open wordRow}
            Sparkline values=[1,2,2,4,6,9,14] h=32
            Text.muted "Tap to read its history"
            Button.outline "Open" onTap={go word}
      if isEmpty(find(lists,x=>x.name==activeList).words)
        Card gap=8
          Text.muted "This list is empty. Save a word from its page and it appears here."
          Button "Find a word" onTap={go search}
  Footer
    Row between
      Text.strong "Etymon"
      Text.muted "Word histories from the Oxford and Wiktionary corpora"
  Dialog#newListDialog "New list" "Group words you want to keep"
    Input "Name" "Weird words" @newName
    Button "Create" onTap={newName!="" ? {lists=append(lists,{"name":newName,"words":[]}); activeList=newName; newName=""; close} : toast "Give the list a name"}
  Menu#wordRow end
    MenuItem "Move to another list" icon="drive_file_move" onTap={open moveSheet}
    MenuItem "Remove from list" icon="delete" onTap={lists=updateWhere(lists,x=>x.name==activeList,x=>with(x,"words",without(x.words,sel))); saved=without(saved,sel); toast "Removed {sel}"}
  Menu#listMenu
    MenuItem "Rename" icon="edit" onTap={open newListDialog}
    MenuItem "Delete list" icon="delete" onTap={lists=removeWhere(lists,x=>x.name==activeList); activeList=lists[0].name; toast "List deleted"}
  Sheet#moveSheet "Move {sel}"
    each l in lists
      ListTile "{l.name}" icon="folder" onTap={lists=updateWhere(updateWhere(lists,x=>x.name==activeList,x=>with(x,"words",without(x.words,sel))),y=>y.name==l.name,y=>with(y,"words",append(y.words,sel))); toast "Moved to {l.name}"; close}
