### Player
state words=["Once","upon","a","time,","a","sleepy","fox","curled","up","beneath","the","silver","moon","and","listened","to","the","quiet","hills","hum","goodnight."]
state cur=0
state playing=true
state lullaby=false
state voice="Luna"
state timerLeft=900
state fav=false
state dim=0
every 450 when playing onTick={cur=min(cur+1,length(words)); dim=clamp(cur/length(words),0,1)}
every 1s when playing onTick={timerLeft=max(0,timerLeft-1)}
App
  Main
    Box fill=#FF0B1026 pad=16 gap=16 grow
      Row between
        IconButton "arrow_back" tooltip="Back to shelf" color=#FFFFFFFF onTap={go shelf}
        Column center gap=4
          MoonProgress value={cur/length(words)} dim={dim} size=72
          Text.caption "{duration(timerLeft)} until sleep" color=#FFB0B8D0
        IconButton "{fav ? 'favorite' : 'favorite_border'}" tooltip="Favourite" color=#FFFFFFFF onTap={fav=!fav; toast fav ? "Removed from favourites" : "Added to favourites"}
      Text.title "The Sleepy Fox" color=#FFFFFFFF center
      Text.caption "Mira Bell · read by {voice}" color=#FFB0B8D0 center
      Row gap=10 center wrap pad=16
        each w in words
          Text "{w}" size=34 weight={index==cur ? 800 : 500} color={index==cur ? "#FFFFD54F" : (index<cur ? "#FF6E7699" : "#FFE8EAF6")} animate={index==cur ? "pulse" : "fade"}
      Slider @cur min=0 max={length(words)}
      Row between
        Text.caption "{cur} of {length(words)} words" color=#FFB0B8D0
        Text.caption "{round(cur/length(words)*100,0)}%" color=#FFB0B8D0
      Row gap=16 center middle
        IconButton "replay_10" tooltip="Back" color=#FFFFFFFF onTap={cur=max(0,cur-10)}
        IconButton "{playing ? 'pause_circle' : 'play_circle'}" tooltip="Play or pause" size=72 color=#FFFFD54F onTap={playing=!playing}
        IconButton "forward_10" tooltip="Forward" color=#FFFFFFFF onTap={cur=min(length(words),cur+10)}
    Footer
      Row between pad=8
        Toggle "Lullaby mode" @lullaby onChange={emit lullaby result}
        Button.outline "Voice: {voice}" icon="record_voice_over" onTap={open voiceMenu}
        Button.outline "Timer" icon="timer" onTap={open timerPop}
  Menu#voiceMenu above
    MenuItem "Luna (soft)" icon="check" onTap={voice="Luna"; emit voice voice}
    MenuItem "Orion (deep)" icon="check" onTap={voice="Orion"; emit voice voice}
    MenuItem "Willow (whisper)" icon="check" onTap={voice="Willow"; emit voice voice}
  Popover#timerPop
    Column gap=8
      Text.strong "Add time"
      Row gap=8
        Button.outline "+5 min" onTap={timerLeft=timerLeft+300}
        Button.outline "+10 min" onTap={timerLeft=timerLeft+600}
      Button.ghost "Stop timer" onTap={timerLeft=0; playing=false; close}
