### Round
state tab="round"
state sel=""
state areas=[{"id":"a1","name":"Helideck","deck":"Top","result":"Pass","severity":"","note":"","photo":""},{"id":"a2","name":"Derrick floor","deck":"Drill","result":"","severity":"","note":"","photo":""},{"id":"a3","name":"Mud pits","deck":"Main","result":"","severity":"","note":"","photo":""},{"id":"a4","name":"Pump room","deck":"Main","result":"","severity":"","note":"","photo":""},{"id":"a5","name":"Lifeboat station B","deck":"Cellar","result":"","severity":"","note":"","photo":""},{"id":"a6","name":"Generator hall","deck":"Cellar","result":"","severity":"","note":"","photo":""}]
state gas={"h2s":2,"co":18,"lel":4,"o2":20.9}
App
  Sidebar
    Brand "Derrick Daily" icon="oil_barrel"
    NavItem "Round" icon="checklist" badge="{count(areas,a=>a.result==\"\")}" selected={tab=="round"} onTap={tab="round"; go round}
    NavItem "Deck map" icon="map" selected={tab=="map"} onTap={tab="map"; go deck-map}
    NavItem "Muster" icon="groups" selected={tab=="muster"} onTap={tab="muster"; go muster}
    NavItem "Sign-off" icon="draw" selected={tab=="signoff"} onTap={tab="signoff"; go sign-off}
    Spacer grow
    Text.muted "Day shift, 06:00 to 18:00"
    Text.muted "J. Okafor, HSE lead"
  Main
    Nav
      Heading "Safety round" level=1
      Row gap=8
        Badge.info "{count(areas,a=>a.result!=\"\")} of {length(areas)} answered"
        Button.outline "Reset round" onTap={areas=map(areas,a=>with(a,"result",""))}
    Column pad=24 gap=16 scroll
      Progress {count(areas,a=>a.result!="")/length(areas)}
      Card gap=12
        Text.strong "Gas readings"
        Grid cols=4 gap=12
          Input "H2S (ppm)" "limit 10" type=number bind=gas.h2s invalid={gas.h2s>10}
          Input "CO (ppm)" "limit 35" type=number bind=gas.co invalid={gas.co>35}
          Input "LEL (%)" "limit 10" type=number bind=gas.lel invalid={gas.lel>10}
          Input "O2 (%)" "19.5 to 23.5" type=number bind=gas.o2 invalid={gas.o2<19.5 || gas.o2>23.5}
        if gas.h2s>10 || gas.co>35 || gas.lel>10 || gas.o2<19.5 || gas.o2>23.5
          Row gap=8
            Icon "warning" color=#FFD32F2F
            Text.danger "A reading is over its limit. Stop work and report to the control room."
            Spacer grow
            Button.destructive "Call control room" onTap={emit callControl gas}
      each a in areas
        Card gap=12
          Row between
            Column gap=2
              Text.strong "{a.name}"
              Text.muted "{a.deck} deck"
            Row gap=8
              if a.result=="Fail"
                Badge.error "Fail, {a.severity}"
              if a.result=="Pass"
                Badge.success "Pass"
              if a.result=="N/A"
                Badge.neutral "Not applicable"
          SegmentedButton items=["Pass","Fail","N/A"] bind=a.result onChange={result=="Fail" ? {sel=a.id; open failure} : toast "{a.name} marked {result}"}
          if a.result=="Fail"
            Row gap=8
              Text.muted "{a.note}"
              Spacer grow
              Button.outline "Edit report" onTap={sel=a.id; open failure}
  Sheet#failure "Failed check"
    each a in filter(areas,x=>x.id==sel)
      Column gap=12 pad=16
        Text.strong "{a.name}, {a.deck} deck"
        Select "Severity" items=["Minor","Major","Critical"] bind=a.severity
        Input "What did you find?" multiline bind=a.note
        Row gap=12
          Button.outline "Take photo" icon="photo_camera" onTap={emit takePhoto a.id}
          Button.outline "Choose from gallery" icon="image" onTap={emit pickPhoto a.id}
        if a.photo!=""
          Image "{a.photo}" h=160 radius=8
        Row end gap=8
          Button.ghost "Cancel" onTap={close}
          Button "Save report" onTap={a.severity=="" ? toast "Choose a severity" : {emit failReported a.id; toast "Report saved"; close}}

### Sign-off
state tab="signoff"
state areas=[{"id":"a1","name":"Helideck","result":"Pass"},{"id":"a2","name":"Derrick floor","result":"Fail"},{"id":"a3","name":"Mud pits","result":"Pass"},{"id":"a4","name":"Pump room","result":""},{"id":"a5","name":"Lifeboat station B","result":"N/A"},{"id":"a6","name":"Generator hall","result":""}]
state name=""
state declared=false
state signed=false
Page
  Nav
    Brand "Derrick Daily" icon="oil_barrel"
    Row gap=4
      Link "Round" onTap={go round}
      Link "Deck map" onTap={go deck-map}
      Link "Muster" onTap={go muster}
      Link "Sign-off" onTap={go sign-off}
  Section narrow
    Column gap=16
      Heading "Sign off the round" level=1
      Text.muted "The round can be signed once every area has an answer."
      Card gap=8
        Row between
          Text.strong "Areas answered"
          Text.strong "{count(areas,a=>a.result!=\"\")} of {length(areas)}"
        Progress {count(areas,a=>a.result!="")/length(areas)}
        each a in areas
          Row between
            Text "{a.name}"
            if a.result==""
              Badge.warning "Unanswered"
            if a.result=="Pass"
              Badge.success "Pass"
            if a.result=="Fail"
              Badge.error "Fail"
            if a.result=="N/A"
              Badge.neutral "N/A"
      if count(areas,a=>a.result=="")>0
        Row gap=8
          Icon "lock" color=#FFD32F2F
          Text.danger "Locked: {count(areas,a=>a.result==\"\")} areas still to answer."
      if count(areas,a=>a.result=="Fail")>0
        Text.muted "{count(areas,a=>a.result==\"Fail\")} failed checks will go to the HSE lead with this sign-off."
      Input "Your full name" bind=name disabled={count(areas,a=>a.result=="")>0}
      Checkbox "I walked every area and the answers are true" bind=declared disabled={count(areas,a=>a.result=="")>0}
      Button "Sign off round" icon="draw" disabled={count(areas,a=>a.result=="")>0 || trim(name)=="" || !declared} onTap={open confirm}
      if signed
        Row gap=8
          Icon "verified" color=#FF2E7D32
          Text.success "Signed by {name}"
  Dialog#confirm "Sign off this round?" "This cannot be changed afterwards."
    Text "{name} signs off {length(areas)} areas, {count(areas,a=>a.result==\"Fail\")} failed."
    Row end gap=8
      Button.ghost "Go back" onTap={close}
      Button "Confirm and sign" onTap={signed=true; emit signedOff name; toast "Round signed"; close}
