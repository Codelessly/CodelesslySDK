### Weigh desk
state tab="desk"
state t=0
state locked=false
state target=41280
state plate=""
state product="Crushed granite 20mm"

  Main
    Nav
      Heading "Weigh desk" level=1
      Row gap=8
        Badge.success "Scale connected" hide=phone
        Button.outline "Zero platform" onTap={t=0; locked=false; toast "Platform zeroed"}
    Column pad=24 gap=16 scroll
      Row gap=16
        Card grow gap=12
          Row between
            Text.panelTitle "Live platform weight"
            if locked
              Badge.success "LOCKED"
            if !locked
              Badge.warning "SETTLING"
          Stack center middle h=140
            Text "{locked ? target : target + ((t*37)%120) - 60} kg" size=64 weight=700 animate={locked ? "pulse" : "shimmer"}
          Row gap=8 center
            Button "Lock and record" onTap={locked ? (current==null ? toast "Pick a truck first" : (firstWeigh==0 ? firstWeigh=target : secondWeigh=target)) : toast "Wait for a steady reading"}
            Button.outline "Re-weigh" onTap={t=0; locked=false}
        Card grow gap=12
          Text.panelTitle "Truck"
          Row gap=8
            Input "Plate" "e.g. KX 4821" @plate grow
            IconButton "photo_camera" tooltip="Capture plate" onTap={open plateCam}
          Select "Product" @product items={map(products,p=>p.name)}
          Text.muted "Price per tonne: {money(find(products,p=>p.name==product).price,"$",2)}"
          Button "Start ticket" onTap={plate=="" ? toast "Enter a plate" : (current=plate; firstWeigh=0; secondWeigh=0; t=0; locked=false; toast "Ticket opened for {plate}")}
      Row gap=16
        Card grow gap=8
          Text.panelTitle "First weigh (in)"
          Text "{firstWeigh==0 ? "-" : firstWeigh + " kg"}" size=32 weight=700
          Text.muted "Loaded truck on arrival"
        Card grow gap=8
          Text.panelTitle "Second weigh (out)"
          Text "{secondWeigh==0 ? "-" : secondWeigh + " kg"}" size=32 weight=700
          Text.muted "Empty truck on departure"
        Card grow gap=8 fill=#FF14213D
          Text.panelTitle "Net" color=#FFFFFFFF
          Text "{firstWeigh>0 && secondWeigh>0 ? fixed(abs(firstWeigh-secondWeigh)/1000,2) + " t" : "-"}" size=48 weight=700 color=#FFFFFFFF
          Text "{firstWeigh>0 && secondWeigh>0 ? money(abs(firstWeigh-secondWeigh)/1000*find(products,p=>p.name==product).price,"$",2) : "Awaiting both weights"}" color=#FFE5E5E5
      if firstWeigh>gvmLimit
        Card gap=12 fill=#FFFDECEA animate=shake
          Row gap=8
            Icon "warning" color=#FFB3261E
            Text.strong "Overweight: {firstWeigh-gvmLimit} kg over the {gvmLimit} kg gross limit" color=#FFB3261E
          each a in axles
            Row gap=12
              Text a.name w=110
              Progress {min(1,a.kg/a.limit)} grow
              Text.strong "{a.kg} / {a.limit} kg" w=150
              if a.kg>a.limit
                Badge.error "Over by {a.kg-a.limit}"
              if a.kg<=a.limit
                Badge.success "OK"
          Row gap=8
            Button.destructive "Reject load" onTap={emit rejectLoad current; toast "Load rejected"; current=null}
            Button.outline "Allow with permit" onTap={open permit}
      Row end gap=8
        Button.outline "Cancel" onTap={current=null; firstWeigh=0; secondWeigh=0}
        Button "Print ticket" onTap={firstWeigh==0 || secondWeigh==0 ? toast "Record both weights" : (lastTicket={"code":"WB-" + (1040+length(tickets)),"plate":current,"product":product,"net":abs(firstWeigh-secondWeigh),"amount":abs(firstWeigh-secondWeigh)/1000*find(products,p=>p.name==product).price}; tickets=append(tickets,lastTicket); queue=removeWhere(queue,q=>q.plate==current); emit printTicket lastTicket; go ticket)}
    Pane w=320 pad=16 gap=12
      Text.panelTitle "Trucks waiting ({length(queue)})"
      each q in queue
        ListTile "{q.plate}" subtitle="{q.driver}, {q.firm}" onTap={plate=q.plate; toast "Loaded {q.plate}"}
  Dialog#plateCam "Capture plate" "Hold the plate inside the frame"
    Camera facing=back onCapture={plate=result.text; close; toast "Read {result.text}"}
    Button.outline "Cancel" onTap={close}
  Dialog#permit "Overweight permit" "Record the permit that allows this load"
    Input "Permit number" @permitNo
    Input "Issued by" @permitBy
    Button "Allow load" onTap={permitNo=="" ? toast "Permit number required" : (toast "Permit {permitNo} recorded"; close)}
