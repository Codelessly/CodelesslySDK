### Calibration
state tab="calibration"
state peaks=[]
state step=1
state ref1=0
state ref2=0
state lampOn=true
App
  Sidebar
    Brand "Stellarline" icon="auto_awesome"
    NavItem "Spectrum" icon="show_chart" selected={tab=="spectrum"} onTap={go spectrum}
    NavItem "Calibration" icon="tune" selected={tab=="calibration"} onTap={go calibration}
    NavItem "Exposure" icon="timer" selected={tab=="exposure"} onTap={go exposure}
    NavItem "Targets" icon="my_location" badge="8" selected={tab=="targets"} onTap={go targets}
    Spacer grow
    Text.muted "Mount Alder Observatory"
  Main
    Nav
      Heading "Lamp calibration" level=1
      Row gap=8
        Toggle "Neon lamp" bind=lampOn
        Button.outline "Start over" onTap={peaks=[]; step=1; ref1=0; ref2=0}
    Column pad=24 gap=16
      Card gap=12
        Row between
          Text.panelTitle "Step {step} of 3"
          Progress {step/3} w=160
        if step==1
          Text.body "Tap two bright peaks on the neon lamp spectrum."
        if step==2
          Text.body "Give each peak its known wavelength."
        if step==3
          Text.success "Calibrated. Dispersion 0.47 nm per pixel."
        SpectrumChart values={lampSpectrum} h=260 peaks={peaks} onPeakTap={peaks=length(peaks)<2 ? append(peaks,result) : peaks; length(peaks)==2 ? step=2 : step=step}
      if step>=2
        Grid cols=2 gap=16
          Card gap=8
            Text.fieldLabel "Peak A at pixel {peaks[0].px}"
            Input "Known wavelength (nm)" "585.249" type=number bind=ref1
          Card gap=8
            Text.fieldLabel "Peak B at pixel {peaks[1].px}"
            Input "Known wavelength (nm)" "640.225" type=number bind=ref2
        Row end gap=8
          Button.outline "Use neon table" onTap={ref1=585.249; ref2=640.225}
          Button "Apply calibration" onTap={ref1>0 && ref2>ref1 ? step=3; emit calibrate {"a":ref1,"b":ref2}; toast "Calibration applied" : toast "Enter both wavelengths"}

### Exposure
state tab="exposure"
state exposure=120
state remaining=120
state running=false
state frames=[{"n":1,"noise":9.4},{"n":2,"noise":6.8},{"n":3,"noise":5.5},{"n":4,"noise":4.8},{"n":5,"noise":4.2},{"n":6,"noise":3.9}]
every 1s when running onTick={remaining=remaining-1; remaining<=0 ? running=false; frames=append(frames,{"n":length(frames)+1,"noise":round(9.4/(length(frames)+1)*1.6,1)}); remaining=exposure; toast "Frame {length(frames)+1} stored" : }
App
  Sidebar
    Brand "Stellarline" icon="auto_awesome"
    NavItem "Spectrum" icon="show_chart" selected={tab=="spectrum"} onTap={go spectrum}
    NavItem "Calibration" icon="tune" selected={tab=="calibration"} onTap={go calibration}
    NavItem "Exposure" icon="timer" selected={tab=="exposure"} onTap={go exposure}
    NavItem "Targets" icon="my_location" badge="8" selected={tab=="targets"} onTap={go targets}
    Spacer grow
    Text.muted "Mount Alder Observatory"
  Main
    Nav
      Heading "Exposure" level=1
      Badge.info "{length(frames)} frames stacked"
    Column pad=24 gap=16
      Row gap=16
        Card grow gap=12 center
          Stack center middle
            Ring {1-remaining/exposure} w=200 h=200
            Column center
              Text "{duration(remaining)}" size=40 weight=700
              Text.caption "remaining"
          Row gap=8 center
            Button "Start" icon="play_arrow" hidden={running} onTap={running=true; remaining=exposure}
            Button.destructive "Stop" icon="stop" visible={running} onTap={running=false; remaining=exposure}
        Card grow gap=12
          Text.panelTitle "Exposure time"
          Slider bind=exposure min=10 max=600 onChange={remaining=result}
          Text.mono "{exposure} s"
          Row gap=8
            ChoiceChip "30 s" selected={exposure==30} onTap={exposure=30; remaining=30}
            ChoiceChip "120 s" selected={exposure==120} onTap={exposure=120; remaining=120}
            ChoiceChip "300 s" selected={exposure==300} onTap={exposure=300; remaining=300}
          Select "Binning" items=["1x1","2x2","4x4"] bind=binning
          Text.caption "Total on target {duration(length(frames)*exposure)}"
      Card gap=12
        Row between
          Text.panelTitle "Stack of previous exposures"
          Button.outline "Clear stack" icon="delete" onTap={open clearConfirm}
        LineChart values={map(frames,f=>f.noise)} labels={map(frames,f=>"#{f.n}")} h=180
        Text.muted "Noise falls from {first(frames).noise} to {last(frames).noise} counts as frames are averaged"
        each f in reverse(frames)
          ListTile "Frame {f.n}" subtitle="Noise {f.noise}"
            IconButton "close" tooltip="Drop from stack" onTap={frames=removeWhere(frames,x=>x.n==f.n)}
Dialog#clearConfirm "Clear the stack?" "All stored frames will be removed."
  Row end gap=8
    Button.outline "Cancel" onTap={close}
    Button.destructive "Clear" onTap={frames=[]; close; toast "Stack cleared"}
