### Bench
state tool="select"
state running=true
state parts=[{"id":"b1","kind":"battery","label":"Battery","value":9,"unit":"V","x":1,"y":3},{"id":"r1","kind":"resistor","label":"R1","value":220,"unit":"ohm","x":4,"y":1},{"id":"l1","kind":"lamp","label":"Lamp","value":60,"unit":"ohm","x":7,"y":3},{"id":"s1","kind":"switch","label":"Switch","value":1,"unit":"","x":4,"y":5}]
state wires=[{"id":"w1","from":"b1","to":"r1"},{"id":"w2","from":"r1","to":"l1"},{"id":"w3","from":"l1","to":"s1"},{"id":"w4","from":"s1","to":"b1"}]
state sel=""
state probeAt=""
state probeVolts=0
state current=0.032
state volts=[0,2,4,6,7,8,8,9]
state shorted=false
state time=0
every 500 when running onTick={time=time+0.5; volts=append(take(reverse(volts),39),current*220)}
every 1s when running onTick={shorted=any(wires,w=>w.from==w.to) ? true : shorted}
if shorted
  every 100 onTick={running=false; open shortCircuit}
App
  Sidebar
    Brand "Solderless" icon="electrical_services"
    Text.muted "Parts tray"
    Column gap=8
      Card drag=part("battery") gap=4
        Row gap=8 middle
          IconTile "battery_full"
          Text.strong "Battery"
      Card drag=part("resistor") gap=4
        Row gap=8 middle
          IconTile "polyline"
          Text.strong "Resistor"
      Card drag=part("lamp") gap=4
        Row gap=8 middle
          IconTile "lightbulb"
          Text.strong "Lamp"
      Card drag=part("switch") gap=4
        Row gap=8 middle
          IconTile "toggle_on"
          Text.strong "Switch"
      Card drag=part("capacitor") gap=4
        Row gap=8 middle
          IconTile "graphic_eq"
          Text.strong "Capacitor"
    Spacer grow
    Button.outline "Examples" icon="school" onTap={go examples}
    Text.muted "{length(parts)} parts, {length(wires)} wires"
  Main
    Nav
      Heading "Series lamp circuit"
      Row gap=8
        SegmentedButton @tool items=["select","wire","probe"]
        IconButton "undo" tooltip="Undo"
        IconButton "delete_sweep" tooltip="Clear bench" onTap={parts=[]; wires=[]; sel=""}
        Button "{running ? 'Pause' : 'Run'}" icon="{running ? 'pause' : 'play_arrow'}" onTap={running=!running}
    Column pad=24 gap=16
      Row gap=16 top
        Card grow gap=8
          Row between
            Text.strong "Bench"
            Row gap=8
              Badge.success "{fixed(current*1000,0)} mA"
              if tool=="probe"
                Badge.info "Probe: {probeAt=="" ? 'tap a point' : fixed(probeVolts,2) + ' V'}"
          CircuitCanvas @parts wires=wires cols=12 rows=8 running={running} tool={tool} h=420 drop=part onDrop={parts=append(parts,{"id":concat("p",length(parts)),"kind":result.data,"label":result.data,"value":10,"unit":"","x":result.x,"y":result.y})} onWire={wires=append(wires,result)} onProbe={probeAt=result.node; probeVolts=result.volts} onPartTap={sel=result.id; open partSheet} onSwitch={emit switchToggled result.id; toast result.on ? "Circuit closed" : "Circuit open"} onShort={shorted=true; running=false; open shortCircuit}
          each p in parts
            if p.kind=="lamp"
              Row gap=12 middle
                Icon "lightbulb"
                Text.strong "{p.label}"
                Progress {clamp(current*20,0,1)}
                Text.muted "{fixed(current*current*p.value,2)} W"
        Pane w=320 pad=16 gap=12
          Heading "Scope" level=3
          LineChart values={volts} h=180
          Row between
            Text.muted "Supply"
            Text.strong "{fixed(last(volts),2)} V"
          Row between
            Text.muted "Elapsed"
            Text.strong "{duration(time)}"
          Divider
          Text.strong "Probe readings"
          if probeAt==""
            Text.muted "Choose the probe tool and tap a node."
          if probeAt!=""
            Row between
              Text "{probeAt}"
              Text.strong "{fixed(probeVolts,2)} V"
          Button.outline "Open full scope" icon="show_chart" onTap={go scope}
  Sheet#partSheet "Edit part"
    each p in filter(parts,x=>x.id==sel)
      Column gap=12
        Input "Label" @p.label
        Row between middle
          Text.strong "Value ({p.unit})"
          Row gap=8 middle
            IconButton "remove" tooltip="Decrease" onTap={parts=updateWhere(parts,x=>x.id==sel,x=>with(x,"value",max(1,x.value-1)))}
            Text.strong "{p.value}"
            IconButton "add" tooltip="Increase" onTap={parts=updateWhere(parts,x=>x.id==sel,x=>with(x,"value",x.value+1))}
        if p.kind=="switch"
          Toggle "Closed" @p.value
        Row gap=8
          Button.destructive "Remove part" onTap={parts=removeWhere(parts,x=>x.id==sel); wires=removeWhere(wires,w=>w.from==sel||w.to==sel); close}
          Spacer grow
          Button "Done" onTap={close}
  Dialog#shortCircuit "Short circuit" "Current is unlimited through a wire with no load."
    Icon "warning" color=#FFD32F2F size=40
    Text "The run was stopped before anything burned out. Add a resistor or lamp between the battery terminals."
    Row gap=8
      Button.outline "Show the wire" onTap={tool="select"; close}
      Spacer grow
      Button "Fix and resume" onTap={shorted=false; running=true; close}
