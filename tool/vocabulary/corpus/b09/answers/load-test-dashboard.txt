### Scenario

state name="Checkout peak"
state target="https://staging.shop.example.com"
state points=[{"t":0,"vus":0},{"t":60,"vus":200},{"t":180,"vus":800},{"t":420,"vus":800},{"t":480,"vus":150},{"t":540,"vus":0}]

  Sidebar
    Brand "Hammer" icon="hardware"
    NavItem "Scenario" icon="tune" selected={tab=="scenario"} onTap={tab="scenario"; go scenario}
    NavItem "Live run" icon="monitoring" badge="live" selected={tab=="live"} onTap={tab="live"; go live-run}
    NavItem "Compare" icon="compare_arrows" selected={tab=="compare"} onTap={tab="compare"; go compare}
    NavItem "Report" icon="fact_check" selected={tab=="report"} onTap={tab="report"; go report}
    Spacer grow
    Text.muted "Dana Whitfield"
  Main
    Nav
      Heading "Scenario" level=1
      Row gap=8
        Button.outline "Save" icon="save" onTap={emit saveScenario points; toast "Scenario saved"}
        Button "Start run" icon="play_arrow" onTap={open countdown}
    Column pad=24 gap=16
      Grid cols=2 gap=16
        Input "Scenario name" @name
        Input "Target" "https://" @target
      Card gap=12
        Row between
          Column gap=4
            Text.strong "Virtual users over time"
            Text.muted "Drag a point to reshape the ramp. Tap the line to add a point."
          Row gap=8
            Button.outline "Add point" icon="add" onTap={points=append(points,{"t":max(points)+30,"vus":100})}
            Button.ghost "Remove point" icon="remove" onTap={sel>=0 ? {points=removeAt(points,sel); sel=-1} : toast "Select a point first"}
        RampEditor @points selected=sel xLabel="seconds" yLabel="virtual users" max=1000 h=260 snap=10 onSelect={sel=result} onChange={points=result}
        Row gap=24
          Text.caption "Peak {max(map(points,p=>p.vus),0)} users"
          Text.caption "Length {last(points).t} s"
      Grid cols=3 gap=16
        Card gap=8
          Text.fieldLabel "Fail the run if p95 latency crosses"
          Slider @thresholdMs min=100 max=3000
          Text.strong "{thresholdMs} ms"
        Card gap=8
          Text.fieldLabel "Fail the run if errors exceed"
          Slider @maxErr min=0 max=20
          Text.strong "{maxErr}%"
        Card gap=8
          Text.fieldLabel "Countdown before start"
          SegmentedButton @count items=["5","10","30"]
          Text.muted "seconds"
  Dialog#countdown "Starting {name}" "Virtual users begin ramping when the count reaches zero."
    Column center gap=12
      Text "{count}" size=72 weight=700 center
      Progress {1 - count/10}
    Row gap=8 end
      Button.outline "Cancel" onTap={running=false; close}
      Button "Start now" onTap={emit startRun {"name":name,"points":points,"thresholdMs":thresholdMs,"maxErr":maxErr}; close; go live-run}
every 1s when running onTick={count=count-1}
