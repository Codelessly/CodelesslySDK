state zone="B2"

every 1s when running onTick={elapsed=elapsed+1}
App
  Main
    Nav
      Brand "Chalkmark" icon="local_parking"
      Row gap=8
        Badge.info "Zone {zone}"
        IconButton "history" tooltip="Day's tally" onTap={go tally}
    Column pad=16 gap=16 scroll
      Card gap=12
        Row between
          Text.sectionTitle "Plate"
          Button.ghost "Scan" icon="photo_camera" onTap={scanning=true; open scan}
        Input "Plate number" "e.g. 7KLM402" @plate type=text
        Row gap=8
          Button "Check permit" grow onTap={result=any(permits,p=>p.plate==upper(replace(plate," ",""))&&p.zone==zone&&p.until>=formatDate(today(),"yyyy-MM-dd")) ? "valid" : "invalid"; tallyChecked=tallyChecked+1; firstSeen=elapsed; running=true; elapsed=0}
          Button.outline "Clear" onTap={plate=""; result="none"; running=false; elapsed=0}
      if result=="valid"
        Box fill=#FF1B8A3E radius=12 pad=16 gap=4 animate=pop
          Row gap=12 middle
            Icon "check_circle" color=#FFFFFFFF size=40
            Column gap=2
              Text "PERMIT VALID" color=#FFFFFFFF size=24 weight=700
              Text "{find(permits,p=>p.plate==upper(replace(plate,' ','')))?.holder} - zone {zone}" color=#FFFFFFFF
      if result=="invalid"
        Box fill=#FFC62828 radius=12 pad=16 gap=4 animate=shake
          Row gap=12 middle
            Icon "cancel" color=#FFFFFFFF size=40
            Column gap=2
              Text "NO VALID PERMIT" color=#FFFFFFFF size=24 weight=700
              Text "{upper(plate)} not on the zone {zone} list" color=#FFFFFFFF
      if result=="invalid"
        Card gap=12
          Row between
            Text.sectionTitle "Grace period"
            Badge.warning "{graceMin} min"
          Progress {clamp(elapsed/(graceMin*60),0,1)}
          Row between
            Text.mono "{duration(elapsed)} since first seen"
            Text.muted "{elapsed >= graceMin*60 ? 'Grace over' : 'Left: ' + duration(graceMin*60-elapsed)}"
          Row gap=8
            Button.outline "Pause" grow onTap={running=false}
            Button.outline "Resume" grow onTap={running=true}
          Button "Write notice" disabled={elapsed < graceMin*60} onTap={go evidence}
          if elapsed < graceMin*60
            Text.caption "A notice can be written once the grace time has run out."
      Card gap=8
        Row between
          Text.sectionTitle "Zone map"
          Button.ghost "Open" icon="map" onTap={go zone}
        MapView h=140 center={[47.6062,-122.3321]} zoom=17 pin={[47.6062,-122.3321]} pinLabel="You are here"
        Text.caption "Pinned at 1420 Alder St, zone {zone}"
  Footer
    Row between pad=8
      NavItem "Check" icon="search" selected onTap={go check-plate}
      NavItem "Zone" icon="map" onTap={go zone}
      NavItem "Evidence" icon="photo_camera" onTap={go evidence}
      NavItem "Tally" icon="bar_chart" onTap={go tally}
  Dialog#scan "Scan plate" "Hold the camera steady on the plate"
    CameraView h=220 overlay="plate" onCapture={plate=result.text; scanning=false; close; toast "Read {result.text}"}
    Row gap=8
      Button.outline "Cancel" grow onTap={scanning=false; close}
      Button "Capture frame" grow onTap={emit captureFrame}

  Main
    Nav
      IconButton "arrow_back" tooltip="Back" onTap={go check-plate}

    Column pad=16 gap=16 scroll
      Card gap=12
        Text.sectionTitle "Photos"
        Grid cols=4 gap=8
          each i in range(4)
            Box h=72 radius=8 fill=#FFE0E0E0 onTap={emit takePhoto}
              if i < length(shots)
                Image "{shots[i].url}"
              if i >= length(shots)
                Icon "add_a_photo"
        Row gap=8
          Button "Take photo" icon="photo_camera" grow disabled={length(shots) >= 4} onTap={shots=append(shots,{"url":"https://img.chalkmark.app/e/" + plate + "-" + (length(shots)+1) + ".jpg"}); toast "Photo saved"}
          Button.outline "Retake last" onTap={shots=removeAt(shots,length(shots)-1)}
        Toggle "Wheel chalk visible" @wheelOn
        Toggle "Sign in frame" @signOn
        if wheelOn && signOn && length(shots) >= 2
          Text.success "Evidence complete"
      Card gap=8
        Text.sectionTitle "Fine type"
        Select "Violation" @fine items={map(fines,f=>f.name)}
        if fine != ""
          Row between
            Text.muted "Fine amount"
            Text.strong "{money(find(fines,f=>f.name==fine).amount,'$',2)}"
        Button "Preview notice" disabled={fine==""||length(shots)<4} onTap={open notice}
        if length(shots) < 4
          Text.caption "Four photos are needed before a notice."
  Footer
    Row between pad=8
      NavItem "Check" icon="search" onTap={go check-plate}
      NavItem "Zone" icon="map" onTap={go zone}

      Text.mono "Issued: {formatDate(today(),'MMM d, yyyy')}"
      Text.mono "Zone: B2, 1420 Alder St"
