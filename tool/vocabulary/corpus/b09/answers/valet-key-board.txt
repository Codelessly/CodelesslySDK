### Retrieve
state eta=420
state running=true
state tip=0
state otherTip=""
state arrived=false
state floor="B"
state minutesOver=0
App
  Sidebar
    Brand "Keyhook" icon="key"
    NavItem "Key board" icon="grid_view" onTap={go key-board}
    NavItem "Retrievals" icon="directions_car" selected={true}
    Spacer grow
    Text.muted "Evening shift, Marta O."
  Main
    Nav
      Heading "Retrieve: ticket 48190" level=1
      Row gap=8
        Button.outline "Cancel request" onTap={running=false; toast "Request cancelled"; go key-board}
    Column pad=24 gap=16
      every 1s when running onTick={eta=max(0,eta-1); arrived=eta==0; running=eta>0}
      every 60s onTick={minutesOver=minutesOver+1}
      Row gap=16 top
        Card grow gap=8
          Text.muted "Guest wants the car in"
          Text "{duration(eta)}" size=48 weight=700
          Progress {1-eta/420}
          Row gap=8
            Avatar "Dario Costa" size=40
            Column
              Text.strong "Runner Dario Costa"
              Text.muted "On the way to bay C07"
            Spacer grow
            IconButton "call" tooltip="Call runner" onTap={emit callRunner}
        Card grow gap=8
          Text.muted "Late fee"
          Text "{money(max(0,minutesOver-15)*2,"$",2)}" size=48 weight=700
          Text.muted "Free for 15 minutes, then $2 a minute"
          Badge.warning "Clock running" visible={minutesOver>15}
      Card gap=8
        Row between
          Heading "Where the car is" level=3
          SegmentedButton bind=floor items=["A","B","C"]
        GaragePlan bind=floor highlight="C07" h=240 onTap={emit pickBay result}
        Text.muted "Bay C07, level {floor}, 40 m from the lobby"
      Card gap=8
        Heading "Tip for the runner" level=3
        Row gap=8
          ChoiceChip "$5" selected={tip==5} onTap={tip=5}
          ChoiceChip "$10" selected={tip==10} onTap={tip=10}
          ChoiceChip "$20" selected={tip==20} onTap={tip=20}
          Input "Other" type=number bind=otherTip w=100 onChange={tip=coalesce(result,0)}
        Button "Add {money(tip,"$",2)} tip" disabled={tip==0} onTap={emit tip tip; toast "Tip added to the room bill"; tip=0}
      if arrived
        Card gap=8 fill=#FFE7F5EC
          Text.strong "Car is at the door"
          Button "Hand over key" onTap={emit handover; toast "Key handed over"; go key-board}
