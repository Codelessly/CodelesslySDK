### Quiz
state score=0
state asked=0
state current=null
state answer=""
state revealed=false
state difficulty="Symbols"
App
  Sidebar
    Brand "Elementary" icon="science"
    NavItem "Table" icon="grid_view" onTap={go table}
    NavItem "Compare" icon="compare_arrows" onTap={go compare}
    NavItem "Quiz" icon="quiz" selected={true}
  Main
    Nav
      Heading "Quiz"
      Row gap=12 middle
        Badge.success "Score {score} / {asked}"
        Button.ghost "Restart" icon="restart_alt" onTap={score=0; asked=0; answer=""; revealed=false; current=elements[floor(random()*length(elements))]}
    Column pad=24 gap=16
      SegmentedButton bind=difficulty items=["Symbols","Families","Numbers"]
      Card gap=12
        Text.muted "Names are hidden. Tap a cell to answer."
        PeriodicTable elements={elements} hideNames=true highlight={current.sym} onSelect={answer=result.name; revealed=true; asked=asked+1; score=result.name==current.name ? score+1 : score}
      Card gap=12
        Text.caption "Find the element"
        Text.heading "{difficulty=="Numbers" ? "Number " + current.n : current.sym}"
        Input "Or type its name" bind=answer
        Row gap=8
          Button "Check" onTap={revealed=true; asked=asked+1; lower(answer)==lower(current.name) ? score=score+1 : toast "It is {current.name}"}
          Button.outline "Skip" onTap={answer=""; revealed=false; current=elements[floor(random()*length(elements))]}
        if revealed
          Text.success "{current.name}: {current.fam}, discovered {current.disc}"
