      Radio "50 m, 72 arrows" bind=round

### Score sheet

state round="18 m"
state ends=[{"n":1,"shots":[10,9,9,8,8,7],"total":51},{"n":2,"shots":[10,10,9,9,8,6],"total":52},{"n":3,"shots":[9,9,9,8,7,7],"total":49}]

App
  Sidebar
    Brand "Gold Ring" icon="adjust"
    NavItem "Plot" icon="gps_fixed" selected={tab=="plot"} onTap={go plot}
    NavItem "Score sheet" icon="table_rows" selected={tab=="sheet"} onTap={go score-sheet}
    NavItem "Records" icon="emoji_events" selected={tab=="records"} onTap={go records}
    Spacer grow
    Text.muted "Round: {round}"
  Main
    Nav
      Heading "Score sheet" level=1
      Row gap=8
        Button.outline "Export" icon="download" onTap={emit exportSheet}
        Button "Share" icon="share" onTap={emit shareSheet}
    Column pad=24 gap=16
      Row gap=16
        Card grow gap=4
          Text.muted "Running total"
          Text.strong "{sum(ends,e=>e.total)}" size=40
        Card grow gap=4
          Text.muted "Average per arrow"
          Text.strong "{fixed(sum(ends,e=>e.total)/(length(ends)*6),2)}" size=40
        Card grow gap=4
          Text.muted "Tens"
          Text.strong "{sum(ends,e=>count(e.shots,s=>s==10))}" size=40
      Card gap=8
        Row gap=8
          Text.fieldLabel "End" w=50
          Text.fieldLabel "Arrows" grow
          Text.fieldLabel "Total" w=70
          Text.fieldLabel "Running" w=80 hide=phone
        Divider
        each e in ends
          Row gap=8 middle onTap={sel=index; open endDetail}
            Text.strong "{e.n}" w=50
            Row grow gap=4
              each s in e.shots
                Badge.neutral "{s}"
            Text.strong "{e.total}" w=70
            Text "{sum(slice(ends,0,index+1),x=>x.total)}" w=80 hide=phone
      Card gap=8
        Text.muted "Score by end"
        BarChart values={map(ends,e=>e.total)} labels={map(ends,e=>"E{e.n}")} h=160
  Dialog#endDetail "End details" "Arrows in the order they were shot"
    Column gap=12
      Row gap=8
        each s in ends[sel].shots
          Badge.primary "{s}"
      Row between
        Text.muted "Total"
        Text.strong "{ends[sel].total}"
      SpreadPlot points={[]} showCenter=true h=140
      Row gap=8
        Button.outline "Edit arrows" onTap={emit editEnd ends[sel].n; close}
        Button.destructive "Delete end" onTap={ends=removeAt(ends,sel); close; toast "End deleted"}
