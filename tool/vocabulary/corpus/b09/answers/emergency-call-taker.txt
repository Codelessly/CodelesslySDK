  Sidebar

  Main

      Row gap=12 middle

      Row gap=8
        Button.outline "Hold" icon="pause" hidden={onHold} onTap={onHold=true; holdSecs=0; timeline=append(timeline,{"t":elapsed,"who":"System","text":"Caller placed on hold"})}
        Button "Resume" icon="play_arrow" hidden={!onHold} onTap={onHold=false; timeline=append(timeline,{"t":elapsed,"who":"System","text":"Caller resumed after {duration(holdSecs)}"})}
        Button.destructive "End call" onTap={open endCall}
    Row grow top gap=0
      Column grow gap=16 pad=24 scroll
        Card gap=12
          Row between
            Text.panelTitle "Caller location"

          Row gap=8

            Button.outline "Verify" icon="pin_drop" onTap={toast "Address verified"; timeline=append(timeline,{"t":elapsed,"who":"Call taker","text":"Address verified: {address}"})}
        Card gap=12
          Row between
            Text.panelTitle "Script"
            Text.caption "Step {q=="q1" ? 1 : q=="q2" ? 2 : q=="q3" ? 3 : 4} of 4"
          Progress {q=="q1" ? 0.25 : q=="q2" ? 0.5 : q=="q3" ? 0.75 : 1}
          if q=="q1"
            Text.heading3 "What is the emergency? Is anyone hurt?"
            Row gap=8
              Button.outline "Medical" onTap={ansType="Medical"; incType="Medical"; priority=2; q="q2"; timeline=append(timeline,{"t":elapsed,"who":"Caller","text":"Medical: man on the floor"})}
              Button.outline "Fire" onTap={ansType="Fire"; incType="Fire"; priority=2; q="q3"; timeline=append(timeline,{"t":elapsed,"who":"Caller","text":"Reports smoke"})}
              Button.outline "Police" onTap={ansType="Police"; incType="Police"; priority=3; q="q4"; timeline=append(timeline,{"t":elapsed,"who":"Caller","text":"Reports a disturbance"})}
          if q=="q2"
            Text.heading3 "Is he breathing?"
            Row gap=8
              Button.outline "Yes, normally" onTap={ansBreathing="Yes"; priority=2; q="q3"; timeline=append(timeline,{"t":elapsed,"who":"Caller","text":"He is breathing normally"})}
              Button.outline "Not breathing" onTap={ansBreathing="No"; priority=1; q="q3"; timeline=append(timeline,{"t":elapsed,"who":"Caller","text":"He is not breathing"}); toast "Start CPR instructions"}
              Button.outline "Gasping" onTap={ansBreathing="Agonal"; priority=1; q="q3"; timeline=append(timeline,{"t":elapsed,"who":"Caller","text":"He is gasping"})}
          if q=="q3"
            Text.heading3 "{ansBreathing=="No" ? "Tell me when you are ready to start chest compressions. Is he awake?" : "Is he awake and answering you?"}"
            Row gap=8
              Button.outline "Awake" onTap={ansConscious="Yes"; q="q4"; timeline=append(timeline,{"t":elapsed,"who":"Caller","text":"He is awake"})}
              Button.outline "Unresponsive" onTap={ansConscious="No"; priority=1; q="q4"; timeline=append(timeline,{"t":elapsed,"who":"Caller","text":"He is unresponsive"})}
          if q=="q4"
            Text.heading3 "Is there any weapon or danger at the scene?"
            Row gap=8
              Button.outline "No danger" onTap={ansWeapon="No"; q="done"; timeline=append(timeline,{"t":elapsed,"who":"Caller","text":"No danger at the scene"})}
              Button.outline "Weapon present" onTap={ansWeapon="Yes"; priority=1; q="done"; timeline=append(timeline,{"t":elapsed,"who":"Caller","text":"A weapon is present"}); toast "Police needed"}
          if q=="done"
            Text.success "Script complete. Assign units."
          Row gap=8
            Button.ghost "Restart script" onTap={q="q1"; ansType=""; ansBreathing=""; ansConscious=""; ansWeapon=""; priority=3}
        Card gap=12
          Text.panelTitle "Incident type"
          Grid cols=4 gap=12
            Button.outline "Medical" icon="medical_services" h=72 selected={incType=="Medical"} onTap={incType="Medical"}
            Button.outline "Fire" icon="local_fire_department" h=72 selected={incType=="Fire"} onTap={incType="Fire"}
            Button.outline "Police" icon="local_police" h=72 selected={incType=="Police"} onTap={incType="Police"}
            Button.outline "Traffic" icon="car_crash" h=72 selected={incType=="Traffic"} onTap={incType="Traffic"}
        Card gap=8
          Text.panelTitle "Notes"
          Input "Notes" "Add details for responders" bind=notes multiline
          Row end
            Button.secondary "Add to timeline" onTap={notes!="" ? {timeline=append(timeline,{"t":elapsed,"who":"Note","text":notes}); notes=""} : toast "Write a note first"}
      Pane w=340 pad=16 gap=12
        Box radius=12 pad=16 gap=4 fill={priority==1 ? "#FFC62828" : priority==2 ? "#FFEF6C00" : "#FF2E7D32"}
          Text "PRIORITY" color=#FFFFFFFF size=12 weight=700
          Text "P{priority}" color=#FFFFFFFF size=48 weight=700 animate={priority==1 ? "pulse" : "fade"}
          Text "{priority==1 ? "Life threatening" : priority==2 ? "Urgent" : "Routine"}" color=#FFFFFFFF
        Text.panelTitle "Timeline"
        Timeline items={timeline} timeKey="t" titleKey="who" textKey="text" h=220 scroll
        Divider
        Row between
          Text.panelTitle "Units"
          Link "All units" onTap={go units}
        each u in sortBy(units, x=>x.km)
          ListTile "{u.name}" subtitle="{u.kind}, {u.status}, {u.km} km"
            Button.outline "Assign" hidden={u.assigned || u.status=="Busy"} onTap={units=updateWhere(units,x=>x.id==u.id,x=>with(x,"assigned",true)); timeline=append(timeline,{"t":elapsed,"who":"Dispatch","text":"{u.name} assigned"}); toast "{u.name} assigned"}
            Badge.success "Sent" hidden={!u.assigned}
Dialog#endCall "End this call?" "The timeline is saved to the call log."
  Text.muted "{length(filter(units,u=>u.assigned))} unit(s) assigned. Priority P{priority}."
  Row end gap=8
    Button.ghost "Keep going" onTap={close}
    Button.destructive "End call" onTap={emit endCall; toast "Call ended"; close; go call-log}
