        Badge.info "NEXT {find(cues,c=>c.id==nextId).num}"
        Button.outline "Record cue" icon="fiber_manual_record" onTap={open record}
        Button.destructive "Blackout" animate={blackout ? "pulse" : "none"} onTap={blackout=!blackout; toast blackout ? "Blackout off" : "Blackout on"}
    Column pad=24 gap=16
      if blackout
        Card gap=4 fill=#FF3B0A0A animate=pulse duration=900
          Text.strong "BLACKOUT" color=#FFFFFFFF
          Text.caption "All outputs held at zero. Press Blackout again to restore." color=#FFFFFFFF
      Card gap=12
        Row between
          Column gap=2
            Text.caption "NOW FADING"
            Text.heading "{find(cues,c=>c.id==prevId).name}  to  {find(cues,c=>c.id==live).name}"
          Text.mono "{fixed(fade*find(cues,c=>c.id==live).fade,1)}s / {find(cues,c=>c.id==live).fade}s"
        Progress value={fade}
        Row gap=12 middle
          Button "GO" size=large icon="play_arrow" onTap={prevId=live; live=nextId; nextId=min(7,nextId+1); fade=0; fading=true}
          Button.outline "Back" icon="skip_previous" onTap={nextId=live; live=prevId; prevId=max(1,prevId-1); fade=0; fading=true}
          Button.ghost "Snap to live" onTap={fade=1; fading=false}
          Spacer grow
          Text.muted "Fade {fading ? 'running' : 'complete'}"
      Card gap=0
        Row gap=8 pad=[8,8,8,8] border-bottom=#FF444444
          Text.caption "Q" w=56
          Text.caption "Name" grow
          Text.caption "Fade" w=70
          Text.caption "Wait" w=70
          Text.caption "" w=36
        each c in cues
          Row gap=8 middle pad=[8,10,8,10] fill={c.id==live ? "#FF14532D" : (c.id==nextId ? "#FF3F3F0A" : "#00000000")} onTap={sel=c.id} menu=cueMenu
            Text.mono "{c.num}" w=56
            Row grow gap=8 middle
              Text.strong "{c.name}"
              if c.id==live
                Badge.success "LIVE"
              if c.id==nextId
                Badge.warning "NEXT"
            Text.mono "{c.fade}s" w=70
            Text.mono "{c.wait}s" w=70
            IconButton "more_vert" tooltip="Cue options" onTap={sel=c.id; open cueMenu}
  Menu#cueMenu end
    MenuItem "Set as next" icon="skip_next" onTap={nextId=sel}
    MenuItem "Fire now" icon="play_arrow" onTap={prevId=live; live=sel; fade=0; fading=true}
    MenuItem "Delete cue" icon="delete" onTap={cues=removeWhere(cues,c=>c.id==sel)}
  Dialog#record "Record current look" "Stores the stage as a new cue"
    Input "Cue number" "3.7" @newNum
    Input "Name" "Singer special 2" @newName
    Select "Insert after" items=["3 Band intro","3.5 Singer special","4 Full stage warm"] @afterId
    Row gap=8 end

      Button "Record" onTap={newName=="" ? toast "Name the cue" : {cues=append(cues,{"id":length(cues)+1,"num":newNum,"name":newName,"fade":3,"wait":0}); toast "Cue {newNum} recorded"; newName=""; newNum=""; close}}
