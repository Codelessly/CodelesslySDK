### Editor
state tool="draw"
state pageIndex=0
state sel=""
state showBleed=true
state showOrder=true
state pages=[{"id":"p1","title":"Page 1","panels":[{"id":"a","x":0,"y":0,"w":1,"h":0.4,"order":1,"caption":"Dawn over the harbor.","bubbles":[]},{"id":"b","x":0,"y":0.4,"w":0.5,"h":0.6,"order":2,"caption":"","bubbles":[{"id":"b1","text":"You're late, Mara.","speaker":"Odd","tailX":0.3,"tailY":0.8}]},{"id":"c","x":0.5,"y":0.4,"w":0.5,"h":0.6,"order":3,"caption":"","bubbles":[]}]},{"id":"p2","title":"Page 2","panels":[]},{"id":"p3","title":"Page 3","panels":[]}]
state draft=""
state exportFormat="PDF"
state exportScope="all"
App
  Sidebar w=160
    Brand "Gutters" icon="grid_view"
    Text.caption "Pages"
    Column scroll gap=8
      each p in pages
        Card pad=6 gap=4 fill={pageIndex==index ? "#FFE8EEFF" : "#FFFFFFFF"} onTap={pageIndex=index; sel=""} drag=page(p.id) drop=page onDrop={pages=movePage(pages,result.data,index)}
          PageThumb page={p} showOrder={showOrder} h=120
          Row between
            Text.small "{p.title}"
            Text.caption "{length(p.panels)} panels"
    Button.outline "Add page" icon="add" onTap={pages=append(pages,{"id":"p"+(length(pages)+1),"title":"Page "+(length(pages)+1),"panels":[]}); pageIndex=length(pages)}
    Spacer grow
    Button.ghost "Page turn preview" icon="auto_stories" onTap={go spread}
  Main
    Nav
      Heading "Issue 1: Low Tide" level=3
      Row gap=8
        SegmentedButton @tool items=["Draw","Cut","Select"]
        Toggle "Bleed and safe area" @showBleed
        Toggle "Reading order" @showOrder
        Button.outline "Spread" icon="auto_stories" onTap={go spread}
        Button "Export" icon="ios_share" onTap={open export}
    Row grow top
      Column grow pad=24 center middle
        PanelCanvas page={pages[pageIndex]} size="letter" tool={tool} showBleed={showBleed} showOrder={showOrder} selected={sel} gutter=12 onDrawPanel={pages=updateWhere(pages,p=>p.id==pages[pageIndex].id,p=>with(p,"panels",append(p.panels,result)))} onCut={pages=updateWhere(pages,p=>p.id==pages[pageIndex].id,p=>with(p,"panels",result))} onResizeEdge={pages=updateWhere(pages,p=>p.id==pages[pageIndex].id,p=>with(p,"panels",result))} onSelect={sel=result.id} onTailMove={pages=updateWhere(pages,p=>p.id==pages[pageIndex].id,p=>with(p,"panels",result))} onOrderChange={pages=updateWhere(pages,p=>p.id==pages[pageIndex].id,p=>with(p,"panels",result))}
        Row gap=8 center
          IconButton "undo" tooltip="Undo" onTap={emit undo}
          IconButton "redo" tooltip="Redo" onTap={emit redo}
          Text.caption "Drag on the page to draw a panel. Drag a gutter edge to resize its neighbours."
      Pane w=300 pad=16 gap=12
        if sel==""
          Text.muted "Select a panel to add a sketch, a caption or a speech bubble."
          Card gap=8
            Text.strong "Reading order"
            each p in sortBy(pages[pageIndex].panels,x=>x.order)
              ListTile "Panel {p.order}" subtitle={p.caption==""?"No caption":p.caption} leading="drag_indicator" drag=order(p.id) drop=order onDrop={pages=updateWhere(pages,q=>q.id==pages[pageIndex].id,q=>with(q,"panels",reorderPanels(q.panels,result.data,p.order)))} onTap={sel=p.id}
        if sel!=""
          Row between
            Text.strong "Panel {find(pages[pageIndex].panels,x=>x.id==sel).order}"
            IconButton "close" tooltip="Deselect" onTap={sel=""}
          Card gap=8
            Text.fieldLabel "Sketch"
            SketchPad h=140 panel={sel} onChange={emit sketch result}
            Row gap=8
              Button.outline "Upload" icon="upload" onTap={emit uploadSketch sel}
              Button.outline "Clear" icon="delete" onTap={emit clearSketch sel}
          Card gap=8
            Text.fieldLabel "Caption box"
            Input "Caption" "Narration text" multiline @draft
            Button.outline "Set caption" onTap={pages=updateWhere(pages,p=>p.id==pages[pageIndex].id,p=>with(p,"panels",updateWhere(p.panels,x=>x.id==sel,x=>with(x,"caption",draft)))); draft=""}
          Card gap=8
            Row between
              Text.fieldLabel "Speech bubbles"
              Button.ghost "Add bubble" icon="chat_bubble" onTap={pages=updateWhere(pages,p=>p.id==pages[pageIndex].id,p=>with(p,"panels",updateWhere(p.panels,x=>x.id==sel,x=>with(x,"bubbles",append(x.bubbles,{"id":"b"+(length(x.bubbles)+1),"text":"New line","speaker":"","tailX":0.5,"tailY":0.9})))))}
            each b in find(pages[pageIndex].panels,x=>x.id==sel).bubbles
              Column gap=4
                Input "Line" @b.text
                Row gap=8
                  Select @b.style items=["Speech","Thought","Shout","Whisper"]
                  IconButton "delete" tooltip="Remove bubble" onTap={pages=updateWhere(pages,p=>p.id==pages[pageIndex].id,p=>with(p,"panels",updateWhere(p.panels,x=>x.id==sel,x=>with(x,"bubbles",removeWhere(x.bubbles,y=>y.id==b.id)))))}
                Text.caption "Drag the tail on the page to the speaker."
          Button.destructive "Delete panel" icon="delete" onTap={pages=updateWhere(pages,p=>p.id==pages[pageIndex].id,p=>with(p,"panels",removeWhere(p.panels,x=>x.id==sel))); sel=""}
  Dialog#export "Export" "Choose what to send out"
    Column gap=12
      SegmentedButton @exportFormat items=["PDF","PNG","CBZ"]
      Select "Pages" @exportScope items=["all","current","spreads"]
      Toggle "Include bleed" @showBleed
      Row end gap=8
        Button.outline "Cancel" onTap={close}
        Button "Export set" icon="download" onTap={emit export {"format":exportFormat,"scope":exportScope,"bleed":showBleed}; toast "Exporting {exportFormat}"; close}
