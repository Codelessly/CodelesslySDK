    Column gap=20 center
      Heading "Check in to pitch 7" level=1
      Text.muted "Enter the six digit code from your confirmation email."
      CodeInput bind=code length=6 onComplete={codeOk = code=="482915"; tries=tries+1}
      if !codeOk && tries>0
        Text.danger "That code does not match. {tries} tries so far."
      if codeOk
        Card gap=12
          Row gap=8
            Icon "check_circle"
            Text.success "Checked in. Welcome to Riverside Meadow."
          Text "Gate code for tonight: 7421"

      Card gap=12
        Text.strong "Running late?"
        Toggle "I will arrive after 8 pm" bind=late

          Input "Note for the warden" multiline bind=lateNote
