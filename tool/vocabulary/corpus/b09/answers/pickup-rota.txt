### Requests
state tab="incoming"
state incoming=[{"id":1,"from":"Priya N.","day":"Tue","slot":"Afternoon","note":"Dentist appointment, can you take Noor?","status":"open"},{"id":2,"from":"Luis M.","day":"Fri","slot":"Morning","note":"Early shift, could we trade for Mon?","status":"open"}]
state sent=[{"id":3,"to":"Tom Hale","day":"Thu","slot":"Morning","note":"Can you cover Leo?","status":"waiting"}]
state sel=0
App
  Sidebar
    Brand "Gate Duty" icon="school"
    NavItem "Week" icon="calendar_view_week" onTap={go week}
    NavItem "Month" icon="calendar_month" onTap={go month}
    NavItem "Requests" icon="swap_horiz" badge={count(incoming,r=>r.status=="open")} selected={true}
    NavItem "My code" icon="qr_code_2" onTap={go code}
  Main
    Nav
      Heading "Swap requests" level=1
      SegmentedButton bind=tab items=["incoming","sent"]
    Column pad=24 gap=12 scroll
      if tab=="incoming"
        each r in incoming
          Card gap=8
            Row between
              Text.strong "{r.from} asks you to cover {r.day} {r.slot}"
              Badge.warning "{r.status}" visible={r.status=="open"}
              Badge.success "{r.status}" visible={r.status=="accepted"}
              Badge.neutral "{r.status}" visible={r.status=="declined"}
            Text.muted "{r.note}"
            if r.status=="open"
              Row gap=8
                Button "Accept" onTap={incoming=updateWhere(incoming,x=>x.id==r.id,x=>with(x,"status","accepted")); emit swapAnswer {"id":r.id,"answer":"accepted"}; toast "Accepted, the rota is updated"}
                Button.outline "Decline" onTap={sel=r.id; open decline}
      if tab=="sent"
        each r in sent
          Card gap=8
            Row between
              Text.strong "To {r.to}: {r.day} {r.slot}"
              Badge.info "{r.status}"
            Text.muted "{r.note}"
            Button.ghost "Withdraw" onTap={sent=removeWhere(sent,x=>x.id==r.id); emit swapWithdraw r.id; toast "Request withdrawn"}
  Dialog#decline "Decline this request?" "Say why, if you like"
    Input "Reply" "Sorry, I am at work then" bind=reply
    Button.destructive "Decline" onTap={incoming=updateWhere(incoming,x=>x.id==sel,x=>with(x,"status","declined")); emit swapAnswer {"id":sel,"answer":"declined","reply":reply}; toast "Declined"; close}
