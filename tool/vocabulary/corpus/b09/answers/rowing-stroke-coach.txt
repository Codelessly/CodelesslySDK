### Piece
state tab="piece"
state running=false
state clock=0
state idx=0
state boatM=0
state ghostM=0
state intervals=[{"label":"Work 1","kind":"work","sec":240,"rate":"22-24"},{"label":"Rest","kind":"rest","sec":90,"rate":""},{"label":"Work 2","kind":"work","sec":240,"rate":"24-26"},{"label":"Rest","kind":"rest","sec":90,"rate":""},{"label":"Work 3","kind":"work","sec":240,"rate":"26-28"},{"label":"Rest","kind":"rest","sec":90,"rate":""},{"label":"Work 4","kind":"work","sec":240,"rate":"28-30"}]
state ghostPace=3.7
state boatPace=3.9
every 1s when running onTick={clock=clock+1; boatM=boatM+boatPace; ghostM=ghostM+ghostPace; running=clock<sum(intervals,i=>i.sec)}
App
  Sidebar
    Brand "Catch & Drive" icon="rowing"
    NavItem "Live" icon="speed" onTap={go live}
    NavItem "Boat" icon="groups" onTap={go boat}
    NavItem "Piece" icon="timer" selected={tab=="piece"} onTap={tab="piece"}
    NavItem "Summary" icon="insights" onTap={go summary}
  Main
    Nav
      Heading "Timed piece" level=1
      Row gap=8
        Button "{running ? 'Pause' : clock>0 ? 'Resume' : 'Start piece'}" onTap={running=!running}
        Button.outline "Reset" onTap={running=false; clock=0; boatM=0; ghostM=0}
        IconButton "tune" tooltip="Edit intervals" onTap={open editor}
    Column pad=24 gap=16
      Card gap=8
        Row between
          Text.strong "4 x 4 min, 90 s rest"
          Text.strong "{duration(clock)} of {duration(sum(intervals,i=>i.sec))}"
        Progress {clock/sum(intervals,i=>i.sec)}
      Card gap=12
        Row between
          Text.strong "Pace boat"
          Row gap=12
            Badge.primary "You {round(boatM,0)} m"
            Badge.neutral "Ghost {round(ghostM,0)} m"
            Badge.{boatM>=ghostM ? "success" : "warning"} "{boatM>=ghostM ? 'Ahead' : 'Behind'} {abs(round(boatM-ghostM,0))} m"
        RaceTrack boat={boatM} ghost={ghostM} length=3000 boatLabel="You" ghostLabel="Last Tuesday"
      Card gap=8
        Text.strong "Intervals"
        each i in intervals
          Row gap=12 middle
            Text.strong "{i.label}" w=80
            Progress {clamp((clock-sum(slice(intervals,0,index),x=>x.sec))/i.sec,0,1)} grow
            Text.muted "{duration(i.sec)}" w=60
            Text.muted "{i.kind=='work' ? i.rate + ' spm' : 'Rest'}" w=90
  Dialog#editor "Edit intervals" "Change a length and the piece updates."
    each i in intervals
      Row gap=8 middle
        Text i.label w=80
        Input "Seconds" bind=i.sec type=number w=120
    Row gap=8 end
      Button.ghost "Cancel" onTap={close}
      Button "Save" onTap={emit saveIntervals intervals; toast "Saved"; close}
