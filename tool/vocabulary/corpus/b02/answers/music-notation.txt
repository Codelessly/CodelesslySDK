          PianoKeyboard from="C4" to="C5" labels=false highlight={answer} correct={flash=="right" ? target : ""} wrong={flash=="wrong" ? answer : ""} h=160 onKey={answer=result; flash=result==target?"right":"wrong"; streak=result==target?streak+1:0; best=max(best,result==target?streak+1:best); asked=asked+1; correct=result==target?correct+1:correct; history=append(history,{"note":target,"ms":1500,"ok":result==target}); emit answered result}
      if answerMode=="Names"
        Card gap=12
          Grid cols=7 gap=8
            Button.outline "C" onTap={answer="C4"; flash=target=="C4"?"right":"wrong"; streak=target=="C4"?streak+1:0; asked=asked+1; emit answered "C4"}
            Button.outline "D" onTap={answer="D4"; flash=target=="D4"?"right":"wrong"; streak=target=="D4"?streak+1:0; asked=asked+1; emit answered "D4"}

            Button.outline "F" onTap={answer="F4"; flash=target=="F4"?"right":"wrong"; streak=target=="F4"?streak+1:0; asked=asked+1; emit answered "F4"}
            Button.outline "G" onTap={answer="G4"; flash=target=="G4"?"right":"wrong"; streak=target=="G4"?streak+1:0; asked=asked+1; emit answered "G4"}
            Button.outline "A" onTap={answer="A4"; flash=target=="A4"?"right":"wrong"; streak=target=="A4"?streak+1:0; asked=asked+1; emit answered "A4"}
            Button.outline "B" onTap={answer="B4"; flash=target=="B4"?"right":"wrong"; streak=target=="B4"?streak+1:0; asked=asked+1; emit answered "B4"}
      Row gap=8 end
        Button.ghost "Skip" onTap={flash=""; emit nextNote}
        Button "Next note" onTap={flash=""; answer=""; emit nextNote}
  Sheet#settings "Clef and range"

      Slider "Seconds per note" bind=limit min=2 max=20
      Row gap=8 end
        Button.outline "Reset streak" onTap={streak=0; toast "Streak reset"}
        Button "Done" onTap={emit settingsChanged; close}

### Rhythm
state tab="rhythm"
state bpm=84
state running=false
state beat=0
state taps=[{"beat":1,"offset":-12},{"beat":2,"offset":8},{"beat":3,"offset":-64},{"beat":4,"offset":41}]
state pattern="Quarter notes"
state tolerance=50
every 1s when running onTick={beat=beat+1}
App
  Sidebar
    Brand "Stave" icon="music_note"
    NavItem "Practice" icon="piano" selected={tab=="practice"} onTap={tab="practice"; go practice}
    NavItem "Rhythm" icon="graphic_eq" selected={tab=="rhythm"} onTap={tab="rhythm"; go rhythm}
    NavItem "Summary" icon="insights" selected={tab=="summary"} onTap={tab="summary"; go summary}
    Spacer grow
    Text.muted "Mia Okafor"
  Main
    Nav
      Heading "Rhythm" level=1
      Row gap=8 middle
        Select bind=pattern items=["Quarter notes","Eighth notes","Syncopated","Triplets"] w=180
        Button {running ? "Stop" : "Start"} onTap={running=!running; beat=0}
    Column pad=24 gap=16 maxw=820
      Card gap=16
        Row between middle
          Column gap=4
            Text.caption "Tempo"
            Text "{bpm} BPM" size=32 weight=700
          Metronome bpm={bpm} running={running} beats=4 bind=beat
        Slider bind=bpm min=40 max=200
        Staff clef="percussion" rhythm={pattern} cursor={beat} h=120
        Text.caption "Tap along on the pad or press the space bar."
      Card gap=12 center
        TapPad "Tap" bpm={bpm} running={running} tolerance={tolerance} h=140 onTap={taps=append(taps,{"beat":length(taps)+1,"offset":result.offset}); emit tapped result.offset}
        Row gap=8 center
          Badge.error "Early {count(taps,t=>t.offset < 0-tolerance)}"
          Badge.success "On time {count(taps,t=>abs(t.offset)<=tolerance)}"
          Badge.warning "Late {count(taps,t=>t.offset > tolerance)}"
      Card gap=8
        Row between
          Text.strong "Your taps"
          Button.ghost "Clear" onTap={taps=[]}
        Row scroll gap=8
          each t in taps
            Column gap=4 center w=72
              Badge.neutral "{t.beat}"
              Text.small "{t.offset > tolerance ? 'Late' : (t.offset < 0-tolerance ? 'Early' : 'On time')}" color={abs(t.offset)<=tolerance ? #FF2E7D32 : #FFC62828}
              Text.caption "{t.offset > 0 ? '+' : ''}{t.offset} ms"
        Slider "Timing window (ms)" bind=tolerance min=20 max=120
