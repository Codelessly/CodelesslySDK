          Text.muted "{fixed(response.size/1024,1)} KB"

          JsonTree value={response.body} collapsible expanded=2
        if resTab=="Headers"
          Column gap=4
            each h in response.headers
              Row gap=12
                Text.mono "{h.key}" w=180
                Text.mono "{h.value}" grow

          each h in history
            ListTile "{h.method} {h.url}" subtitle="{h.at}" selected={sel==h.id} onTap={sel=h.id; go request-builder}
              Badge.neutral "{h.status}"
              IconButton "more_vert" onTap={sel=h.id; open rowMenu}
          Button.ghost "Clear history" icon="delete_sweep" onTap={history=[]; toast "History cleared"}

    MenuItem "Open" icon="open_in_new" onTap={go request-builder}
    MenuItem "Save to collection" icon="bookmark" onTap={open newCollection}
    MenuItem "Remove" icon="delete" onTap={history=removeWhere(history,h=>h.id==sel)}

    MenuItem "Rename" icon="edit" onTap={emit renameNode sel}
