### Search
state pq=""
state pcursor=0
Dialog#palette "" top w=560
  Input "Search notes, notebooks, actions" bind=pq autofocus icon="search" onChange={pcursor=0} onKeyDown={key=="ArrowDown"?pcursor=pcursor+1:(key=="ArrowUp"?pcursor=max(0,pcursor-1):pcursor=pcursor)}
  CommandList query={pq} bind=pcursor fuzzy highlight onEnter={emit paletteRun result} onPick={close}
    Group "Notes"
      each n in filter(notes,x=>!x.trashed)
        CommandItem "{n.title}" hint="{first(map(filter(notebooks,b=>b.id==n.nb),b=>b.name))}" icon="description" keywords="{n.body} {join(n.tags," ")}" onTap={openId=n.id; view="workspace"; pane="note"; close}
    Group "Notebooks"
      each b in notebooks
        CommandItem "{b.name}" icon="folder" onTap={nb=b.id; view="workspace"; pane="list"; close}
    Group "Actions"
      CommandItem "New note" icon="add" shortcut="Ctrl N" onTap={emit newNote; close}
      CommandItem "Open settings" icon="settings" onTap={view="settings"; close}
      CommandItem "Open trash" icon="delete" onTap={view="trash"; close}
      CommandItem "Toggle dark theme" icon="dark_mode" onTap={theme=theme=="Dark"?"Paper":"Dark"; close}

    Text.caption "Up and down to move, Enter to open, Esc to close"
    Text.caption "{length(notes)} notes"

### Trash
App
  Sidebar w=240 fill=#FFF3EBDA
    Brand "Inkwell" icon="auto_stories"
    NavItem "Notes" icon="notes" onTap={go workspace}
    NavItem "Tags" icon="sell" onTap={go tags}
    NavItem "Trash" icon="delete" selected badge="{count(notes,x=>x.trashed)}"
    NavItem "Settings" icon="settings" onTap={go settings}
  Main
    Nav
      Heading "Trash" level=1
      Button.outline "Empty trash" icon="delete_forever" disabled={count(notes,x=>x.trashed)==0} onTap={open confirmEmpty}
    Column pad=24 gap=12 maxw=760
      if count(notes,x=>x.trashed)==0
        EmptyState "The trash is empty" "Deleted notes wait here until you empty it." icon="delete_outline"
      each n in filter(notes,x=>x.trashed)
        Card
          Row between
            Column gap=2
              Text.strong "{n.title}"
              Text.muted "{slice(n.body,0,80)}"
            Row gap=8
              Button.outline "Restore" icon="restore" onTap={notes=updateWhere(notes,x=>x.id==n.id,x=>with(x,"trashed",false)); toast "Restored"}
              Button.destructive "Delete for ever" onTap={sel=n.id; open confirmForever}
  Dialog#confirmForever "Delete for ever?" "This cannot be undone."
    Row end gap=8
      Button.ghost "Cancel" onTap={close}
      Button.destructive "Delete for ever" onTap={notes=removeWhere(notes,x=>x.id==sel); close}
  Dialog#confirmEmpty "Empty the trash?" "{count(notes,x=>x.trashed)} notes will be removed for ever."
    Row end gap=8
      Button.ghost "Cancel" onTap={close}
      Button.destructive "Empty trash" onTap={notes=removeWhere(notes,x=>x.trashed); toast "Trash emptied"; close}
