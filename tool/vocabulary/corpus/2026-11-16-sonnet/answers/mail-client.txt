### Search
state q=""
state results={filter(msgs, m => q!="" && (fuzzy(m.subject,q) || fuzzy(m.from,q) || fuzzy(m.body,q)))}
Dialog#search "Search mail" maxw=640
  Column gap=12
    Input "Search" "Sender, subject or words in a message" @q icon="search" autofocus clearable
    if q == ""
      Text.muted "Typos are fine: 'robmap' finds 'roadmap'."
    if q != "" && length(results) == 0
      EmptyState icon="search_off" title="No results for \"{q}\"" text="Check the spelling or try fewer words."
    if length(results) > 0
      Column scroll h=360 gap=8
        each f in folders
          if count(results, m => m.folder==f.id) > 0
            Text.sectionTitle "{f.name} ({count(results, m => m.folder==f.id)})"
            each m in filter(results, x => x.folder==f.id)
              ListTile onTap={folder=m.folder; openId=m.id; pane="read"; q=""; close; go mailbox}
                Row between
                  Highlight "{m.from}" match={q} weight=600
                  Text.caption "{m.time}" color=#FF64748B
                Highlight "{m.subject}" match={q}
                Highlight "{m.preview}" match={q} maxLines=1 color=#FF64748B
