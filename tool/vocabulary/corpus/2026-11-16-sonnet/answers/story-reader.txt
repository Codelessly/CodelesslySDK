### Search
App
  Sidebar w=200
    Brand "Dispatch" icon="newspaper"
    NavItem "Today" icon="auto_stories" onTap={go today}
    NavItem "Saved" icon="bookmark" onTap={go saved}
    NavItem "Search" icon="search" selected=true
    NavItem "Settings" icon="tune" onTap={go settings}
  Main
    Nav
      Input "Search stories" "Try 'chip' or 'ports'" bind=q fuzzy=true autofocus icon="search" w=420
    Column pad=24 gap=16
      if q==""
        Text.muted "Search headlines and summaries across all eight sections."
      if q!=""
        each sec in settings.sections
          if count(stories,s=>s.section==sec && fuzzyMatch(s.head+" "+s.sum,q))>0
            Text.sectionTitle "{sec}"
            each s in filter(stories,x=>x.section==sec && fuzzyMatch(x.head+" "+x.sum,q))
              ListTile subtitle="{s.mins} min read" onTap={pickedId=s.id; go today; open story}
                Highlight "{s.head}" match={q}
                Highlight.muted "{s.sum}" match={q}
        if count(stories,s=>fuzzyMatch(s.head+" "+s.sum,q))==0
          Text.muted "No stories match '{q}'."
