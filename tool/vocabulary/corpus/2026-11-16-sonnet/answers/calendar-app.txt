### Search
state q=""
state hiddenCals=[] persist
state events=[]
state cals=[]
state draft={"id":0,"title":"","cal":"work","start":"","end":"","from":"09:00","to":"10:00","allDay":false,"repeat":"Never","guests":[],"note":""}
shortcut key=Escape onPress={go month}
App
  Sidebar w=260 hide=tablet
    Brand "Margin" icon="calendar_month"
    NavItem "Back to calendar" icon="arrow_back" onTap={go month}
  Main
    Nav border-bottom=#FFE5E7EB
      Input "Search" "Title, guest or note" icon="search" @q autofocus fuzzy grow
    Column pad=24 gap=12 maxw=720
      if trim(q)==""
        Text.muted "Type to search. Typos are fine: \"denist\" still finds Dentist."
      if trim(q)!=""&&count(events,e=>fuzzy(e.title+" "+join(e.guests," ")+" "+e.note,q))==0
        EmptyState "No matches" "Nothing found for \"{q}\"." icon="search_off"
      each day in unique(map(sortBy(filter(events,e=>fuzzy(e.title+" "+join(e.guests," ")+" "+e.note,q)&&!contains(hiddenCals,e.cal)),e=>e.start),e=>e.start))
        Text.fieldLabel "{formatDate(day,'EEE, MMM d yyyy')}" tabular
        each e in filter(events,x=>x.start==day&&fuzzy(x.title+" "+join(x.guests," ")+" "+x.note,q))
          ListTile e.title subtitle={e.allDay?"All day":"{e.from} - {e.to}"} leading={find(cals,c=>c.id==e.cal).color} onTap={draft=e; go month}
