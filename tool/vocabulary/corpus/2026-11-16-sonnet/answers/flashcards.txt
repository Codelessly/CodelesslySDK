### Quiz
state qs=[{"kind":"choice","word":"la ventana","meaning":"the window","options":["the door","the window","the table","the street"]},{"kind":"typed","word":"cansado","meaning":"tired"},{"kind":"match","pairs":[{"a":"casa","b":"house"},{"a":"perro","b":"dog"},{"a":"agua","b":"water"},{"a":"libro","b":"book"},{"a":"luz","b":"light"}]},{"kind":"choice","word":"aprender","meaning":"to learn","options":["to eat","to learn","to wait","to run"]}]
state qi=0
state score=0
state picked=-1
state answer=""
state verdict=""
state pl=""
state pr=""
state matched=[]
state shake=false
App
  Sidebar fill=#FFFFFFFF border-right=#FFEFE9FB
    Brand "Lexi" icon="translate"
    NavItem "Decks" icon="style" onTap={go decks}
    NavItem "Study" icon="school" onTap={go study}
    NavItem "Quiz" icon="quiz" selected={true} onTap={go quiz}
    NavItem "Progress" icon="insights" onTap={go progress}
    NavItem "Settings" icon="settings" onTap={go settings}
  Main
    Nav
      Heading "Quiz" level=1 color=#FF4C1D95
      Badge.primary "Score {score}"
    Column pad=24 gap=20 center maxw=640
      if qi<length(qs)
        Row gap=12 middle
          Progress {qi/length(qs)} grow color=#FF6D28D9
          Text.strong "{qi+1} of {length(qs)}"
        if qs[qi].kind=="choice"
          Shortcut keys="1" onPress={picked=0; verdict=qs[qi].options[0]==qs[qi].meaning?"right":"wrong"; score=score+(verdict=="right"?1:0)}
          Shortcut keys="2" onPress={picked=1; verdict=qs[qi].options[1]==qs[qi].meaning?"right":"wrong"; score=score+(verdict=="right"?1:0)}
          Shortcut keys="3" onPress={picked=2; verdict=qs[qi].options[2]==qs[qi].meaning?"right":"wrong"; score=score+(verdict=="right"?1:0)}
          Shortcut keys="4" onPress={picked=3; verdict=qs[qi].options[3]==qs[qi].meaning?"right":"wrong"; score=score+(verdict=="right"?1:0)}
          Card gap=16 radius=40 center pad=32
            Text.caption "WHAT DOES THIS MEAN?"
            Text "{qs[qi].word}" size=48 weight=800 color=#FF4C1D95
          Grid cols=2 gap=12
            each o in qs[qi].options
              Button.outline "{index+1}. {o}" size=large radius=24 disabled={picked>=0} fill={picked==index ? (o==qs[qi].meaning ? #FFBBF7D0 : #FFFECACA) : #FFFFFFFF} onTap={picked=index; verdict=o==qs[qi].meaning?"right":"wrong"; score=score+(o==qs[qi].meaning?1:0)}
        if qs[qi].kind=="typed"
          Card gap=16 radius=40 center pad=32
            Text.caption "TYPE THE SPANISH WORD"
            Text "{qs[qi].meaning}" size=40 weight=800 color=#FF4C1D95
            TypedAnswer "Your answer" bind=answer answer={qs[qi].word} tolerance=1 showDiff autofocus onResult={verdict=result.correct?"right":"wrong"; score=score+(result.correct?1:0)}
        if qs[qi].kind=="match"
          Text.title "Match the five pairs" center
          Row gap=24 top
            Column grow gap=8
              each p in qs[qi].pairs
                Button.outline "{p.a}" radius=20 disabled={contains(matched,p.a)} selected={pl==p.a} onTap={pl=p.a; matched=pr!="" && find(qs[qi].pairs,x=>x.a==p.a).b==pr ? append(matched,p.a) : matched; score=score+(pr!="" && find(qs[qi].pairs,x=>x.a==p.a).b==pr ? 1 : 0); verdict=pr==""?"":(find(qs[qi].pairs,x=>x.a==p.a).b==pr?"right":"wrong"); pl=pr!=""?"":p.a; pr=pr!=""?"":pr}
            Column grow gap=8
              each p in reverse(qs[qi].pairs)
                Button.outline "{p.b}" radius=20 disabled={contains(map(filter(qs[qi].pairs,x=>contains(matched,x.a)),x=>x.b),p.b)} selected={pr==p.b} onTap={pr=p.b; matched=pl!="" && find(qs[qi].pairs,x=>x.a==pl).b==p.b ? append(matched,pl) : matched; score=score+(pl!="" && find(qs[qi].pairs,x=>x.a==pl).b==p.b ? 1 : 0); verdict=pl==""?"":(find(qs[qi].pairs,x=>x.a==pl).b==p.b?"right":"wrong"); pr=pl!=""?"":p.b; pl=pl!=""?"":pl}
        if verdict=="right"
          Row gap=8 center animate=pop announce="Correct"
            Icon "check_circle" color=#FF16A34A
            Text.success "Correct!"
        if verdict=="wrong"
          Row gap=8 center animate=shake announce="Not quite"
            Icon "cancel" color=#FFDC2626
            Text.danger "Not quite. The answer is {qs[qi].kind=="typed" ? qs[qi].word : qs[qi].meaning}"
        Row end
          Button "Next" icon="arrow_forward" disabled={verdict=="" && !(qs[qi].kind=="match" && length(matched)==5)} onTap={qi=qi+1; picked=-1; answer=""; verdict=""; pl=""; pr=""; matched=[]}
      if qi>=length(qs)
        Card gap=16 radius=40 center animate=pop pad=32
          Icon "emoji_events" size=64 color=#FFF59E0B
          Heading "Quiz complete" level=1 color=#FF4C1D95
          Text "{score}" size=64 weight=800 color=#FF6D28D9
          Text.muted "points across {length(qs)} questions"
          Button "Play again" onTap={qi=0; score=0; picked=-1; answer=""; verdict=""; matched=[]}
