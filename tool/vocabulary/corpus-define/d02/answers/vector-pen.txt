// A ruler along one edge of the canvas with tick labels; pulling from it drops a guide. Drawn as a labelled strip: the real tick marks and the drag out of the ruler need a canvas.
define Ruler orient="h" zoom=100 guides=[] onPull
  Box w={orient=="h" ? fill : 24} h={orient=="h" ? 24 : fill} fill=#FFF3F4F6 border-bottom=#FFD1D5DB drag=guide(orient)
    Text.caption "{orient=="h" ? "0 ... 640 px" : "0 ... 480"} at {zoom}%  ({length(guides)} guides)"

// The drawing surface: shapes, pen path with draggable points and handles, guides and snapping. A real vector canvas needs pointer drags and curves, so it shows as a labelled box.
define VectorCanvas shapes=[] points=[] guides=[] showGuides=true tool="select" zoom=100 snap=true selected=[] selectedPoint=0 onSelect onAddPoint onClosePath onMovePoint onMoveHandle onPickPoint onMoveGuide onRemoveGuide onZoom
  Box fill=#FFE5E7EB pad=16 grow center middle
    Box w=320 h=240 fill=#FFFFFFFF radius=2 border=1 center middle onTap={tool=="pen" ? onAddPoint({"x":0,"y":0,"hx":20,"hy":-30,"smooth":true}) : onSelect(0)}
      Text.muted "Canvas: {length(shapes)} shapes, {length(points)} path points, tool {tool}, {zoom}%{snap ? ", snapping" : ""}"

// A layers row: icon by kind, name indented by depth, lock and hide toggles.
define LayerRow "name" kind="path" depth=0 selected=false locked=false hidden=false onTap onToggleLock onToggleHide
  Row middle gap=6 pad=[{depth*16+6},4,6,4] fill={selected ? "#FFDBEAFE" : "#00000000"} radius=4 onTap={onTap}
    Icon "{kind=="group" ? "folder" : kind=="rect" ? "crop_square" : kind=="ellipse" ? "circle" : "gesture"}"
    Text "{name}" grow strike={hidden}
    IconButton "{locked ? "lock" : "lock_open"}" tooltip="Lock" onTap={onToggleLock}
    IconButton "{hidden ? "visibility_off" : "visibility"}" tooltip="Hide" onTap={onToggleHide}

// A gradient bar with a draggable marker per stop. The real gradient and the marker drags need a canvas, so it shows as a labelled bar with a chip per stop.
define GradientBar bind=stops active=0 kind="linear" angle=90 onPickStop onMoveStop onAddStop
  state stops=[]
  Column gap=6
    Box h=28 radius=6 fill={stops[0].color} border=1 onTap={onAddStop({"at":0.5})}
      Text.caption "{kind} gradient, {angle}" color=#FFFFFFFF
    Row gap=6
      each stop in stops
        Chip "{round(stop.at*100,0)}%" selected={active==index} drag=stop(index) onTap={onPickStop(index)}
