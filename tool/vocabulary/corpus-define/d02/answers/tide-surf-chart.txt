// A wind arrow pointing the way the wind blows, with its strength.
define WindArrow dir="N" speed=10 unit="mph"
  state angles={"N":0,"NE":45,"E":90,"SE":135,"S":180,"SW":225,"W":270,"NW":315}
  Row gap=8 middle
    Icon "navigation" size=32 rotate={coalesce(angles[dir],0)}
    Column gap=2
      Text "{speed} {unit}" size=32 weight=700
      Text.caption "{dir}, {speed < 8 ? "light" : (speed < 15 ? "moderate" : "strong")}"

// A water temperature reading with its trend.
define WaterTemp value=55 unit="F" trend="steady"
  Row gap=8 middle
    Icon "thermostat"
    Text "{value} {unit}" size=32 weight=700
    Text.caption "{trend}"

// A row of star icons; read only or tapped to set the rating.
define StarRating @value max=5 readonly=false
  Row gap=2
    each i in range(max)
      Icon {i < value ? "star" : "star_border"} onTap={readonly ? "" : value=i+1}

// A tide curve across the day as a smooth wave with a now line and high and low markers.
define TideCurve values=[] labels=[] now=12 highs=[] lows=[] h=200
  Stack h={h}
    AreaChart values={values} labels={labels} smooth h={h}
    Box w=2 h={h} fill=#FF2563EB left={now/24*100}
    Row between pad=8
      each hl in highs
        Badge.info "High {hl}"
      each lw in lows
        Badge.neutral "Low {lw}"

// Surf height by hour as bars, each with its period and a direction arrow.
define SurfBars hours=[] h=200
  Column gap=8
    BarChart values={map(hours, x => x.ft)} labels={map(hours, x => x.h)} h={h}
    Row between
      each x in hours
        Column center gap=2
          Icon "arrow_upward" size=16
          Text.caption "{x.period}s"
          Text.caption "{x.dir}"

// A highlighted green band for the best surf window.
define BestWindow label="Best window" start="" end="" score=80
  Row gap=12 middle pad=12 fill=#FFBBF7D0 radius=8
    Icon "check_circle" color=#FF166534
    Column grow gap=2
      Text.strong "{label}"
      Text.muted "{start} to {end}"
    Badge.success "{score}"

// A real map is not drawable here: shown as a labelled box with a pin for each spot.
define SpotMap spots=[] @selected h=360 onPick
  state selected=""
  Stack h={h} fill=#FFDBEAFE radius=8
    Text.muted "Map of spots" center
    each s in spots
      Icon "location_on" color={s.name==selected ? "#FFDC2626" : "#FF2563EB"} left={s.x*100} top={s.y*100} onTap={selected=s.name; onPick(s.name)}
