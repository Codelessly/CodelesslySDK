state idle=0

state lapStart=0

  Sidebar
    Brand "Cadence" icon="pedal_bike"

    NavItem "Route" icon="route" onTap={go route}
    NavItem "Summary" icon="insights" onTap={go summary}

    Toggle "Auto pause" @autoPause
  Main
    Nav
      Heading "Morning loop" level=1
      Row gap=8
        if autoPause && idle>=3
          Badge.warning "Auto paused"
        Badge.success "GPS locked"
        Button.outline "Lap" icon="flag" onTap={laps=append(laps,{"n":length(laps)+1,"time":duration(elapsed-lapStart),"avg":speed}); lapStart=elapsed; toast "Lap {length(laps)+1} started"}
    Column pad=24 gap=16
      Card gap=4 center
        Text.muted "Speed"
        Text "{fixed(speed,1)}" size=120 weight=700
        Text.muted "km/h"
        Row gap=8 center
          Button.secondary "Simulate stop" onTap={speed=0}
          Button.secondary "Pedal on" onTap={speed=29.8; idle=0}
      Grid cols=3 gap=12
        each key in slots
          MetricTile label={find(metrics,m=>m.key==key).label} value={find(metrics,m=>m.key==key).value} unit={find(metrics,m=>m.key==key).unit}
      Card gap=8
        Row between
          Text.strong "Laps"
          Text.muted "Lap {length(laps)+1} running {duration(elapsed-lapStart)}"
        each lap in laps
          ListTile "Lap {lap.n}" "{lap.time} at {lap.avg} km/h avg"

### Layout and Sensors
project slots=["time","distance","cadence","power","hr"] saved
state pool=[{"key":"time","label":"Time"},{"key":"distance","label":"Distance"},{"key":"cadence","label":"Cadence"},{"key":"power","label":"Power"},{"key":"hr","label":"Heart rate"},{"key":"speedavg","label":"Avg speed"},{"key":"elev","label":"Elevation"}]
state sensors=[{"id":"hrm","name":"Wahoo TICKR","kind":"Heart rate","bars":4,"on":true},{"id":"pwr","name":"Stages Power L","kind":"Power","bars":3,"on":true},{"id":"cad","name":"Garmin Cadence 2","kind":"Cadence","bars":2,"on":true},{"id":"spd","name":"Bryton Speed","kind":"Speed","bars":1,"on":false}]
App
  Sidebar
    Brand "Cadence" icon="pedal_bike"
    NavItem "Ride" icon="speed" onTap={go ride}
    NavItem "Layout" icon="dashboard_customize" selected
    NavItem "Route" icon="route" onTap={go route}
    NavItem "Summary" icon="insights" onTap={go summary}
  Main
    Nav
      Heading "Ride screen layout" level=1
      Button "Done" onTap={go ride}
    Column pad=24 gap=16
      Text.muted "Drag a metric onto a slot to place it. Dropping on a slot replaces what was there."
      Card gap=8
        Text.strong "Slots"
        Grid cols=3 gap=12
          each slot in slots
            Box pad=12 radius=8 fill=#FF1F2937 gap=4 drop=metric onDrop={slots=setAt(slots,index,result.data)} drag=metric(slot)
              Text.caption "Slot {index+1}" color=#FF9CA3AF
              Text.strong "{find(pool,p=>p.key==slot).label}" color=#FFFFFFFF
      Card gap=8
        Text.strong "Available metrics"
        Row scroll gap=8
          each p in pool
            Chip "{p.label}" icon="drag_indicator" drag=metric(p.key)
      Card gap=8
        Row between
          Text.strong "Sensors"
          Button.outline "Scan" icon="bluetooth_searching" onTap={toast "Scanning for sensors"}
        each s in sensors
          ListTile "{s.name}" "{s.kind}"
            Row gap=12 middle
              SignalBars level={s.bars} max=4
              Toggle @s.on

// A map with a drawn line; shown as a labelled box because the core has no map
define RouteMap points=[] progress=0 warning=false h=240
  Box h={h} radius=12 fill=#FFE5E7EB pad=12 gap=4 border={warning ? 2 : 0}
    Text.caption "Route map, {length(points)} points, {round(progress*100,0)}% ridden"
    Progress {progress}
    if warning
      Badge.warning "Turn ahead"

define MetricTile label="" value="" unit=""
  Card gap=2
    Text.caption "{label}"
    Row gap=4 middle
      Text "{value}" size=40 weight=700
      Text.muted "{unit}"

define SignalBars level=0 max=4
  Row gap=2 bottom
    each i in range(max)
      Box w=5 h={6+i*5} radius=1 fill={i<level ? #FF16A34A : #FFD1D5DB}

define ElevationProfile values=[] here=0 unit="m" h=160
  Stack h={h}
    AreaChart values={values} h={h}
    Badge.info "{values[here]} {unit}"
