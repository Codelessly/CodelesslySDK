// a live frequency plot: a line over a grid, a fainter peak line, a limit line, a marker, pinch and drag; drawn here as a labelled box
define SpectrumPlot trace=[] peak=[] centre=433.92 span=2.0 ref=-20 limit=-60 marker={} h=320 onPinch onPick onRefDrag
  Stack h={h} fill=#FF0B1120 radius=6
    LineChart series=[{"name":"Peak hold","values":peak},{"name":"Live","values":trace}] h={h}
    Box h=2 fill=#FFDC2626 hide=false
    Text.caption "{limit} dBm limit" color=#FFDC2626
    Row between top pad=8
      Text.caption "{fixed(centre-span/2,2)} MHz" color=#FF9CA3AF
      Text.caption "{fixed(centre,2)} MHz" color=#FFE5E7EB
      Text.caption "{fixed(centre+span/2,2)} MHz" color=#FF9CA3AF
    Box h=14 fill=#33FFFFFF drag=refline(ref) onDrop={onRefDrag(result.data)}
    Icon "place" color=#FFFBBF24 onTap={onPick(marker)}

// a scrolling colour strip of past sweeps, strongest as brightest; drawn here as a labelled box
define Waterfall rows=[] min=-100 max=-20 h=140
  Box h={h} fill=#FF1E1B4B radius=6
    Sparkline values={map(rows, r => max(r))}
    Text.caption "waterfall: {length(rows)} sweeps, {min} to {max} dBm" color=#FFE5E7EB

### Signals
state q=""
state band="All"
state sel=""
App
  Sidebar
    Brand "Bandscape" icon="graphic_eq"
    NavItem "Analyzer" icon="show_chart" onTap={go analyzer}
    NavItem "Signals" icon="table_rows" selected={true}
    NavItem "Settings" icon="tune" onTap={go settings}
  Main
    Nav
      Heading "Signals" level=1
      Row gap=8
        Input "Search" bind=q w=220
        Select bind=band items=["All","Over limit","Under limit"] w=150
    Column pad=24 gap=16
      Card gap=8
        Row gap=8
          Text.caption "Frequency" w=120
          Text.caption "Level" w=90
          Text.caption "Seen" w=90
          Text.caption "Name" grow
          Text.caption "" w=40
        each s in filter(signals, x => contains(lower(x.name), lower(q)) && (band == "All" || (band == "Over limit" && x.level > limit) || (band == "Under limit" && x.level <= limit)))
          Row gap=8 middle
            Text.mono "{fixed(s.freq,3)} MHz" w=120
            Text.mono "{s.level} dBm" w=90 color={s.level > limit ? "#FFDC2626" : "#FF111827"}
            Text.muted "{s.name == 'Unknown' ? 'new' : 'known'}" w=90
            Text "{s.name}" grow
            IconButton "more_vert" tooltip="Actions" onTap={sel=s.freq; open signalMenu}
        if length(signals) == 0
          Text.muted "No signals yet. Run a sweep on the Analyzer."
Menu#signalMenu end
  MenuItem "Place marker" icon="place" onTap={marker=find(signals, x => x.freq == sel); centre=sel; go analyzer}
  MenuItem "Rename" icon="edit" onTap={open renameSignal}
  MenuItem "Delete" icon="delete" onTap={signals=removeWhere(signals, x => x.freq == sel)}
Dialog#renameSignal "Rename signal" "Give it a name you will recognise"
  state newName=""
  Input "Name" bind=newName
  Button "Save" onTap={signals=updateWhere(signals, x => x.freq == sel, x => with(x,"name",newName)); toast "Renamed"; close}
