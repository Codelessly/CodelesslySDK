### Application
state step=0
state housing="House"
state yard=true
state otherPets=false
state adults=2
state kids=0
state notes=""
state refs=[{"name":"","phone":""},{"name":"","phone":""}]
state name=""
state email=""
state phone=""
state dob=""
Page
  Nav
    Brand "Paws & Co" icon="pets"
    Link "Back to discover" onTap={go discover}
  Section narrow
    Column gap=16
      Heading "Adoption application for Biscuit" level=1
      Stepper @step steps=["About you","Your home","References","Review"]
      if step==0
        Card gap=12
          Input "Full name" @name
          Input "Email" "you@example.com" @email type=email
          Input "Phone" @phone type=phone
          DatePicker "Date of birth" @dob
      if step==1
        Card gap=12
          Select "Home type" @housing items=["House","Apartment","Condo","Farm"]
          Toggle "Fenced yard" @yard
          Toggle "Other pets at home" @otherPets
          Row gap=12
            Input "Adults" @adults type=number grow
            Input "Children" @kids type=number grow
          Input "Anything else we should know?" @notes multiline
      if step==2
        Card gap=12
          each r in refs
            Text.strong "Reference {index+1}"
            Input "Name" @r.name
            Input "Phone" @r.phone type=phone
      if step==3
        Card gap=8
          Text.heading "Review"
          Text "{name} - {email}"
          Text "{housing}, {adults} adults, {kids} children, fenced yard: {yard ? "yes" : "no"}"
          Text.muted "References: {join(map(refs, r => r.name), ", ")}"
      Row between
        Button.outline "Back" onTap={step=max(0,step-1)}
        if step<3
          Button "Continue" onTap={step=min(3,step+1)}
        if step==3
          Button "Submit application" onTap={emit submitApplication; toast "Application sent"; go status}

  Nav

      Heading "Application status" level=1

      StatusTracker steps={stages} current={stage}
      Card gap=8
        Text.heading "{stages[stage]}"
        if stage==1
          Text "A volunteer will visit your home this week. We will email you to pick a time."

          Text "Bring everyone in your household to meet Biscuit."
        Row gap=8
          Button.outline "Message the shelter" onTap={open message}

  Button "Send" onTap={emit sendMessage msg; toast "Sent"; close}

  Row gap=8
    Button.outline "Keep it" onTap={close}
    Button.destructive "Withdraw" onTap={emit withdraw; close}

define Stepper @step steps=["One","Two"]
  state step=0
  Row gap=8 middle
    each s in steps
      Row gap=6 middle grow
        Box w=28 h=28 radius=14 fill={index<=step ? "#FF2563EB" : "#FFE5E7EB"} center middle
          Text "{index+1}" color={index<=step ? "#FFFFFFFF" : "#FF374151"}
        Text.small "{s}"

define StatusTracker steps=["A","B"] current=0
  Column gap=0
    each s in steps
      Row gap=12 middle
        Icon {index<current ? "check_circle" : (index==current ? "radio_button_checked" : "radio_button_unchecked")}
        Text.strong "{s}"
