// Counts down from a number of seconds as days, hours, minutes and seconds in four boxes.
define Countdown seconds=0
  Row gap=8
    each part in [{"n":floor(seconds/86400),"u":"days"},{"n":floor(seconds%86400/3600),"u":"hours"},{"n":floor(seconds%3600/60),"u":"min"},{"n":seconds%60,"u":"sec"}]
      Box center pad=8 radius=8 fill=#FFF3F4F6 w=64
        Text.strong "{part.n}"
        Text.caption "{part.u}"

// A map of the yard that draws labelled markers by position (x and y from 0 to 1); shown as a labelled box, as a real map needs a drawing surface.
define YardMap bind=selected markers=[] filter="All" h=300 onSelect
  state selected=""
  Stack h={h} fill=#FFDCFCE7 radius=12
    each m in markers
      if filter=="All" || m.kind==filter
        Box pad=6 radius=8 fill={selected==m.label ? "#FF16A34A" : "#FFFFFFFF"} onTap={selected=m.label; onSelect(m.label)}
          Text.small "{m.label}"
