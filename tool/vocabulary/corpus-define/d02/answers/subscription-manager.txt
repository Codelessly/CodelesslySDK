// A month grid with a mark on the days that carry a payment; a tap picks a day.
define PaymentCalendar "title" marks=[] startOffset=0 days=30 selected=0 onPick
  Card gap=8
    Text.strong "{title}"
    Grid cols=7 gap=4
      each d in ["Mon","Tue","Wed","Thu","Fri","Sat","Sun"]
        Text.caption "{d}" center
      each i in range(startOffset)
        Box h=48
      each n in range(days)
        Box h=48 radius=8 center middle fill={selected==n+1 ? '#FFDBEAFE' : (any(marks, m=>m.day==n+1) ? '#FFF3F4F6' : '#00000000')} onTap={onPick(n+1)}
          Text "{n+1}"
          if any(marks, m=>m.day==n+1)
            Box w=6 h=6 radius=3 fill=#FF2563EB
