    Row between middle pad=[8,2,8,2] fill={focused ? "#FF1F7A4D" : "#FF21262D"}

// One line of scrollback; the cursor line and selected lines show in reverse colour
define CopyLine "text" cursor=false selected=false onTap
  Row pad=[8,1,8,1] fill={cursor ? "#FFC9D1D9" : selected ? "#FF264F78" : "#00000000"} onTap={onTap}
    Text.mono "{text}" color={cursor ? "#FF0D1117" : "#FFC9D1D9"}
