define ChessBoard pieces=[] flip=false selected=-1 legal=[] lastFrom=-1 lastTo=-1 checkSquare=-1 onSquare onMove
  Grid cols=8 gap=0 w=480
    each i in range(64)
      Box w=60 h=60 center middle fill={(i == lastFrom || i == lastTo) ? "#FFF6E58D" : ((floor(i/8) + i%8) % 2 == 0 ? "#FFEEEED2" : "#FF769656")} border={i == selected ? 3 : 0} onTap={onSquare(i)} drag=piece(i) drop=piece onDrop={onMove({"from":result.data,"to":i})}
        Stack center middle
          if i == checkSquare
            Box w=48 h=48 radius=24 fill=#66FF0000
          if contains(legal, i)
            if pieces[i] == ""
              Box w=18 h=18 radius=9 fill=#55000000
            else
              Box w=52 h=52 radius=26 border=4
          Text "{replace(replace(replace(replace(replace(replace(replace(replace(replace(replace(replace(replace(pieces[i],'K','♔'),'Q','♕'),'R','♖'),'B','♗'),'N','♘'),'P','♙'),'k','♚'),'q','♛'),'r','♜'),'b','♝'),'n','♞'),'p','♟')}" size=44

define Clock time=600 active=false
  Box pad=[16,8,16,8] radius=8 fill={active ? (time < 10 ? "#FFDC2626" : "#FF1F2937") : "#FFE5E7EB"}
    if active && time < 10
      Text "{floor(time/60)}:{time%60 < 10 ? '0' : ''}{time%60}" size=28 weight=700 color=#FFFFFFFF animate=pulse duration=500
    else
      Text "{floor(time/60)}:{time%60 < 10 ? '0' : ''}{time%60}" size=28 weight=700 color={active ? "#FFFFFFFF" : "#FF374151"}

define MoveList moves=[] @view onStep
  state view=0
  Column gap=2 h=220 scroll
    each m in moves
      Row gap=8 middle
        Text.caption "{m.n}." w=28
        Box grow pad=6 radius=4 fill={view == m.n*2-1 ? "#FFDBEAFE" : "#00FFFFFF"} onTap={view=m.n*2-1; onStep(m.n*2-1)}
          Text.mono "{m.w}"
        Box grow pad=6 radius=4 fill={view == m.n*2 ? "#FFDBEAFE" : "#00FFFFFF"} onTap={view=m.n*2; onStep(m.n*2)}
          Text.mono "{coalesce(m.b,'')}"

define CapturedPieces pieces=[] color="w"
  Row gap=0 middle
