### View in room
state placed=false
state turn=0
state scale=100
state shots=[]
state flash=false
App
  Main
    Nav
      Row gap=8 middle
        IconButton "arrow_back" tooltip="Back" onTap={go product}
        Heading "View in your room"
      Button.outline "Planner" onTap={go room-planner}
    Column pad=24 gap=16 maxw=720
      CameraView "Point your phone at the floor" ring={placed} item="Oslo Lounge Chair" turn={turn} scale={scale} onPlace={placed=true} onMove={emit moveItem result}
      if !placed
        Text.muted center "Move your phone slowly until the ring appears, then tap the floor to place the chair."
        Button "Place the chair here" onTap={placed=true; toast "Chair placed"}
      if placed
        Card gap=12
          Row between middle
            Text.strong "Rotate"
            Row gap=8 middle
              IconButton "rotate_left" tooltip="Rotate left" onTap={turn=(turn+345)%360}
              Text.muted w=60 center "{turn} deg"
              IconButton "rotate_right" tooltip="Rotate right" onTap={turn=(turn+15)%360}
          Row gap=12 middle
            Icon "zoom_out_map"
            Slider @scale min=70 max=130
            Text.muted w=50 "{scale}%"
        Row gap=12 center
          IconButton "refresh" tooltip="Reset" onTap={placed=false; turn=0; scale=100}
          Button "Take snapshot" icon="photo_camera" onTap={shots=append(shots,{"id":length(shots)+1,"turn":turn}); flash=true; toast "Snapshot saved"}
          Button.outline "Add to cart" onTap={emit addToCart; toast "Added to cart"}
        if length(shots)>0
          Text.strong "Snapshots ({length(shots)})"
          Row scroll gap=8
            each s in shots
              Box w=96 h=96 radius=8 fill=#FFE5E7EB center middle onTap={open shot}
                Icon "image"
                Text.caption "View {s.id}"
  Sheet#shot "Snapshot"
    Box h=200 radius=8 fill=#FFE5E7EB center middle
      Icon "image"
    Row gap=8
      Button "Share" icon="share" onTap={toast "Shared"; close}
      Button.outline "Delete" onTap={shots=removeAt(shots,length(shots)-1); close}

// A 3D model viewer cannot be drawn; it shows a labelled tinted box that turns with the swipe.
define Viewer3D "name" angle=0 color=#FFD8CBB5 dims=true onRotate
  Stack h=360 radius=12 fill=#FFF3F4F6 center middle onSwipeLeft={onRotate((angle+30)%360)} onSwipeRight={onRotate((angle+330)%360)}
    Column center gap=4
      Box w={160-abs(90-angle%180)} h=150 radius=24 fill={color}
      Text.caption "3D model: {name}, view {angle} deg"
    if dims
      Column end bottom pad=12
        Text.caption "W 78 cm   D 84 cm   H 72 cm"

define Swatch "name" fill=#FFD8CBB5 selected=false onTap
  Column center gap=4 onTap={onTap}
    Box w=36 h=36 radius=18 fill={fill} border={selected ? 3 : 1}
    Text.caption "{name}"

// A live camera cannot be drawn; it shows a labelled frame with a placement ring and handles.
define CameraView "hint" ring=false item="" turn=0 scale=100 onPlace onMove
  Stack h=420 radius=12 fill=#FF1F2937 center middle onTap={onPlace}
    Text.caption color=#FFD1D5DB "Camera: {hint}"
    if ring
      Column center middle gap=4
        Box w={1.2*scale} h={1.2*scale} radius={0.6*scale} border=2 center middle
          Icon "chair"
        Text.caption color=#FFFFFFFF "{item}, {turn} deg"
    Row between pad=12 top
      Icon "videocam"
      Text.caption color=#FFD1D5DB "Floor detected"

define FloorPlan cols=8 rows=6 items=[] placed={} rotation={} selected="" onMove onSelect
  Grid cols={cols} gap=2 border=4
    each cell in range(cols*rows)
      Box h=56 fill=#FFF9FAFB border=1 center middle drop=piece onDrop={onMove({"id":result.data,"cell":cell})}
        each it in items
          if placed[it.id]==cell
            Box pad=6 radius=6 fill={selected==it.id ? #FF2563EB : #FF6B7280} onTap={onSelect(it.id)}
              Icon "{it.icon}" color=#FFFFFFFF
              Text.caption color=#FFFFFFFF "{rotation[it.id]}"
