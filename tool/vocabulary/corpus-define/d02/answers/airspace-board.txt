define StatusPill "status"
  Badge.neutral "{status}" fill={status == "On time" ? "#FFDCFCE7" : status == "Boarding" ? "#FFDBEAFE" : status == "Delayed" ? "#FFFEF3C7" : status == "Gate change" ? "#FFFFEDD5" : status == "Cancelled" ? "#FFFEE2E2" : "#FFE5E7EB"} color={status == "On time" ? "#FF166534" : status == "Boarding" ? "#FF1E40AF" : status == "Delayed" ? "#FF92400E" : status == "Gate change" ? "#FF9A3412" : status == "Cancelled" ? "#FF991B1B" : "#FF374151"}

define FlightRow flight={} onTap
  Row gap=8 pad=[16,12,16,12] middle border-bottom=#FFF3F4F6 animate=slide-up duration=300 onTap={onTap}
    Text.strong "{flight.id}" w=80
    Text "{flight.city}" grow
    Text "{flight.gate}" w=60 fill={flight.changed ? "#FFFEF08A" : "#00000000"}
    Text "{flight.sched}" w=70
    Text "{flight.est}" w=70 fill={flight.changed ? "#FFFEF08A" : "#00000000"} animate={flight.changed ? "pulse" : "none"}
    Box w=110
      StatusPill status={flight.status}

define RunwayStrip movements=[]
  Card gap=8
    Row between
      Text.panelTitle "Runway 27L / 09R: next ten movements"
      Text.caption "time order"
    Row gap=4 scroll
      each m in movements
        Box w=96 pad=8 gap=2 radius=4 fill={m.dir == "Departures" ? "#FFDBEAFE" : "#FFDCFCE7"} onTap={emit focusFlight m.id}
          Text.caption "{m.est}"
          Text.strong "{m.id}"
          Row gap=4 middle
            Icon "{m.dir == "Departures" ? "flight_takeoff" : "flight_land"}"
            Text.small "{m.term}"

define WorldClock zone="Local" offset=0 secs=0
  Column gap=0
    Text.caption "{zone}"
    Text.mono "{fixed(floor((50400 + secs + offset * 3600) / 3600) % 24, 0)}:{fixed(floor((secs + 300) / 60) % 60, 0)}:{fixed(secs % 60, 0)}"

// A real wind compass cannot be drawn; the arrow is a labelled box.
define WeatherPanel temp=0 condition="" visibility=0 gust=0 windDir=0 windSpeed=0 runway=""
  Column gap=12
    Row between middle
      Text size=40 weight=700 "{temp}°C"
      Text.muted "{condition}"
    Box w=120 h=120 radius=60 border=1 center middle
      Text.caption "Wind from {windDir}°"
      Text.strong "{windSpeed} kt"
    Gauge bind=gust min=0 max=60 limit=40 unit="kt"
    Row between
      Text.muted "Visibility"
      Text.strong "{visibility} km"
    Row between
      Text.muted "Active runway"
      Text.strong "{runway}"
    Sparkline values=[18,20,22,25,31,34,28,22]
    Text.caption "Gusts, last 8 hours"
