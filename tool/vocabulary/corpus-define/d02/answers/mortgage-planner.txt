### Compare
App
  Sidebar
    Brand "Hearthloan" icon="home"
    NavItem "Planner" icon="calculate" onTap={go planner}
    NavItem "Compare" icon="compare_arrows" badge="{length(scenarios)}" selected={true}
    Spacer grow
    Text.muted "Draft plan, not an offer"
  Main
    Nav
      Heading "Saved scenarios" level=1
      Button "New scenario" onTap={go planner}
    Column pad=24 gap=16 scroll
      if length(scenarios)==0
        Card gap=8 center
          Icon "folder_open"
          Text.strong "Nothing saved yet"
          Text.muted "Save a scenario from the planner to compare it here."
          Button "Open planner" onTap={go planner}
      Row scroll gap=16 top
        each s in scenarios
          Card w=300 gap=10
            Row between middle
              Text.heading "{s.name}"
              IconButton "delete" tooltip="Remove" onTap={scenarios=removeWhere(scenarios,x=>x.name==s.name)}
            Text "{money(s.payment,"$",0)}" size=40 weight=700
            Text.caption "a month"
            Divider
            Row between
              Text.muted "Price"
              Text.strong "{money(s.price,"$",0)}"
            Row between
              Text.muted "Deposit"
              Text.strong "{money(s.deposit,"$",0)}"
            Row between
              Text.muted "Term"
              Text.strong "{s.term} years"
            Row between
              Text.muted "Rate"
              Text.strong "{s.rate}%"
            Row between
              Text.muted "Total interest"
              Text.strong "{money(s.interest,"$",0)}"
            Row between
              Text.muted "Vs cheapest"
              Badge.neutral "+{money(s.payment-min(s.payment,scenarios[0].payment),"$",0)} / mo"
            Button.outline "Load into planner" onTap={price=s.price; deposit=s.deposit; term=s.term; rate=s.rate; emit recalc; go planner}
      if length(scenarios)>1
        Card gap=8
          Text.strong "Total interest by scenario"
          BarChart values={map(scenarios,s=>s.interest)} labels={map(scenarios,s=>s.name)} h=200
