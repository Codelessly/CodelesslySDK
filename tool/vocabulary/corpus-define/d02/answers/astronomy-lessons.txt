### Moon quiz
state qs=[{"q":"How many moons does Mars have?","options":["0","2","14","79"],"answer":1},{"q":"Which planet has the most known moons?","options":["Jupiter","Earth","Saturn","Neptune"],"answer":2},{"q":"What is the name of Earth's only moon?","options":["Luna","Titan","Europa","Io"],"answer":0},{"q":"Which moon of Jupiter may hide an ocean under its ice?","options":["Phobos","Europa","Triton","Charon"],"answer":1}]
state i=0
state picked=-1
state score=0
state done=false
App
  Sidebar
    Brand "Orbit School" icon="public"
    NavItem "Solar system" icon="brightness_5" onTap={go solar-system}
    NavItem "Compare planets" icon="compare" onTap={go compare}
    NavItem "Moon quiz" icon="quiz" selected={true}
    NavItem "Sky tonight" icon="nights_stay" onTap={go sky}
  Main
    Nav
      Heading "Moon quiz"
      Badge.primary "Score {score}"
    Column pad=24 gap=16 maxw=720
      if !done
        Progress {(i + 1) / length(qs)}
        Text.caption "Question {i + 1} of {length(qs)}"
        each q in qs
          if index == i
            Card gap=12
              Text.heading "{q.q}"
              each o in q.options
                ChoiceChip "{o}" selected={picked == index} onTap={picked == -1 ? picked = index : toast "Already answered"}
              if picked != -1
                Text.success "{picked == q.answer ? 'Correct!' : 'Not quite. The answer is ' + q.options[q.answer] + '.'}"
              Row end gap=8
                Button "{i + 1 == length(qs) ? 'Finish' : 'Next'}" onTap={picked == -1 ? toast "Pick an answer first" : {score = score + (picked == q.answer ? 1 : 0); picked = -1; i + 1 == length(qs) ? done = true : i = i + 1}}
      if done
        Card gap=12 center
          Text.heading "You scored {score} out of {length(qs)}"
          Button "Try again" onTap={i=0; picked=-1; score=0; done=false}
          Button.outline "Back to the solar system" onTap={go solar-system}
