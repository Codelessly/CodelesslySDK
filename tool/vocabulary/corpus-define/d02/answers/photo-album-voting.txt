define Uploader "name"
  Row gap=6 middle
    Avatar "{name}" size=24
    Text.small "{name}"

define LikeButton @liked onChange
  state liked=false
  IconButton {liked ? "favorite" : "favorite_border"} tooltip="Like" onTap={liked=!liked; onChange(liked)}

define Countdown "label" seconds=0
  state left=seconds
  every 1s when left>0 onTick={left=left-1}
  Column gap=2 pad=12
    Text.caption "{label}"
    Text.strong "{duration(left)}"

define Rating @value max=5 onChange
  state value=0
  Row gap=2 middle
    each i in range(max)
      IconButton {i<value ? "star" : "star_border"} tooltip="{i+1} of {max}" onTap={value=i+1; onChange(i+1)}
