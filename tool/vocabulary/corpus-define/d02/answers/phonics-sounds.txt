// Sound cannot play here; the waves are drawn bars that pulse while a sound is playing.
define SoundWaves playing=false color=#FF2563EB
  Row gap=4 middle center h=56
    each i in range(9)
      Box w=6 h={8 + (playing ? ((i*5)%4+1)*10 : 0)} radius=3 fill={color} animate=pulse delay={i*80}

define LetterCard letter="a" word="apple" picture="apple" fill=#FFFDE68A playing=false onTap
  Row gap=32 middle center pad=24 fill={fill} radius=32 onTap={onTap}
    Column center gap=0
      Text "{upper(letter)}{letter}" size=140 weight=800
    Column center gap=8
      Icon "{picture}" size=96
      Text.strong "{word}"
      if playing
        Icon "volume_up" animate=pulse

define StarShelf count=0 total=10
  Card gap=8 pad=12
    Row between middle
      Text.strong "My stars"
      Text.muted "{count} of {total}"
    Row gap=6
      each i in range(total)
        Icon "{i<count ? 'star' : 'star_border'}" size=28 color={i<count ? '#FFF59E0B' : '#FF9CA3AF'}

define FloatingPic "word" icon="star" delay=0 drag
  Box w=120 h=120 radius=60 fill=#FFE0F2FE center middle animate=slide-up duration=700 delay={delay} drag={drag}
    Icon "{icon}" size=56
    Text.strong "{word}"

define BlendTile "letter" wide=false onTap
  Box w={wide ? 140 : 64} h=72 radius=14 fill=#FF2563EB center middle onTap={onTap}
    Text "{letter}" size=40 weight=800 color=#FFFFFFFF

define SoundProgress letter="a" score=0 plays=0
  Row gap=12 middle
    Text.strong "{upper(letter)}" w=24
    Progress {score} grow
    Text.muted "{plays} plays" w=70
