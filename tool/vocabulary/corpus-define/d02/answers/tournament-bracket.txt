// A bracket match: two team lines with scores, the winner in bold, the followed team highlighted.
define MatchCard a="" b="" sa="" sb="" follow="" onTap
  Card gap=4 pad=8 onTap={onTap} border={(a==follow || b==follow) ? 2 : 1}
    Row between
      Text "{a == "" ? "TBD" : a}" weight={sa != "" && sa > sb ? 700 : 400} color={a==follow ? #FF2563EB : #FF111827}
      Text.strong "{sa}"
    Row between
      Text "{b == "" ? "TBD" : b}" weight={sb != "" && sb > sa ? 700 : 400} color={b==follow ? #FF2563EB : #FF111827}
      Text.strong "{sb}"

// Lines joining pairs of matches to the next round.
define Connector count=1
  Column w=40 around
    each i in range(count)
      Box h=80 w=40 border-top=#FF9CA3AF border-right=#FF9CA3AF border-bottom=#FF9CA3AF

// Vertical timeline of match events: minute, an icon for the kind, the name, sides alternating by team.
define Timeline events=[]
  Column gap=8
    each e in events
      Row gap=12 middle
        Text.strong "{e.min}'" w=40
        Icon "{e.kind=="goal" ? "sports_soccer" : (e.kind=="yellow" ? "square" : "swap_horiz")}" color={e.kind=="yellow" ? #FFEAB308 : #FF111827"}
        Text "{e.who}"
        Spacer grow
        Badge.neutral "{e.team=="home" ? "Brazil" : "Japan"}"
