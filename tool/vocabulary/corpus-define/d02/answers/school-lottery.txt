### Draw day
state called=[]
state drum="idle"
state mine=[17,203,388]
state all=[88,17,251,402,203,19,388,64,330,120]
state n=0
every 1500 when drum=="spinning" onTick={n=n+1; called=take(all,n); drum=n>=length(all) ? "done" : "spinning"}
App
  Sidebar
    Brand "Seatdraw" icon="casino"
    NavItem "Rank schools" icon="format_list_numbered" onTap={go rank-schools}
    NavItem "Draw day" icon="stars" selected={true}
    NavItem "My offer" icon="mail" badge="1" onTap={go offer}
    NavItem "Zone map" icon="map" onTap={go rank-schools}
  Main
    Nav
      Heading "Draw day: March 21" level=1
      Row gap=8 middle
        Badge.error "Live"
        Button disabled={drum=="spinning"} onTap={drum="spinning"; n=0; called=[]} "Start the draw"
    Column pad=24 gap=16
      Grid cols=2 gap=16
        Card gap=12 center
          Text.strong "The drum"
          Drum spinning={drum=="spinning"} balls={all} current={last(called)}
          if drum=="idle"
            Text.muted "Waiting for the registrar to start the draw."
          if drum=="spinning"
            Text.muted "Drawing number {n} of {length(all)}"
          if drum=="done"
            Text.success "The draw is complete."
        Card gap=12
          Text.strong "Your family numbers"
          Row gap=8
            each m in mine
              Chip "#{m}" selected={contains(called,m)}
          if any(mine, m=>contains(called,m))
            Row gap=8 middle animate=pop
              Icon "celebration"
              Text.success "A number of yours was just called."
          Text.muted "Your numbers light up as they are called."
          Progress {n/length(all)}
      Card gap=12
        Row between
          Text.strong "Numbers called, in order"
          Text.muted "{length(called)} of {length(all)}"
        Row scroll gap=8
          each c in called
            Box w=56 h=56 radius=28 center middle fill={contains(mine,c) ? "#FF2563EB" : "#FFE5E7EB"} animate=scale stagger=80
              Text.strong "{c}" color={contains(mine,c) ? "#FFFFFFFF" : "#FF111827"}
        if length(called)==0
          Text.muted "No numbers called yet."

// A map with pins and shaded zones is not drawable from the core words; it shows as a labelled box.
define MapView center=[0,0] zoom=10 h=400
  Box h={h} fill=#FFE0F2FE radius=8 center middle
    Text.muted "Map"
    ...

// Draws a shaded region on the map; shown as a labelled tag.
define ZoneLayer zoneId="" visible=true
  if visible
    Badge.info "Zone {upper(zoneId)}"

// A marker on the map; shown as a tappable labelled badge.
define MapPin lat=0 lng=0 label="" selected=false onTap
  Badge.primary "{label}" onTap={onTap}

define Countdown seconds=0 format="h m s"
  Row gap=4 middle
    Icon "schedule"
    Text.strong "{floor(seconds/86400)}d {floor(seconds%86400/3600)}h {floor(seconds%3600/60)}m {seconds%60}s"

define SeatBar taken=0 seats=1
  Row gap=8 middle
    Progress {taken/seats} w=120
    Text.caption "{seats-taken} of {seats} seats left"

define PriorityTag "label"
  Badge.warning "{label} priority"

define ReorderList bind=items disabled=false
  state items=[]
  Column gap=8
    ...

// A spinning drum with numbered balls is not drawable from the core words; it shows as a labelled box that spins.
define Drum spinning=false balls=[] current=0
  Stack center middle
    Box w=200 h=200 radius=100 fill=#FFFEF3C7 border=3 animate={spinning ? "spin" : "none"}
      Text.muted "drum"
    Text "{current}" size=48 weight=700
