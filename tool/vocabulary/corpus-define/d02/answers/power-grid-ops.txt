define SubstationNode "name" load=0 mw=0 onTap
  Box w=150 pad=10 gap=4 radius=8 fill={load<60 ? "#FF166534" : load<85 ? "#FFB45309" : "#FF991B1B"} onTap={onTap}
    Text.strong "{name}" color=#FFFFFFFF
    Text "{mw} MW" color=#FFFFFFFF
    Text.small "{load}% loaded" color=#FFFFFFFF

define LineSegment "label" load=0 mw=0
  Column center gap=2 w=90
    Text.small "{mw} MW"
    Box h=6 w=90 radius=3 fill={load<60 ? "#FF16A34A" : load<85 ? "#FFF59E0B" : "#FFDC2626"}
    Text.small "{load}%"

define BreakerSymbol closed=true onTap
  Box w=44 h=44 radius=6 border=2 center middle fill={closed ? "#FF16A34A" : "#FFFFFFFF"} onTap={onTap}
    Icon {closed ? "power" : "power_off"}

// a dial that wobbles around its nominal value; shown as a gauge with a limit band
define FrequencyDial bind=freq min=49.5 max=50.5 nominal=50
  state freq=50
  Gauge {freq} min={min} max={max} limit={nominal+0.2} unit="Hz"
