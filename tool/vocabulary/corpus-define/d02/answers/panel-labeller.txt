    Nav
      Heading "Warnings" level=1
      Button.outline "Back to panel" onTap={go panel}
    Column pad=24 gap=16
      Card gap=12
        Text.strong "Wire does not suit the rating"
        if count(circuits, c => c.amps > (c.wire == "14" ? 15 : c.wire == "12" ? 20 : c.wire == "10" ? 30 : 40)) == 0
          Row gap=8 middle
            Icon "check_circle"
            Text.success "Every wire gauge suits its breaker."
        each c in filter(circuits, c => c.amps > (c.wire == "14" ? 15 : c.wire == "12" ? 20 : c.wire == "10" ? 30 : 40))
          ListTile "Slot {c.slot}: {c.name}" subtitle="{c.amps} A breaker on {c.wire} AWG wire (limit {c.wire == \"14\" ? 15 : c.wire == \"12\" ? 20 : c.wire == \"10\" ? 30 : 40} A)" icon="warning"
            Button.outline "Lower to limit" onTap={circuits=updateWhere(circuits, x => x.id == c.id, x => with(x, "amps", x.wire == "14" ? 15 : x.wire == "12" ? 20 : x.wire == "10" ? 30 : 40)); toast "Lowered"}
      Card gap=12
        Text.strong "Spare breakers"
        each c in filter(circuits, c => c.spare)
          Row between middle
            Text "Slot {c.slot} ({c.amps} A)"
            Badge.neutral "Spare"
        if count(circuits, c => c.spare) == 0
          Text.muted "No spare breakers."

// shows a card sized in true inches as a bordered box with corner cut marks, since the notation cannot measure physical size
define PrintCard "title" "subtitle" w=312 h=480 cut=true
  Column w={w} h={h} pad=16 gap=6 fill=#FFFFFFFF border=1 border-color={cut ? "#FF111827" : "#00000000"} radius=0
    Row between
      Text.strong "{title}"
      Text.caption "{subtitle}"
    Divider
    ...
    Spacer grow
    if cut
      Text.caption "- - - - cut here - - - -" center

define Breaker "slot" amps=20 name="" room="" phase="A" spare=false warn=false selected=false onTap
  Row gap=8 middle pad=8 radius=6 border={selected ? 2 : 1} fill={spare ? "#FFF3F4F6" : warn ? "#FFFEE2E2" : "#FFFFFFFF"} onTap={onTap}
    Text.mono "{slot}" w=24
    Box w=44 h=32 radius=4 fill={spare ? "#FF9CA3AF" : "#FF1F2937"} center middle
      Text.small "{amps}A" color=#FFFFFFFF
    Column grow gap=0
      if spare
        Text.muted "Spare"
      if !spare
        Text.strong "{name}"
        Text.caption "{room} / phase {phase}"
    if warn
      Icon "warning"

define BalanceBar "label" value=0 max=100
  Row gap=8 middle
    Text.strong "{label}" w=20
    Progress {clamp(value / max, 0, 1)} grow
    Text.mono "{value} A" w=56
