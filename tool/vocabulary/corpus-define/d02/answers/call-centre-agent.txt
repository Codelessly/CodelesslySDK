// A supervisor whisper indicator: a pulsing badge with the supervisor's name; the audio itself cannot be drawn.
define WhisperIndicator "name" "caption"
  Row gap=8 middle pad=[10,6,12,6] fill=#FFFEF3C7 radius=16 animate=pulse
    Icon "hearing"
    Column
      Text.strong "Whisper: {name}"
      Text.caption "{caption}"

define CallTimer seconds=0 label=""
  Row gap=8 middle
    Icon "timer"
    Text.mono "{floor(seconds/60)}:{seconds%60<10 ? "0" : ""}{seconds%60}"
    Text.caption "{label}"

define StatTile "label" "value"
  Column gap=2
    Text "{value}" size=28 weight=700
    Text.caption "{label}"

define Timeline items=[]
  Column gap=0
    each item in items
      Row gap=10 top
        Column center w=24
          IconTile "{item.kind=="chat" ? "chat" : item.kind=="call" ? "call" : item.kind=="visit" ? "home_repair_service" : "shopping_bag"}"
          Divider
        Column grow gap=2 pad=[0,0,0,12]
          Row between
            Text.strong "{item.title}"
            Text.caption "{item.when}"
          Text.muted "{item.detail}"
