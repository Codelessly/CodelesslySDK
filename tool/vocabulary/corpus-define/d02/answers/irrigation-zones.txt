// Tapping a tile picks the zone; the garden outline is drawn as a green box with a tile grid.
define GardenMap zones=[] onPick
  Box fill=#FFDCFCE7 radius=24 pad=16 border=2
    Grid cols=3 gap=12
      each z in zones
        ZoneTile name={z.name} running={z.running} left={z.left} fault={z.fault} onTap={onPick(z.id)}

define ZoneTile name="" running=false left=0 fault=false onTap
  Card gap=6 fill={running ? #FFDBEAFE : #FFFFFFFF} onTap={onTap}
    Row between middle
      Icon "{fault ? 'warning' : (running ? 'water_drop' : 'power_settings_new')}"
      if fault
        Badge.error "Fault"
      if !fault && running
        Badge.info "Running"
      if !fault && !running
        Badge.neutral "Idle"
    Text.strong "{name}"
    if running
      Text.muted "{left} min left"
      Progress {left/30}
    if !running
      Text.muted "Off"

// A tube shaped level gauge that fills from the bottom.
define MoistureTube bind=level
  state level=50
  Box w=44 h=180 radius=22 border=2 clip
    Column end h=fill
      Box h={level*1.8} w=fill fill=#FF3B82F6

define Stepper bind=value min=0 max=100 step=1 unit=""
  state value=0
  Row gap=8 middle
    IconButton "remove" tooltip="Less" onTap={value=max(min,value-step)}
    Text.strong "{value} {unit}"
    IconButton "add" tooltip="More" onTap={value=min(max,value+step)}

// Rows are days, columns are hours from 5 to 12; the bar is dragged onto another hour to move the start.
define DayGrid bind=start days=[] on=[] dur=20 onChange
  state start=6
  Column gap=4
    Row gap=2
      Text.caption "" w=44
      each h in [5,6,7,8,9,10,11,12]
        Text.caption "{h}:00" grow center
    each d in days
      Row gap=2 middle
        Text.caption "{d}" w=44
        each h in [5,6,7,8,9,10,11,12]
          Box grow h=32 radius=4 fill=#FFF3F4F6 drop=slot onDrop={onChange(h)}
            if contains(on,d) && h==start
              Box fill=#FF2563EB radius=4 h=24 w=fill drag=slot(h) animate=pop
                Text.small "{dur}m" color=#FFFFFFFF center
