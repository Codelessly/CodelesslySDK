define Breadcrumb items=[] onPick
  Row gap=4 middle
    each crumb in items
      if index > 0
        Icon "chevron_right" size=16
      Text.muted "{crumb.label}" onTap={onPick(crumb.id)}

define Bullet text="" depth=0 task=false done=false folded=false hasChildren=false id="" onZoom onFold onCheck onEdit onIndent onOutdent onEnter onMore
  Row gap=4 middle pad=[{depth*24},0,0,0] drag=bullet(id)
    Icon "drag_indicator" size=16 hide=phone
    if hasChildren
      IconButton "{folded ? 'arrow_right' : 'arrow_drop_down'}" tooltip="Fold" onTap={onFold}
    if !hasChildren
      Spacer w=40
    if task
      Checkbox "" selected={done} onChange={onCheck}
    if !task
      Icon "circle" size=8 onTap={onZoom}
    Input "" @text grow strike={task && done} onChange={onEdit} onTab={onIndent} onShiftTab={onOutdent} onEnter={onEnter}
    IconButton "more_horiz" tooltip="More" onTap={onMore}

define Backlinks "title" target="" items=[] onPick
  Column gap=8
    Text.strong "{title}"
    if length(items)==0
      Text.muted "Nothing links to {target} yet."
    each hit in items
      Card gap=2 onTap={onPick(hit.id)}
        Text "{hit.text}"
        Text.caption "{hit.path}"
