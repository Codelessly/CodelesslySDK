// A countdown to a sailing; turns red below warnBelow seconds. Shows as a labelled figure.
define Countdown "label" seconds=0 warnBelow=600
  Row gap=8 middle
    Icon "schedule" color={seconds<warnBelow ? "#FFDC2626" : "#FF475569"}
    Column gap=0
      Text.caption "{label}"
      Text.strong "{duration(seconds)}" color={seconds<warnBelow ? "#FFDC2626" : "#FF111827"}

// A box for one vehicle, scaled by its length and tinted by its kind.
define VehicleBox vehicle={} scale=10 onTap
  Column center middle gap=2 w={max(36,vehicle.len*scale)} h=44 radius=4 fill={vehicle.kind=="lorry" ? "#FFF59E0B" : vehicle.kind=="van" ? "#FF60A5FA" : vehicle.kind=="bike" ? "#FF86EFAC" : "#FFA5B4FC"} onTap={onTap}
    Text.small "{vehicle.plate}" color=#FF111827
    Text.small "{vehicle.len} m" color=#FF111827

// One lane of the deck: a long strip with its vehicles in order, a gap marker and a full marker. Vehicles are dropped onto it.
define Lane lane={} vehicles=[] selected=0 onDrop onPick
  Row gap=8 middle
    Text.strong "{lane.id}" w=32
    Row scroll grow gap=2 h=52 pad=4 radius=6 fill=#FFE5E7EB drop=veh onDrop={onDrop}
      each v in sortBy(filter(vehicles,v=>v.lane==lane.id),v=>v.pos)
        VehicleBox vehicle={v} onTap={onPick(v.id)}
    if sum(filter(vehicles,v=>v.lane==lane.id),v=>v.len)>=lane.cap-2
      Badge.error "Full"
    else
      Text.muted "{fixed(lane.cap-sum(filter(vehicles,v=>v.lane==lane.id),v=>v.len),1)} m" w=60

// A needle between two ends (port and starboard); a real dial is drawn as a labelled track with a marker.
define TrimNeedle value=0 min=-6 max=6 leftLabel="" rightLabel="" limit=1.5
  Column gap=6
    Stack h=24
      Box h=8 radius=4 fill=#FFE5E7EB
      Box w={(2*limit)/(max-min)*260} h=8 radius=4 fill=#FFBBF7D0
      Icon "arrow_drop_down" color={abs(value)>limit ? "#FFDC2626" : "#FF111827"}
    Row between
      Text.muted "{leftLabel}"
      Text.strong "{fixed(value,1)}" color={abs(value)>limit ? "#FFDC2626" : "#FF111827"}
      Text.muted "{rightLabel}"
