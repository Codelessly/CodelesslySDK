// a coin that spins in the air then lands on heads or tails from the seed
define CoinFlip bind=result seed=0 size=100 heads="Heads" tails="Tails" onLanded
  state result=""
  state flipping=false
  state n=0
  every 120 when flipping onTick={n=n+1; n>=12 ? {flipping=false; n=0; result=seed%2==0 ? heads : tails; onLanded(result)} : }
  Column center gap=10
    Stack center middle w={size} h={size}
      Box w={size} h={size} radius={size/2} fill={result==tails ? '#FF9CA3AF' : '#FFF59E0B'} animate={flipping ? 'spin' : 'pop'} duration=240
      Text "{flipping ? '?' : (result==tails ? 'T' : (result=='' ? '' : 'H'))}" size={size/2} weight=700 color=#FF111827
    Button "{flipping ? 'Flipping' : 'Flip the coin'}" onTap={result=""; flipping=true}

// a wheel of names that spins and stops, leaving the names in the order drawn; drawn as a labelled disc
define PickOrderWheel bind=order names=[] seed=0 size=200 onSpin
  state order=[]
  state spinning=false
  state k=0
  every 100 when spinning onTick={k=k+1; order=map(range(length(names)),i=>names[(i+k+seed)%length(names)]); k>=12 ? {spinning=false; k=0; onSpin(order)} : }
  Column center gap=10
    Stack center middle w={size} h={size}
      Box w={size} h={size} radius={size/2} fill=#FF2563EB animate={spinning ? 'spin' : 'fade'} duration=500
      Text "{coalesce(first(order),'Spin')}" color=#FFFFFFFF weight=700
    Button.secondary "Spin the wheel" icon="autorenew" onTap={spinning=true}
