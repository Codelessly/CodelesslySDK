// A small red dot that pulses; marks a room as live.
define LiveDot color=#FFDC2626 size=10
  Box w={size} h={size} radius={size/2} fill={color} animate=pulse

define AvatarStack names=[] size=32 max=3
  Row middle
    each n in take(names, max)
      Avatar "{n}" size={size} border=2
    if length(names) > max
      Text.caption "+{length(names)-max}"

// Speakers are shown in a wrapped ring-like layout; a true circular placement is approximated by a centered grid.
define SpeakerCircle speakers=[] active="" size=72 onTap
  Grid cols=3 gap=20 center
    each s in speakers
      Column center gap=6 onTap={onTap(s)}
        Stack center middle
          Box w={size+16} h={size+16} radius={(size+16)/2} fill={s.name==active ? #3322C55E : #00000000} animate={s.name==active ? "pulse" : ""}
          Avatar "{s.name}" size={size}
        Row gap=4 middle
          Text.strong "{s.name}"
          if s.role=="host"
            Badge.primary "Host"
        Icon "{s.name==active ? 'graphic_eq' : 'mic_off'}"

// Emoji drift upward from the bottom of the circle and fade; each item rises when it appears.
define FloatingReactions items=[] h=160
  Stack h={h} right bottom
    each r in take(reverse(items), 8)
      Text "{r.emoji}" size=32 animate=slide-up duration=1200
