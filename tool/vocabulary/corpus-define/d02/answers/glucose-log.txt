// a coloured band behind a big reading: low, in range or high
define ReadingBand value=100 low=70 high=180
  Box pad=12 radius=12 fill={value < low ? #FFFEE2E2 : (value > high ? #FFFEF3C7 : #FFDCFCE7)}
    Row gap=8 middle
      Text "{value}" size=64 weight=700 color={value < low ? #FFB91C1C : (value > high ? #FFB45309 : #FF15803D)}
      Column
        Text.strong "mg/dL"
        Badge "{value < low ? "Low" : (value > high ? "High" : "In range")}"

// a number pad with digits, a delete key and a clear key
define NumberPad bind=value max=3
  state value=""
  Grid cols=3 gap=8
    each d in ["1","2","3","4","5","6","7","8","9","C","0","<"]
      Button.outline "{d}" onTap={d=="C" ? value="" : (d=="<" ? value=slice(value,0,length(value)-1) : (length(value) < max ? value=value+d : toast "Three digits at most"))}

// a day chart: dots joined by a line, a green band between low and high, a small plate over meal readings
define DayChart readings=[] low=70 high=180 h=240
  Stack h={h}
    Box fill=#2216A34A radius=4 h={h*(high-low)/300} top={h*(300-high)/300}
    LineChart values={map(readings,r=>r.v)} labels={map(readings,r=>r.t)} h={h} dots=true
    Row between
      each r in readings
        if r.meal != "Fasting"
          Icon "restaurant"

// a table of hours by day; each cell is coloured by its reading, blank when none
define HourTable rows=[] columns=[] low=70 high=180 onPick
  Column gap=4
    Row gap=4
      Text.caption "" w=48
      each c in columns
        Text.caption "{c}" w=64 center
    each row in rows
      Row gap=4 middle
        Text.strong "{row.day}" w=48
        each cell in row.cells
          Box w=64 h=36 radius=6 fill={cell == 0 ? #FFF3F4F6 : (cell < low ? #FFFECACA : (cell > high ? #FFFDE68A : #FFBBF7D0))} onTap={onPick(cell)}
            Text "{cell == 0 ? "" : cell}" center
