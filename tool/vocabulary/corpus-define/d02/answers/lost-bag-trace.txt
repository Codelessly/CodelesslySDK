// A barcode camera is not available; the scan button is shown as a labelled box.
define TagScanner "label" bind=tag onScan
  state tag=""
  Row gap=8 middle
    Input "{label}" bind=tag grow
    Box w=96 h=48 radius=8 fill=#FF111827 center middle onTap={onScan("LH 482913")}
      Icon "qr_code_scanner"
      Text.caption "Scan" color=#FFFFFFFF

define BagShapePicker bind=shape shapes=[{"id":"hard","name":"Hard case","icon":"luggage"},{"id":"soft","name":"Soft duffel","icon":"work"},{"id":"back","name":"Backpack","icon":"backpack"},{"id":"tote","name":"Tote","icon":"shopping_bag"}]
  state shape=""
  Grid cols=4 gap=12
    each s in shapes
      Card center gap=4 border={s.id==shape ? 2 : 1} onTap={shape=s.id}
        Icon "{s.icon}" size=40
        Text.small "{s.name}"

define ColorSwatches bind=color colors=["#FF111827","#FF6B7280","#FFFFFFFF","#FFDC2626","#FF2563EB","#FF16A34A","#FFF59E0B"] size=32
  state color=""
  Row gap=8 middle
    each c in colors
      Box w={size} h={size} radius={size/2} fill={c} border={c==color ? 3 : 1} onTap={color=c}

define StatusSteps steps=["Reported","In transit","Located","Delivered"] current=0
  Row gap=8 middle
    each s in steps
      Row gap=8 middle grow
        Box w=28 h=28 radius=14 center middle fill={index<=current ? "#FF16A34A" : "#FFE5E7EB"}
          Text "{index+1}" color=#FFFFFFFF
        Text.small "{s}"
        Divider grow

define RouteLine stops=["A","B"] current="" bind=sel
  state sel=0
  Row middle gap=0
    each s in stops
      Row middle grow
        Column center gap=4
          Box w=16 h=16 radius=8 fill={s==current ? "#FF16A34A" : "#FF2563EB"}
          Text.strong "{s}"
        Divider grow

define ScanTimeline events=[] bind=sel onSelect
  state sel=0
  Column gap=0
    each e in events
      Row gap=12 onTap={onSelect(index)}
        Column center gap=0
          Box w=14 h=14 radius=7 fill={e.done ? "#FF16A34A" : "#FF9CA3AF"} border={index==sel ? 3 : 0}
          Divider w=2 h=36
        Column gap=2 grow
          Text.strong "{e.airport} - {e.label}"
          Text.caption "{e.city}, {e.time}"

define ChatBubble "text" mine=false time=""
  Row {mine ? "end" : "start"}
    Box pad=12 radius=12 fill={mine ? "#FF2563EB" : "#FFF3F4F6"} maxw=420 gap=2
      Text "{text}" color={mine ? "#FFFFFFFF" : "#FF111827"}
      Text.caption "{time}" color={mine ? "#FFDBEAFE" : "#FF6B7280"}

define ChatLog messages=[]
  Column pad=24 gap=8 scroll grow
    each m in messages
      ChatBubble "{m.text}" mine={m.mine} time={m.time}

define TimeWindowPicker bind=window slots=["09:00-11:00"]
  state window=""
  Row gap=8
    each s in slots
      ChoiceChip "{s}" selected={window==s} onTap={window=s}
