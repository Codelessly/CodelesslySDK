// A real 3D renderer cannot be drawn with the core words, so it shows as a labelled box that stands for the model.
define Model3D "src" yaw=30 pitch=20 zoom=1 panX=0 panY=0 view="iso" wireframe=false sectionAt=100 light=[0.5,0.7,0.5] measuring=false hidden=[] isolated="" selected="" color="#FFB45309" roughness=0.4 metalness=0.8 onOrbit onZoom onPan onPick onMeasure
  Box h=fill fill=#FF1F2937 radius=8 center middle
    Text "3D view of {src}: yaw {yaw}, pitch {pitch}, zoom {zoom}" color=#FF9CA3AF

// A real cube widget is drawn as a grid of the six face buttons.
define ViewCube @view onPick
  state view="iso"
  Card gap=4 pad=6
    Row gap=4
      Chip "Top" selected={view == "top"} onTap={view="top"; onPick("top")}
      Chip "Front" selected={view == "front"} onTap={view="front"; onPick("front")}
    Row gap=4
      Chip "Left" selected={view == "left"} onTap={view="left"; onPick("left")}
      Chip "Right" selected={view == "right"} onTap={view="right"; onPick("right")}
    Row gap=4
      Chip "Back" selected={view == "back"} onTap={view="back"; onPick("back")}
      Chip "Bottom" selected={view == "bottom"} onTap={view="bottom"; onPick("bottom")}

// A draggable ball needs pointer capture the core words lack, so it shows as a labelled ball.
define LightBall @light size=96 onChange
  state light=[0.5,0.7,0.5]
  Stack w={size} h={size}
    Box w={size} h={size} radius={size/2} fill=#FF374151
    Box w=14 h=14 radius=7 fill=#FFFBBF24 left={light[0]*size/2+size/2-7} top={size/2-light[1]*size/2-7}

define ColorPicker @color swatches=["#FF111827","#FFDC2626","#FF2563EB"] size=28 onPick
  state color=""
  Row gap=8
    each swatch in swatches
      Box w={size} h={size} radius={size/2} fill={swatch} border={swatch == color ? 2 : 0} onTap={color=swatch; onPick(swatch)}

Dialog#export "Export image" "Render the current view"
  Select "Size" items=["1024 x 1024","2048 x 2048","4096 x 4096"] @exportSize
  Toggle "Transparent background" @exportAlpha
  Button "Export" onTap={emit exportImage; toast "Saved to Downloads"; close}

    Nav
      Heading "Parts" level=1

          Text.strong "Material" w=140

            Text.muted "{part.material}" w=140
            Text.muted "{part.tris}" w=90

      Text.muted "{sum(parts, p => p.tris)} triangles in total"
  Sheet#partSheet "Part details"
    Text.heading "{find(parts, p => p.id == picked).name}"
    Text.muted "{find(parts, p => p.id == picked).material}"
    Button "Isolate in viewer" onTap={emit isolate picked; close; go viewer}
    Button.outline "Hide part" onTap={parts=updateWhere(parts, p => p.id == picked, p => with(p,"hidden",true)); close}

### Measurements
state measures=[{"id":1,"label":"Body width","mm":182.4},{"id":2,"label":"Spout to handle","mm":264.0},{"id":3,"label":"Lid height","mm":38.5}]
state unit="mm"
state sel=0
state newLabel=""

App
  Sidebar
    Brand "Turntable" icon="view_in_ar"
    NavItem "Viewer" icon="view_in_ar" onTap={go viewer}
    NavItem "Parts" icon="account_tree" onTap={go parts}
    NavItem "Measurements" icon="straighten" selected={true}
  Main
    Nav
      Heading "Measurements" level=1
      SegmentedButton items=["mm","in"] @unit
    Column pad=24 gap=16
      if length(measures) == 0
        Text.muted "No measurements yet. Use Measure in the viewer to place two points."
      each m in measures
        Card
          Row between middle
            Column gap=2
              Text.strong "{m.label}"
              Text.muted "{unit == 'mm' ? fixed(m.mm,1) + ' mm' : fixed(m.mm/25.4,2) + ' in'}"
            Row gap=4
              IconButton "edit" tooltip="Rename" onTap={sel=m.id; open rename}
              IconButton "delete" tooltip="Delete" onTap={measures=removeWhere(measures, x => x.id == m.id); toast "Deleted"}
      Button.outline "Clear all" onTap={measures=[]}
  Dialog#rename "Rename measurement"
    Input "Label" @newLabel
    Button "Save" onTap={measures=updateWhere(measures, x => x.id == sel, x => with(x,"label",newLabel)); close}
