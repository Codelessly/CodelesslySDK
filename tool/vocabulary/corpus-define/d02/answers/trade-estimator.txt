define DiffCell a=0 b=0 money=false w=100
  Row w={w} gap=6 middle pad=4 radius=4 fill={a != b ? (b > a ? "#FFFEF3C7" : "#FFFEE2E2") : "#00000000"}
    Text.muted strike={a != b} "{money ? money(a,'$',2) : a}"
    if a != b
      Icon "arrow_forward" size=14
      Text.strong "{money ? money(b,'$',2) : b}"
