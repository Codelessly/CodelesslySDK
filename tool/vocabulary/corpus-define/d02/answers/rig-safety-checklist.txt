// a camera capture, shown as a labelled box
define PhotoCapture bind=photo label="Add a photo"
  state photo=""
  Box h=120 radius=8 fill=#FFF3F4F6 center middle onTap={photo = photo=="" ? "captured" : ""}
    Column center gap=4
      Icon name={photo=="" ? "photo_camera" : "image"}
      Text.muted "{photo=="" ? label : "Photo attached, tap to remove"}"

// a deck diagram with a dot per area; a real drawing is shown as a labelled stack of bands
define DeckMap decks=["Upper","Main"] dots=[] onSelect
  Column gap=6
    each deck in decks
      Stack h=70 radius=6 fill=#FFE5E7EB
        Text.small "{deck} deck" left top pad=4
        each dot in filter(dots,d=>d.deck==deck)
          Icon name="circle" color={dot.done ? #FF16A34A : #FF9CA3AF} size=18 onTap={onSelect(dot.id)}

### Gas Readings
App
  Sidebar
    Brand "Derrick Daily" icon="oil_barrel"
    NavItem "Safety round" icon="checklist" onTap={go round}
    NavItem "Gas readings" icon="air" selected={true}
    NavItem "Muster" icon="groups" onTap={go muster}
    NavItem "Sign-off" icon="draw" onTap={go sign-off}
  Main
    Nav
      Heading "Gas readings" level=1
      Row gap=8
        Badge.error "{count(gas,g=>g.id!='g4' && g.value>g.limit) + count(gas,g=>g.id=='g4' && g.value<g.limit)} over limit"
        Button "Back to round" onTap={go round}
    Column pad=24 gap=16 maxw=720
      Text.muted "Type the detector figure at each point. A box turns red beyond its limit."
      Grid cols=2 gap=16
        each row in gas
          GasReading "{row.label}" bind=row.value limit={row.limit} low={row.id=="g4"} unit={row.id=="g4" ? "% vol" : row.id=="g3" ? "% LEL" : "ppm"}
      if any(gas,g=>g.id!="g4" && g.value>g.limit)
        Card gap=8 fill=#FFFEE2E2
          Text.strong color=#FFB91C1C "Toxic or flammable gas above limit"
          Text color=#FF7F1D1D "Stop work in the area, alert the control room and log a failure on the round."
          Button.destructive "Call control room" onTap={emit controlRoom; toast "Control room alerted"}

define GasReading "label" bind=value limit=10 low=false unit="ppm"
  state value=0
  Card gap=6 fill={(low ? value<limit : value>limit) ? #FFFEE2E2 : #FFFFFFFF}
    Text.fieldLabel "{label}"
    Input "Reading ({unit})" type=number bind=value
    Row between
      Text.small "Limit {low ? 'below' : 'above'} {limit} {unit}"
      Text.strong color={(low ? value<limit : value>limit) ? #FFDC2626 : #FF16A34A} "{(low ? value<limit : value>limit) ? 'OVER LIMIT' : 'OK'}"

### Muster
state phase="idle"
state seconds=0
every 1s when phase=="running" onTick={seconds=seconds+1}
App
  Sidebar
    Brand "Derrick Daily" icon="oil_barrel"
    NavItem "Safety round" icon="checklist" onTap={go round}
    NavItem "Gas readings" icon="air" onTap={go gas-readings}
    NavItem "Muster" icon="groups" selected={true}
    NavItem "Sign-off" icon="draw" onTap={go sign-off}
  Main
    Nav
      Heading "Muster headcount" level=1
      Badge.info "{onboard} persons on board"
    Column pad=24 gap=16 center maxw=720
      Card gap=12 center
        Text.muted "Time since alarm"
        Text "{duration(seconds)}" size=48 weight=700
        Row gap=8
          Button "Start drill" onTap={phase="running"; seconds=0; mustered=0; missing=[]}
          Button.outline "Stop" onTap={phase="idle"}
      Card gap=12 center
        Ring {mustered/onboard}
        Text "{mustered} of {onboard} at muster stations" size=24 weight=600
        Row gap=8
          Button.outline "-5" onTap={mustered=max(0,mustered-5)}
          Button.outline "-1" onTap={mustered=max(0,mustered-1)}
          Button "+1" onTap={mustered=min(onboard,mustered+1)}
          Button "+5" onTap={mustered=min(onboard,mustered+5)}
        Button.destructive "Mark unaccounted" onTap={open unaccounted}
      if mustered==onboard
        Card fill=#FFDCFCE7 gap=4
          Text.strong color=#FF166534 "All persons accounted for in {duration(seconds)}"
      if length(missing)>0
        Card gap=6
          Text.strong "Unaccounted"
          each name in missing
            ListTile "{name}" icon="person_search"
              Button.ghost "Found" onTap={missing=without(missing,name); mustered=min(onboard,mustered+1)}
Dialog#unaccounted "Unaccounted person" "Enter a name or badge number"
  Input "Name or badge" bind=who
  Button "Add to list" onTap={who=="" ? toast "Enter a name" : {missing=append(missing,who); who=""; close}}

          each area in areas
            if count(checks,c=>c.area==area.id && c.answer=="")>0

                Text "{area.name}"

      Card gap=12
        Input "Full name" bind=signer
        Input "Comments" multiline bind=comments
        Button "Sign off the round" disabled={count(checks,c=>c.answer=="")>0 || trim(signer)==""} onTap={signed=true; emit signoff signer; open done}

  Text "Signed by {signer}. {count(checks,c=>c.answer=='Fail')} failures are logged."
  Button "Close" onTap={close; go round}
