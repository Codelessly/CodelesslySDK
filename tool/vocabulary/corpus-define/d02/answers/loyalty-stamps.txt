// A card of one shop: its name, a grid of stamps and a redeem button once full.
define StampCard shop="" stamps=0 total=10 color=#FF7C4A2D reward="" onRedeem
  Column gap=12 pad=20 fill={color} radius=16
    Row between middle
      Text.heading "{shop}" color=#FFFFFFFF
      Text.strong "{stamps}/{total}" color=#FFFFFFFF
    StampGrid filled={stamps} total={total}
    Row between middle
      Text.small "Reward: {reward}" color=#FFFFFFFF
      Button.elevated "Redeem" visible={stamps >= total} onTap={onRedeem}

define StampGrid filled=0 total=10 cols=5
  Grid cols={cols} gap=8
    each i in range(total)
      if i < filled
        Box w=44 h=44 radius=22 fill=#FFFFFFFF center middle animate=pop
          Icon "check" color=#FF15803D
      if i == filled
        Box w=44 h=44 radius=22 border=2 center middle animate=pulse
          Icon "add" color=#FFFFFFFF
      if i > filled
        Box w=44 h=44 radius=22 border=1
