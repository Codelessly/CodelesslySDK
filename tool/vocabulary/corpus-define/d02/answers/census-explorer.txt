define ChoroplethMap measure="Median age" year=2020 bind=selected regions=[] breaks=[] h=360 onSelect
  // a real map is a labelled box here: regions are listed as tiles shaded by break
  Box h={h} fill=#FFF1F5F9 radius=8 pad=12 gap=8
    Text.caption "Map: {measure}, {year}"
    Grid cols=3 gap=8
      each r in regions
        Box pad=10 radius=6 fill={r.age < breaks[0] ? "#FFDBEAFE" : r.age < breaks[1] ? "#FF93C5FD" : r.age < breaks[2] ? "#FF3B82F6" : "#FF1D4ED8"} border={selected==r.name ? 2 : 0} onTap={selected=r.name; onSelect(r.name)}
          Text.small "{r.name}"

define BreakLegend bind=breaks min=0 max=100 colors=[] onChange
  Column gap=6
    Row gap=2
      each c in colors
        Box grow h=12 fill={c}
    Row gap=8
      each b in breaks
        Slider bind=b min={min} max={max} grow onChange={onChange(breaks)}
    Text.caption "Drag a break to change the shading: {join(breaks," / ")}"

define PopulationPyramid ages=[] left=[] right=[] leftLabel="Left" rightLabel="Right" h=260
  Column gap=4 h={h}
    Row between
      Text.caption "{leftLabel}"
      Text.caption "{rightLabel}"
    each a in ages
      Row gap=4 middle
        Row grow right
          Box h=14 w={left[indexOf(ages,a)]*12} fill=#FF3B82F6 radius=2
        Text.caption "{a}" w=44 center
        Row grow
          Box h=14 w={right[indexOf(ages,a)]*12} fill=#FFF59E0B radius=2

define GroupedBarChart labels=[] series=[] h=260
  Column gap=8
    Row gap=12
      each s in series
        Text.caption "{s.name}"
    Row gap=6 bottom h={h}
      each l in labels
        Column grow center gap=2
          Row gap=2 bottom
            each s in series
              Box w=14 h={s.values[indexOf(labels,l)]*10} fill={indexOf(series,s)==0 ? "#FF3B82F6" : "#FFF59E0B"} radius=2
          Text.caption "{l}"

define DataTable bind=sortCol dir="asc" rows=[] onSort
  Column gap=0
    Row gap=8 pad=12 border-bottom=#FFE5E7EB
      ...
    each r in rows
      Row gap=8 pad=12 border-bottom=#FFF3F4F6
        Text "{r.name}" grow
        Text "{r.pop}" w=110
        Text "{r.age}" w=90
        Text "{r.hh}" w=90
        Text "{r.spanish}" w=90

define Column "title" key="" align="left"
  Text.strong "{title}" grow={align=="left"} w={align=="left" ? 0 : 110}
