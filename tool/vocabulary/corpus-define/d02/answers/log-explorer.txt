// A query field that colours field names and operators as it is typed.
define QueryBar @query placeholder="service:checkout AND level:error" onSubmit
  state query=""
  Column gap=4
    Input "Query" "{placeholder}" @query onSubmit={onSubmit}
    Row gap=6
      each tok in split(query," ")
        Text.mono "{tok}" color={contains(tok,":") ? "#FF2563EB" : (tok=="AND" || tok=="OR" || tok=="NOT") ? "#FFDC2626" : "#FF111827"}

// A time range menu with presets and a custom range.
define TimeRangePicker @range presets=["Last 15 min"] onPick
  state range=""
  state openMenu=false
  Column gap=4
    Button.outline "{range}" icon="schedule" onTap={openMenu=!openMenu}
    if openMenu
      Card gap=4
        each p in presets
          ListTile "{p}" selected={range==p} onTap={range=p; openMenu=false; onPick(p)}

// Bars of log counts; a drag across them selects a span (shown as the highlighted bars).
define Histogram values=[] labels=[] h=100 @range onSelect
  state range=""
  state lo=-1
  state hi=-1
  Column gap=4
    Row gap=2 bottom h={h}
      each v in values
        Box grow h={max(4, v/max(1,max(values[0],max(values[5],values[6])))*h)} radius=2 fill={index>=lo && index<=hi ? "#FF2563EB" : "#FF93C5FD"} onTap={lo=index; hi=index; onSelect({"from":index*5,"to":index*5+5})}
    Row between
      each l in labels
        Text.caption "{l}"

// A single log line with a level colour, a time and an expandable field list.
define LogRow log={} expanded=false onToggle onInclude onExclude
  Card gap=6 onTap={onToggle}
    Row gap=10 middle
      Box w=4 h=20 radius=2 fill={log.level=="error" ? "#FFDC2626" : log.level=="warn" ? "#FFD97706" : log.level=="info" ? "#FF2563EB" : "#FF9CA3AF"}
      Text.mono "{log.time}" color=#FF6B7280
      Badge "{upper(log.level)}"
      Text "{log.msg}" grow
      Icon "{expanded ? 'expand_less' : 'expand_more'}"
    if expanded
      Column gap=4 pad=[14,4,0,0]
        each f in log.fields
          Row gap=8 middle
            Text.mono "{f.k}" color=#FF2563EB w=110
            Text.mono "{f.v}" grow
            IconButton "add_circle" tooltip="Add as filter" onTap={onInclude(f)}
            IconButton "do_not_disturb_on" tooltip="Exclude from results" onTap={onExclude(f)}

// A field with its most common values and their shares.
define FieldSummary name="" top=[] total=1 onPick
  Accordion "{name}"
    each t in top
      Row gap=8 middle onTap={onPick(t.v)}
        Text.mono "{t.v}" grow
        Text.caption "{round(t.n/total*100,0)}%"
        Progress {t.n/total} w=60
