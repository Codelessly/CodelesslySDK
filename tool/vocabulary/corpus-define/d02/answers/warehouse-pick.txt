// A scan field: a text field with a scan button; a real camera or laser scanner is shown as a labelled box.
define ScanInput "label" @value onScan
  state value=""
  Column gap=8
    Row gap=8 middle
      Input "{label}" @value grow
      Button.outline "Scan" icon="qr_code_scanner" onTap={onScan(value)}
    Box h=64 radius=8 fill=#FF111827 center middle
      Text.caption "Camera viewfinder" color=#FFD1D5DB

define Stepper @value min=0 max=99 step=1 onChange
  state value=0
  Row gap=8 middle
    IconButton "remove" tooltip="Less" onTap={value=max(min,value-step); onChange(value)}
    Text.strong "{value}" w=40 center
    IconButton "add" tooltip="More" onTap={value=min(max,value+step); onChange(value)}

// A floor plan of the aisles with the walking path drawn between stops; shown as a labelled box with a stop list since lines cannot be drawn.
define AisleMap aisles=["A","B","C"] stops=[] current=0 start="Dock" h=360 onSelect
  Box h={h} radius=8 fill=#FFF3F4F6 pad=16 gap=8
    Row between
      Text.caption "{start}"
      Text.caption "Floor plan"
    Row gap=12 grow
      each a in aisles
        Column grow gap=6 center
          Text.strong "Aisle {a}"
          Box grow radius=6 fill=#FFE5E7EB
            each s in stops
              if startsWith(s,a)
                Chip "{s}" selected={indexOf(stops,s)==current} onTap={onSelect(indexOf(stops,s))}
    Text.caption "Path: {start} to {join(stops,\" > \")}"
