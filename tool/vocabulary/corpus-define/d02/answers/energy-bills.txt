// Shows a row of single digit boxes that fill from one reading; the value is the digits joined.
define DigitBoxes @value digits=5 onChange
  state cells=[{"d":""},{"d":""},{"d":""},{"d":""},{"d":""}]
  Row gap=8
    each c in cells
      Input @c.d type=number w=48 center onChange={value=join(map(cells,x=>x.d),""); onChange(value)}

// A camera or file picker cannot be drawn here, so it shows a labelled box that fills in once a photo is chosen.

  Box h=120 radius=8 border=1 center middle gap=4 onTap={photo="meter.jpg"}
    Icon "photo_camera"
    Text.muted "{label}"
    if photo!=null
      Text.caption "{photo} attached"
