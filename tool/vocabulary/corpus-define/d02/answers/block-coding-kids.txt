// a labelled tile standing for a level on the adventure map
define LevelNode "n" name="" stars=0 locked=false onTap
  Card gap=8 center animate=pop onTap={locked ? toast "Finish the earlier levels first" : onTap}
    if locked
      IconTile "lock"
      Text.muted "Level {n}"
    if !locked
      Text.heading "{n}"
      Text.strong "{name}"
      Row gap=2 center
        Icon "star" color={stars >= 1 ? "#FFF59E0B" : "#FFD1D5DB"}
        Icon "star" color={stars >= 2 ? "#FFF59E0B" : "#FFD1D5DB"}
        Icon "star" color={stars >= 3 ? "#FFF59E0B" : "#FFD1D5DB"}

### Play
App
  Pane w=240 pad=12 gap=8
    Text.strong "Pieces"
    Text.muted "Drag a piece into your script."
    BlockPiece "Forward" icon="arrow_upward" fill=#FF2563EB drag=piece("forward")

      Badge.primary "{sum(script, s => s.cmd == 'repeat' ? 1 + length(s.body) : 1)} / {levels[curIdx].limit}"
    HintBuddy "{hintOpen ? (hintNo == 0 ? 'Start by walking forward.' : 'Wrap the same pieces in a Repeat to save space!') : 'Stuck? Tap Hint and I will help.'}"
    Button.outline "Hint" onTap={hintOpen=true; hintNo=(hintNo+1)%2}
  Main
    Nav
      Row gap=8 middle
        IconButton "arrow_back" tooltip="Back to the map" onTap={go level-map}
        Heading "Level {levels[curIdx].n}: {levels[curIdx].name}" level=1
      Row gap=8
        IconButton "replay" tooltip="Start over" onTap={pc=0; bi=0; lap=0; rx=levels[curIdx].rx; ry=levels[curIdx].ry; rdir=1; running=false; go1=false}
        IconButton "delete" tooltip="Clear script" onTap={script=[]; pc=0; bi=0; lap=0; rx=levels[curIdx].rx; ry=levels[curIdx].ry; rdir=1; running=false; go1=false}
    Row pad=24 gap=24 top
      Column grow gap=12
        Text.strong "Your script"
        Column#scriptArea gap=6 pad=12 radius=12 fill=#FFF3F4F6 h=360 scroll drop=piece onDrop={length(script) >= levels[curIdx].limit ? toast "That is all the pieces you may use" : {script=append(script,{"id":nextId,"cmd":result.data,"n":3,"body":[]}); nextId=nextId+1}}
          if length(script) == 0
            Text.muted "Drop pieces here, one under another"
          each s in script
            if s.cmd != "repeat"
              BlockPiece "{replace(s.cmd, '_', ' ')}" icon={s.cmd == "forward" ? "arrow_upward" : (s.cmd == "turn_left" ? "rotate_left" : "rotate_right")} fill={s.cmd == "forward" ? "#FF2563EB" : "#FFF97316"} active={running && pc == index}
            if s.cmd == "repeat"
              LoopBlock n={s.n} active={running && pc == index} onMinus={script=updateWhere(script, x => x.id == s.id, x => with(x, "n", max(1, x.n-1)))} onPlus={script=updateWhere(script, x => x.id == s.id, x => with(x, "n", min(9, x.n+1)))} onDropIn={script=updateWhere(script, x => x.id == s.id, x => with(x, "body", append(x.body, result.data)))}
                each b in s.body
                  BlockPiece "{replace(b, '_', ' ')}" icon={b == "forward" ? "arrow_upward" : (b == "turn_left" ? "rotate_left" : "rotate_right")} fill={b == "forward" ? "#FF2563EB" : "#FFF97316"} active={running && pc == index && bi == indexOf(s.body, b)}
        Row gap=8
          Button "Play" icon="play_arrow" onTap={script == [] ? toast "Add some pieces first" : {pc=0; bi=0; lap=0; rx=levels[curIdx].rx; ry=levels[curIdx].ry; rdir=1; running=true}}
          Button.outline "Step" icon="skip_next" onTap={go1=true}
          Button.ghost "Stop" icon="stop" onTap={running=false; go1=false}
      Column w=320 gap=12
        Text.strong "Stage"
        StageGrid cols=5 rows=5 robotX={rx} robotY={ry} robotDir={rdir} starX={levels[curIdx].sx} starY={levels[curIdx].sy} rocks={levels[curIdx].rocks}
        Row gap=8 middle
          Icon "flag" color=#FFF59E0B
          Text.small "Goal: help the robot reach the star"

every 600 when (running || go1) && pc < length(script) onTick={nx=rx+(rdir==1 ? 1 : (rdir==3 ? -1 : 0)); ny=ry+(rdir==2 ? 1 : (rdir==0 ? -1 : 0)); cmd=script[pc].cmd == "repeat" ? script[pc].body[bi] : script[pc].cmd; blocked=nx<0 || nx>4 || ny<0 || ny>4 || any(levels[curIdx].rocks, r => r.x==nx && r.y==ny); rx=(cmd=="forward" && !blocked) ? nx : rx; ry=(cmd=="forward" && !blocked) ? ny : ry; rdir=cmd=="turn_left" ? (rdir+3)%4 : (cmd=="turn_right" ? (rdir+1)%4 : rdir); script[pc].cmd == "repeat" ? (bi+1 < length(script[pc].body) ? {bi=bi+1} : (lap+1 < script[pc].n ? {bi=0; lap=lap+1} : {bi=0; lap=0; pc=pc+1})) : {pc=pc+1}; go1=false}
every 600 when (running || go1) && length(script) > 0 && pc >= length(script) onTick={running=false; go1=false; won=(rx==levels[curIdx].sx && ry==levels[curIdx].sy); won ? {levels=updateWhere(levels, l => l.n == levels[curIdx].n, l => with(l, "stars", max(l.stars, sum(script, s => s.cmd=='repeat' ? 1+length(s.body) : 1) <= l.par ? 3 : (sum(script, s => s.cmd=='repeat' ? 1+length(s.body) : 1) <= l.limit - 1 ? 2 : 1)))); unlocked=max(unlocked, levels[curIdx].n + 1); open win} : open oops}

// a grid with a robot, a star and rocks drawn from the cells
define StageGrid cols=5 rows=5 robotX=0 robotY=0 robotDir=1 starX=4 starY=4 rocks=[]
  Grid cols={cols} gap=4
    each i in range(cols*rows)
      Box h=56 radius=8 fill={(i%cols + floor(i/cols)) % 2 == 0 ? "#FFDCFCE7" : "#FFBBF7D0"} center middle
        if i%cols == robotX && floor(i/cols) == robotY
          Icon "smart_toy" size=36 color=#FF2563EB rotate={robotDir*90-90}
        if (i%cols != robotX || floor(i/cols) != robotY) && i%cols == starX && floor(i/cols) == starY
          Icon "star" size=36 color=#FFF59E0B animate=pulse
        if count(rocks, r => r.x == i%cols && r.y == floor(i/cols)) > 0
          Icon "landscape" size=32 color=#FF6B7280
