// A countdown timer that turns urgent below a threshold; uses a plain Text with a warning style.
define Countdown seconds=0 warnAt=300 onWarn
  Row gap=6 middle
    Icon "timer"
    Text.strong "{floor(seconds/60)}:{seconds%60<10 ? "0" : ""}{seconds%60}" color={seconds<=warnAt ? "#FFDC2626" : "#FF111827"} animate={seconds<=warnAt ? "pulse" : ""}
    if seconds==warnAt
      Badge.warning "5 min left"

define QuestionPalette bind=cur count=0 answers=[] flagged=[] seen=[]
  Grid cols=5 gap=6
    each i in range(count)
      Box w=36 h=36 radius=18 center middle fill={contains(flagged,i) ? "#FFFDE68A" : !isEmpty(answers[i]) ? "#FFBBF7D0" : contains(seen,i) ? "#FFE5E7EB" : "#FFFFFFFF"} border={cur==i ? 2 : 1} onTap={cur=i}
        Text.small "{i+1}"

// Drag to reorder: each row can be dragged onto another row.
define OrderList items=[] onChange
  Column gap=8
    each item in items
      Card drag=ord(item) drop=ord onDrop={onChange(concat(append(take(without(items,result.data),indexOf(items,item)),result.data),slice(without(items,result.data),indexOf(items,item),length(items))))}
        Row gap=8 middle
          Icon "drag_indicator"
          Text "{index+1}. {item}"

// Matching by lines cannot be drawn; a pair is made by tapping a left item then a right item and shown as a connected row.
define MatchPairs left=[] right=[] bind=pairs onChange
  state pairs=[]
  state pick=-1
  Row gap=24 top
    Column grow gap=8
      each l in left
        Button.outline "{l}" selected={pick==index} onTap={pick=index}
    Column grow gap=8
      each r in right
        Button.outline "{r}" onTap={pick>=0 ? {pairs=append(without(pairs,find(pairs,p=>p.l==left[pick])),{"l":left[pick],"r":r}); onChange(pairs); pick=-1} : toast "Pick a term first"}
  each p in pairs
    Row gap=6 middle
      Text.small "{p.l}"
      Icon "arrow_right_alt"
      Text.small "{p.r}"

// An image whose parts can be tapped; the tappable zones are drawn as a grid of boxes over the picture.
define Hotspot src="" regions=[] onPick
  state chosen=""
  Stack
    Image "{src}" h=260
    Grid cols=3 gap=0
      each r in regions
        Box h=130 border={chosen==r ? 2 : 0} onTap={chosen=r; onPick(r)}
  if chosen!=""
    Text.muted "Selected: {chosen}"
