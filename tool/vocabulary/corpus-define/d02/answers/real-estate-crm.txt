// a week strip of seven day buttons with a dot under days that have something booked
define WeekStrip bind=day marks=[] labels=["Sun","Mon","Tue","Wed","Thu","Fri","Sat"] onPick
  Row gap=4 between
    each label in labels
      Column center gap=2 grow onTap={day=index; onPick(index)}
        Text.caption "{label}"
        Box w=36 h=36 radius=18 fill={day==index ? "#FF2563EB" : "#00000000"} center middle
          Text "{index+13}" color={day==index ? "#FFFFFFFF" : "#FF111827"}
        Box w=6 h=6 radius=3 fill={contains(marks,index) ? "#FF16A34A" : "#00000000"}
