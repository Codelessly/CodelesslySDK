        Card gap=12
          Text.strong "Client"

          Row gap=8 scroll
            each c in filter(clients,c=>clientQ=="" || contains(lower(c),lower(clientQ)))
              ChoiceChip "{c}" selected={client==c} onTap={client=c}
          if client!=""
            Text.muted "Billing to {client}"
        Card gap=12
          Row between middle
            Text.strong "Line items"
            Button.outline "Add line" onTap={lines=append(lines,{"id":length(lines)+100+sum(lines,l=>l.id),"desc":"","qty":1,"rate":0,"tax":0})}
          each row in lines
            Row gap=8 middle top
              Column gap=0
                IconButton "keyboard_arrow_up" tooltip="Move up" onTap={index>0 ? lines=setAt(setAt(lines,index-1,row),index,lines[index-1]) : toast "Already first"}
                IconButton "keyboard_arrow_down" tooltip="Move down" onTap={index<length(lines)-1 ? lines=setAt(setAt(lines,index+1,row),index,lines[index+1]) : toast "Already last"}
              Input "Description" @row.desc grow
              Input "Qty" @row.qty type=number w=70

define StatusBadge status="draft"
  Badge.neutral "{status}" hidden={status!="draft"}
  Badge.info "{status}" hidden={status!="sent"}
  Badge.success "{status}" hidden={status!="paid"}
  Badge.error "{status}" hidden={status!="overdue"}
