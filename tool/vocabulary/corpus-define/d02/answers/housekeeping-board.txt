// A drag-and-tap tile of the floor grid, coloured by the state of the room.
define RoomTile "number" status="dirty" time="" early=false staff="" onTap
  Column gap=4 pad=12 radius=8 fill={status=="dirty" ? "#FFFCA5A5" : status=="progress" ? "#FFFDE68A" : status=="clean" ? "#FFA7F3D0" : status=="inspected" ? "#FF93C5FD" : "#FFD1D5DB"} onTap={onTap}
    Row between
      Text.strong "{number}" color=#FF111827
      if early
        Icon "priority_high" color=#FFB91C1C
    Text.caption "{status=="progress" ? "In progress" : status=="dnd" ? "Do not disturb" : upper(status)}" color=#FF111827
    Text.caption "{time == "" ? "-" : time}" color=#FF374151
    Text.caption "{staff == "" ? "Unassigned" : staff}" color=#FF374151

define StaffCard "name" assigned=0 cap=6 done=0 drop onDrop
  Card gap=8 drop=room onDrop={onDrop}
    Row middle gap=8
      Avatar "{name}" size=32
      Text.strong "{name}"
    Progress {assigned / cap}
    Text.caption "{assigned} of {cap} rooms assigned, {done} finished"

### Staff Load
App
  Sidebar
    Brand "Turndown" icon="bed"
    NavItem "Floor board" icon="grid_view" onTap={go floor-board}
    NavItem "Staff" icon="groups" selected={true}
    NavItem "Inspection" icon="fact_check" onTap={go inspection}
  Main
    Nav
      Heading "Staff" level=1
      Button "Clear all assignments" onTap={rooms=map(rooms,r=>with(r,"staff","")); toast "Assignments cleared"}
    Column pad=24 gap=16
      Row gap=24 top
        Column grow gap=12
          Text.strong "Housekeepers"
          each s in staff
            Card gap=8 drop=room onDrop={rooms=updateWhere(rooms,r=>r.id==result.data,r=>with(r,"staff",s.name))}
              Row between middle
                Row gap=8 middle
                  Avatar "{s.name}" size=36
                  Text.strong "{s.name}"
                Badge.neutral "{count(rooms, r=>r.staff==s.name)} / {s.cap}"
              Progress {count(rooms, r=>r.staff==s.name) / s.cap}
              Row gap=8 scroll
                each r in filter(rooms, x=>x.staff==s.name)
                  Chip "{r.id}{r.early ? " !" : ""}" drag=room(r.id)
              if count(rooms, r=>r.staff==s.name) > s.cap
                Text.danger "Over capacity"
        Column w=320 gap=12
          Text.strong "Unassigned"
          Card gap=8
            if count(rooms, r=>r.staff=="") == 0
              Text.muted "Everyone is covered."
            each r in filter(rooms, x=>x.staff=="")
              Row between middle
                Text "Room {r.id}"
                Badge.neutral "{r.status}"
                Chip "Drag" drag=room(r.id)

### Inspection
state active="202"
App
  Sidebar
    Brand "Turndown" icon="bed"
    NavItem "Floor board" icon="grid_view" onTap={go floor-board}
    NavItem "Staff" icon="groups" onTap={go staff-load}
    NavItem "Inspection" icon="fact_check" selected={true}
  Main
    Nav
      Heading "Inspector's checklist" level=1
      Row gap=8 middle
        Text.muted "Room"
        Select @active items=["203","208","210"]
    Column pad=24 gap=16 maxw=720
      Card gap=4
        Row between middle
          Text.strong "Room {active}"
          Badge.info "{count(checks, c=>c.result=="pass")} pass, {count(checks, c=>c.result=="fail")} fail"
        Progress {count(checks, c=>c.result!="") / length(checks)}
      each c in checks
        CheckRow "{c.label}" result={c.result} onPass={checks=updateWhere(checks,x=>x.id==c.id,x=>with(x,"result","pass"))} onFail={checks=updateWhere(checks,x=>x.id==c.id,x=>with(x,"result","fail"))}
      Row gap=8
        Button "Sign off room" onTap={count(checks, c=>c.result!="pass") > 0 ? toast "Every item must pass" : {rooms=updateWhere(rooms,r=>r.id==active,r=>with(with(r,"status","inspected"),"time",formatDate(today(),"h:mm a"))); checks=map(checks,c=>with(c,"result","")); toast "Room {active} inspected"}}
        Button.destructive "Send back dirty" onTap={count(checks, c=>c.result=="fail") == 0 ? toast "Mark at least one failure" : {rooms=updateWhere(rooms,r=>r.id==active,r=>with(r,"status","dirty")); checks=map(checks,c=>with(c,"result","")); toast "Room {active} returned to housekeeping"}}

define CheckRow "label" result="" onPass onFail
  Card
    Row between middle
      Text "{label}"
      Row gap=8
        Button.outline "Pass" selected={result=="pass"} onTap={onPass}
        Button.destructive "Fail" selected={result=="fail"} onTap={onFail}
