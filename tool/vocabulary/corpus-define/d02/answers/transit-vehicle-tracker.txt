// A real map is not available: drawn as a labelled box with a tram marker per tram.
define LiveMap lines=[] trams=[] selectedLine="" closedFrom="" closedTo="" h=360 onPickTram
  Stack h={h} fill=#FFE8EEF4 radius=8
    Text.muted "Map: line {selectedLine}"
    each t in trams
      Box w={t.line==selectedLine ? 64 : 48} h=28 radius=14 fill={t.line==selectedLine ? "#FF2563EB" : "#FF6B7280"} onTap={onPickTram(t.id)}
        Row gap=4 center middle
          Icon "navigation"
          Text.small "{t.id}" color=#FFFFFFFF
    if closedFrom!=""
      Badge.error "Closed {closedFrom} to {closedTo}"

define CrowdingMeter level=1 w=100
  Row gap=2 w={w} middle
    Icon "person" color={level>=1 ? "#FF16A34A" : "#FFD1D5DB"}
    Icon "person" color={level>=2 ? "#FFF59E0B" : "#FFD1D5DB"}
    Icon "person" color={level>=3 ? "#FFDC2626" : "#FFD1D5DB"}

define SortHeader "label" key="" w=100 sortKey="" sortDesc=false onSort
  Row gap=2 middle w={w} onTap={onSort({"key": key, "desc": sortKey==key ? !sortDesc : false})}
    Text.fieldLabel "{label}"
    Icon "{sortKey==key ? (sortDesc ? 'arrow_downward' : 'arrow_upward') : 'unfold_more'}"
