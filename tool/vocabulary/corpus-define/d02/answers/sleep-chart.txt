### Wind down
state tab="wind"
state secs=1800
state running=false
state warmth=0
every 1s when running onTick={secs=max(0,secs-1); warmth=round((1800-secs)/1800*100,0); running=secs>1}
App
  Sidebar
    Brand "Nightshade" icon="bedtime"
    NavItem "Last night" icon="nightlight" selected={tab=="night"} onTap={go last-night}
    NavItem "This week" icon="insights" selected={tab=="week"} onTap={go this-week}
    NavItem "Wind down" icon="self_improvement" selected={tab=="wind"} onTap={go wind-down}
    NavItem "Alarm" icon="alarm" selected={tab=="alarm"} onTap={go alarm}
  Main
    Nav
      Heading "Wind down" level=1
      Button.ghost "Add note" onTap={open note}
    Column pad=24 gap=16 center
      Box gap=16 pad=32 radius=16 center fill={warmth<34 ? "#FF1E293B" : (warmth<67 ? "#FF7C2D12" : "#FF1C0A04")}
        Text "Screen warmth {warmth}%" color=#FFFDE68A
        Text "{duration(secs)}" size=72 weight=700 color=#FFFFFFFF
        Text "until lights out at 11:10 pm" color=#FFFDE68A
        Row gap=8
          Button "Start" onTap={running=true}
          Button.outline "Pause" onTap={running=false}
          Button.ghost "Reset" onTap={running=false; secs=1800; warmth=0}
      Card gap=8
        Text.strong "Tonight's routine"
        Checkbox "Phone on the charger across the room" bind=c1
        Checkbox "Lights low, warm bulbs only" bind=c2
        Checkbox "No caffeine after 3 pm" bind=c3
        Checkbox "Stretch for five minutes" bind=c4
        Progress {count([c1,c2,c3,c4],x=>x)/4}
      Select "Length" items=["15 min","30 min","45 min"] onChange={secs=(result=="15 min" ? 900 : (result=="30 min" ? 1800 : 2700)); running=false}
  Dialog#note "Note for tonight" "Caffeine and screens"
    Toggle "Had caffeine after 3 pm" bind=lateCaffeine
    Toggle "Screens in the last hour" bind=lateScreens
    Button "Save note" onTap={emit saveNote; toast "Note saved"; close}

### Alarm
state tab="alarm"
state wake="6:30 am"
state window=20
state days=["Mon","Tue","Wed","Thu","Fri"]
state gentle=true
App
  Sidebar
    Brand "Nightshade" icon="bedtime"
    NavItem "Last night" icon="nightlight" selected={tab=="night"} onTap={go last-night}
    NavItem "This week" icon="insights" selected={tab=="week"} onTap={go this-week}
    NavItem "Wind down" icon="self_improvement" selected={tab=="wind"} onTap={go wind-down}
    NavItem "Alarm" icon="alarm" selected={tab=="alarm"} onTap={go alarm}
  Main
    Nav
      Heading "Alarm" level=1
      Button "Edit alarm" icon="edit" onTap={open alarmWindow}
    Column pad=24 gap=16
      Card gap=8 center
        Text.muted "Gentle wake"
        Text "{wake}" size=64 weight=700
        Text.muted "Wakes you in the lightest sleep between {wake} and {window} minutes later"
        Row gap=8
          each d in ["Mon","Tue","Wed","Thu","Fri","Sat","Sun"]
            FilterChip "{d}" selected={contains(days,d)} onTap={days=(contains(days,d) ? without(days,d) : append(days,d))}
      Card gap=8
        Row between
          Text.strong "Sunrise light and soft chime"
          Toggle bind=gentle
        Text.muted "Light builds over the last ten minutes of the window."
  Dialog#alarmWindow "Wake alarm" "Set the time and how long it can look for light sleep"
    TimePicker "Wake no later than" bind=wake
    Text.fieldLabel "Wake range: {window} minutes"
    Slider bind=window min=0 max=45
    Text.caption "Earliest wake {wake} minus {window} minutes"
    Row end gap=8
      Button.ghost "Cancel" onTap={close}
      Button "Save" onTap={emit saveAlarm; toast "Alarm set"; close}

define StageChart bind=at stages=[] h=140
  Row gap=0 bottom
    each st in stages
      Column w={st.m*1.6} h={h} bottom onTap={at=st.t+st.m/2}
        Box h={st.s=="awake" ? h : (st.s=="dream" ? h*0.72 : (st.s=="light" ? h*0.48 : h*0.24))} fill={st.s=="awake" ? "#FFF59E0B" : (st.s=="dream" ? "#FFA78BFA" : (st.s=="light" ? "#FF60A5FA" : "#FF1D4ED8"))} border={at>=st.t && at<st.t+st.m ? 2 : 0}
