### Timers
state tab="timers"
state running=true
state timers=[{"id":1,"name":"Fold 2","left":1500,"total":1800,"done":false},{"id":2,"name":"Bulk proof","left":9000,"total":10800,"done":false},{"id":3,"name":"Preheat oven","left":0,"total":2700,"done":true}]
state nextId=4
state label=""
state minutes=10
every 1s when running onTick={timers=map(timers,t=>t.done||t.left<=0 ? t : with(with(t,"left",t.left-1),"done",t.left-1<=0))}
App
  Sidebar
    Brand "Crumb" icon="bakery_dining"
    NavItem "Formula" icon="calculate" selected={tab=="formula"} onTap={go formula}
    NavItem "Schedule" icon="schedule" selected={tab=="schedule"} onTap={go schedule}
    NavItem "Timers" icon="timer" badge="{count(timers,t=>!t.done)}" selected={tab=="timers"} onTap={go timers}
    Spacer grow
    Text.muted "Country loaf, 2 loaves"
  Main
    Nav
      Heading "Timers" level=1
      Row gap=8
        Toggle "Running" bind=running
        Button "New timer" onTap={open newTimer}
    Column pad=24 gap=16 scroll
      if count(timers,t=>t.done)>0
        Card gap=8 fill=#FFFEF3C7
          Row between middle
            Row gap=8 middle
              Icon "notifications_active"
              Text.strong "{count(timers,t=>t.done)} ready: {join(map(filter(timers,t=>t.done),t=>t.name),', ')}"
            Button.outline "Dismiss all" onTap={timers=removeWhere(timers,t=>t.done)}
      Grid cols=3 gap=16
        each t in timers
          Card gap=12 center animate={t.done ? 'pulse' : ''}
            Text.strong "{t.name}"
            Stack center middle
              Ring {1-t.left/t.total}
              Text "{t.done ? 'Done' : duration(t.left)}" size=24 weight=700
            Row gap=8
              Button.outline "-5 min" onTap={timers=updateWhere(timers,x=>x.id==t.id,x=>with(x,"left",max(0,x.left-300)))}
              Button.outline "+5 min" onTap={timers=updateWhere(timers,x=>x.id==t.id,x=>with(with(x,"left",x.left+300),"done",false))}
              IconButton "close" tooltip="Remove" onTap={timers=removeWhere(timers,x=>x.id==t.id)}
      if isEmpty(timers)
        Text.muted "No timers yet. Start one from the schedule."
  Dialog#newTimer "New timer" "Rings when the time is up"
    Input "What for" "Fold 3" bind=label
    Slider bind=minutes min=1 max=240
    Text.muted "{minutes} minutes"
    Row gap=8 end
      Button.outline "Cancel" onTap={close}
      Button "Start" onTap={timers=append(timers,{"id":nextId,"name":coalesce(label,"Timer"),"left":minutes*60,"total":minutes*60,"done":false}); nextId=nextId+1; label=""; close}
