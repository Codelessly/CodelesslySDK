// A coloured read-only line of the source text, with term parts highlighted.
define SourceText parts=[]
  Row wrap gap=0
    each p in parts
      if p.term
        Text "{p.t}" fill=#FFFEF3C7 color=#FF92400E weight=700 tooltip="{p.term}"
      if !p.term
        Text "{p.t}"

define StatusChip "status"="new"
  if status=="new"
    Badge.neutral "New"
  if status=="draft"
    Badge.warning "Draft"
  if status=="approved"
    Badge.success "Approved"

define MemoryMatch pct=100 source=[] target="" onInsert
  Card gap=8
    Row between middle
      Badge.info "{pct}% match"
      Button.outline "Insert" icon="subdirectory_arrow_left" onTap={onInsert}
    Row wrap gap=0
      each p in source
        Text "{p.t}" color={coalesce(p.c,"#FF111827")} weight={p.c ? 700 : 400}
    Text.muted "{target}"
