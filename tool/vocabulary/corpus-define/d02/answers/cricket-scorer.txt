// The six circles of the over: each fills with its runs, a W or an extra mark
define OverTracker balls=[]
  Row gap=10 middle
    each i in range(6)
      Box w=44 h=44 radius=22 center middle fill={i<length(balls) ? (balls[i].w ? #FFDC2626 : (balls[i].r>=4 ? #FF16A34A : (balls[i].k!="run" ? #FFF59E0B : #FF1F2937))) : #FFE5E7EB}
        Text "{i<length(balls) ? (balls[i].w ? "W" : (balls[i].k=="wide" ? "Wd" : (balls[i].k=="noball" ? "Nb" : (balls[i].k=="bye" ? "B" : balls[i].r)))) : ""}" color=#FFFFFFFF weight=700

define RunPad onRun onExtra onWicket
  Column gap=12
    Row gap=12
      each n in range(7)
        Button.elevated "{n}" size=large grow border={n>=4 ? 2 : 0} onTap={onRun(n)}
    Row gap=12
      Button.outline "Wide" grow onTap={onExtra("wide")}
      Button.outline "No ball" grow onTap={onExtra("noball")}
      Button.outline "Bye" grow onTap={onExtra("bye")}
      Button.destructive "Wicket" grow onTap={onWicket}

define PlayerRow "name" runs=0 balls=0 strike=false
  Row between middle
    Row gap=8 middle
      Icon "sports_cricket" visible={strike}
      Text.strong "{name}"
    Row gap=16
      Text.strong "{runs}"
      Text.muted "({balls})"
      Text.muted "SR {fixed(runs*100/max(1,balls),1)}"

define PartnershipBar a="Batter A" b="Batter B" aRuns=0 bRuns=0
  Column gap=6
    Progress {aRuns/max(1,aRuns+bRuns)}
    Row between
      Text.small "{a} {aRuns}"
      Text.small "{b} {bRuns}"
