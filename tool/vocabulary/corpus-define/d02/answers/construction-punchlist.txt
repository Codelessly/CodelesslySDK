  Stack h=360 fill=#FFF8FAFC radius=8 border=1 onTap={onPlace(result)}
    Box x=20 y=20 w=260 h=140 border=2
      Text.muted "Living"
    Box x=280 y=20 w=200 h=140 border=2
      Text.muted "Bedroom 1"
    Box x=480 y=20 w=140 h=140 border=2
      Text.muted "Bedroom 2"
    Box x=20 y=160 w=600 h=40 border=2
      Text.muted "Corridor"
    Box x=20 y=200 w=300 h=140 border=2
      Text.muted "Kitchen"
    Box x=320 y=200 w=300 h=140 border=2
      Text.muted "Bathroom"
    each pin in pins
      Box x={pin.x} y={pin.y} w=28 h=28 radius=14 fill={pin.status=="Fixed" ? "#FF16A34A" : "#FFDC2626"} center middle border={selected==index ? 3 : 0} onTap={selected=index; onPick(index)}
        Text "{pin.id}" color=#FFFFFFFF weight=700

// Shows a photo; drawing arrows on it is stored as a list of strokes.
define PhotoMarkup src="" @marks tool="arrow" color=#FFDC2626 h=200
  state marks=[]
  Stack h={h} radius=8 fill=#FFE5E7EB drop=none onTap={marks=append(marks,{"tool":tool,"at":result})}
    Image "{src}" h={h}
    each m in marks
      Icon "north_east" color={color} x={m.at.x} y={m.at.y}

// Two photos split by a draggable handle.
define BeforeAfter before="" after="" h=240
  state at=50
  Stack h={h} radius=8
    Image "{after}" h={h}
    Box w="{at}%" h={h}
      Image "{before}" h={h}
    Slider @at min=0 max=100
    Row between
      Badge.neutral "Before"
      Badge.success "After"
