// MonthCalendar draws a month as a grid of day cells coloured by status.
define MonthCalendar days=[] @day firstWeekday="Mon" onPick
  state day=1
  Column gap=8
    Grid cols=7 gap=6
      each w in ["Sun","Mon","Tue","Wed","Thu","Fri","Sat"]
        Text.caption "{w}" center
      each c in days
        Box h=44 radius=8 fill={c.s=="taken" ? #FFDCFCE7 : c.s=="missed" ? #FFFEE2E2 : #FFFEF3C7} border={day==c.d ? 2 : 0} center middle onTap={day=c.d; onPick(c.d)}
          Text "{c.d}"

define Banner "title" "text" icon="info" action="" onAction onDismiss
  Row gap=12 pad=12 radius=8 fill=#FFFEF3C7 middle
    Icon "{icon}"
    Column grow gap=2
      Text.strong "{title}"
      Text.small "{text}"
    Button.outline "{action}" hidden={action==""} onTap={onAction}
    IconButton "close" tooltip="Dismiss" onTap={onDismiss}

define DoseCard "name" detail="" time="" status="due" late=false onTake onSkip onSnooze
  Card gap=8 fill={late ? #FFFEE2E2 : #FFFFFFFF}
    Row between middle
      Text.strong "{name}"
      Badge.error "Late" hidden={!late}
      Badge.success "Taken" hidden={status!="taken"}
      Badge.neutral "Skipped" hidden={status!="skipped"}
    Text.muted "{detail}"
    Row gap=6 middle
      Icon "schedule"
      Text.danger "{time}" hidden={!late}
      Text.small "{time}" hidden={late}
    if status=="due"
      Row gap=8
        Button "Take" onTap={onTake}
        Button.outline "Snooze" onTap={onSnooze}
        Button.ghost "Skip" onTap={onSkip}

define Refill left=0 total=1 perDay=1 onRefill
  Column gap=6
    Row between
      Text.strong "{floor(left/perDay)} days until refill"
      Text.muted "{left} pills left"
    Progress {left/total}
    Button.outline "Request refill" onTap={onRefill}
