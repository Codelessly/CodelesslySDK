state sig=""
state code=""

state stops=[]

  Nav
    Brand "Dropoff" icon="local_shipping"
    Row gap=4
      Link "Route" onTap={go route}
      Link "Day summary" onTap={go summary}
  Section narrow
    Column gap=16
      Heading "Proof of delivery" level=1
      Text.muted "Harlow Bakery, 14 Mill Lane. 2 parcels."
      SegmentedButton @method items=["photo","signature","code"]
      if method=="photo"
        Card gap=8
          Text.strong "Photo of the parcel at the door"
          CameraCapture @photo h=240
          if photo!=""
            Badge.success "Photo attached"
      if method=="signature"
        Card gap=8
          Text.strong "Recipient signs below"
          SignaturePad @sig h=180
          Button.ghost "Clear" onTap={sig=""}
      if method=="code"
        Card gap=8
          Text.strong "Ask the customer for the 4-digit code"
          Input "Delivery code" type=number @code
          if code!="" && length(code)==4 && code!=expected
            Text.danger "That code does not match."
      Row gap=8
        Button.outline "Back" grow onTap={go route}
        Button "Confirm delivery" grow onTap={(method=="photo" && photo=="") || (method=="signature" && sig=="") || (method=="code" && code!=expected) ? toast "Proof is missing" : {emit delivered; toast "Delivered"; go route}}
