define PanZoom bind=zoom h=400
  state zoom=1
  Box h={h} fill=#FFF8FAFC radius=8 border=1
    Text.caption "Canvas, zoom {zoom}"
    ...

define PersonCard name="" born="" died="" photo="" notes=false selected=false onTap onNotes onAddParent onAddPartner onAddChild
  Column center gap=4
    IconButton "add" tooltip="Add parent" onTap={onAddParent}
    Row middle gap=4
      IconButton "add" tooltip="Add partner" onTap={onAddPartner}
      Card w={160} gap=6 center border={selected ? 2 : 1} onTap={onTap}
        Avatar src="{photo}" size=56
        Text.strong "{name}" center
        Text.muted "{born}{died != "" ? " - " + died : ""}" center
        if notes
          IconButton "sticky_note_2" tooltip="Notes" onTap={onNotes}
      IconButton "add" tooltip="Add partner" onTap={onAddPartner}
    IconButton "add" tooltip="Add child" onTap={onAddChild}
