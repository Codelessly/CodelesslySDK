define MonthTile name=0 year=2026 tasks=[] selected=false onPick
  Card gap=6 pad=10 fill={selected ? "#FFE0ECFF" : "#FFFFFFFF"} onTap={onPick}
    Text.strong "{formatDate(name,"MMM")}"
    Row gap=4
      each t in tasks
        Box w=10 h=10 radius=5 fill={t.done ? "#FF16A34A" : (t.days<0 ? "#FFDC2626" : "#FF2563EB")}

define TaskRow task={} onDone onCall
  Card gap=6
    Row between middle
      Column gap=2
        Text.strong "{task.name}"
        Text.muted "Every {task.every} {task.unit}, last done {task.last == "" ? "never" : formatDate(task.last,"MMM d")}"
      Badge.neutral "{task.done ? "done" : (task.days<0 ? "overdue " + abs(task.days) + "d" : "in " + task.days + " days")}"
    Row between middle
      Text.muted "Call: {task.who}"
      Row gap=8
        Button.ghost "Contact" icon="call" onTap={onCall}
        Button "Done" icon="check" onTap={onDone}

define Rating value=0 readonly
  Row gap=2
    each i in range(5)
      Icon {i<value ? "star" : "star_border"}

// shows the rooms as labelled boxes; a real drawn plan is not possible
define FloorPlan @room rooms=[] pins=[] onPick
  Card gap=8
    Grid cols=10 gap=2
      each r in rooms
        Box pad=8 fill={room==r.name ? "#FFDBEAFE" : "#FFF3F4F6"} radius=4 border=1 onTap={room=r.name; onPick(r.name)}
          Text.strong "{r.name}"
          Row gap=4
            each p in filter(pins,p=>p.room==r.name)
              Box w=10 h=10 radius=5 fill={p.urgent ? "#FFDC2626" : "#FF2563EB"}
