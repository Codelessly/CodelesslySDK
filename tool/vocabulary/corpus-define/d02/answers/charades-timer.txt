### Play
state left=60
state running=false
state revealed=false
state idx=0
state team=0
state streak=0
state best=0
state scores=[0,0]
state teams=[{"name":"Red Foxes"},{"name":"Blue Herons"}]
state pool=[{"word":"Titanic","cat":"film","d":"Easy"}]
state played=[]
every 1s when running onTick={left=left-1; left<=1 ? {running=false; open timeup} : {}}
App
  Main
    Nav
      Button.ghost "Quit" icon="arrow_back" onTap={running=false; go setup}
      Text.strong "{teams[team].name}"
      Row gap=8
        Badge.warning "Streak {streak}"
        IconButton "help_outline" tooltip="Rules" onTap={open rules}
    Column pad=24 gap=20 center maxw=720
      Stack center middle
        if left>10
          Ring {left/60} size=160
        if left<=10
          Ring {left/60} size=160 color=#FFDC2626 animate=pulse
        Text "{left}" size=56 weight=700 color={left<=10 ? "#FFDC2626" : "#FF111827"}
      Text.muted "seconds left"
      HoldReveal word={pool[idx % length(pool)].word} category={pool[idx % length(pool)].cat} @revealed
      Row gap=12 center
        Button.outline "Pass" icon="skip_next" size=large disabled={!running} onTap={played=append(played,{"word":pool[idx % length(pool)].word,"ok":false}); streak=0; idx=idx+1; revealed=false}
        Button "Got it" icon="check" size=large disabled={!running} onTap={scores=setAt(scores,team,scores[team]+1); streak=streak+1; best=max(best,streak+1); played=append(played,{"word":pool[idx % length(pool)].word,"ok":true}); idx=idx+1; revealed=false; toast "+1 for {teams[team].name}"}
      if !running
        Button "Start the clock" icon="play_arrow" size=large onTap={running=true}
      if running
        Button.ghost "Pause" icon="pause" onTap={running=false}
      TiltGesture "Tilt the phone down for Got it" disabled={!running} onTilt={scores=setAt(scores,team,scores[team]+1); streak=streak+1; played=append(played,{"word":pool[idx % length(pool)].word,"ok":true}); idx=idx+1; revealed=false; toast "Tilt! +1"}
    Footer
      Scoreboard teams={teams} scores={scores} active={team}
Dialog#timeup "Time is up"
  Buzzer sound="horn" playing={left<=1} label="Buzzer"
  Column gap=8 center
    Text.heading "Round over"
    Text.muted "{count(played,p=>p.ok)} words guessed by {teams[team].name}"
    Button "See the summary" onTap={close; go summary}

// A card that hides its word until the player presses and holds; the hold itself (press and release events) cannot be drawn, so it shows the word on a tap toggle.
define HoldReveal word="" category="film" @revealed
  state revealed=false
  Card gap=8 center fill={revealed ? "#FFFEF3C7" : "#FF1F2937"} h=180 onTap={revealed=!revealed}
    Badge.neutral "{upper(category)}"
    if revealed
      Text.title "{word}" size=32 weight=700 center color=#FF111827
    if !revealed
      Icon "visibility_off"
      Text.muted "Press and hold to reveal" color=#FFE5E7EB

// A device tilt sensor cannot be drawn here, so it shows as a labelled box; a tap stands in for the tilt.
define TiltGesture "label" disabled=false onTilt
  Box pad=12 radius=8 fill=#FFEEF2FF center onTap={disabled ? {} : onTilt}
    Row gap=8 middle
      Icon "screen_rotation"
      Text.small "{label}"

// A sound cannot be played here, so it shows as a labelled box that lights up while it plays.
define Buzzer sound="horn" playing=false label="Buzzer"
  Box pad=8 radius=8 fill={playing ? "#FFFEE2E2" : "#FFF3F4F6"}
    Row gap=8 middle
      Icon "volume_up"
      Text.small "{label}: {sound}"

define Scoreboard teams=[] scores=[] active=0
  Column gap=8
    each t in teams
      Row gap=12 middle between border-bottom=#FFE5E7EB
        Row gap=8 middle
          Avatar "{t.name}" size=32
          Text.strong "{t.name}"
          Badge.primary "Up" hidden={index != active}
        Text.heading "{scores[index]}"
