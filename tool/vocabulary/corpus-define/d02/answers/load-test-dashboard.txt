// A draggable line editor: shown as a line chart with a drag handle for each point.
define RampEditor @points maxUsers=2000 unit="min" h=240 onChange
  Column gap=8
    LineChart values={map(points,p=>p.users)} labels={map(points,p=>p.min)} h={h}
    Row gap=8 between top
      each p in points
        Column grow gap=4 center
          Icon "drag_indicator"
          Slider @p.users min=0 max={maxUsers} drag=ramp(p.min)
          Text.small "{p.users}"
          Text.caption "{p.min} {unit}"
