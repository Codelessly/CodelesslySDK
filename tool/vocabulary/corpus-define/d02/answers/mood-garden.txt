define PinPad bind=pin length=4 onComplete
  state pin=""
  Column center gap=16
    Row gap=12
      each i in range(length)
        Box w=14 h=14 radius=7 fill={i < length(pin) ? "#FF2563EB" : "#FFE2E8F0"}
    Grid cols=3 gap=10 w=240
      each d in ["1","2","3","4","5","6","7","8","9","","0","<"]
        Button.outline "{d}" hidden={d==""} onTap={d=="<" ? pin=slice(pin,0,length(pin)-1) : {pin=pin+d; length(pin)+1==length ? onComplete(pin+d) : ""}}

define MoodWheel bind=mood size=240 onPick
  state mood=""
  // a true colour wheel is drawn as a ring of tappable swatches
  Grid cols=4 gap=8 w={size}
    each m in [{"mood":"joyful","color":"#FFFACC15"},{"mood":"excited","color":"#FFF97316"},{"mood":"angry","color":"#FFDC2626"},{"mood":"anxious","color":"#FFEC4899"},{"mood":"sad","color":"#FF6366F1"},{"mood":"calm","color":"#FF60A5FA"},{"mood":"tired","color":"#FF94A3B8"},{"mood":"content","color":"#FF4ADE80"}]
      Box w={size/4-8} h={size/4-8} radius={size/8} fill={m.color} border={mood==m.mood ? 3 : 0} onTap={mood=m.mood; onPick(m)}

define PlantGlyph mood="" color=#FF4ADE80 size=64
  Stack w={size} h={size} center middle
    Box w={size/8} h={size/2} radius={size/16} fill=#FF3F6212
    Box w={size*0.7} h={size*0.7} radius={mood=="sharp" || mood=="angry" ? 4 : size/2} fill={color}

define GardenPlot entries=[] days=30 bind=sel onPick
  state sel=null
  Row scroll gap=10 pad=12 fill=#FF5B3A29 radius=12
    each d in range(days)
      Column center gap=4 w=72 h=120 end
        if any(entries,e=>e.day==d+1)
          PlantGlyph mood={find(entries,e=>e.day==d+1).mood} color={find(entries,e=>e.day==d+1).color} size=56 onTap={onPick(find(entries,e=>e.day==d+1))}
        Text.small "{d+1}" color=#FFE7D5C3

define SeasonGraph months=[] values=[] h=240
  Column gap=8
    LineChart values={values} labels={months} h={h}
    Row between
      Text.caption "low"
      Text.caption "high"
