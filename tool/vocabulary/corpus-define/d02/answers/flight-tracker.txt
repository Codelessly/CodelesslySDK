state mode="Flight"
state recent=[{"id":"BA117","route":"LHR to JFK","status":"On time","dep":"09:40","arr":"12:35"},{"id":"DL404","route":"JFK to LAX","status":"Delayed","dep":"14:10","arr":"17:45"},{"id":"LH401","route":"FRA to EWR","status":"Landed","dep":"07:55","arr":"10:20"},{"id":"AF11","route":"CDG to JFK","status":"Boarding","dep":"11:30","arr":"14:05"}]

  Sidebar

    NavItem "Search" icon="search" selected onTap={go search}
    NavItem "My flight" icon="flight_takeoff" onTap={go flight}
    NavItem "Departures" icon="departure_board" onTap={go departures}
    NavItem "Alerts" icon="notifications" badge="2" onTap={go alerts}
    Spacer grow
    Text.muted "Signed in as Dana Kim"
  Main
    Nav
      Heading "Find a flight" level=1
      Avatar "Dana Kim" size=36
    Column pad=24 gap=16 maxw=720
      SegmentedButton @mode items=["Flight","Route"]
      if mode=="Flight"
        Input "Flight number" "e.g. BA117" @q
      if mode=="Route"
        Row gap=12
          Input "From" "LHR" grow
          Icon "swap_horiz"
          Input "To" "JFK" grow
      Button "Track" onTap={go flight}
      Text.sectionTitle "Recent"
      each f in filter(recent, r => q=="" || contains(lower(r.id), lower(q)) || contains(lower(r.route), lower(q)))
        Card gap=8 onTap={go flight}
          Row between middle
            Column gap=2
              Text.strong "{f.id}"
              Text.muted "{f.route}"
            Badge.info "{f.status}" visible={f.status=="Boarding"}
            Badge.warning "{f.status}" visible={f.status=="Delayed"}
            Badge.success "{f.status}" visible={f.status=="On time" || f.status=="Landed"}
          Row between
            Text.caption "Departs {f.dep}"
            Text.caption "Arrives {f.arr}"

### Flight

state progress=0.58
state tracking=true
every 5s when tracking onTick={progress=min(1,progress+0.01)}
App
  Sidebar
    Brand "Contrail" icon="flight"
    NavItem "Search" icon="search" onTap={go search}
    NavItem "My flight" icon="flight_takeoff" selected onTap={go flight}
    NavItem "Departures" icon="departure_board" onTap={go departures}
    NavItem "Alerts" icon="notifications" badge="2" onTap={go alerts}
  Main
    Nav
      Row gap=8 middle
        IconButton "arrow_back" tooltip="Back to search" onTap={go search}
        Heading "BA117  London to New York" level=1
      Row gap=8
        Toggle "Follow" @tracking
        IconButton "more_vert" tooltip="More" onTap={open flightMenu}
    Column pad=24 gap=20 scroll
      Row between middle
        Column gap=2
          Text.caption "LHR  Heathrow, Terminal 5"
          Row gap=12 middle
            Text "{delayed ? '10:05' : '09:40'}" size=56 weight=700
            Text.muted "09:40" size=24 strike visible={delayed}
        Icon "flight" size=32
        Column gap=2 end
          Text.caption "JFK  Kennedy, Terminal 7"
          Row gap=12 middle
            Text "{delayed ? '13:00' : '12:35'}" size=56 weight=700
            Text.muted "12:35" size=24 strike visible={delayed}
      if delayed
        Badge.warning "Delayed 25 min: late inbound aircraft"
      FlightProgress value={progress} from="LHR" to="JFK" label="{round(progress*100,0)}% flown"
      Row gap=12
        ChangeChip "Gate" value={gate} changed={gateChanged}
        ChangeChip "Terminal" value={terminal} changed={terminalChanged}
        ChangeChip "Belt" value={belt} changed={beltChanged}
      Card gap=8
        Row between middle
          Text.panelTitle "Route"
          Text.muted "5,540 km  7h 25m"
        FlightMap from="LHR" to="JFK" progress={progress} h=260
      Grid cols=3 gap=12
        Card gap=4
          Text.caption "Altitude"
          Text.strong "37,000 ft"
        Card gap=4
          Text.caption "Ground speed"
          Text.strong "902 km/h"
        Card gap=4
          Text.caption "Aircraft"
          Text.strong "Boeing 787-9"
      Row gap=8
        Button.outline "Simulate gate change" onTap={gate="B24"; gateChanged=true; toast "Gate changed to B24"}
        Button.outline "Clear flashes" onTap={gateChanged=false; terminalChanged=false; beltChanged=false}
        Button "Set alert" onTap={go alerts}
  Menu#flightMenu end
    MenuItem "Share flight" icon="share" onTap={toast "Link copied"}
    MenuItem "Add to calendar" icon="event" onTap={toast "Added to calendar"}
    MenuItem "Stop following" icon="notifications_off" onTap={tracking=false}

define FlightProgress value=0.5 from="" to="" label=""
  Column gap=6
    Stack h=32
      Progress {value}
      Row w=fill middle start pad=[0,0,0,0]
        Spacer w={value*100}
        Icon "flight" size=28
    Row between
      Text.caption "{from}"
      Text.caption "{label}"
      Text.caption "{to}"

define ChangeChip "label" value="" changed=false
  Chip gap=6 animate={changed ? "pulse" : "none"} duration=600
    Text.caption "{label}"
    Text.strong "{value}"
    if changed
      Badge.warning "changed"

// A real map is shown as a labelled box with the route arc drawn over it.
define FlightMap from="" to="" progress=0.5 h=240
  Stack h={h} radius=8 fill=#FFE8F0F8
    Text.caption "Map: {from} to {to}"
    Row between middle pad=16
      Column center gap=2
        Icon "trip_origin"
        Text.small "{from}"
      Sparkline values=[0,3,5,6,5,3,0]
      Column center gap=2
        Icon "place"
        Text.small "{to}"
