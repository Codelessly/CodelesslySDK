### Question reader
state qi=0
state questions=[{"round":"Geography","q":"Which river flows through both Vienna and Budapest?","a":"The Danube"},{"round":"Geography","q":"What is the only country in the world with a non-rectangular national flag?","a":"Nepal"},{"round":"Music","q":"Which band released the 1975 album Wish You Were Here?","a":"Pink Floyd"}]
state left=10
state running=true
state revealed=false
state clipPlaying=false
every 1s when running onTick={left=left-1; left<=1 ? {running=false; revealed=true} : toast ""}
App
  Sidebar
    Brand "Round Table" icon="quiz"
    NavItem "Scoresheet" icon="grid_on" onTap={go scoresheet}
    NavItem "Question reader" icon="campaign" selected={true}
    NavItem "Leaderboard" icon="leaderboard" onTap={go leaderboard}
    Spacer grow
    Text.muted "Question {qi+1} of {length(questions)}"
  Main
    Nav
      Heading "{questions[qi].round}" level=1
      Row gap=8
        Button.outline "Previous" icon="chevron_left" onTap={qi=max(0,qi-1); left=10; running=true; revealed=false}
        Button "Next question" icon="chevron_right" onTap={qi=min(length(questions)-1,qi+1); left=10; running=true; revealed=false}
    Column pad=24 gap=24 center
      Progress {left/10}
      Text "{questions[qi].q}" size=44 weight=700 center
      if questions[qi].round=="Music"
        MusicClip "Mystery track" title={questions[qi].a} hideTitle={!revealed} playing={clipPlaying} onPlay={clipPlaying=true} onPause={clipPlaying=false}
      Row gap=16 middle center
        Ring {left/10}
        Text.strong "{max(left,0)}s to answer" size=24
      if revealed
        Card gap=8 center animate=pop
          Text.muted "The answer"
          Text "{questions[qi].a}" size=40 weight=700 color=#FF16A34A
      if !revealed
        Button.outline "Reveal now" onTap={revealed=true; running=false}
    Footer
      Row between pad=12
        Text.muted "Pens down when the timer ends"
        Button.ghost "Restart timer" icon="replay" onTap={left=10; running=true; revealed=false}

### Leaderboard
state standings=[{"name":"Quizzy Rascals","points":38},{"name":"The Cluedo Crew","points":39},{"name":"Pie Hards","points":44},{"name":"Les Quizerables","points":31}]
state shown=2
every 4s onTick={standings=updateWhere(standings,s=>s.name=="Quizzy Rascals",s=>with(s,"points",s.points+(shown==2?7:0))); shown=shown+1}
Page
  Nav
    Brand "Round Table" icon="quiz"
    Row gap=4
      Link "Scoresheet" onTap={go scoresheet}
      Link "Question reader" onTap={go reader}
    Badge.primary "After round {shown}"
  Section center
    Heading "Leaderboard" level=1
    Text.muted "Standings as the rounds are scored"
  Section narrow
    RankBoard items={standings} labelKey="name" valueKey="points" unit="pts"

// The fixed rows of a ranking: a name and a value, drawn from the highest down
define ScoreCell value=0 joker=false onChange
  state v=value
  Input type=number @v w=70 center onChange={onChange(result)}

// A mystery clip: the player cannot play real sound here, so it is a labelled box that hides the title until revealed
define MusicClip "label" title="" hideTitle=true playing=false onPlay onPause
  Card gap=12 pad=16
    Row gap=12 middle
      IconButton {playing ? "pause_circle" : "play_circle"} tooltip="Play" onTap={playing ? onPause : onPlay}
      Column grow gap=4
        Text.strong "{label}"
        Text.muted "{hideTitle ? "Title hidden" : title}"
      Icon "graphic_eq"
    Progress {playing ? 0.5 : 0}

// A ranked list sorted by value with each row sliding to its new place
define RankBoard items=[] labelKey="name" valueKey="value" unit=""
  Column gap=8
    each row in sortByDesc(items,x=>x[valueKey])
      Card pad=16 animate=slide-up stagger=80 fill={index==0 ? "#FFFEF3C7" : "#FFFFFFFF"}
        Row gap=16 middle
          Text.strong "{index+1}" size=28 w=40
          Text.strong "{row[labelKey]}" size=24
          Spacer grow
          Text.strong "{row[valueKey]} {unit}" size=28
