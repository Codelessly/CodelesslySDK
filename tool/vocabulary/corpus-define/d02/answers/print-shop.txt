// shows a framed picture with a ratio frame; the frame drags and the zoom scales the picture, drawn here as a stack with sliders for the offset
define CropFrame src="" ratio="3:2" zoom=1 x=@0.5 y=@0.5
  state fx=0.5
  state fy=0.5
  Column gap=8
    Stack h=280 radius=6 fill=#FF111827
      Image "{src}" h={280*zoom}
      Box w={ratio=="1:1" ? 200 : 240} h={ratio=="2:3" ? 260 : 170} border=2 drag=frame(1)
    Row gap=8 middle
      Text.caption "Left to right"
      Slider @fx min=0 max=1 grow onChange={x=result}
    Row gap=8 middle
      Text.caption "Top to bottom"
      Slider @fy min=0 max=1 grow onChange={y=result}

define Stepper @value min=0 max=99 onChange
  Row gap=4 middle
    IconButton "remove" tooltip="Fewer" onTap={value=max(min,value-1); onChange(value)}
    Text.strong "{value}" w=32 center
    IconButton "add" tooltip="More" onTap={value=min(max,value+1); onChange(value)}

define TierTable tiers=[] count=0

        Badge.success "{t.off*100}% off" selected={count>=t.min}
