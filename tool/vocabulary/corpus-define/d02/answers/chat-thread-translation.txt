### Glossary
state gl=["Bridge","Okonomiyaki","Abuela Rosa","Shibuya"]
state newTerm=""
state note=""
App
  Sidebar
    Brand "Bridge" icon="translate"
    NavItem "Family trip" icon="groups" onTap={go thread}
    NavItem "Work lunch" icon="restaurant" onTap={go thread}
    NavItem "Glossary" icon="menu_book" selected={true}
  Main
    Nav
      Heading "Glossary" level=3
      Button "Add a name" onTap={open addTerm}
    Column pad=24 gap=16
      Text.muted "Names and words here are never translated, in any language."
      Card gap=8
        each g in gl
          Row between middle
            Column
              Text.strong "{g}"
              Text.caption "Stays as written for JP, ES and EN"
            IconButton "delete" tooltip="Remove" onTap={gl=without(gl,g)}
        Text.muted "Nothing here yet" hidden={length(gl)>0}
      Card gap=8
        Text.strong "Try it"
        Input "Type a sentence" @note
        if note != ""
          Row gap=8 middle
            LangFlag "JP"
            Text "{note}"
          Text.caption "Words in the glossary are kept: {join(filter(gl, g=>contains(note,g)), ', ')}"
  Dialog#addTerm "Add a name" "It will stay as written everywhere"
    Input "Name or word" @newTerm
    Button "Add" onTap={newTerm=="" ? toast "Write the name" : {gl=append(gl,newTerm); newTerm=""; close}}

### Voice note
state playing=false
state pos=0
state lang="EN"
every 1s when playing onTick={pos=pos>=14 ? 0 : pos+1; playing=pos>=14 ? false : playing}
Page
  Nav
    Brand "Bridge" icon="translate"
    Link "Back to thread" onTap={go thread}
  Section narrow
    Column gap=16
      Row gap=8 middle
        Avatar "Kenji Mori" size=40
        Column
          Text.strong "Kenji"
          Row gap=4 middle
            LangFlag "JP"
            Text.caption "Voice note, 0:14"
      Card gap=12
        Row gap=12 middle
          IconButton {playing ? "pause" : "play_arrow"} tooltip="Play" onTap={playing=!playing}
          Waveform values=[3,5,8,6,9,4,7,10,6,3,5,8,4,2] progress={pos/14} grow
          Text.caption "{duration(pos)} / {duration(14)}"
        SegmentedButton @lang items=["EN","JP"]
        Text "There is a small place near the station, they do not take reservations, so we should go early." hidden={lang!="EN"}
        Text "駅の近くに小さなお店があります。予約はできないので、早めに行きましょう。" hidden={lang!="JP"}
        Row gap=8
          Button.outline "Copy transcript" onTap={copy "transcript"; toast "Copied"}
          Button.outline "Reply" onTap={go thread}

// A small round language mark: a flag-style code on a tinted dot.
define LangFlag "code" size=18
  Box w={size+10} h={size} radius={size/2} fill="#FFE5E7EB" center middle
    Text.caption "{code}"

define TypingIndicator "who" onTap
  Row gap=6 middle animate=pulse
    Avatar "{who}" size=20
    Text.caption "{who} is typing"
    Row gap=3
      Box w=5 h=5 radius=3 fill="#FF9CA3AF"
      Box w=5 h=5 radius=3 fill="#FF9CA3AF"
      Box w=5 h=5 radius=3 fill="#FF9CA3AF"

define ReplyQuote sender="" text="" onJump
  Row gap=8 border-left="#FF2563EB" pad=[8,4,8,4] onTap={onJump}
    Column
      Text.caption "{sender}"
      Text.small "{text}"

define ChatBubble "text" sender="" lang="" original="" mine=false showOriginal=false onTap onLongPress
  Row {mine ? "end" : "start"} gap=8 top
    if !mine
      Avatar "{sender}" size=32
    Column gap=4 pad=12 radius=14 fill={mine ? "#FFDBEAFE" : "#FFF3F4F6"} onTap={onTap} onLongPress={onLongPress}
      Row gap=6 middle
        Text.strong "{sender}"
        LangFlag "{lang}"
      Text "{text}"
      if showOriginal
        Divider
        Text.muted "{original}"
      ...

define VoiceNote sender="" lang="" seconds=0 transcripts=[] onLongPress
  Row start gap=8 top
    Avatar "{sender}" size=32
    Card gap=8 w=320 onLongPress={onLongPress}
      Row gap=6 middle
        Text.strong "{sender}"
        LangFlag "{lang}"
      Row gap=8 middle
        Icon "play_arrow"
        Waveform values=[3,5,8,6,9,4,7,10,6,3,5,8] grow
        Text.caption "{duration(seconds)}"
      each t in transcripts
        Row gap=6 top
          LangFlag "{t.lang}"
          Text.small "{t.text}"

// Sound is not available, so this is a row of bars.
define Waveform values=[] progress=0 barW=3
  Row gap=2 middle h=32
    each v in values
      Box w={barW} h={v*3} radius=2 fill={index/length(values) < progress ? "#FF2563EB" : "#FF9CA3AF"}
