// A step indicator: a numbered list of steps with done marks and the current one lit; a tap goes to that step.
define Stepper steps=[] @step done=[] orientation="horizontal" onPick
  Column gap=4
    each s in steps
      Row gap=10 middle pad=[8,6,8,6] radius=8 fill={index == step ? "#FFEFF6FF" : "#00000000"} onTap={step=index; onPick(index)}
        Box w=24 h=24 radius=12 center middle fill={contains(done,index) ? "#FF16A34A" : (index == step ? "#FF2563EB" : "#FFE5E7EB")}
          if contains(done,index)
            Icon "check" size=14 color=#FFFFFFFF
          if !contains(done,index)
            Text "{index + 1}" size=12 color={index == step ? "#FFFFFFFF" : "#FF374151"}
        Text.small "{s}" weight={index == step ? 700 : 400}

define YesNoCard "label" icon="check" selected=false onTap
  Card grow gap=6 center border={selected ? 2 : 1} fill={selected ? "#FFEFF6FF" : "#FFFFFFFF"} onTap={onTap}
    Icon "{icon}" size=28
    Text.strong "{label}"

define CurrencyInput "label" @amount currency="$" hint=""
  state amount=0
  Column gap=4
    Text.fieldLabel "{label}"
    Row gap=8 middle
      Text.strong "{currency}"
      Input "" "{hint}" @amount type=number grow
    Text.muted "{money(amount, currency, 2)}"

define RefundMeter value=0
  Row gap=8 middle pad=[12,6,12,6] radius=20 fill={value >= 0 ? "#FFECFDF5" : "#FFFEF2F2"}
    Icon {value >= 0 ? "trending_up" : "trending_down"} size=18 color={value >= 0 ? "#FF16A34A" : "#FFDC2626"}
    Text.muted "{value >= 0 ? "Refund" : "You owe"}"
    Text.strong "{money(abs(value), "$", 0)}"

// A file drop target cannot read real files here, so it shows a labelled box.
define FileDropzone "label" accept=[] multiple=false onUpload
  Box pad=24 gap=8 center middle radius=12 border=2 fill=#FFF9FAFB onTap={onUpload({"name":"scan_0042.jpg"})}
    Icon "upload_file" size=32
    Text.strong "{label}"
    Text.muted "or click to browse ({join(accept,", ")})"

// A drawn signature needs a touch surface, so it shows a labelled box.
define SignaturePad @signature h=160 onChange
  state signature=""
  Box h={h} radius=8 border=1 fill=#FFFFFFFF center middle onTap={signature="signed"; onChange("signed")}
    Text.muted "{signature == "" ? "Signature area" : "Signed"}"
