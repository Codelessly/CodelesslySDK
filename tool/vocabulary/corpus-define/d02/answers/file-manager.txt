define FolderTree bind=path folders=[]
  Column gap=2
    each fo in folders
      NavItem "{fo.name}" icon="folder" selected={last(path) == fo.name} pad=[{fo.depth*16},0,0,0] drop=file onTap={path=fo.path; sel=[]} onDrop={files=updateWhere(files,x=>contains(sel,x.id) || x.id==result.data,x=>with(x,"folder",fo.name)); toast "Moved to {fo.name}"}

define Breadcrumb crumbs=[] onPick
  Row gap=4 middle
    each c in crumbs
      if index > 0
        Icon "chevron_right" size=16
      Text.strong "{c}" onTap={onPick(index)}

// a coloured file-type glyph drawn from an Icon
define FileIcon kind="doc" size=24
  Icon "{kind=='folder' ? 'folder' : kind=='image' ? 'image' : kind=='video' ? 'movie' : kind=='pdf' ? 'picture_as_pdf' : kind=='sheet' ? 'table_chart' : kind=='archive' ? 'folder_zip' : 'description'}" size={size} color={kind=='folder' ? '#FFF59E0B' : '#FF6B7280'}

// the real file contents (an image, a PDF page) are not drawable: a labelled box stands in
define Preview file large=false
  Box h={large ? 360 : 180} fill=#FFF3F4F6 radius=8 center middle
    FileIcon kind={file.kind} size=56
    Text.caption "Preview of {file.name}"

define StorageBar used=0 quota=1
  Column gap=4 pad=12
    Progress {used/quota}
    Text.caption "{fixed(used,1)} GB of {quota} GB used"

define UploadTray uploads=[] onClear
  Card gap=8 pad=12
    Row between
      Text.strong "Uploads ({count(uploads,u=>u.pct>=100)} of {length(uploads)} done)"
      IconButton "close" tooltip="Dismiss" onTap={onClear}
    each u in uploads
      Row gap=8 middle
        Text "{u.name}" grow
        Progress {u.pct/100} w=140
        Icon "{u.pct >= 100 ? 'check_circle' : 'upload'}" size=16

define Snackbar "text" action="" onAction onDismiss
  Row between middle pad=12 fill=#FF111827 radius=6
    Text "{text}" color=#FFFFFFFF
    Row gap=8
      Button.ghost "{action}" onTap={onAction}
      IconButton "close" tooltip="Dismiss" onTap={onDismiss}

### Search
state q=""
state kindFilter="Any type"
state dateFilter="Any time"
state sel=[]
App
  Sidebar
    Brand "Shelf" icon="folder_open"
    NavItem "My files" icon="folder" onTap={go files}
    NavItem "Search" icon="search" selected={true}
    NavItem "Trash" icon="delete" onTap={go trash}
    Spacer grow
    StorageBar used={used} quota={quota}
  Main
    Nav
      Heading "Search" level=1
      Input "Search files" bind=q w=320 autofocus
    Column pad=24 gap=16
      Row gap=12 wrap
        Select "Type" bind=kindFilter items=["Any type","folder","pdf","doc","sheet","image","video","archive"]
        Select "Modified" bind=dateFilter items=["Any time","Today","Past week","Past month"]
        FilterChip "Images only" selected={kindFilter=="image"} onTap={kindFilter=kindFilter=="image" ? "Any type" : "image"}
        FilterChip "Past week" selected={dateFilter=="Past week"} onTap={dateFilter=dateFilter=="Past week" ? "Any time" : "Past week"}
        Button.ghost "Clear filters" onTap={q=""; kindFilter="Any type"; dateFilter="Any time"}
      Text.muted "{length(filter(files,f=>contains(lower(f.name),lower(q)) && (kindFilter=='Any type' || f.kind==kindFilter) && (dateFilter=='Any time' || f.age <= (dateFilter=='Today' ? 0 : dateFilter=='Past week' ? 7 : 30))))} results"
      Card gap=0 pad=0
        each f in sortBy(filter(files,x=>contains(lower(x.name),lower(q)) && (kindFilter=="Any type" || x.kind==kindFilter) && (dateFilter=="Any time" || x.age <= (dateFilter=="Today" ? 0 : dateFilter=="Past week" ? 7 : 30))),x=>x.age)
          ListTile "{f.name}" subtitle="{f.folder} - modified {f.mod}" onTap={path=append(["My files"],f.folder); sel=[f.id]; go files}
            FileIcon kind={f.kind}
      if length(filter(files,f=>contains(lower(f.name),lower(q)) && (kindFilter=="Any type" || f.kind==kindFilter) && (dateFilter=="Any time" || f.age <= (dateFilter=="Today" ? 0 : dateFilter=="Past week" ? 7 : 30)))) == 0
        Column center gap=8 pad=48
          Icon "search_off" size=48
          Text.muted "Nothing matches. Try removing a filter."
