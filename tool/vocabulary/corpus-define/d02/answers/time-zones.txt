// A card for one city: its name, its local time, a 24 hour strip shaded for day and night with the working hours marked, and a drag handle.
define CityRow name="" tz="" hour=12 from=9 to=17 h24=false onRemove onTap
  Card gap=8 onTap={onTap}
    Row between middle
      Row gap=12 middle
        Icon "drag_indicator"
        Column gap=2
          Text.strong "{name}"
          Text.muted "{tz}"
      Row gap=12 middle
        Badge.success "Working" visible={hour >= from && hour < to}
        Badge.neutral "Off hours" visible={!(hour >= from && hour < to)}
        Icon name={hour >= 6 && hour < 18 ? "wb_sunny" : "dark_mode"}
        Text.heading "{h24 ? fixed(floor(hour), 0) + ':00' : (floor(hour) % 12 == 0 ? 12 : floor(hour) % 12) + (hour < 12 ? ' AM' : ' PM')}" fill={hour >= 6 && hour < 18 ? #FFFEF9C3 : #FF1E293B} color={hour >= 6 && hour < 18 ? #FF854D0E : #FFE2E8F0}
        IconButton "close" tooltip="Remove city" onTap={onRemove}
    HourStrip hour={hour} from={from} to={to}

define HourStrip hour=12 from=9 to=17
  Row gap=1 h=14
    each i in range(24)
      Box grow h=14 fill={i >= from && i < to ? (i == floor(hour) ? #FF15803D : #FF86EFAC) : (i >= 6 && i < 18 ? #FFFDE68A : #FF64748B)} border={i == floor(hour) ? 2 : 0}
