### Print view
App
  Sidebar
    Brand "Tablecloth" icon="table_restaurant"
    NavItem "Floor plan" icon="grid_view" onTap={go floor-plan}
    NavItem "Print view" icon="print" selected={true}
  Main
    Nav
      Heading "Seating list"
      Row gap=8
        Select "Sort" bind=printSort items=["By table","By name"]
        Button "Print" icon="print" onTap={emit print}
    Column pad=24 gap=16 scroll
      if printSort!="By name"
        each t in tables
          Card gap=8
            Row between
              Heading "{t.name}" level=3
              Text.muted "{count(guests,g=>g.table==t.id)} of {t.cap} seats"
            each g in sortBy(filter(guests,x=>x.table==t.id),x=>x.name)
              Row gap=12
                Text.strong "{g.name}" w=180
                Text.muted "{g.side}" w=80
                Text "{g.diet}"
        Card gap=8
          Heading "Unseated" level=3
          each g in filter(guests,x=>x.table=="")
            Row gap=12
              Text "{g.name}" w=180
              Text.muted "{g.side}"
      if printSort=="By name"
        Card gap=8
          each g in sortBy(guests,x=>x.name)
            Row gap=12
              Text.strong "{g.name}" w=180
              Text "{coalesce(find(tables,t=>t.id==g.table).name,\"Unseated\")}"
