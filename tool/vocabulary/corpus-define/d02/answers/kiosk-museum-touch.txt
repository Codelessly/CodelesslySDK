### Attract
state lang="en"
state started=false
state works=[{"id":"w1","url":"https://images.museum.example/vermeer-pearl.jpg"},{"id":"w2","url":"https://images.museum.example/nightwatch.jpg"},{"id":"w3","url":"https://images.museum.example/crows.jpg"},{"id":"w4","url":"https://images.museum.example/venus.jpg"}]

  Section center bleed

      DriftingWall items={works} speed=6 columns=4 dim=0.55

        Heading "Touch to begin" level=1 animate=pulse duration=1800

        Button "Start exploring" size=large onTap={go gallery-wall}

define LanguageSwitch @lang languages=["en"] onChange
  state lang="en"
  SegmentedButton items={languages} @lang onChange={onChange(result)}

// Drifting thumbnails: a canvas that slowly moves and follows a drag; shown as a grid of tiles here.
define DriftingWall items=[] speed=10 columns=6 dim=0 onDrag onPick
  state offset=0
  every 1s onTick={offset=offset+speed}
  Stack onSwipeLeft={offset=offset+200; onDrag(offset)} onSwipeRight={offset=offset-200; onDrag(offset)}
    Grid cols={columns} gap=8 pad=8
      each w in items
        Image "{w.url}" h=160 radius=6 onTap={onPick(w.id)}
    Box fill=#FF000000 w=fill h=fill
    ...

// A circle of one visitor's own touch space on the wall, shown as a draggable ring with a label.
define TouchCircle x=0.5 y=0.5 color="#FF2563EB" label="" radius=100 onMove onClose
  Stack
    Ring 1 size={radius*2} color={color}
    Row gap=4
      Text.small "{label}" color={color}
      IconButton "close" tooltip="Remove circle" onTap={onClose}

// A century range slider with tick labels; the draggable handles are shown as two sliders.
define TimelineSlider min=0 max=100 step=1 from=0 to=100 marks=[] onChange
  state lo=0
  state hi=100
  Column gap=4
    Slider @lo min={min} max={max} step={step} onChange={lo=result; onChange({"from":lo,"to":hi})}
    Slider @hi min={min} max={max} step={step} onChange={hi=result; onChange({"from":lo,"to":hi})}
    Row between
      each m in marks
        Text.small "{m}" color=#FF9CA3AF

// A picture you pinch to zoom, with dots over it that open short texts.
define HotspotImage src="" hotspots=[] maxZoom=4 onHotspot
  state zoom=1
  Stack onTap={zoom=min(maxZoom,zoom+1)}
    Image "{src}" h=fill radius=8
    each s in hotspots
      Box w=24 h=24 radius=12 fill=#FFFFFFFF border=2 onTap={onHotspot(indexOf(hotspots,s))}
        Icon "add"
    Text.small "x{zoom}" color=#FFFFFFFF

// A horizontal strip of related works.
define RelatedStrip items=[] onPick
  Row scroll gap=12
    each r in items
      Card gap=4 w=160 onTap={onPick(r.id)}
        Box h=100 fill=#FFE5E7EB radius=6
        Text.small "{r.title}"

// A QR code, shown as a labelled box since no code can be drawn here.
define QRCode value="" size=160
  Box w={size} h={size} fill=#FFFFFFFF border=1 pad=8
    Text.caption "{value}"
