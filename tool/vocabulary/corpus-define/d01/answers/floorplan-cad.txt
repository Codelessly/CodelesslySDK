// A canvas of boxes: draws walls, openings, rooms and furniture as labelled boxes on a grid; real hit testing, snapping and rotation handles are not possible, so onPoint, onSnap and onRotate are given by the host.
define PlanCanvas walls=[] openings=[] rooms=[] furniture=[] dims=[] layers=[] tool="select" units="m" zoom=1 sel="" onPoint onSnap onRotate onDrop
  Stack h=420 fill=#FFF8FAFC radius=8 border=#FFCBD5E1 onTap={onPoint}
    each r in rooms
      Box x={r.x*60*zoom} y={r.y*60*zoom} w={r.w*60*zoom} h={r.d*60*zoom} fill=#FFE0F2FE center middle onTap={onPoint}
        Text.strong "{r.name}"
        Text.caption "{units=="m" ? fixed(r.w*r.d,1) : fixed(r.w*r.d*10.764,0)} {units=="m" ? "m²" : "ft²"}"
    each w in walls
      Box x={w.x1*60*zoom} y={w.y1*60*zoom} w={max(4,abs(w.x2-w.x1)*60*zoom)} h={max(4,abs(w.y2-w.y1)*60*zoom)} fill={sel==w.id ? "#FF2563EB" : "#FF111827"} onTap={onPoint}
        Text.caption "{units=="m" ? fixed(w.len,2) : fixed(w.len*3.281,1)} {units}" color=#FFDC2626
    each o in openings
      Box w={o.width*60*zoom} h=8 fill={o.kind=="door" ? "#FFF59E0B" : "#FF38BDF8"}
    each f in furniture
      Box w={f.w*60*zoom} h={f.d*60*zoom} fill=#FFE2E8F0 radius=4 border=#FF64748B center middle onTap={onPoint}
        Text.caption "{f.kind}"
        IconButton "rotate_right" tooltip="Rotate" onTap={onRotate}
    each d in dims
      Row gap=4 middle
        Icon "straighten"
        Text.caption "{units=="m" ? fixed(d.len,2) : fixed(d.len*3.281,1)} {units}"

define ScaleBar length=1 unit="m" zoom=1
  Row gap=6 middle
    Box w={60*zoom*length} h=6 fill=#FF111827
    Text.caption "1 {unit}"
