// Placeholder: a real sky disc (projected stars, drag, phone orientation) cannot be drawn with these words. It shows the disc as a labelled box with its layers and a tappable list of dots for the objects.
define SkyDisc stars=[] lines=[] azimuth=0 altitude=0 grid=true names=true deep=false night=false pointTo="" compass=false onDrag onPick
  Card gap=8 fill={night ? "#FF1A0000" : "#FF0B1026"} radius=16 drag=sky(azimuth) onDrop={onDrag(result.data)}
    Row between
      Text.small "Sky disc, facing {azimuth} deg, up {altitude} deg" color={night ? "#FFFF4D4D" : "#FFCBD5E1"}
      Text.small "{compass ? 'Compass on' : 'Drag to move'}" color={night ? "#FFFF4D4D" : "#FFCBD5E1"}
    Row gap=16 center middle h=240 wrap
      each s in stars
        Column center gap=2 onTap={onPick(s.id)}
          Box w={max(6,14-s.mag*3)} h={max(6,14-s.mag*3)} radius={14} fill={night ? "#FFFF4D4D" : (s.id==pointTo ? "#FFFACC15" : "#FFFFFFFF")}
          if names
            Text.caption "{s.name}" color={night ? "#FFFF4D4D" : "#FFCBD5E1"}
          if s.id==pointTo
            Icon "arrow_upward" color="#FFFACC15"
    Row gap=12
      if grid
        Text.caption "Grid on" color={night ? "#FFFF4D4D" : "#FFCBD5E1"}
      if deep
        Text.caption "Deep sky on" color={night ? "#FFFF4D4D" : "#FFCBD5E1"}
      Text.caption "{length(lines)} constellation lines" color={night ? "#FFFF4D4D" : "#FFCBD5E1"}

define TimeScrubber bind=value min=-720 max=720 step=10 label="" onChange
  Column gap=4
    Text.caption "{label}"
    Slider bind=value min={min} max={max} step={step} onChange={onChange}
