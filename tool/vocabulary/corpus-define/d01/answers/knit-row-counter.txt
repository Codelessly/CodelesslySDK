            Text.muted "tap to count a row"

        Text.caption "Heard: {heard}"

            Row between middle
              Column gap=2
                Text.strong "{r.name} every {r.every} rows"
                Text.muted "{r.left} more rows"
              Progress {1-(r.left/r.every)} w=120

        Progress {rowsToday/goal}
        Text.muted "About {fixed(seconds/60/max(1,rowsToday),1)} minutes a row"

define VoiceCommand phrases=["next"] onHeard
  // Speech recognition is not available here: it is a labelled button that simulates hearing the first phrase.
  state listening=false
  Button.outline "{listening ? 'Listening...' : 'Voice'}" icon="mic" onTap={listening=!listening; listening ? onHeard(first(phrases)) : toast "Voice off"}

define PatternChart rows=[] current=0 cell=20 onTapRow
  Column gap=2
    each r in rows
      Row gap=2 onTap={onTapRow(r.n)}
        Text.caption "{r.n}" w=28
        each c in r.cells
          Box w={cell} h={cell} radius=3 fill={r.n<current ? "#FFD1D5DB" : (c==1 ? "#FF1F6F8B" : "#FFF2C14E")} border={r.n==current ? "#FF111827" : "#00000000"}

define Magnifier rows=[] current=0 zoom=2
  // A magnifier strip over the chart: the current row drawn larger.
  Row scroll gap=2 middle
    each c in find(rows,r=>r.n==current).cells
      Box w={12*zoom*2} h={12*zoom*2} radius=4 fill={c==1 ? "#FF1F6F8B" : "#FFF2C14E"}
