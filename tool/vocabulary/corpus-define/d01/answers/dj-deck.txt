### Recordings
state recording=false
state secs=0
state takes=[{"name":"Friday warm-up","len":"1:12:30","date":"Nov 14"},{"name":"Rooftop set","len":"58:04","date":"Nov 8"}]
every 1s when recording onTick={secs=secs+1}
App
  Sidebar
    Brand "Crossfade" icon="album"

    NavItem "Library" icon="library_music" onTap={go library}

  Main
    Nav
      Heading "Recordings" level=1
      Button "{recording ? 'Stop' : 'Start recording'}" onTap={recording ? {recording=false; takes=append(takes,{"name":"Untitled take","len":duration(secs),"date":"Today"}); secs=0; toast "Take saved"} : {recording=true}}

        Card gap=8

          Waveform values=[3,5,4,7,6,8,5,4,6,9,7,5] position=0
      each tk in takes
        Card
          Row gap=12 middle

            Column grow
              Text.strong "{tk.name}"
              Text.muted "{tk.date}, {tk.len}"
            IconButton "download" tooltip="Download" onTap={toast "Exporting {tk.name}"}

// Platter drawn as a ring with a record icon; it spins only while playing. No real audio.
define Platter spinning=false accent=#FF22D3EE size=120
  Stack center middle
    Ring 1 w={size} h={size}
    if spinning
      Icon "album" size={size/2} color={accent} animate=spin duration=2000
    if !spinning
      Icon "album" size={size/2} color={accent}

// Waveform is a bar chart stand-in with a position readout; real scrolling audio is not drawn.
define Waveform values=[4,6,5,8,7,9,6,5] position=0 cues=[] color=#FF22D3EE
  Column gap=4
    BarChart values={values} h=64
    Row between
      Text.caption "{duration(position)}"
      Text.caption "Cues: {length(cues)}"

define Knob "label" bind=value min=-12 max=12
  state value=0
  Column gap=4 center
    Stack center middle
      Ring {(value-min)/(max-min)} w=56 h=56
      Text.small "{value}"
    Slider bind=value min={min} max={max} w=64
    Text.caption "{label}"

// Vertical fader shown as a labelled level column with a horizontal slider; no vertical slider is drawn.
define Fader "label" bind=level
  state level=75
  Column gap=4 center
    Progress {level/100} w=48
    Slider bind=level min=0 max=100 w=80
    Text.caption "CH {label}"

define Deck "name" title="" bpm=120 key="" accent=#FF22D3EE
  state playing=false
  state tempo=0
  state pos=0
  state loop=0
  state cues=[]
  every 1s when playing onTick={pos=pos+1}
  Card gap=12
    Row between
      Row gap=8 middle
        Badge.primary "DECK {name}"
        Column
          Text.strong "{title}"
          Text.muted "{bpm + round(bpm*tempo/100,1)} BPM, key {key}"
      Text.mono "{duration(pos)}"
    Row gap=16 middle
      Platter spinning={playing} accent={accent}
      Column grow gap=8
        Waveform values=[3,5,4,7,6,8,5,4,6,9,7,5,4,6] position={pos} cues={cues} color={accent}
        Row gap=8
          Button.outline "Bend -" onTap={tempo=clamp(tempo-1,-8,8)}
          Button.outline "Bend +" onTap={tempo=clamp(tempo+1,-8,8)}
          Text.small "Tempo {tempo}%"
    Slider bind=tempo min=-8 max=8
    Row gap=8
      Button "{playing ? 'Pause' : 'Play'}" onTap={playing=!playing}
      Button.secondary "Cue" onTap={pos=0; playing=false}
      Spacer grow
      ChoiceChip "1/4" selected={loop==0.25} onTap={loop=loop==0.25 ? 0 : 0.25}
      ChoiceChip "1/2" selected={loop==0.5} onTap={loop=loop==0.5 ? 0 : 0.5}
      ChoiceChip "1" selected={loop==1} onTap={loop=loop==1 ? 0 : 1}
      ChoiceChip "4" selected={loop==4} onTap={loop=loop==4 ? 0 : 4}
    Grid cols=4 gap=8
      each i in range(4)
        Button.outline "Cue {i+1}" selected={contains(cues,i)} onTap={contains(cues,i) ? {pos=i*30} : {cues=append(cues,i); toast "Cue {i+1} set"}}
