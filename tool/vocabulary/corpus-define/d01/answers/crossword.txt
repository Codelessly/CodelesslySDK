// A 13 by 13 board of cells. Keys are read on the grid; typing fills the cell and steps along the word.
define CrosswordGrid
  Grid cols=13 gap=1 w=520 fill=#FF111827 onKey={result=="Backspace" ? {vals=setAt(vals,cur,""); cur=max(0,cur-(dir=="a"?1:13))} : result=="ArrowRight" ? {cur=min(168,cur+1)} : result=="ArrowLeft" ? {cur=max(0,cur-1)} : result=="ArrowDown" ? {cur=min(168,cur+13)} : result=="ArrowUp" ? {cur=max(0,cur-13)} : result=="Tab" ? {cur=nextStart(cur)} : result==" " ? {dir=(dir=="a"?"d":"a")} : {vals=setAt(vals,cur,upper(result)); pen=setAt(pen,cur,pencil); bad=setAt(bad,cur,false); cur=min(168,cur+(dir=="a"?1:13))}}
    each i in range(169)
      Cell i={i}

define Cell i=0
  Stack h=40 fill={slice(sol,i,i+1)=="#" ? "#FF111827" : i==cur ? "#FFFDE047" : (dir=="a" ? (floor(i/13)==floor(cur/13)) : (i%13==cur%13)) ? "#FFBFDBFE" : "#FFFFFFFF"} onTap={slice(sol,i,i+1)=="#" ? {} : i==cur ? {dir=(dir=="a"?"d":"a")} : {cur=i}}
    if find(clues,k=>k.s==i)
      Text "{find(clues,k=>k.s==i).n}" size=9 left top
    Text "{vals[i]}" center middle size=22 weight=600 color={bad[i] ? "#FFDC2626" : pen[i] ? "#FF6B7280" : "#FF111827"}
    if bad[i]
      Icon "close" size=12 right bottom color=#FFDC2626

define ClueList which="a"
  Column gap=2 scroll h=240
    each k in filter(clues,x=>x.dir==which)
      Row gap=8 pad=6 radius=4 fill={k.dir==dir && k.s==cur ? "#FFBFDBFE" : "#00000000"} onTap={cur=k.s; dir=which}
        Text.strong "{k.n}" w=24
        Text.body "{k.t}" grow
