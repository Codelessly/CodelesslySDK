    Spacer grow
    Text.muted "Ray Li, hcp 12.4"

    Nav
      Heading "Scorecard - Saturday round" level=1
      SegmentedButton @view items=["mine","friend"]
    Column pad=24 gap=16
      Row gap=16
        Card grow gap=4
          Text.muted "Ray"
          Text "{sum(mine,s=>s)}" size=40 weight=700
          Text.muted "{sum(mine,s=>s)-sum(pars,p=>p)} over par"
        Card grow gap=4
          Text.muted "Dana Whitfield"
          Text "{sum(friend,s=>s)}" size=40 weight=700
          Text.muted "{sum(friend,s=>s)-sum(pars,p=>p)} over par"
      Card gap=12
        Heading "Front nine" level=3
        Grid cols=9 gap=6 keep
          each p in take(pars,9)
            Column center gap=2
              Text.caption "{index+1}"
              Text.caption "Par {p}"
              ParCell score={view=="mine" ? mine[index] : friend[index]} par={p}
        Heading "Back nine" level=3
        Grid cols=9 gap=6 keep
          each p in slice(pars,9,18)
            Column center gap=2
              Text.caption "{index+10}"
              Text.caption "Par {p}"
              ParCell score={view=="mine" ? mine[index+9] : friend[index+9]} par={p}
      Card gap=8
        Heading "Side by side" level=3
        Row gap=8
          Text.strong "Hole" w=60
          Text.strong "Par" w=60
          Text.strong "Ray" w=60
          Text.strong "Dana" grow
        each p in pars
          Row gap=8
            Text "{index+1}" w=60
            Text "{p}" w=60
            Text "{mine[index]}" w=60
            Text "{friend[index]}" grow
            Badge.success "Ray" visible={mine[index]<friend[index]}
            Badge.info "Dana" visible={friend[index]<mine[index]}

// Draws a hole from above on the green fairway tint with the tee, the green, two bunkers and the shots as numbered dots joined by lines;
// it is a flat drawing, not a measured course. A tap reports its place as result.x and result.y (0 to 100).
define HoleMap hole=1 shots=[] h=460 onTap
  Stack w=fill h={h} fill=#FF86C27A radius=12 onTap={onTap}
    Box w=140 h=60 fill=#FF2E7D32 radius=30
    Box w=60 h=40 fill=#FFE9D8A6 radius=20
    Box w=50 h=34 fill=#FFE9D8A6 radius=17
    Text.caption "Green" color=#FFFFFFFF
    Text.caption "Tee" color=#FFFFFFFF
    Text.caption "Hole {hole}" color=#FFFFFFFF
    each s in shots
      Stack
        Box w=22 h=22 radius=11 fill=#FFFFFFFF
        Text.caption "{index+1} - {s.yds}y"

// A scorecard cell tinted by score against par: birdie green, par plain, bogey amber, double or worse red.
define ParCell score=0 par=4
  Box w=fill pad=6 radius=6 fill={score==0 ? "#FFF3F4F6" : (score<par ? "#FFBBF7D0" : (score==par ? "#FFFFFFFF" : (score==par+1 ? "#FFFDE68A" : "#FFFECACA")))}
    Text.strong "{score==0 ? '-' : score}" center
