// Photo tile with a water-drop meter: how full the drop is shows how recently it was watered.
define PlantTile "name" species="" photo="" water=0.5 onTap
  Card gap=8 pad=0 onTap={onTap}
    Image "{photo}" h=160
    Column gap=4 pad=12
      Row between middle
        Text.strong "{name}"
        WaterDrop value={water}
      Text.muted "{species}"

// Horizontal strip of dated photos joined by a line; tapping one fires onSelect with that item as result.
define GrowthTimeline items=[] onSelect
  Row scroll gap=24
    each g in items
      Column gap=6 w=160
        Row middle gap=0
          Box w=12 h=12 radius=6 fill=#FF2E7D32
          Box h=2 grow fill=#FFA5D6A7
        Image "{g.url}" h=120 radius=8 onTap={onSelect(g)}
        Text.strong "{g.date}"
        Text.muted "{g.note}"
