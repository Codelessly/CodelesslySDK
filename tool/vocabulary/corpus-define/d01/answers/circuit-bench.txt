// Honest version: the board lists parts in a grid of cells, marks the selected one, and shows each wire as a labelled line of text under the grid; real freehand wire drawing, drifting dots and part icons need a canvas, so a labelled box stands in for them. Lamp glow follows `amps`.
define CircuitBoard parts=[] wires=[] cols=8 rows=5 selected="" amps=0 running=true wireFrom="" onSelect onOpen onDrop
  Column gap=8
    Box fill=#FF0F172A radius=8 pad=12 gap=8 h=300 onDrop={onDrop}
      Text.caption "Canvas grid {cols} x {rows}" color=#FF94A3B8
      Grid cols=4 gap=8
        each p in parts
          Box pad=10 radius=8 gap=4 fill={p.id==selected ? "#FF1E3A8A" : "#FF1E293B"} border={p.id==wireFrom ? "#FFFACC15" : "#FF334155"} menu=partMenu onTap={onSelect(p.id)} animate={running && p.kind=="lamp" ? "pulse" : "fade"}
            Row gap=6 middle
              Icon "{p.kind=="battery" ? "battery_full" : p.kind=="resistor" ? "linear_scale" : p.kind=="lamp" ? "lightbulb" : p.kind=="switch" ? "toggle_on" : "sync_alt"}" color={p.kind=="lamp" && amps>0 ? "#FFFACC15" : "#FFE2E8F0"}
              Text.small "{p.kind}" color=#FFE2E8F0
            Text.caption "{p.value}{p.kind=="battery" ? " V" : p.kind=="switch" ? (p.value==1 ? " on" : " off") : " Ω"}" color=#FF94A3B8
    Row scroll gap=6
      each w in wires
        Chip "{w.from} to {w.to}"

// A lamp whose light grows with glow from 0 to 1.
define Lamp glow=0 size=48
  Box w={size} h={size} radius={size/2} fill={glow>0.66 ? "#FFFACC15" : glow>0.33 ? "#FFFDE68A" : glow>0.02 ? "#FFFEF3C7" : "#FFE2E8F0"} border="#FF94A3B8" animate={glow>0.02 ? "pulse" : "fade"}

// A number with minus and plus buttons.
define Stepper "label" bind=value min=0 max=100 step=1 unit="" onChange
  Row gap=8 middle
    Button.outline "-" onTap={value=max(min,value-step); onChange(value)}
    Text.strong "{value} {unit}"
    Button.outline "+" onTap={value=min(max,value+step); onChange(value)}

### Scope
state source=9
state scale="5 s"
state samples=[0,2,5,7,9,9,9,8,9,9,0,0,9,9,9]
state cursor=6
state hold=false
every 1s when !hold onTick={samples=length(samples)>=60 ? append(removeAt(samples,0),source) : append(samples,source)}
