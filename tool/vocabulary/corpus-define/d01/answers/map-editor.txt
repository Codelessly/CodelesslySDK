// A map surface cannot be drawn with the words taught: shown as a labelled box listing what it holds.
define MapCanvas features=[] layers=[] tool="select" snap=false @selected onMove onMeasure onVertexDrag onAdd
  Box grow fill=#FFE8F0E4 pad=16 gap=8 drop=feature onDrop={onAdd(result.data)}
    Text.caption "Map preview (placeholder) - tool: {tool}, snap: {snap ? 'on' : 'off'}"
    each f in features
      Chip "{f.name}" selected={f.id==selected} onTap={selected=f.id}
    Spacer grow
    Row gap=8
      Text.muted "{length(features)} features on {count(layers,l=>l.visible)} visible layers"

// Shows a coordinate pair as text; the live cursor position is supplied by the map.
define CoordinateReadout "text" zoom=12
  Row gap=8 middle
    Icon "my_location"
    Text.mono "{text}"
    Text.muted "z{zoom}"

define ColorPicker @color swatches=["#FF111827","#FFDC2626","#FF2563EB"] size=28 onPick
  state color=""
  Row gap=8 middle
    each swatch in swatches
      Box w={size} h={size} radius={size/2} fill={swatch} border={swatch == color ? "#FF111827" : "#00000000"} onTap={color=swatch; onPick(swatch)}

define IconPicker @icon icons=["place"] onPick
  state icon=""
  Row gap=8 middle
    each name in icons
      IconButton "{name}" selected={name==icon} onTap={icon=name; onPick(name)}

// A file chooser cannot read files here: drawn as a dashed drop zone box.
define FileDrop "label" accept=[] @fileName onPick
  Box pad=24 gap=8 center radius=8 border=#FF9CA3AF onTap={onPick(fileName)}
    Icon "upload_file"
    Text "{label}"
    Text.caption "{fileName}"

define Table columns=[] rows=[]
  Column gap=0
    Row gap=8 pad=8 border-bottom=#FFD1D5DB
      each c in columns
        Text.strong "{c}" w=110
    each r in rows
      Row gap=8 pad=8
        each c in columns
          Text "{r[c]}" w=110
