// A curve editor with draggable bezier handles needs a canvas; this draws the curve as a LineChart of the preset and shows the handle values as two sliders.
define CurveEditor @curve h=200 onChange
  state c1=0.25
  state c2=0.75
  Column gap=8
    LineChart values={map(range(11), i => 3*(1-i/10)*(1-i/10)*(i/10)*c1 + 3*(1-i/10)*(i/10)*(i/10)*c2 + (i/10)*(i/10)*(i/10))} h={h}
    Row gap=8 middle
      Text.caption "Handle 1" w=60
      Slider @c1 min=0 max=1 grow onChange={onChange}
    Row gap=8 middle
      Text.caption "Handle 2" w=60
      Slider @c2 min=0 max=1 grow onChange={onChange}

// A time ruler with ticks every frame and the loop region shaded; scrubbing is a slider under it, drawn as a labelled strip.
define Ruler duration=24 loopStart=0 loopEnd=24 @time
  Row gap=8 middle
    Text.caption "Frames" w=100
    Row grow gap=0
      each i in range(duration + 1)
        Box grow h=20 center middle border-left=#FF374151 fill={i >= loopStart && i <= loopEnd ? "#332563EB" : "#00000000"} onTap={time = i}
          Text.caption "{i % 4 == 0 ? i : ""}"
