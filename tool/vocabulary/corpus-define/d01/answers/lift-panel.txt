### Lift panel
state floors=[9,8,7,6,5,4,3,2,1,0]
state restricted=[8,9]
state lit=[]
state current=0
state dir="idle"
state doors="closed"
state weight=420
state fire=false
state keycard=false
state beep=false
state idle=0
state alarm=false
every 1s onTick={idle=idle+1; idle==20 && length(lit)==0 && doors=="closed" ? {go ads} : }
every 1s when length(lit)>0 && weight<=800 && doors=="closed" onTick={current==first(lit) ? {lit=removeAt(lit,0); doors="open"; dir="idle"; play chime; speak "Floor {current}"} : {current=current+(first(lit)>current ? 1 : -1); dir=first(lit)>current ? "up" : "down"}}
every 3s when doors=="open" && weight<=800 onTick={doors="closed"}
every 1s when weight>800 onTick={beep=!beep; play beep}
App
  Main
    Nav
      Brand "Floors" icon="elevator"
      Row gap=8
        Badge.error "OVERLOAD" hide={weight<=800}
        Badge.error "FIRE MODE" hide={!fire}
        Badge.warning "ALARM" hide={!alarm}
        Button.outline "Key card" icon="badge" onTap={open keycard}
    Column pad=24 gap=16 center maxw=480
      if fire
        Card fill=#FFB91C1C gap=4 animate=pulse
          Text.strong "FIRE: use the exit floor only" color=#FFFFFFFF
          Text "Doors will open at floor 0. Do not use the lift once out." color=#FFFFFFFF
      if weight>800
        Card fill=#FFFEF3C7 gap=4 animate=shake
          Text.strong "Overload: {weight} kg of 800 kg"
          Text.muted "Please step out until the beeping stops."
      Card gap=8
        Row between
          Column gap=2
            Text.caption "Current floor"
            Text "{current}" size=64 weight=700
          Column center gap=2
            Icon "arrow_upward" hide={dir!="up"}
            Icon "arrow_downward" hide={dir!="down"}
            Icon "pause" hide={dir!="idle"}
            Text.caption "{dir == "idle" ? "Doors " + doors : "Going " + dir}"
        Text.muted "Queue: {length(lit)==0 ? "empty" : join(lit,", ")}"
      Row gap=16 top
        Column grow gap=10 center
          each f in floors
            FloorPad "{f}" lit={contains(lit,f)} locked={contains(restricted,f) && !keycard} onTap={contains(restricted,f) && !keycard ? toast "Key card needed for floor {f}" : fire && f!=0 ? toast "Fire mode: exit floor only" : weight>800 ? toast "Overloaded" : {lit=contains(lit,f) ? lit : append(lit,f); idle=0; play click}}
        Column gap=12 w=140
          Button.outline "Open doors" icon="keyboard_double_arrow_left" onTap={weight>800 || doors=="closed" && length(lit)>0 && dir!="idle" ? toast "Cannot open while moving" : {doors="open"; idle=0}}
          Button.outline "Close doors" icon="keyboard_double_arrow_right" onTap={doors="closed"; idle=0}
          Button.destructive "Alarm" icon="notifications_active" onTap={alarm=true; emit alarm; toast "Alarm sent to the control room"}
          Button.ghost "Cancel all" onTap={lit=[]}
          Divider
          Text.caption "Load: {weight} kg"
          Slider bind=weight min=0 max=1000
          Toggle "Fire mode" bind=fire onChange={result ? {lit=[0]; speak "Fire. Proceed to the exit floor."} : {lit=[]}}
          Toggle "Voice" bind=voice
  Dialog#keycard "Key card" "Floors 8 and 9 are staff only"
    Column gap=12 center
      Icon "contactless" size=48
      Text.muted "Hold your card to the reader."
      Button "Tap card" onTap={keycard=true; toast "Card accepted"; close}
      Button.ghost "Remove card" onTap={keycard=false; lit=without(without(lit,8),9); close}

### Idle ads
state slide=0
state ads=[{"brand":"Cafe Ostia","line":"Flat white and a pastry, ground floor, $6","fill":"#FF92400E"},{"brand":"Hartwell Dental","line":"Level 7. New patients welcome this month.","fill":"#FF1D4ED8"},{"brand":"Sky Yoga","line":"Level 9. Free first class with your key card.","fill":"#FF047857"}]
every 4s onTick={slide=(slide+1)%length(ads)}
App
  Main
    Column pad=0 gap=0
      Pages bind=slide h=360
        each a in ads
          Box fill={a.fill} pad=32 center middle gap=12 animate=fade
            Text "{a.brand}" size=40 weight=700 color=#FFFFFFFF
            Text "{a.line}" size=20 color=#FFFFFFFF
      Dots bind=slide count={length(ads)}
      Row between pad=16
        Column
          Text.caption "Floor"
          Text "{current}" size=40 weight=700
        Text.muted "Touch to use the lift"
        Button "Wake" onTap={idle=0; go lift-panel}

define FloorPad "label" lit=false locked=false size=56 onTap
  Stack center middle w={size} h={size} radius={size/2} fill={lit ? "#FFFBBF24" : "#FFE5E7EB"} border={locked ? "#FFDC2626" : "#FF9CA3AF"} onTap={onTap}
    Text.strong "{label}" color={lit ? "#FF111827" : "#FF374151"}
    Icon "lock" size=14 hide={!locked}
