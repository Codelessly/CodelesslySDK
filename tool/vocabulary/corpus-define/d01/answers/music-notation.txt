state offsets=[]

    NavItem "Practice" icon="piano" onTap={go practice}
    NavItem "Rhythm" icon="graphic_eq" selected={true}

    NavItem "Settings" icon="tune" onTap={open settings}
  Main
    Nav
      Heading "Tap along" level=1
      Row gap=8
        Text.muted "{bpm} BPM"
        Button "{running ? 'Stop' : 'Start'}" onTap={running=!running; beat=0; taps=[]; offsets=[]}
    Column pad=24 gap=16
      Card gap=12
        Text.strong "Tempo"
        Slider bind=bpm min=40 max=160
        Row gap=12 middle
          Metronome bpm={bpm} running={running} beat={beat}
          Text.muted "Beat {beat+1} of 4"
      Card gap=12
        Text.strong "Pattern"
        BeatTrack pattern=["q","q","e e","q"] beat={beat} taps={taps}
        Row gap=8
          each o in offsets
            Badge.neutral "{o < 0 ? 'early' : 'late'} {abs(o)} ms"
      Button "Tap" size=large onTap={taps=append(taps,beat); offsets=append(offsets,(beat*37)%90-45)}
      Row gap=24
        Column gap=4
          Text.muted "Average offset"
          Text.heading "{length(offsets)==0 ? 0 : round(sum(offsets,o=>o)/length(offsets),0)} ms"
        Column gap=4
          Text.muted "Early taps"
          Text.heading "{count(offsets,o=>o<0)}"
        Column gap=4
          Text.muted "Late taps"
          Text.heading "{count(offsets,o=>o>0)}"
every 1s when running onTick={beat=(beat+1)%4}

// Draws five staff lines and the note as a labelled round mark; no real notation glyphs or positions
define Staff "note" clef="Treble" flash=""
  Column gap=0 pad=12
    Text.caption "{clef}"
    Stack h=100
      Column gap=18 pad=[0,6,0,6]
        Divider
        Divider
        Divider
        Divider
        Divider
      Box w=28 h=22 radius=11 fill={flash=="right" ? "#FF16A34A" : flash=="wrong" ? "#FFDC2626" : "#FF111827"} center middle
        Text "{note}" color=#FFFFFFFF size=11

// A row of white keys; black keys are not drawn. Tapping a key gives its letter as result
define Keyboard from="C4" octaves=1 flash="" onKey
  Row gap=2 h=120
    each k in ["C","D","E","F","G","A","B"]
      Box grow fill={flash=="wrong" ? "#FFFEE2E2" : "#FFFFFFFF"} border="#FF111827" radius=4 bottom onTap={onKey(k)}
        Text.caption "{k}"

// A blinking dot per beat; the pulse is a fade, no sound
define Metronome bpm=90 running=false beat=0
  Row gap=8
    each b in range(4)
      Box w=20 h=20 radius=10 fill={running && b==beat ? "#FF2563EB" : "#FFD1D5DB"}

// A row of beat cells for the pattern, marking the beats you tapped
define BeatTrack pattern=[] beat=0 taps=[]
  Row gap=8
    each p in pattern
      Box grow h=48 radius=6 pad=8 fill={indexOf(pattern,p)==beat ? "#FFDBEAFE" : "#FFF3F4F6"} center middle
        Text "{p}"
        if contains(taps,indexOf(pattern,p))
          Icon "check"
