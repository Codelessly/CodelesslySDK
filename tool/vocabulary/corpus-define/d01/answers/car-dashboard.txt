### Navigation
state mode="normal"
state night=false
state distance=450
state step=0
state steps=[{"icon":"turn_right","street":"Harbour Road","dist":450},{"icon":"turn_left","street":"King Street","dist":1200},{"icon":"roundabout_left","street":"Ring Road, 2nd exit","dist":3100},{"icon":"flag","street":"Arrive at Pier 9","dist":0}]
state speed=62
every 1s onTick={distance=max(0,distance-17)}
every 1s when distance==0 onTick={step=min(length(steps)-1,step+1); distance=steps[min(length(steps)-1,step+1)].dist}
