// A key shortcut has no visual form: shown as a labelled box.
define KeyListener key="space" onPress
  Box hidden
    Text.caption "Key: {key}"

// A hue wheel needs a canvas; shown as a labelled round box filled with the colour.
define ColorWheel bind=color size=200
  Box w={size} h={size} radius={size/2} fill={color}
    Text.caption "hue wheel" center

define HexInput "label" bind=value onChange
  state value="#FF000000"
  Row gap=8 middle
    Box w=28 h=28 radius=6 fill={value}
    Input "{label}" bind=value onChange={onChange}

// Syntax highlighting is not drawn; the code is plain monospaced text.
define CodeBlock lang="css" code=""
  Box fill=#FF111827 radius=8 pad=14
    Text.mono "{code}" color=#FFE5E7EB
