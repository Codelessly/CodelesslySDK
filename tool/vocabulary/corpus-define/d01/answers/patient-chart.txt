// A trend line over a shaded normal band, drawn as a labelled chart with the range stated: the band itself is not drawn, the range is.
define VitalsTrend values=[] labels=[] low=0 high=100 h=160
  Column gap=4
    LineChart values={values} labels={labels} h={h}
    Row between
      Text.caption "Normal band {low} to {high}"
      Badge.neutral "latest {last(values)}"

define VitalTile "title" value=0 unit="" low=0 high=100 icon="favorite"
  Card gap=4 fill={value<low || value>high ? "#FFFEF2F2" : "#FFFFFFFF"}
    Row gap=6 middle
      Icon "{icon}"
      Text.caption "{title}"
    Row gap=6 bottom
      Text "{value}" size=40 weight=700 color={value<low || value>high ? "#FFDC2626" : "#FF111827"}
      Text.muted "{unit}"
    Text.caption "Normal {low} to {high}"

define MarCell "status" w=80 onTap
  Box w={w} h=40 radius=6 center middle onTap={onTap} fill={status=="given" ? "#FF16A34A" : status=="due" ? "#FFF59E0B" : status=="missed" ? "#FFDC2626" : "#FFE5E7EB"}
    Icon {status=="given" ? "check" : status=="due" ? "schedule" : status=="missed" ? "close" : "remove"} color={status=="none" ? "#FF6B7280" : "#FFFFFFFF"}
