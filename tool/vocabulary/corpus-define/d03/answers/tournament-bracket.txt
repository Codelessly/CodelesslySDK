// a line joining two ties into the one they feed
define Elbow h=88 color=#FF9CA3AF
  Box w=24 h={h} border-top={1} border-right={1} border-bottom={1} border-color={color}

// one match of a bracket: two teams, two score boxes, outlined when it is on the followed team's path
define BracketTie "title" a="TBD" b="TBD" sa="" sb="" mine=false onScoreA onScoreB
  Card gap=4 pad=8 h=72 border={mine ? 2 : 1} fill={mine ? #FFFEF3C7 : #FFFFFFFF}
    Row between middle
      Text.strong "{a}"
      Input type=number w=48 value={sa} disabled={a=="TBD"||b=="TBD"} onChange={onScoreA(result)}
    Row between middle
      Text.strong "{b}"
      Input type=number w=48 value={sb} disabled={a=="TBD"||b=="TBD"} onChange={onScoreB(result)}
