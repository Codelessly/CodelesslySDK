      Card gap=8

        SegmentedButton @channel items=["Push","Email","SMS"]

        if quiet
          Row gap=12
            TimePicker "From" @quietFrom
            TimePicker "Until" @quietTo
      Button "Save" onTap={emit saveAlerts; toast "Alerts saved"}

// A drawn map: the great-circle arc between two airports with a plane marker at the given share, shown as a labelled box with the arc.
define FlightMap from="SFO" to="NRT" progress=0.5 h=240
  Stack h={h} radius=12 fill=#FFE0F2FE
    Row between pad=16
      Text.strong "{from}"
      Text.strong "{to}"
    Text.muted "Route map {from} to {to}" center
    Box w={progress*100} h=4 fill=#FF2563EB radius=2 animate=pulse
    Icon "flight"

// Shows a gate, terminal or belt value; flashes when it has changed.
define GateChip "label" value="" changed=false
  Box pad=12 gap=2 radius=10 fill={changed ? '#FFFEF3C7' : '#FFF3F4F6'} animate={changed ? 'pulse' : 'fade'}
    Text.caption "{label}"
    Text.strong "{value}" size=24
