  Sidebar

        Card gap=12
          Heading "About you" level=3
          Input "Full name" bind=fullName
          Input "Email" "you@example.com" bind=email type=email
          Input "Phone" bind=phone type=phone

        Card gap=12
          Heading "Your home" level=3
          Text.fieldLabel "Home type"
          SegmentedButton bind=homeType items=["House","Apartment","Condo"]
          Text.fieldLabel "Do you own or rent?"
          SegmentedButton bind=ownRent items=["Own","Rent"]
          Toggle "Yard" bind=yard
          if yard
            Toggle "Fenced yard" bind=fenced
          Input "People in household" bind=household type=number
          Toggle "Children under 12 at home" bind=kids
          Input "Other pets (species and age)" bind=otherPets multiline
          Text.fieldLabel "Hours the pet would be alone: {hoursAlone}"
          Slider bind=hoursAlone min=0 max=12
      if step==2
        Card gap=12
          Heading "Two references" level=3
          each r in refs
            Row gap=8 top
              Input "Name" bind=r.name grow
              Input "Relationship" bind=r.relation grow
              Input "Phone" bind=r.phone type=phone grow
              IconButton "delete" tooltip="Remove" onTap={refs=removeAt(refs,index)}
          Button.outline "Add reference" icon="add" visible={length(refs)<3} onTap={refs=append(refs,{"name":"","relation":"","phone":""})}
      if step==3
        Card gap=8
          Heading "Review" level=3
          Text "{fullName}, {email}, {phone}"
          Text "{homeType} ({ownRent}), household of {household}, {yard ? (fenced ? 'fenced yard' : 'unfenced yard') : 'no yard'}"
          Text "{length(refs)} references"
          Checkbox "I confirm the details above are true" bind=agree
      Row between
        Button.outline "Back" visible={step>0} onTap={step=step-1}
        Spacer grow
        if step<3
          Button "Continue" onTap={(step==0 && (trim(fullName)=="" || trim(email)=="")) ? toast "Name and email are needed" : step=step+1}
        if step==3
          Button "Submit application" onTap={agree ? (submitted=true; emit submitApplication; open sent) : toast "Please confirm the details"}
