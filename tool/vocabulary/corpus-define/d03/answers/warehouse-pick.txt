// A camera barcode scanner cannot be drawn here; it shows as a labelled box with a field to type the code.
define Scanner "hint" bind=code onScan
  state typed=""
  Column gap=8
    Box h=120 radius=8 fill=#FF111827 center middle
      Icon "qr_code_scanner" color=#FFFFFFFF
      Text "{hint}" color=#FFFFFFFF
    Input "Barcode" bind=typed
    Button.outline "Verify" onTap={code=typed; onScan(typed)}

// A floor plan with the pick path; drawn as aisle boxes with the stops highlighted.
define AisleMap aisles=["A","B"] path=[] current="" onSelect
  Column gap=12
    each a in aisles
      Row gap=8 middle
        Text.strong "{a}" w=24
        Row gap=6 grow
          each s in path
            if startsWith(s,a)
              Box pad=10 radius=6 fill={s==current ? "#FF2563EB" : "#FFE5E7EB"} onTap={onSelect(s)}
                Text "{s}" color={s==current ? "#FFFFFFFF" : "#FF111827"}
