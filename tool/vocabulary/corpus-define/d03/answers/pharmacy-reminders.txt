// A refill countdown: pills left, days remaining at the daily rate, a bar, and a refill button. Draws a labelled meter, no real pharmacy link.
define RefillMeter left=30 perDay=1 onRefill
  Column gap=6
    Row between
      Text.muted "{left} pills left"
      Text.strong "{floor(left / perDay)} days until empty" color={left / perDay <= 10 ? "#FFDC2626" : "#FF111827"}
    Progress {clamp(left / 60, 0, 1)}
    if left / perDay <= 10
      Row between middle
        Text.danger "Refill soon"
        Button.outline "Request refill" onTap={onRefill}

define DoseCard "med" strength="" form="" time="" status="due" late=false onTake onSkip onSnooze
  Card gap=10 fill={late ? "#FFFEE2E2" : "#FFFFFFFF"} border={late ? 1 : 0}
    Row between middle
      Column gap=2
        Text.strong "{med}" strike={status != "due"}
        Text.muted "{strength} · {form}"
      Column gap=2 right
        Text.strong "{time}" color={late ? "#FFDC2626" : "#FF111827"}
        if late
          Badge.danger "Late"
        if status == "taken"
          Badge.success "Taken"
        if status == "skipped"
          Badge.neutral "Skipped"
    if status == "due"
      Row gap=8
        Button "Take" icon="check" onTap={onTake}
        Button.outline "Snooze" icon="snooze" onTap={onSnooze}
        Button.ghost "Skip" onTap={onSkip}

define MonthCalendar "title" days=[] lead=0 @picked onPick
  state picked=0
  Column gap=8
    Text.strong "{title}"
    Grid cols=7 gap=6
      each w in ["Mon","Tue","Wed","Thu","Fri","Sat","Sun"]
        Text.caption "{w}" center
      each i in range(lead)
        Box h=44
      each d in days
        Box h=44 radius=8 center middle fill={d.state == "taken" ? "#FFDCFCE7" : (d.state == "missed" ? "#FFFEE2E2" : "#FFFEF3C7")} border={picked == d.day ? 2 : 0} onTap={picked=d.day; onPick(d.day)}
          Text "{d.day}"
    Row gap=12
      Row gap=4 middle
        Box w=12 h=12 radius=6 fill=#FFDCFCE7
        Text.caption "Taken"
      Row gap=4 middle
        Box w=12 h=12 radius=6 fill=#FFFEE2E2
        Text.caption "Missed"
      Row gap=4 middle
        Box w=12 h=12 radius=6 fill=#FFFEF3C7
        Text.caption "Partly taken"
