// A coloured pill for an invoice status.
define StatusPill "status" w
  Badge.neutral "{upper(status)}" w={w} hidden={status!="draft"}
  Badge.info "{upper(status)}" w={w} hidden={status!="sent"}
  Badge.success "{upper(status)}" w={w} hidden={status!="paid"}
  Badge.error "{upper(status)}" w={w} hidden={status!="overdue"}

define StatCard "label" value="" caption="" tone="neutral"
  Card gap=4
    Text.caption "{label}"
    Text.heading "{value}" color={tone=="danger" ? "#FFDC2626" : tone=="success" ? "#FF16A34A" : "#FF111827"}
    Text.muted "{caption}"

define InvoiceTotals lines=[] discType="%" disc=0
  Column gap=6
    Row between
      Text.muted "Subtotal"
      Text "{money(sum(lines,l=>l.qty*l.rate),'$',2)}"
    Row between
      Text.muted "Discount"
      Text "-{money(discType=='%' ? sum(lines,l=>l.qty*l.rate)*disc/100 : disc,'$',2)}"
    Row between
      Text.muted "Tax"
      Text "{money(sum(lines,l=>l.qty*l.rate*l.tax/100),'$',2)}"
    Divider
    Row between
      Text.strong "Total due"
      Text.heading "{money(sum(lines,l=>l.qty*l.rate)-(discType=='%' ? sum(lines,l=>l.qty*l.rate)*disc/100 : disc)+sum(lines,l=>l.qty*l.rate*l.tax/100),'$',2)}"
