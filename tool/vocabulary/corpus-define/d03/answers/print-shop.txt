// shows the live price with bulk discount tiers
define PriceSummary qty=0 subtotal=0
  Card gap=10
    Text.strong "Order total"
    Row between
      Text.muted "{qty} prints"
      Text "{money(subtotal,"$",2)}"
    Row between
      Text.muted "Bulk discount ({(qty>=50 ? 20 : qty>=25 ? 15 : qty>=10 ? 10 : 0)}%)"
      Text.success "-{money(subtotal*(qty>=50 ? 0.2 : qty>=25 ? 0.15 : qty>=10 ? 0.1 : 0),"$",2)}"
    Divider
    Row between
      Text.strong "Total"
      Text.strong "{money(subtotal*(1-(qty>=50 ? 0.2 : qty>=25 ? 0.15 : qty>=10 ? 0.1 : 0)),"$",2)}"
    PriceTiers qty={qty}

define PriceTiers qty=0
  Column gap=6
    Text.caption "Buy more, save more"
    each tier in [{"min":10,"off":10},{"min":25,"off":15},{"min":50,"off":20}]
      Row between middle
        Row gap=6 middle
          Icon "{qty>=tier.min ? "check_circle" : "radio_button_unchecked"}"
          Text.small "{tier.min}+ prints"
        Badge.success "{tier.off}% off" visible={qty>=tier.min}
        Text.muted "{tier.off}% off" visible={qty<tier.min}

// a real draggable crop needs pointer tracking; shown as the photo with a ratio frame over it
define CropFrame src="" ratio="3:2" @zoom
  state zoom=1
  Stack h=280 radius=8
    Image "{src}" h=280 fill=#FF111827
    Box w={200*zoom/2+120} h={ratio=="1:1" ? 200*zoom/2+120 : ratio=="3:2" ? 133*zoom/2+80 : ratio=="4:5" ? 250*zoom/2+150 : 280} border=2 fill=#22FFFFFF drag=crop(ratio)
    Text.caption "{ratio} frame" color=#FFFFFFFF
