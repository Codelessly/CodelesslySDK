// A floor plan drawing with numbered pins; shown here as a labelled box with pins positioned by share of its width and height.
define FloorPlan "name" pins=[] selected=0 placing=false onPick onPlace
  Stack h=420 fill=#FFF3F4F6 radius=8 border=1 onTap={placing ? onPlace({"x":0.5,"y":0.5}) : close}
    Text.muted "Floor plan: {name}" center middle
    each p in pins
      Box w=28 h=28 radius=14 fill={p.status=="Fixed" ? "#FF16A34A" : (p.priority=="High" ? "#FFDC2626" : "#FFF59E0B")} border={p.id==selected ? 3 : 0} left={p.x*100}% top={p.y*100}% center middle onTap={onPick(p.id)}
        Text "{p.id}" color=#FFFFFFFF weight=700

// A photo with drawn markup (arrows, circles, text); a real drawing surface shown as a labelled box.
define PhotoMarkup "label" src="" bind=marks tools=["arrow"] color=#FFDC2626 h=200
  state marks=[]
  state tool="arrow"
  Column gap=6
    Text.fieldLabel "{label}"
    Stack h={h} fill=#FF1F2937 radius=8
      Image "{src}" h={h}
      Text.muted "Draw on the photo" center middle
    Row gap=8 middle
      each t in tools
        Chip "{t}" selected={tool==t} onTap={tool=t}
      Spacer grow
      Button.ghost "Undo" onTap={marks=removeAt(marks,length(marks)-1)}
      Button.ghost "Clear" onTap={marks=[]}
