          Text.strong "Open Settings"
          Text "Sign in at notion.so, open Settings and choose Plans."
        if step==2
          Text.strong "Choose Downgrade to Free"
          Text "Select Downgrade, then confirm. Your workspace stays; paid features end at the next billing date."
        if step==3
          Text.strong "Confirm by email"
          Text "Look for the confirmation email, then mark this subscription as cancelled here."
          Button "Mark as cancelled" onTap={toast "Notion marked as cancelled"; go subscriptions}
        Row between
          Button.outline "Back" onTap={step=max(0,step-1)}
          Button "Next" hidden={step==3} onTap={step=min(3,step+1)}

// A month grid of days; marked days show a dot of the payment's colour
define MonthCalendar "title" marks=[] onPick
  state picked=0
  Column gap=8
    Text.strong "{title}"
    Grid cols=7 gap=4
      each d in range(30)
        Box h=56 radius=8 pad=4 border={picked==d+1 ? 2 : 1} onTap={picked=d+1; onPick(d+1)}
          Text.small "{d+1}"
          each m in filter(marks,x=>x.day==d+1)
            Dot color={m.color}
