state day="Mon 16"
state time=""
state booked=false
state days=["Mon 16","Tue 17","Wed 18","Thu 19","Fri 20"]
state slots=[{"t":"09:00","h":9,"m":0},{"t":"10:00","h":10,"m":0},{"t":"11:00","h":11,"m":0},{"t":"13:00","h":13,"m":0},{"t":"14:00","h":14,"m":0},{"t":"15:00","h":15,"m":0}]
state taken=day=="Mon 16"?["10:00","13:00"]:(day=="Tue 17"?["09:00","14:00"]:(day=="Wed 18"?["11:00","15:00"]:(day=="Thu 19"?["09:00","10:00","13:00"]:["11:00","14:00"])))
state s=find(slots,x=>x.t==time)
state endH=s.h+floor((s.m+30)/60)
state endM=(s.m+30)%60
state endT=(endH<10?"0":"")+endH+":"+(endM<10?"0":"")+endM
ChatCard "Team meeting" "30 minutes · next week" icon:"event_rounded" status:=booked?"Booked":(length(slots)-length(taken))+" slots free" tone:=booked?"good":"accent"
  if !booked
    Row gap:8
      PersonChip "Priya"
      PersonChip "Marcus"
      PersonChip "Elena"
    DayPicker#days =day days:=days -> time=""
    SectionLabel "Time" "Struck-through times are taken"
    ChoiceChips#slots =time options:=map(slots,x=>x.t) disabled:=taken columns:3
    TotalBar#confirm "{day} Nov · 30 min" amount:"{time==""?"Pick a time":time+" to "+endT}" cta:"Confirm" enabled:=time!="" -> booked=true; emit book_meeting day+" "+time
  if booked
    Receipt "Meeting booked" "{day} Nov, {time} to {endT}, with Priya, Marcus and Elena." icon:"event_available_rounded"
