let sum = sum(matches, i => i.amount)

  InputField "Search" "Customer" value:query icon:search
    onChanged: query = value
  ChoiceChips value:status options:["All","Paid","Due","Overdue"] columns:4
    onChanged: status = value
  TableHeader value:"amount" desc:sortAsc == false columns:["Customer","Amount","Status"] keys:["customer","amount","status"]
    onPressed: sortAsc = sortAsc == true ? false : true
  for i in rows
    TableRow cells:["{i.customer}","${i.amount}"] badge:i.status
