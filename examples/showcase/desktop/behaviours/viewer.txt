// A photo viewer over the page, three ways. With Cards (--design-system design_systems/cards):
//   --tap trip/cell#2                 the PhotoGallery opens its third photo large over the window (the grid stays)
//   --tap shot#1                      a gallery of pictures of your own, opened at the one tapped
//   --tap cover                       one picture
// then, on the viewer (its parts are named after its id: tripsViewerPhoto here is <screen>ViewerPhoto):
//   --double-tap shotsPhoto           2x where it was tapped; again: back to 1x
//   --pinch shotsPhoto=3              a pinch; --wheel shotsPhoto=-300 the wheel
//   --drag shotsPhoto=-300,0          at rest: the next picture; zoomed: pans, kept over its place
//   --drag shotsPhoto=0,300           at rest: closes; --key Escape, --tap shotsClose and a tap beside the picture too
//   --key ArrowRight, --tap shotsNext, --tap shotsThumb#2
// photo.png and tall.png are any pictures beside site.json (ratio is their width over their height).
state open=0
state at=0
state shots=[{"url":"photo.png","caption":"The valley","ratio":1.5},{"url":"tall.png","caption":"The tower","ratio":0.6},{"url":"photo.png","caption":"The valley again","ratio":1.5}]

  Section
    Heading "Trip photos" level=1
    PhotoGallery#trip @open columns=4 photos=[{"kind":"harbor","caption":"Harbour at sunrise"},{"kind":"market","caption":"Old town market"},{"kind":"forest","caption":"Forest trail"},{"kind":"dunes","caption":"Desert dunes"},{"kind":"skyline","caption":"Evening skyline"},{"kind":"lagoon","caption":"Hidden lagoon"},{"kind":"mountain","caption":"Mountain hut"},{"kind":"festival","caption":"Street festival"}]
    Text.muted#under "Nothing under the gallery moves when a photo opens."
    Row gap=12
      each s in shots
        Image#shot "{s.url}" w=160 h=110 radius=10 onTap={at=index; open shots}
      Image#cover "photo.png" w=160 h=110 radius=10 onTap={open zoom}

Lightbox#shots photos={shots} @at
