// The words of the folded behaviours on one page: a shortcut, a search with what matched marked, a table worked by keys,
// a list reordered by its handles, rows with swipe actions, a log that follows its end, a calendar with a range, a range
// slider, a stepping number, a one-time code, a rating, a splitter, a hover card with an arrow, a guarded
// dialog, a sheet with stops, confirm, announce, a live text, a saved value and device.hover.
// tool/notation/notation.sh expand words.txt --design-system design_systems/cards --named --id words --width 1280 --out <site>/words.json

//   then, for a state: --key Tab, --tap ordersTable, --tap stayCal@2026-11-23, --tap editBtn, --tap filtersBtn, --tap removeBtn, --hover author
state q=""
state draft="" saved

state arrival="2026-11-20"
state departure="2026-11-22"
state low=40
state high=160
state qty=2
state code=""
state stars=4
state paneW=200
state name=""
state preview="Chur"
key mod+k onPress={open search}
key alt+shift+l onPress={lines=append(lines,"Rebuilt at step {length(lines)+1}")}
Page
  Nav
    Brand "Tidewater" icon="waves"
    Row gap=4
      Link "Orders" href="#orders"
      Link "Trip" href="#trip"

    Row gap=8 middle
      Text.muted "Search" shortcut="mod+k"
      IconButton#searchBtn "search" tooltip="Search orders" onTap={open search}

      Button#editBtn "Edit profile" onTap={open edit}
  Section#orders
    Heading "Orders" level=1
    Row gap=12 middle
      Input#find "Find a customer" bind=q w=280
      Text.muted#count "{length(fuzzy(orders,q,x=>x.name))} of {length(orders)} orders" live
      Link#author "About Ada" preview=bio
    each o in fuzzy(orders,q,x=>x.name)
      Row gap=12 middle
        Text#match "{o.name}" mark={q} grow
        Badge "{o.status}"
    DataTable#ordersTable rows={orders} columns=[{"key":"name","label":"Customer"},{"key":"status","label":"Status"},{"key":"total","label":"Total"}] bind=row h=220 onSort={announce "Sorted by {result.column}"} onRowTap={toast "Row {result.rowIndex + 1}"}
    Row gap=12 middle
      Text.muted "Row chosen: {row}"
      Button.outline#removeBtn "Remove all orders" onTap={confirm "Remove every order?" then {orders=[]; row=-1; announce "Orders removed"}}
  Section#trip tint
    Heading "Trip"
    Row gap=32 top
      Column grow gap=12
        Text.strong "Stops, in order"
        each stop in trip reorder
          Row gap=8 middle
            Handle
            Text "{stop}" grow
            Button.ghost "Preview" onTap={preview=stop}
        if !device.hover
          Text.muted "Tap Preview to see a stop."
        Text.muted#previewing "Previewing {preview}"
      Column grow gap=12
        Text.strong "Classes"
        each c in classes
          SwipeAction "Book" side=end when={!c.full} onTap={booked=result.item.name; announce "Booked {result.item.name}"}

          Row gap=8 middle
            Text "{c.name}" grow
            IconButton "more_vert" tooltip="Class actions" onTap={booked=c.name}
        Text.muted "Booked: {booked}"
      Column grow gap=12
        Text.strong "Build log"
        Card
          each line in lines follow h=140
            Text.muted "{line}"
        Button.outline#logBtn "Add a line" onTap={lines=append(lines,"Step {length(lines)+1} done")}
  Section#stay

    Row gap=32 top

        Calendar#stayCal bind=arrival bindEnd=departure disabledDates=["2026-11-24","2026-11-25"] marks={"2026-11-20":"£225","2026-11-21":"£240"} label="Your stay"
        Text.muted "{arrival} to {departure}"
      Column grow gap=16
        Text.strong "Price a night: {round(low,0)} to {round(high,0)}"
        Slider#price bind=low bindEnd=high min=0 max=300 step=10 label="Price range"
        Row gap=16 middle
          Input#guests "Guests" bind=qty type=number min=1 max=8 step=1 w=120
          CodeInput#otp "Booking code" bind=code length=6 digits
        Row gap=16 middle
          Rating "Your rating" bind=stars

      Column#notesPane w={paneW} pad=12 fill=#FFF3F4F6
