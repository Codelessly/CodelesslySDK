state remaining=300
state running=false

every 1s when running -> remaining=max(0,remaining-1); running=remaining-1>0

Card gap:16 pad:16
  Text.title "Countdown"
  Row center
    Text "{floor(remaining/60)}:{remaining%60<10?'0':''}{remaining%60}" size:64 weight:700
  Progress =remaining/300
  Row gap:8 center
    Button "Start" visible:=!running&&remaining>0 -> running=true
    Button.secondary "Pause" visible:=running -> running=false
    Button.outline "Reset" -> running=false; remaining=300
  if remaining==0
    Text.success "Time's up!"
