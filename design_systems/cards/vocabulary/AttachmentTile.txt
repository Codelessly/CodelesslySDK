// A file attached to a message: its icon, name and size, with a download or a remove button.
AttachmentTile "name" subtitle icon="description" removable=false onTap onDownload onRemove
  Row gap=10 pad=[10,8,10,8] radius=10 fill=tile border=line middle onTap={onTap}
    Icon "{icon}" color=accent size=20
    Column grow gap=2
      Text.strong "{name}"
      if subtitle != null
        Text.caption "{subtitle}"
    if onDownload
      IconButton "download" onTap={onDownload}
    if removable
      IconButton "close" onTap={onRemove}
