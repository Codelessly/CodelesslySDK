// One message of a chat: its text (and the lines under it), at the right in the accent's tint when mine (or role="user"); the sender over it unless grouped with the one before.
ChatBubble "text" mine=false role sender time grouped=false
  Row w=fill top
    if mine || role == "user"
      Spacer grow
    Column gap=2 maxw=480
      if !grouped && !(mine || role == "user") && sender != null
        Text.caption "{sender}"
      Column gap=4 pad=[12,8,12,8] radius=14 fill={mine || role == "user" ? "#FFDBEAFE" : "#FFFFFFFF"} border={mine || role == "user" ? "#00000000" : "#FFE5E7EB"}
        if text != null
          Text "{text}"
        ...
      if time != null
        Text.caption "{time}"
