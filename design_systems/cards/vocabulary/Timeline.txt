// A clip editor's timeline: the ruler and tracks written under it, with the playhead's time and zoom buttons.
Timeline#tracks zoom=1 playhead=0 onZoom
  Column w=fill gap=0 radius=8 fill=#FF1E1E24 border=#FF2E2E36
    Row w=fill gap=8 pad=[8,6,8,6] middle
      Text.caption "{fixed(playhead, 1)} s" color=#FFFFFFFF
      Spacer grow
      IconButton "zoom_out" onTap={onZoom(max(0.25, zoom / 2))}
      IconButton "zoom_in" onTap={onZoom(min(8, zoom * 2))}
    Column w=fill gap=2
      ...

// A list of dated events down a rail: each line under it is one event.
Timeline.vertical#events
  Column w=fill gap=12 pad=[12,0,0,0] border-left=line
    ...

// A route's stops down a rail, with where the vehicle is: each line under it is a stop.
Timeline.vertical#route bind=at position marker="directions_bus"
  state at=0
  Column w=fill gap=10 pad=[12,0,0,0] border-left=accent
    if position != null
      Row gap=6 middle
        Icon "{marker}" size=18 color=accent
        Text.caption "Between stops {floor(position) + 1} and {floor(position) + 2}"
    ...
