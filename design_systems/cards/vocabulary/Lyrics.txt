// The words of a song, line by line: lines is a list of {"t": seconds, "text": …} (or plain texts); the line sounding at time (or bind) is large and in the highlight; a tap on a line seeks to it (onSeek: result is its time).
Lyrics lines=[] time bind=pos h=200 highlight=accent onSeek
  state pos=0
  Row w=fill h={h} stretch
    Column grow gap=8 pad=[0,4,0,4] scroll
      each line in lines
        Text "{coalesce(line.text, line)}" size={index == count(lines, l => coalesce(l.t, 0) <= coalesce(time, pos)) - 1 ? 20 : 16} weight={index == count(lines, l => coalesce(l.t, 0) <= coalesce(time, pos)) - 1 ? 700 : 400} color={index == count(lines, l => coalesce(l.t, 0) <= coalesce(time, pos)) - 1 ? highlight : "inkFaint"} onTap={pos=coalesce(line.t, 0); onSeek(coalesce(line.t, 0))}
